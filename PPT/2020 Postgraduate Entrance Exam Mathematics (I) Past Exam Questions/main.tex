\section{选择题}

\begin{frame}{第1题}
\begin{block}{题目}
当 $x\to 0^+$ 时，下列无穷小量中最高阶的是
$\mathrm{A.}\int_{0}^{x}(e^{t^{2}}-1)\mathrm{d}t \quad
\mathrm{B.}\int_{0}^{x}\ln (1+\sqrt{t^{3}})\mathrm{d}t \quad
\mathrm{C.}\int_{0}^{\sin x}\sin t^{2}\mathrm{d}t \quad
\mathrm{D.}\int_{0}^{1-\cos x}\sqrt{\sin^{3}t}\mathrm{d}t$.
\end{block}
\textcolor{red}{\textbf{答案：}$\mathbf{D}$} \\
\textbf{解析：}
\begin{align*}
&\underset{x\rightarrow 0^+}{\lim}\frac{\int_0^x{\left( e^{t^2}-1 \right) \mathrm{d}t}}{x^3}=\underset{x\rightarrow 0^+}{\lim}\frac{e^{x^2}-1}{3x^2}=\underset{x\rightarrow 0^+}{\lim}\frac{x^2}{3x^2}=\frac{1}{3}
\Longrightarrow \int_0^x{\left( e^{t^2}-1 \right) \mathrm{d}t}\sim x^3\,\,\left( x\rightarrow 0^+ \right) ;
\\
&\underset{x\rightarrow 0^+}{\lim}\frac{\int_0^x{\ln \left( 1+\sqrt{t^3} \right)}\mathrm{d}t}{x^{\frac{5}{2}}}=\underset{x\rightarrow 0^+}{\lim}\frac{2\ln \left( 1+\sqrt{x^3} \right)}{5x^{\frac{3}{2}}}=\underset{x\rightarrow 0^+}{\lim}\frac{2x^{\frac{3}{2}}}{5x^{\frac{3}{2}}}=\frac{2}{5}
\\
&\Longrightarrow \int_0^x{\ln \left( 1+\sqrt{t^3} \right)}\sim x^{\frac{3}{2}}\,\,\left( x\rightarrow 0^+ \right) ;
\end{align*}
\end{frame}

\begin{frame}{第1题}
\begin{align*}
&\underset{x\rightarrow 0^+}{\lim}\frac{\int_0^{\sin x}{\sin t^2\mathrm{d}t}}{x^3}=\underset{x\rightarrow 0^+}{\lim}\frac{\sin \left( \sin ^2x \right) \cdot \cos x}{3x^2}=\underset{x\rightarrow 0^+}{\lim}\frac{\sin ^2x}{3x^2}=\frac{1}{3}
\\
&\Longrightarrow \int_0^{\sin x}{\sin t^2\mathrm{d}t}\sim x^3\,\,\left( x\rightarrow 0^+ \right) ;
\\
&\underset{x\rightarrow 0^+}{\lim}\frac{\int_0^{1-\cos x}{\sqrt{\sin ^3t}\mathrm{d}t}}{x^5}=\underset{x\rightarrow 0^+}{\lim}\frac{\sin x\cdot \sin ^{\frac{3}{2}}\left( 1-\cos x \right)}{5x^4}=\underset{x\rightarrow 0^+}{\lim}\frac{x\left( 1-\cos x \right) ^{\frac{3}{2}}}{5x^4}
\\
&=\underset{x\rightarrow 0^+}{\lim}\frac{x\left( \frac{x^2}{2} \right) ^{\frac{3}{2}}}{5x^4}=\frac{1}{40}\Longrightarrow \int_0^{1-\cos x}{\sqrt{\sin ^3t}\mathrm{d}t}\sim x^5\,\,\left( x\rightarrow 0^+ \right) .
\end{align*}
\end{frame}

\begin{frame}{第2题}
\begin{block}{题目}
设 $f(x)$ 在 $(-1,1)$ 内有定义，且 $\lim\limits_{x\to 0}f(x)=0$，则
\begin{enumerate}[A.]
\item $\lim\limits_{x\to 0}\dfrac{f(x)}{\sqrt{|x|}}=0\Rightarrow f(x)$ 在 $x=0$ 可导；
\item $\lim\limits_{x\to 0}\dfrac{f(x)}{x^{2}}=0\Rightarrow f(x)$ 在 $x=0$ 可导；
\item $\lim\limits_{x\to 0}\dfrac{f(x)}{\sqrt{|x|}}=0$ 是 $f(x)$ 在 $x=0$ 可导的必要条件；
\item $\lim\limits_{x\to 0}\dfrac{f(x)}{x^{2}}=0$ 是 $f(x)$ 在 $x=0$ 可导的必要条件.
\end{enumerate}
\end{block}
\textcolor{red}{\textbf{答案：}$\mathbf{C}$}
\end{frame}

\begin{frame}{第2题}
\textbf{解析：}$f$ 在$x=0$处可导的充要条件是$\underset{x\rightarrow 0}{\lim}\frac{f(x)}{x}=f'(0)$，也即
\begin{align*}
f(x)\sim x\text{或}o(x) \,(x\to 0).
\end{align*}
于是
\[
\underset{x\rightarrow 0}{\lim}\frac{f(x)}{\sqrt{|x|}}=\underset{x\rightarrow 0}{\lim}\frac{f(x)}{x}\cdot \frac{x}{\sqrt{|x|}}=0,
\]
故C正确，A、B、D错误.

A 反例：$f(x)=|x|$，$\dfrac{f(x)}{\sqrt{|x|}}=\sqrt{|x|}\to 0\,(x\to 0)$，但$f$在$x=0$处不可导；

B 反例：$f\left( x \right) =\begin{cases}
1,&x=0,\\
x^2,&x\ne 0,\\
\end{cases}$显然$\lim\limits_{x\to 0}\dfrac{f(x)}{x^{2}}=0$，但$f$在$x=0$不连续;

D 反例：$f(x)=x$ 在$x=0$处可导但 $\dfrac{f(x)}{x^2}=\dfrac1x\not\to 0\,(x\to 0)$.
\end{frame}

\begin{frame}{第3题}
\begin{block}{题目}
设 $f(x,y)$ 在 $(0,0)$ 可微，$f(0,0)=0$，$\boldsymbol{n}=\left.\left(f_x,f_y,-1\right)\right|_{(0,0)}$，非零向量 $\boldsymbol{d}$ 与 $\boldsymbol{n}$ 垂直，则
\begin{enumerate}[A.]
\item $\lim\limits_{(x,y)\to(0,0)}\dfrac{|\boldsymbol{n}\cdot(x,y,f(x,y))|}{\sqrt{x^2+y^2}}=0$；
\item $\lim\limits_{(x,y)\to(0,0)}\dfrac{|\boldsymbol{n}\times(x,y,f(x,y))|}{\sqrt{x^2+y^2}}=0$；
\item $\lim\limits_{(x,y)\to(0,0)}\dfrac{|\boldsymbol{d}\cdot(x,y,f(x,y))|}{\sqrt{x^2+y^2}}=0$；
\item $\lim\limits_{(x,y)\to(0,0)}\dfrac{|\boldsymbol{d}\times(x,y,f(x,y))|}{\sqrt{x^2+y^2}}=0$.
\end{enumerate}
\end{block}
\textcolor{red}{\textbf{答案：}$\mathbf{A}$}
\end{frame}

\begin{frame}{第3题}
\textbf{解析：}由可微定义知$f=f_x\cdot x+f_y\cdot y+o\left(\sqrt{x^2+y^2}\right)$，故
\begin{align*}
&\left| \boldsymbol{n}\cdot \left( x,y,f \right) \right|=\left| f_x\cdot x+f_y\cdot y-f \right|=o\left( \sqrt{x^2+y^2} \right) 
\\
&\Longrightarrow \lim_{(x,y)\rightarrow (0,0)} \frac{|\boldsymbol{n}\cdot (x,y,f(x,y))|}{\sqrt{x^2+y^2}}=\lim_{(x,y)\rightarrow (0,0)} \frac{o\left( \sqrt{x^2+y^2} \right)}{\sqrt{x^2+y^2}}=0.
\end{align*}
故A正确.

由 $f$ 在 $(0,0)$ 可微，设 $f_x(0,0)=a$，$f_y(0,0)=b$，则有
\[
f(x,y)=ax+by+o(\rho),\quad \rho=\sqrt{x^2+y^2}.
\]
并且$f$在$(0,0)$处的切平面$\Gamma$为
\begin{align*}
\Gamma: z = ax+by.
\end{align*}
\end{frame}

\begin{frame}{第3题}
记 $\boldsymbol{v}=(x,y,f(x,y))$，$\boldsymbol{w}=(x,y,ax+by)$，$\boldsymbol{d}=(d_1,d_2,d_3)$，则
\[
\boldsymbol{v}=\boldsymbol{w}+(0,0,o(\rho)).
\]
其中 $\boldsymbol{w}$ 完全落在切平面$\Gamma$内，$\boldsymbol{n}=(a,b,-1)$ 是该切平面的法向量，$\boldsymbol{d}$ 也是该切平面内的非零向量。

对于B：由$\left| \boldsymbol{n}\times \boldsymbol{v} \right|=\left| \boldsymbol{n}\times \boldsymbol{w}+\boldsymbol{n}\times \left( 0,0,o\left( \rho \right) \right) \right|
$及三角不等式可得
\begin{align*}
\left| \left| \boldsymbol{n}\times \boldsymbol{w} \right|-\left| \boldsymbol{n}\times \left( 0,0,o\left( \rho \right) \right) \right| \right|\leqslant \left| \boldsymbol{n}\times \boldsymbol{v} \right|\leqslant \left| \boldsymbol{n}\times \boldsymbol{w} \right|+\left| \boldsymbol{n}\times \left( 0,0,o\left( \rho \right) \right) \right|.
\end{align*}
作极坐标变换
\begin{align*}
\begin{cases}
x=\rho \cos \theta\\
y=\rho \sin \theta\\
\end{cases},\quad \rho =\sqrt{x^2+y^2}\in \mathbb{R} ,\quad \theta \in \left[ 0,2\pi \right].
\end{align*}
\end{frame}

\begin{frame}{第3题}
又因$\boldsymbol{n}\perp\boldsymbol{w}$，故
\begin{align*}
\left| \boldsymbol{n}\times \boldsymbol{w} \right|&=\left| \boldsymbol{n} \right|\left| \boldsymbol{w} \right|=\sqrt{a^2+b^2+1}\sqrt{x^2+y^2+\left( ax+by \right) ^2}
\\
&=\rho \sqrt{a^2+b^2+1}\sqrt{1+\left( a\cos \theta +b\sin \theta \right) ^2};
\end{align*}
\begin{align*}
\left| \boldsymbol{n}\times \left( 0,0,o\left( \rho \right) \right) \right|=\left| \left( b\cdot o\left( \rho \right) ,-a\cdot o\left( \rho \right) ,0 \right) \right|=\sqrt{a^2+b^2}\cdot o\left( \rho \right) = o\left( \rho \right) ,\quad \rho \rightarrow 0.
\end{align*}
因此
\begin{align*}
\left| \boldsymbol{n}\times \boldsymbol{v} \right|\sim \rho \sqrt{a^2+b^2+1}\sqrt{1+\left( a\cos \theta +b\sin \theta \right) ^2}+o\left( \rho \right) ,\quad \rho \rightarrow 0.
\end{align*}
于是
\begin{align*}
\underset{\left( x,y \right) \rightarrow \left( 0,0 \right)}{\lim}\frac{\left| \boldsymbol{n}\times \boldsymbol{v} \right|}{\rho}=\sqrt{a^2+b^2+1}\sqrt{1+\left( a\cos \theta +b\sin \theta \right) ^2},
\end{align*}
这个极限随着趋于原点的方向$\theta$变化而不同，因此这个极限根本不存在.
\end{frame}

\begin{frame}{第3题}
对于C：$\boldsymbol{d}\cdot\boldsymbol{v}=\boldsymbol{d}\cdot\boldsymbol{w}+o(\rho)$。因 $\boldsymbol{d}$ 与 $\boldsymbol{w}$ 均在切平面内，所以作极坐标变换
\begin{align*}
\begin{cases}
x=\rho \cos \theta\\
y=\rho \sin \theta\\
\end{cases},\quad \rho =\sqrt{x^2+y^2}\in \mathbb{R} ,\quad \theta \in \left[ 0,2\pi \right] .
\end{align*}
并令$A=d_1+ad_3,B=d_2+bd_3$，则
\begin{align*}
&\left| \boldsymbol{d}\cdot \boldsymbol{v} \right|=\left| \boldsymbol{d}\cdot \boldsymbol{w}+o\left( \rho \right) \right|=\left| \left( d_1+ad_3 \right) x+\left( d_2+bd_3 \right) y+o\left( \rho \right) \right|
\\
&=\left| Ax+By+o\left( \rho \right) \right|=\rho \left| A\cos \theta +B\sin \theta \right|+o\left( \rho \right) ,\quad \rho \rightarrow 0.
\end{align*}
从而
\begin{align*}
\underset{\left( x,y \right) \rightarrow \left( 0,0 \right)}{\lim}\frac{\left| \boldsymbol{d}\cdot \boldsymbol{v} \right|}{\rho}=\left| A\cos \theta +B\sin \theta \right|,
\end{align*}
这个极限随着趋于原点的方向$\theta$变化而不同，因此这个极限根本不存在.
\end{frame}

\begin{frame}{第3题}
对于D：由$\left| \boldsymbol{d}\times \boldsymbol{v} \right|=\left| \boldsymbol{d}\times \boldsymbol{w}+\boldsymbol{d}\times \left( 0,0,o\left( \rho \right) \right) \right|
$及三角不等式可得
\begin{align*}
\left| \left| \boldsymbol{d}\times \boldsymbol{w} \right|-\left| \boldsymbol{d}\times \left( 0,0,o\left( \rho \right) \right) \right| \right|\leqslant \left| \boldsymbol{d}\times \boldsymbol{v} \right|\leqslant \left| \boldsymbol{d}\times \boldsymbol{w} \right|+\left| \boldsymbol{d}\times \left( 0,0,o\left( \rho \right) \right) \right|.
\end{align*}
先计算 $\boldsymbol{d}\times \boldsymbol{w}$ 的模长。
\begin{align*}
\boldsymbol{d}\times \boldsymbol{w}=
\begin{vmatrix}
\boldsymbol{i} & \boldsymbol{j} & \boldsymbol{k}\\
d_1 & d_2 & d_3\\
x & y & ax+by
\end{vmatrix}=\left( a d_2 x + \left( b d_2 - d_3 \right) y, \left( d_3 - a d_1 \right) x - b d_1 y, -d_2 x + d_1 y \right).
\end{align*}
做极坐标变换 $x=\rho \cos \theta, y=\rho \sin \theta$，则
{\footnotesize \begin{align*}
\left| \boldsymbol{d}\times \boldsymbol{w} \right| = C(\theta)\rho,
\end{align*}}
其中
{\footnotesize 
\begin{align*}
C(\theta)=\sqrt{\left( a d_2 \cos \theta + \left( b d_2 - d_3 \right) \sin \theta \right) ^2 + \left( \left( d_3 - a d_1 \right) \cos \theta - b d_1 \sin \theta \right) ^2 + \left( -d_2 \cos \theta + d_1 \sin \theta \right) ^2}.
\end{align*}}
\end{frame}

\begin{frame}{第3题}
并且
\begin{align*}
\left| \boldsymbol{d}\times \left( 0,0,o\left( \rho \right) \right) \right| =\left| \left( d_2\cdot o\left( \rho \right) ,-d_1\cdot o\left( \rho \right) ,0 \right)\right|= \sqrt{d_1^2+d_2^2}\cdot o\left( \rho \right) = o\left( \rho \right) ,\quad \rho \rightarrow 0.
\end{align*}
因此
\begin{align*}
\left| \boldsymbol{d}\times \boldsymbol{v} \right| = C(\theta)\rho + o\left( \rho \right) ,\quad \rho \rightarrow 0.
\end{align*}
于是
\begin{align*}
\underset{\left( x,y \right) \rightarrow \left( 0,0 \right)}{\lim}\frac{\left| \boldsymbol{d}\times \boldsymbol{v} \right|}{\rho}=C(\theta),
\end{align*}
这个极限随着趋于原点的方向$\theta$变化而不同，因此这个极限根本不存在.
\end{frame}

\begin{frame}{第4题}
\begin{block}{题目}
设 $R$ 为幂级数 $\sum_{n=1}^{\infty} a_n x^n$ 的收敛半径，$r$ 是实数，则
\begin{enumerate}[A.]
\item $\sum_{n=1}^{\infty} a_{2n}r^{2n}$ 发散时，$|r|\geqslant R$；
\item $\sum_{n=1}^{\infty} a_{2n}r^{2n}$ 发散时，$|r|\leqslant R$；
\item $ |r|\geqslant R$时，$\sum_{n=1}^{\infty} a_{2n}r^{2n}$ 发散；
\item $ |r|\leqslant R$时，$\sum_{n=1}^{\infty} a_{2n}r^{2n}$ 发散.
\end{enumerate}
\end{block}
\textcolor{red}{\textbf{答案：}$\mathbf{A}$}
\end{frame}

\begin{frame}{第4题}
\textbf{解析：}若 $|r|<R$，原级数绝对收敛，故
\begin{align*}
\sum_{n=1}^{\infty} a_{2n}r^{2n}\leqslant \sum_{n=1}^{\infty} |a_{n}||r|^{n}<+\infty.
\end{align*}
取逆否命题即得$\sum_{n=1}^{\infty} a_{2n}r^{2n}$发散时，必有 $|r|\geqslant R$.
\end{frame}

\begin{frame}{第4题}
选项 B ：若 \(\sum_{n=1}^{\infty} a_{2n}r^{2n}\) 发散，则 \(|r|\le R\)。但发散只能推出 \(|r|\ge R\)，不能推出 \(|r|\le R\)。

反例：取 \(a_n=1\)，则
\[
\sum_{n=1}^{\infty} a_nx^n=\sum_{n=1}^{\infty}x^n
\]
的收敛半径 \(R=1\)。取 \(r=2\)，则
\[
\sum_{n=1}^{\infty} a_{2n}r^{2n}
=\sum_{n=1}^{\infty}2^{2n}
=\sum_{n=1}^{\infty}4^n
\]
发散，但$|r|=2\leqslant R=1$不成立，故 B 错。
\end{frame}

\begin{frame}{第4题}
选项 C ：若 \(|r|\geqslant R\)，则 \(\sum_{n=1}^{\infty} a_{2n}r^{2n}\) 发散。但 \(|r|\geqslant R\) 时，该子级数不一定发散。

反例：令
\[
a_{2n-1}=1,\qquad a_{2n}=0\quad(n=1,2,\cdots).
\]
则原级数为
\[
\sum_{n=1}^{\infty}a_nx^n=x+x^3+x^5+\cdots,
\]
其收敛半径 \(R=1\)。但
\[
\sum_{n=1}^{\infty}a_{2n}r^{2n}=0
\]
恒收敛。取 \(r=2\)，有
\[
|r|=2\geqslant R=1,
\]
但该子级数并不发散，故 C 错。
\end{frame}

\begin{frame}{第4题}
选项D：若 \(|r|\leqslant R\)，则 \(\sum_{n=1}^{\infty} a_{2n}r^{2n}\) 发散。但 \(|r|\leqslant R\) 时，该子级数不一定发散。

反例：仍取 \(a_n=1\)，则
\[
\sum_{n=1}^{\infty}a_nx^n=\sum_{n=1}^{\infty}x^n
\]
的收敛半径 \(R=1\)。取 \(r=\frac12\)，有
\[
|r|=\frac12\leqslant 1=R,
\]
但
\[
\sum_{n=1}^{\infty}a_{2n}r^{2n}
=\sum_{n=1}^{\infty}\left(\frac12\right)^{2n}
=\sum_{n=1}^{\infty}\frac1{4^n}
\]
收敛，并不发散，故 D 错。
\end{frame}

\begin{frame}{第5题}
\begin{block}{题目}
若矩阵 $\bm{A}$ 经初等列变换化成 $\bm{B}$，则
\begin{enumerate}[A.]
\item 存在矩阵$P$，使得$PA=B$；
\item 存在矩阵$P$，使得$BP=A$；
\item 存在矩阵$P$，使得$PB=A$；
\item 方程组$Ax=0$ 与 $Bx=0$ 同解.
\end{enumerate}
\end{block}
\textcolor{red}{\textbf{答案：}$\mathbf{B}$} \\
\textbf{解析：}初等列变换即右乘可逆矩阵 $P$，$AP=B$，故 $BP^{-1}=A$，令 $P'=P^{-1}$ 即 $BP'=A$.列变换不保持齐次方程组同解（行变换才同解）.
\end{frame}

\begin{frame}{第6题}
\begin{block}{题目}
已知 $L_1:\dfrac{x-a_2}{a_1}=\dfrac{y-b_2}{b_1}=\dfrac{z-c_2}{c_1}$ 与 $L_2:\dfrac{x-a_3}{a_2}=\dfrac{y-b_3}{b_2}=\dfrac{z-c_3}{c_2}$ 相交.设 $\alpha_i=(a_i,b_i,c_i)^T$，则
\begin{enumerate}[A.]
\item $\alpha_1$ 可由 $\alpha_2,\alpha_3$ 线性表示；
\item $\alpha_2$ 可由 $\alpha_1,\alpha_3$ 线性表示；
\item $\alpha_3$ 可由 $\alpha_1,\alpha_2$ 线性表示；
\item $\alpha_1,\alpha_2,\alpha_3$ 线性无关.
\end{enumerate}
\end{block}
\textcolor{red}{\textbf{答案：}$\mathbf{C}$} \\
\textbf{解析：}将直线方程写成向量形式得
\begin{align*}
L_1:r=\alpha _2+t\alpha _1,\quad L_2:r=\alpha _3+t\alpha _2\quad \left( t\in \mathbb{R} \right) .
\end{align*}
相交即存在$s,t\in \mathbb{R}$，使得$\alpha_2+s\alpha_1=\alpha_3+t\alpha_2$，移项得 $\alpha_3=s\alpha_1+(1-t)\alpha_2$，$C$正确.显然$\alpha_1,\alpha_2,\alpha_3$线性相关，$D$错误.当$s=0$或$t=1$时，$A,B$错误.
\end{frame}

\begin{frame}{第7题}
\begin{block}{题目}
设$A,B,C$为三个随机事件，且$P(A)=P(B)=P(C)=\frac14$，$P(AB)=0$，$P(AC)=P(BC)=\frac1{12}$，则 $A,B,C$ 中恰有一个事件发生的概率为

$\mathrm{A.}\frac34. \quad \mathrm{B.}\frac23. \quad \mathrm{C.}\frac12. \quad \mathrm{D.}\frac5{12}$.
\end{block}
\textcolor{red}{\textbf{答案：}$\mathbf{D}$} \\
\textbf{解析：}由$P(AB)=0$知$P(ABC)=0$，故$A,B,C$中恰有一个事件发生的概率为
\begin{align*}
&P\left( A \right) -P\left( AB \right) -P\left( AC \right) +P\left( ABC \right) 
\\
&+P\left( B \right) -P\left( AB \right) -P\left( BC \right) +P\left( ABC \right) 
\\
&+P\left( C \right) -P\left( AC \right) -P\left( BC \right) +P\left( ABC \right) 
\\
&=\frac{3}{4}-2\left( 0+\frac{1}{12}+\frac{1}{12} \right) =\frac{5}{12}.
\end{align*}
\end{frame}

\begin{frame}{第8题}

\begin{frame}{第9题}
\begin{block}{题目}
$\displaystyle\lim_{x\to 0}\left[\frac{1}{e^x-1}-\frac{1}{\ln(1+x)}\right]=$ \underline{\textcolor{red}{$-1$}}.
\end{block}
\textbf{解析：}
\begin{align*}
&\lim_{x\rightarrow 0} \left[ \frac{1}{e^x-1}-\frac{1}{\ln \left( 1+x \right)} \right] =\lim_{x\rightarrow 0} \frac{\ln \left( 1+x \right) -\left( e^x-1 \right)}{\left( e^x-1 \right) \ln \left( 1+x \right)}
\\
&=\lim_{x\rightarrow 0} \frac{x-\frac{x^2}{2}+o\left( x^2 \right) -\left( x+\frac{x^2}{2}+o\left( x^2 \right) \right)}{x^2}=-1.
\end{align*}

\begin{frame}{第10题}
\begin{block}{题目}
设 $\begin{cases}x=\sqrt{t^2+1}\\ y=\ln(t+\sqrt{t^2+1})\end{cases}$，则 $\left.\frac{\mathrm{d}^2y}{\mathrm{d}x^2}\right|_{t=1}=$ \underline{\textcolor{red}{$-\sqrt2$}}.
\end{block}
\textbf{解析：}由链式法则直接计算得
\begin{gather*}
\frac{\mathrm{d}x}{\mathrm{d}t}=\frac{t}{\sqrt{t^2+1}},\quad \frac{\mathrm{d}y}{\mathrm{d}t}=\frac{1+\frac{t}{\sqrt{t^2+1}}}{t+\sqrt{t^2+1}},
\\
\frac{\mathrm{d}y}{\mathrm{d}x}=\frac{\frac{\mathrm{d}y}{\mathrm{d}t}}{\frac{\mathrm{d}x}{\mathrm{d}t}}=\frac{\frac{1+\frac{t}{\sqrt{t^2+1}}}{t+\sqrt{t^2+1}}}{\frac{t}{\sqrt{t^2+1}}}=\frac{\frac{\sqrt{t^2+1}+t}{t+\sqrt{t^2+1}}}{t}=\frac{1}{t},
\\
\frac{\mathrm{d}^2y}{\mathrm{d}x^2}=\frac{\mathrm{d}\frac{\mathrm{d}y}{\mathrm{d}x}}{\mathrm{d}x}=\frac{\frac{\mathrm{d}\frac{\mathrm{d}y}{\mathrm{d}x}}{\mathrm{d}t}}{\frac{\mathrm{d}x}{\mathrm{d}t}}=\frac{-\frac{1}{t^2}}{\frac{t}{\sqrt{t^2+1}}}=-\frac{\sqrt{t^2+1}}{t^3}\Longrightarrow \left.\frac{\mathrm{d}^2y}{\mathrm{d}x^2}\right|_{t=1}=-\sqrt2.
\end{gather*}
\end{frame}

\begin{frame}{第11题}
\begin{block}{题目}

\end{block}
\textbf{解析：}设 $\lambda_1,\lambda_2$ 是特征方程 $\lambda^2+a\lambda+1=0$ 的两个根, 则由 Vieta 定理知
\begin{align*}
\lambda_1+\lambda_2=a<0,\quad \lambda_1\lambda_2=1>0\Longrightarrow \lambda_1<0,\lambda_2<0.
\end{align*}
于是
\begin{align*}
f(x)=C_1e^{\lambda_1x}+C_2e^{\lambda_2x}.
\end{align*}
其中 $C_1,C_2$ 为固定常数.
从而$f'(x)=C_1\lambda_1e^{\lambda_1x}+C_2\lambda_2e^{\lambda_2x},$
故$\lim\limits_{x\to+\infty}f(x)$ $=$ $\lim\limits_{x\to+\infty}f'(x)=0.$
再对原方程两边积分得
\begin{align*}
\int_0^{+\infty}f(x)\mathrm{d}x=-\int_0^{+\infty}f''(x)\mathrm{d}x-a\int_0^{+\infty}f'(x)\mathrm{d}x=n+am.
\end{align*}
\end{frame}

\begin{frame}{第13题}
\begin{block}{题目}
行列式 $\begin{vmatrix}

0 & a & 1 & -1\\
-1 & 1 & a & 0\\
1 & -1 & 0 & a
\end{vmatrix}=$ \underline{\textcolor{red}{$a^4-4a^2$}}.
\end{block}
\textbf{解析：}先证一个引理.
设\(A,B\)是\(n\)阶矩阵,则
\begin{gather*}
\begin{vmatrix}
A & B\\
B & A
\end{vmatrix}=|A + B||A - B|.
\end{gather*}
证明:将行列式的第二行加到第一行上,再将第二列减去第一列,可得
\[
\left| \begin{matrix}
A&		B\\
B&		A\\
\end{matrix} \right|=\left| \begin{matrix}
A+B&		A+B\\
B&		A\\
\end{matrix} \right|=\left| \begin{matrix}
A+B&		O\\
B&		A-B\\
\end{matrix} \right|=|A + B||A - B|.
\]
\end{frame}

\begin{frame}{第14题}

\end{block}
\textbf{解析：}概率密度函数为
$f(x)=
\begin{cases}
\frac{1}{\pi}, & -\frac{\pi}{2}<x<\frac{\pi}{2},\\
0, & \text{其他}.
\end{cases}$
从而$E(X)=\frac{1}{\pi}\int_{-\frac{\pi}{2}}^{\frac{\pi}{2}}x\,\mathrm{d}x=0$，并且
\begin{align*}

\\
&=\frac{2}{\pi}\left[-x\cos x\Big |_0^{\frac{\pi}{2}}+\int_0^{\frac{\pi}{2}}\cos x\,\mathrm{d}x\right]
=\frac{2}{\pi}.
\end{align*}
故$\operatorname{Cov}(X,Y)=E(XY)-E(X)E(Y)=\frac{2}{\pi}.$

\begin{frame}{第15题}
\begin{block}{题目}
求 $f(x,y)=x^3+8y^3-xy$ 的极值.
\end{block}
\textbf{解析：}直接计算得
\begin{align*}
\frac{\partial f}{\partial x} = 3x^2 - y, \quad \frac{\partial f}{\partial y} = 24y^2 - x, \quad \frac{\partial^2 f}{\partial x^2} = 6x, \quad \frac{\partial^2 f}{\partial x \partial y} = -1, \quad \frac{\partial^2 f}{\partial y^2} = 48y.
\end{align*}
令 $\frac{\partial f}{\partial x} = \frac{\partial f}{\partial y} = 0$ 得驻点 $(0,0), \left( \frac{1}{6}, \frac{1}{12} \right)$. 两个驻点处的Hessian矩阵分别为
\begin{align*}
H(f(0,0)) = \begin{pmatrix} 0 & -1 \\ -1 & 0 \end{pmatrix}, \quad H\left( f\left( \frac{1}{6}, \frac{1}{12} \right) \right) = \begin{pmatrix} 1 & -1 \\ -1 & 4 \end{pmatrix}.
\end{align*}
显然 $H(f(0,0))$ 是不定矩阵, $H\left( f\left( \frac{1}{6}, \frac{1}{12} \right) \right)$ 是正定矩阵, 故由多元函数极值定理可知 $(0,0)$ 一定不是极值点,$f\left( \frac{1}{6}, \frac{1}{12} \right)=-\frac{1}{216}$是严格极小值,也是唯一极值.
\end{frame}

\begin{frame}{第16题}
\begin{block}{题目}
计算 $I=\oint_L \frac{4x-y}{4x^2+y^2}\mathrm{d}x+\frac{x+y}{4x^2+y^2}\mathrm{d}y$，其中 $L$ 为 $x^2+y^2=2$，方向为逆时针方向.
\end{block}
\textbf{解析：}令
\[
P(x,y)=\frac{4x-y}{4x^2+y^2},\qquad Q(x,y)=\frac{x+y}{4x^2+y^2}.
\]
计算偏导数得
\begin{align*}
\frac{\partial P}{\partial y}
&=\frac{-(4x^2+y^2)-(4x-y)\cdot 2y}{(4x^2+y^2)^2}
=\frac{-4x^2+y^2-8xy}{(4x^2+y^2)^2},\\
\frac{\partial Q}{\partial x}
&=\frac{(4x^2+y^2)-(x+y)\cdot 8x}{(4x^2+y^2)^2}
=\frac{-4x^2+y^2-8xy}{(4x^2+y^2)^2}.
\end{align*}
故除原点外 $\frac{\partial Q}{\partial x}=\frac{\partial P}{\partial y}$.对充分小的$\varepsilon>0$，取椭圆
$L_\varepsilon :\ 4x^2+y^2=\varepsilon^2$,方向取逆时针方向.
\end{frame}

\begin{frame}{第16题}
由Green公式得
\[
I=\oint_{L+L_{\varepsilon}^{-}}{P\mathrm{d}x+Q\mathrm{d}y}+\oint_{L_{\varepsilon}}{P\mathrm{d}x+Q\mathrm{d}y}=\oint_{L_{\varepsilon}}{P\mathrm{d}x+Q\mathrm{d}y}.
\]
对$L_\varepsilon$极坐标化得$x=\frac{\varepsilon}{2}\cos\theta, y=\varepsilon\sin\theta,(\theta\in[0,2\pi]).$
则
\begin{align*}
\mathrm{d}x=-\frac{\varepsilon}{2}\sin\theta\,\mathrm{d}\theta,\quad
\mathrm{d}y=\varepsilon\cos\theta\,\mathrm{d}\theta,
\end{align*}
于是
\begin{align*}
I&=\int_0^{2\pi}
\left[
\frac{2\varepsilon\cos\theta-\varepsilon\sin\theta}{\varepsilon^2}
\left(-\frac{\varepsilon}{2}\sin\theta\right)
+\frac{\frac{\varepsilon}{2}\cos\theta+\varepsilon\sin\theta}{\varepsilon^2}
\left(\varepsilon\cos\theta\right)
\right]\mathrm{d}\theta\\
&=\int_0^{2\pi}
\left[
-\frac12(2\cos\theta-\sin\theta)\sin\theta
+\left(\frac12\cos\theta+\sin\theta\right)\cos\theta
\right]\mathrm{d}\theta\\
&=\int_0^{2\pi}\frac12\,\mathrm{d}\theta=\pi.
\end{align*}
\end{frame}

\begin{frame}{第19题}
\begin{block}{题目}
设$f\in C^1[0,2]$，$f(0)=f(2)=0$，$M=\max_{x\in [0,2]}|f|$.证明：

(1)存在$\xi\in (0,2)$，使$|f'(\xi)|\geqslant M$；

(2) 若对任意$x\in (0,2)$，有$|f'(x)|\leqslant M$，则 $M=0$.
\end{block}
\textbf{证明：}因为 $f$ 在 $[0,2]$ 上连续，所以存在 $c\in[0,2]$，使得
\[
|f(c)|=M.
\]
若 $c=0$ 或 $c=2$，则 $M=0$，(1)(2)的结论都显然成立.下设 $0<c<2$.

(1) 若 $0<c\leqslant 1$，在 $[0,c]$ 上对 $f$ 用Lagrange中值定理，存在 $\xi\in(0,c)$，使得
\[
f(c)=f(c)-f(0)=f'(\xi)c.
\]
于是
\[
M=|f(c)|=|f'(\xi)|c\leqslant |f'(\xi)|.
\]
\end{frame}

\begin{frame}{第19题}
若 $1<c<2$，在 $[c,2]$ 上用Lagrange中值定理，存在 $\xi\in(c,2)$，使得
\[
-f(c)=f(2)-f(c)=f'(\xi)(2-c).
\]
于是
\[
M=|f(c)|=|f'(\xi)|(2-c)\leqslant |f'(\xi)|.
\]
故总存在 $\xi\in (0,2)$，使 $|f'(\xi)|\geqslant M$.

(2) 反证法.假设 $M>0$.由 (1) 知存在 $\xi\in (0,2)$，使 $|f'(\xi)|\geqslant M$.又已知 $|f'(\xi)|\leqslant M$，故
\[
|f'(\xi)|=M.
\]
但由 (1) 的证明可见：
若 $c<1$，则 $|f'(\xi)|=\frac{M}{c}>M$；
若 $c>1$，则 $|f'(\xi)|=\frac{M}{2-c}>M$.
均与 $|f'|\leqslant M$ 矛盾！因此只能有 $c=1$，即
\[
|f(1)|=M.
\]
\end{frame}

\begin{frame}{第19题}
不妨设 $f(1)=M$（否则以 $-f$ 代替 $f$）.令$g(x)=f(x)-Mx$，则$g(0)=g(1)=0$.

g'(x)=f'(x)-M\leqslant 0.
\]
于是$g(x)=0(x\in [0,1])$,即

综上可得$f'(1^-)=M,\, f'(1^+)=-M$，即
$f$ 在 $x=1$ 处左右导数不相等，与 $f$ 在 $x=1$ 处可导矛盾！故假设不成立，必有 $M=0$.
\end{frame}

\begin{frame}{第21题}

(2) 由
$\bm A^2\alpha+\bm A\alpha-6\alpha=0$
得$\bm A^2\alpha=6\alpha-\bm A\alpha.$于是
\end{frame}

\begin{frame}{第22题}
\begin{block}{题目}
设随机变量$X_1,X_2,X_3$相互独立，其中$X_1$和$X_2$服从标准正态分布，$X_3$的概率分布为$P(X_3=0)=P(X_3=1)=\frac12$，$Y=X_3X_1+(1-X_3)X_2$.

(1) 求二维随机变量$(X_1,Y)$的分布函数，结果用标准正态分布函数$\Phi(x)$表示；

(2) 证明随机变量$Y$服从标准正态分布.
\end{block}
\textbf{解析：}(1) 由全概率公式，又因为 $X_3$ 与 $X_1,X_2$ 独立，所以对任意 $x,y\in\mathbb R$，有
\begin{align*}
&F_{X_1,Y}\left( x,y \right) =P\left( X_1\leqslant x, Y\leqslant y \right) 
\\
&=P\left( X_3=1 \right) P\left( X_1\leqslant x, Y\leqslant y\mid X_3=1 \right) +P\left( X_3=0 \right) P\left( X_1\leqslant x, Y\leqslant y\mid X_3=0 \right) 
\\
&=\frac{1}{2}P\left( X_1\leqslant x, X_1\leqslant y \right) +\frac{1}{2}P\left( X_1\leqslant x, X_2\leqslant y \right) 
\\
&=\frac{1}{2}P\left( X_1\leqslant \min \left\{ x,y \right\} \right) +\frac{1}{2}P\left( X_1\leqslant x \right) P\left( X_2\leqslant y \right) 
=\frac{1}{2}\Phi \left( \min \left\{ x,y \right\} \right) +\frac{1}{2}\Phi \left( x \right) \Phi \left( y \right) .
\end{align*}
\end{frame}

\begin{frame}{第22题}
(2) 对任意 $y\in\mathbb R$，由全概率公式得
\begin{align*}
P\left( Y\leqslant y \right) &=P\left( X_3=1 \right) P\left( Y\leqslant y\mid X_3=1 \right) +P\left( X_3=0 \right) P\left( Y\leqslant y\mid X_3=0 \right) 
\\
&=\frac{1}{2}P\left( X_1\leqslant y \right) +\frac{1}{2}P\left( X_2\leqslant y \right) =\frac{1}{2}\Phi (y)+\frac{1}{2}\Phi (y)=\Phi (y).
\end{align*}
因此随机变量$Y$服从标准正态分布.
\end{frame}

\begin{frame}{第23题}
\begin{block}{题目}
设某种元件的使用寿命 $T$ 的分布函数为
\begin{align*}
F(t)=\begin{cases}
1-e^{-\left( \frac{t}{\theta} \right) ^m},&		t\geqslant 0,\\
0,&		\text{其他},\\
\end{cases}
\end{align*}
其中 $\theta, m$ 为参数且大于零.
\begin{enumerate}[(1)]
\item 求概率 $P\{T>t\}$ 与 $P\{T>s+t \mid T>s\}$, 其中 $s>0$, $t>0$;
\item 任取 $n$ 个这种元件做寿命试验, 测得它们的寿命分别为 $t_1, t_2, \cdots, t_n$. 若 $m$ 已知, 求 $\theta$ 的最大似然估计值 $\hat{\theta}$.
\end{enumerate}
\end{block}
\textbf{解析：}(1) 当 $t>0$ 时，有$P(T>t)=1-F(t)=e^{-\left(\frac{t}{\theta}\right)^m}.$于是
\begin{align*}
P\left( T>s+t\mid T>s \right) =\frac{P\left( T>s+t \right)}{P\left( T>s \right)}=\frac{e^{-\left( \frac{s+t}{\theta} \right) ^m}}{e^{-\left( \frac{s}{\theta} \right) ^m}}=e^{\left( \frac{s}{\theta} \right) ^m-\left( \frac{s+t}{\theta} \right) ^m}.
\end{align*}
\end{frame}

\begin{frame}{第23题}
(2) 概率密度函数为
\[
f(t)=F'(t)=\begin{cases}
\frac{m}{\theta}\left( \frac{t}{\theta} \right) ^{m-1}e^{-\left( \frac{t}{\theta} \right) ^m},&t>0,\\
0,&\text{其他}.\\
\end{cases}
\]
显然$t_i>0(i=1,2,\cdots,n)$，故似然函数为
\[
L(\theta )=\prod_{i=1}^n{f(t_i)}=m^n\theta ^{-mn}\left( \prod_{i=1}^n{t_{i}^{m-1}} \right) e^{-\sum\limits_{i=1}^n{\left( \frac{t_i}{\theta} \right) ^m}}.
\]
取对数得
\[
\ln L(\theta)
=n\ln m-mn\ln\theta+(m-1)\sum_{i=1}^n\ln t_i
-\sum_{i=1}^n\left(\frac{t_i}{\theta}\right)^m.
\]
\end{frame}

\begin{frame}{第23题}
对 $\theta$ 求导并令其为0得
\[
\frac{\mathrm{d}\ln L(\theta)}{\mathrm{d}\theta}
=-\frac{mn}{\theta}
+m\theta^{-m-1}\sum_{i=1}^n t_i^m=0.
\]
化简得
\[
\theta^m=\frac1n\sum_{i=1}^n t_i^m.
\]
因此 $\theta$ 的最大似然估计为
\[
\hat\theta=\left(\frac1n\sum_{i=1}^n t_i^m\right)^{\frac1m}.
\]
\end{frame}
