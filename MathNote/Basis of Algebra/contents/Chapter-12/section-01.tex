\section{双线性函数}

\begin{definition}
设$V$是域$\mathbb{K}$上的线性空间, 若映射$(,): V \times V \to \mathbb{K}$满足: 对$\forall x,x_1,x_2,y,y_1,y_2 \in V$,$\forall \lambda \in \mathbb{K}$, 都有
\begin{align*}
(\lambda x_1 + x_2, y) = \lambda (x_1, y) + (x_2, y), \\
(x, \lambda y_1 + y_2) = \lambda (x, y_1) + (x, y_2),
\end{align*}
则称$(,)$是$V$上的一个\textbf{双线性函数}.
\end{definition}

\begin{definition}
设$(,)$是域$\mathbb{K}$上线性空间$V$上的双线性函数. 
\begin{enumerate}[(1)]
\item 若对$\forall x \in V$, 有
\begin{align*}
\left( x,y \right) =0,\,\,\forall y\in V\Longrightarrow x=0.
\end{align*}
则称$(,)$\textbf{关于第一个变元非退化};

\item 若对$\forall y \in V$,有
\begin{align*}
\left( x,y \right) =0,\,\,\forall x\in V\Longrightarrow y=0.
\end{align*}
则称$(,)$\textbf{关于第二个变元非退化};

\item 若$(,)$关于两个变元都非退化, 则称$(,)$是$V$上的\textbf{非退化双线性函数}.
\end{enumerate}
\end{definition}

\begin{example}
设$k$是一个域, $A$是一个$n$维$k-$代数(即$A$是环, $A$又是$n$维$k-$线性空间, 并且$A$的乘法是$k-$双线性的). 假设$(,): A \times A \to k$是$A$上的某个非退化的双线性函数. 证明:
\begin{enumerate}[(1)]
\item 下式
\begin{align*}
(x,y) = (y,\varphi(x)) \quad (x,y \in A)
\end{align*}
定义了一个$k-$线性同构$\varphi: A \to A$.
\item 进一步, 若$(,)$满足结合律, 即对所有$x,y,z \in A$, 有
\begin{align*}
(xy,z) = (x,yz)
\end{align*}
成立, 则$\varphi: A \to A$保持$k-$代数的乘法, 即$\varphi$是$k-$代数的同构.
\end{enumerate}
\end{example}
\begin{proof}
\begin{enumerate}[(1)]
\item 对$\forall x_1,x_2,y \in A$, $\lambda \in k$, 有
\begin{align*}
(y,\varphi(\lambda x_1)) = (\lambda x_1,y) = \lambda (x_1,y) = \lambda (y,\varphi(x_1)) = (y,\lambda \varphi(x_1)),
\end{align*}
\begin{align*}
(y,\varphi(x_1+x_2)) = (x_1+x_2,y) = (x_1,y) + (x_2,y) = (y,\varphi(x_1)) + (y,\varphi(x_2)) = (y,\varphi(x_1) + \varphi(x_2)),
\end{align*}
因为双线性型非退化, 所以
\begin{align*}
\varphi(\lambda x_1) = \lambda \varphi(x_1), \quad \varphi(x_1+x_2) = \varphi(x_1) + \varphi(x_2).
\end{align*}
因此$\varphi$是线性映射. 再设$x \in \mathrm{Ker}\varphi$, 则对$\forall y \in A$, 有
\begin{align*}
(x,y) = (y,\varphi(x)) = (y,0) = 0,
\end{align*}
由于双线性型非退化, 故$x=0$, 因此$\mathrm{Ker}\varphi = \{0\}$, 因此$\varphi$是单射. 再由线性映射的维数公式知
\begin{align*}
\dim A = \dim \mathrm{Im}\varphi + \dim \mathrm{Ker}\varphi = \dim \mathrm{Im}\varphi,
\end{align*}
故$\varphi$也是满射. 综上可知, $\varphi$是$k-$线性同构.

\item 对$\forall x,y,z \in A$, 由题设可知
\begin{align*}
(z,\varphi(xy)) = (xy,z) = (x,yz) = (yz,\varphi(x)) = (y,z\varphi(x)) = (z\varphi(x),\varphi(y)) = (z,\varphi(x)\varphi(y)).
\end{align*}
因为双线性型非退化, 所以
\begin{align*}
\varphi(xy) = \varphi(x)\varphi(y).
\end{align*}
再由(1)知$\varphi$是$k-$代数的同构.
\end{enumerate}
\end{proof}
