\begin{theorem}[Parseval恒等式]\label{theorem:复分析-Parseval恒等式}
设 $D=\{z\in\mathbb{C}:|z|<1\}$，$u(z)=\sum_{k=0}^{\infty} b_k z^k$ 在 $D$ 内解析，且 $u\in H^2(D)$，即
\begin{align}
\iint_D |u(z)|^2\,\mathrm{d}x\mathrm{d}y<\infty. \nonumber
\end{align}
则成立以下两个等式：
\begin{enumerate}[(1)]
\item 圆周上的 Parseval 等式：对任意 $0<r<1$，有
\begin{align*}
\int_0^{2\pi} |u(r e^{\mathrm{i}\theta})|^2\,\mathrm{d}\theta
=2\pi\sum_{k=0}^{\infty} |b_k|^2 r^{2k}. 
\end{align*}

\item 圆盘上的面积 Parseval 等式：
\begin{align*}
\iint_D |u(z)|^2\,\mathrm{d}x\mathrm{d}y
=\pi\sum_{k=0}^{\infty}\frac{|b_k|^2}{k+1}.
\end{align*}
\end{enumerate}
\end{theorem}
