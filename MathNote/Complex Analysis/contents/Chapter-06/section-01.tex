\section{Schwarz对称原理}

\begin{theorem}[Painlevé原理]\label{theorem:Painlevé原理}
设 \(D\) 是域，\(\gamma_1, \cdots, \gamma_n\) 是 \(\mathbb{C}\) 中的 \(n\) 条可求长曲线。如果 \(f\) 在 \(D\) 上连续，在开集 \(D \setminus \left(\bigcup\limits_{k=1}^n \gamma_k\right)\) 上全纯，那么 \(f\) 必在 \(D\) 上全纯。
\end{theorem}
\begin{proof}
任取圆盘 \(B(a, r) \subset D\)，只须证明 \(f\) 在 \(B(a, r)\) 上全纯即可。由\hyperref[theorem:Morera定理]{Morera定理}，只须证明对 \(B(a, r)\) 中任意可求长简单闭曲线 \(l\)，总有
\begin{align}
\int_l f(z) dz = 0 \label{eq:painleve}
\end{align}
就行了。如果 \(l\) 不与 \(\gamma_1, \cdots, \gamma_n\) 中的任意一条相交，\eqref{eq:painleve} 式当然成立。今设 \(l\) 与 \(\gamma_k\) 相交，那么由 Cauchy 积分定理，只要将 \(f\) 沿 \(l\) 的积分分解成沿\reffig{figure:图6.11111}中箭头方向所表示的两条闭曲线的积分，便知 \eqref{eq:painleve} 式也成立。这里考虑的是最简单的情形，一般情形的证明完全类似。

\centering
\includegraphics[width=0.3\textwidth]{图6.11111.png}
\caption{}
\label{figure:图6.11111}
\end{figure}\end{proof}

\begin{theorem}[Schwarz对称原理]\label{theorem:Schwarz对称原理}
设域 \(D\) 关于实轴对称。如果 \(f\) 满足
\begin{enumerate}[(i)]
\item \(f\) 在 \(D \cap \{z \in \mathbb{C} : \operatorname{Im}z > 0\}\) 上全纯；
\item \(f\) 在 \(D \cap \{z \in \mathbb{C} : \operatorname{Im}z \geqslant 0\}\) 上连续；
\item \(f(D \cap \mathbb{R}) \subset \mathbb{R}\)，
\end{enumerate}
那么
\begin{align*}
F(z) = 
\begin{cases}
f(z), & z \in D \cap \{z \in \mathbb{C} : \operatorname{Im}z \geqslant 0\}, \\
\overline{f(\overline{z})}, & z \in D \cap \{z \in \mathbb{C} : \operatorname{Im}z < 0\}
\end{cases}
\end{align*}
便是 \(f\) 在 \(D\) 上的全纯开拓。
\end{theorem}
\begin{remark}
Schwarz 对称原理可以推广到更一般的情形。为了叙述方便，对 \(\mathbb{C}_{\infty}\) 中的圆周 \(\gamma\)，分别用 \(\mathbb{C}_{\infty}^+(\gamma)\) 和 \(\mathbb{C}_{\infty}^-(\gamma)\) 表示 \(\mathbb{C}_{\infty}\) 被 \(\gamma\) 分成的两个单连通域。实际上，\(\mathbb{C}_{\infty}^+(\gamma)\) 和 \(\mathbb{C}_{\infty}^-(\gamma)\) 就是圆周 \(\gamma\) 的两侧。
\end{remark}
\begin{proof}
我们证明 \(F \in H(D)\)。显然，\(F\) 在 \(D \cap \{z \in \mathbb{C} : \operatorname{Im}z > 0\}\) 中是全纯的。今任取 \(z \in D \cap \{z \in \mathbb{C} : \operatorname{Im}z < 0\}\)，那么 \(\zeta = \overline{z} \in D \cap \{z \in \mathbb{C} : \operatorname{Im}z > 0\}\)，于是
\begin{align*}
\frac{\partial}{\partial \overline{z}} F(z) = \frac{\partial}{\partial \overline{z}} \overline{f(\overline{z})} = \overline{\frac{\partial}{\partial z} f(\overline{z})} = 0.
\end{align*}
这就证明了 \(F\) 在 \(D \cap \{z \in \mathbb{C} : \operatorname{Im}z < 0\}\) 上全纯。今任取 \(x_0 \in D \cap \mathbb{R}\)，由 (iii)，\(f(x_0)\) 取实数值，因而当 \(z\) 在 \(D\) 的下半平面部分趋于 \(x_0\) 时，有
\begin{align*}
\lim\limits_{z \to x_0} F(z) = \lim\limits_{\overline{z} \to x_0} \overline{f(\overline{z})} = \overline{f(x_0)} = f(x_0) = F(x_0),
\end{align*}
故 \(F\) 在 \(D\) 上连续。由 \hyperref[theorem:Painlevé原理]{Painlevé原理}即知 \(F \in H(D)\).
\end{proof}

\begin{theorem}[推广的Schwarz对称原理]\label{theorem:推广的Schwarz对称原理}
设 \(\mathbb{C}_{\infty}\) 中的域 \(D\) 关于圆周 \(\gamma = \{z \in \mathbb{C} : |z - z_0| = r\}\) 对称。如果 \(f\) 满足
\begin{enumerate}[(i)]
\item \(f\) 在 \(D \cap \mathbb{C}_{\infty}^+(\gamma)\) 上全纯；
\item \(f\) 在 \(D \cap (\mathbb{C}_{\infty}^+(\gamma) \cup \gamma)\) 上连续；
\item \(f(D \cap \gamma) \subset \Gamma\)，这里 \(\Gamma = \{w \in \mathbb{C} : |w - w_0| = \rho\}\) 是一个圆周；
\item 对于任意 \(z \in D \cap \mathbb{C}_{\infty}^+(\gamma)\)，\(f(z) \neq w_0\)，
\end{enumerate}
那么 \(f\) 能全纯开拓到 \(D\) 上，成为 \(D\) 上的全纯函数 \(F\)，\(F\) 将 \(D\) 中关于 \(\gamma\) 对称的两点映为 \(F(D)\) 中关于 \(\Gamma\) 对称的两点。
\end{theorem}
\begin{remark}
在上面的证明中，我们假定 \(\gamma\) 和 \(\Gamma\) 都是 \(\mathbb{C}\) 中的圆周。实际上，\(\gamma\) 和 \(\Gamma\) 中有一条（或者两条）是直线，定理当然也成立。特别地，当 \(\gamma\) 和 \(\Gamma\) 都是实轴时，就得到定理 6.1.2。
\end{remark}
\begin{proof}
由分式线性变换的性质知道，\(z\) 与 \(z_0 + \frac{r^2}{\overline{z - z_0}}\) 关于圆周 \(\gamma\) 对称，\(w\) 与 \(w_0 + \frac{\rho^2}{\overline{w - w_0}}\) 关于圆周 \(\Gamma\) 对称。令
\begin{align*}
F(z) = 
\begin{cases}
f(z), & z \in D \cap (\mathbb{C}_{\infty}^+(\gamma) \cup \gamma), \\
w_0 + \dfrac{\rho^2}{\overline{f\left(z_0 + \frac{r^2}{\overline{z - z_0}}\right) - w_0}}, & z \in D \cap \mathbb{C}_{\infty}^-(\gamma),
\end{cases}
\end{align*}
由定义，\(F\) 在 \(D \cap \mathbb{C}_{\infty}^+(\gamma)\) 中全纯；用\hyperref[theorem:Schwarz对称原理]{Schwarz对称原理}证明中的方法即知 \(F\) 也在 \(D \cap \mathbb{C}_{\infty}^-(\gamma)\) 中全纯，这里用到了条件 (iv)。现在证明 \(F\) 在 \(D\) 上连续。为此任取 \(\zeta \in D \cap \gamma\)，当 \(z \in D \cap \mathbb{C}_{\infty}^-(\gamma)\)，且 \(z \to \zeta\) 时，总有 \(z_0 + \frac{r^2}{\overline{z - z_0}} \in D \cap \mathbb{C}_{\infty}^+(\gamma)\)，且 \(z_0 + \frac{r^2}{\overline{z - z_0}} \to \zeta\)。因而
\begin{align*}
f\left(z_0 + \frac{r^2}{\overline{z - z_0}}\right) \to f(\zeta) \in \Gamma,
\end{align*}
于是也有
\begin{align*}
w_0 + \frac{\rho^2}{\overline{f\left(z_0 + \frac{r^2}{\overline{z - z_0}}\right) - w_0}} \to f(\zeta),
\end{align*}
即 \(F(z) \to f(\zeta)\)。这就证明了 \(F \in C(D)\)。于是由 \hyperref[theorem:Painlevé原理]{Painlevé原理}即知 \(F \in H(D)\).
\end{proof}

\begin{theorem}[双全纯映射的Schwarz对称原理]\label{theorem:双全纯映射的Schwarz对称原理}
设 \(\mathbb{C}_{\infty}\) 中的域 \(D\) 关于圆周 \(\gamma = \{z \in \mathbb{C} : |z - z_0| = r\}\) 对称。如果 \(f\) 满足
\begin{enumerate}[(i)]
\item \(f\) 在 \(D \cap \mathbb{C}_{\infty}^+(\gamma)\) 上单叶全纯；
\item \(f\) 在 \(D \cap (\mathbb{C}_{\infty}^+(\gamma) \cup \gamma)\) 上单叶连续；
\item \(f(D \cap \gamma) \subset \Gamma\)，这里 \(\Gamma = \{w \in \mathbb{C} : |w - w_0| = \rho\}\) 是一个圆周；
\item 对于任意 \(z \in D \cap \mathbb{C}_{\infty}^+(\gamma)\)，\(f(z) \neq w_0\)，
\end{enumerate}
那么 \(f\) 能双全纯地开拓到 \(D\) 上，成为 \(D\) 上的单叶全纯函数 \(F\)，\(F\) 将 \(D\) 中关于 \(\gamma\) 对称的两点映为 \(F(D)\) 中关于 \(\Gamma\) 对称的两点。
\end{theorem}

\begin{example}
设 \(0 < a < \infty\)，\(L_k = \left\{ z = k\pi + \frac{\pi}{2} + iy : 0 \leqslant y \leqslant a \right\}\)，\(D = \{ z \in \mathbb{C} : \operatorname{Im}z > 0 \} \setminus \bigcup\limits_{k=-\infty}^{\infty} L_k\)（图 6.2），求一双全纯映射，把 \(D\) 映为上半平面。
\end{example}
\begin{figure}[H]
\centering
\includegraphics[width=0.5\textwidth]{图6.2222222222.png}
\caption{}
\label{figure:图6.2222222222}
\end{figure}
\begin{solution}
已知 \(w = \sin z\) 把半带状域 \(G = \left\{ z \in \mathbb{C} : \operatorname{Im}z > 0,\ -\frac{\pi}{2} < \operatorname{Re}z < \frac{\pi}{2} \right\}\) 双全纯地映为上半平面，把点 \(-\frac{\pi}{2} + ia,\ -\frac{\pi}{2},\ \frac{\pi}{2},\ \frac{\pi}{2} + ia\) 分别映为 \(-\operatorname{ch}a,\ -1,\ 1,\ \operatorname{ch}a\)，因此 \(w = \arcsin \frac{\sin z}{\operatorname{ch}a}\) 便将 \(G\) 双全纯地映为自己。这时，开射线
\begin{align*}
z = -\frac{\pi}{2} + iy : a < y < \infty
\end{align*}
和
\begin{align*}
z = \frac{\pi}{2} + iy : a < y < \infty
\end{align*}
分别被映为开射线
\begin{align*}
\left\{z = -\frac{\pi}{2} + iy: 0 < y < \infty\right\}
\end{align*}
和
\begin{align*}
\left\{z = \frac{\pi}{2} + iy: 0 < y < \infty\right\}.
\end{align*}
反复利用双全纯映射的 Schwarz 对称原理（定理 6.1.4）无穷多次，便可将
\begin{align*}
w = \arcsin \frac{\sin z}{\operatorname{ch}a}
\end{align*}
双全纯地开拓到 \(D\) 上，成为 \(D\) 上的双全纯映射，它把 \(D\) 一一地映为上半平面。
\end{solution}

\begin{proposition}\label{proposition:例6.1.6复变}
圆环 \(D = \{z \in \mathbb{C}: r_1 < |z| < r_2\}\) 与圆环 \(G = \{w \in \mathbb{C}: R_1 < |w| < R_2\}\) 双全纯等价的充分必要条件是
\begin{align}
\frac{r_2}{r_1} = \frac{R_2}{R_1}. \label{eq:annulus}
\end{align}
当 \eqref{eq:annulus} 式成立时，\(w = f(z)\) 是将 \(D\) 映为 \(G\) 的双全纯映射，当且仅当
\begin{align}
f(z) = e^{\mathrm{i}\theta } \frac{R_2}{r_2} z, \quad \theta \in \mathbb{R} \label{eq:annulus_map1}
\end{align}
或
\begin{align}
f(z) = e^{\mathrm{i}\theta } \frac{r_1 R_2}{z}, \quad \theta \in \mathbb{R}. \label{eq:annulus_map2}
\end{align}
特别地，圆环 \(D\) 的全纯自同构群为
\begin{align*}
\operatorname{Aut}(D) = \left\{e^{\mathrm{i}\theta } z \text{ 或 } e^{\mathrm{i}\theta } \frac{r_1 r_2}{z}: \theta \in \mathbb{R}\right\}.
\end{align*}
\end{proposition}
\begin{proof}
如果 \eqref{eq:annulus} 式成立，那么
\begin{align*}
f(z) = e^{\mathrm{i}\theta } \frac{R_2}{r_2} z, \quad \theta \in \mathbb{R}
\end{align*}
或
\begin{align*}
f(z) = e^{\mathrm{i}\theta } \frac{r_1 R_2}{z}, \quad \theta \in \mathbb{R}
\end{align*}
便把 \(D\) 双全纯地映为 \(G\)，因而 \(D\) 和 \(G\) 全纯等价。

现在证明 \eqref{eq:annulus} 式是必要的。如果 \(f\) 将 \(D\) 双全纯地映为 \(G\)，那么由史济怀复变函数(P285)-习题7.3的第2题，\(f\) 也同胚地将 \(D\) 映为 \(G\)。先考虑 \(f\) 将 \(|z| = r_1\) 和 \(|z| = r_2\) 分别映为 \(|w| = R_1\) 和 \(|w| = R_2\) 的情形。令
\begin{align*}
D_1 = \left\{ z \in \mathbb{C} : \frac{r_1^2}{r_2} < |z| < r_1 \right\},\quad
G_1 = \left\{ w \in \mathbb{C} : \frac{R_1^2}{R_2} < |w| < R_1 \right\},
\end{align*}
\begin{figure}[H]
\centering
\includegraphics[width=0.8\textwidth]{图6.333333333333.png}
\caption{}
\label{figure:图6.333333333333}
\end{figure}
那么 \(D_1\) 和 \(G_1\) 分别是 \(D\) 和 \(G\) 关于 \(|z| = r_1\) 和 \(|w| = R_1\) 对称的圆环（\reffig{figure:图6.333333333333}）。根据\hyperref[theorem:Schwarz对称原理]{Schwarz对称原理}，\(f\) 可双全纯地开拓到 \(D_1\) 上，成为 \(D \cup D_1 \cup \{ z \in \mathbb{C} : |z| = r_1 \}\) 上的双全纯函数，而且
\begin{align*}
f(D \cup D_1 \cup \{ z \in \mathbb{C} : |z| = r_1 \}) = G \cup G_1 \cup \{ w \in \mathbb{C} : |w| = R_1 \}.
\end{align*}
这个过程可继续进行，于是 \(f\) 可双全纯地开拓到 \(B(0, r_1)\)，成为 \(B(0, r_2)\) 上的双全纯映射，而且
\begin{align*}
f(B(0, r_2)) = B(0, R_2),\quad f(0) = 0,
\end{align*}
因而有
\begin{align*}
f(z) = e^{\mathrm{i}\theta } \frac{R_2}{r_2} z.
\end{align*}
由于同时成立
\begin{align*}
f(B(0, r_1)) = B(0, R_1),
\end{align*}
所以也有
\begin{align*}
f(z) = e^{\mathrm{i}\theta } \frac{R_1}{r_1} z.
\end{align*}
由此即得
\begin{align*}
\frac{r_2}{r_1} = \frac{R_2}{R_1}.
\end{align*}

再考虑 \(f\) 将 \(|z| = r_1\) 和 \(|z| = r_2\) 分别映为 \(|w| = R_2\) 和 \(|w| = R_1\) 的情形。这时，\(g(z) = \frac{R_1 R_2}{f(z)}\) 也将 \(D\) 双全纯地映为 \(G\)，并且将 \(|z| = r_1\) 和 \(|z| = r_2\) 分别映为 \(|w| = R_1\) 和 \(|w| = R_2\)。由上面的证明便知
\begin{align*}
\frac{r_2}{r_1} = \frac{R_2}{R_1},
\end{align*}
而且
\begin{align*}
g(z) = e^{\mathrm{i}\theta } \frac{R_1}{r_1} z,
\end{align*}
因而
\begin{align*}
f(z) = e^{-\mathrm{i}\theta } \frac{r_1 R_2}{z}.
\end{align*}
\end{proof}

\begin{example}\label{proposition:复变例1112}
设 \(D\) 是由 \(3\) 条与 \(\partial B(0,1)\) 正交的圆弧所围成的单连通域, 如 \cref{figure:复变图111} 所示. 若 \(\varphi\) 双全纯地将 \(D\) 映为上半平面 \(C^{+} = \{z \in \mathbb{C} : \operatorname{Im} z > 0\}\), 且将 \(\overline{D}\) 同胚地映为 \(\overline{C^{+}}\), 对应边界点 \(A,B,C\) 分别映为 \(0,1,\infty\), 则 \(\varphi\) 可全纯开拓到 \(B(0,1)\), 并且满足 \(\varphi(B(0,1)) = \mathbb{C} \setminus \{0,1\}\).
\end{example}
