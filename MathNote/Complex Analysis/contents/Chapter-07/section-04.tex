\section{Schwarz-Christoffel公式}

\begin{lemma}\label{lemma:复变引理7.4.1}
设 \(G\) 是 \(w\) 平面上以 \(w_1,\cdots,w_n\) 为顶点的多角形域，\(w=f(z)\) 是把 \(z\) 平面的上半平面 \(D\) 一一地映为 \(G\) 的双全纯映射,则存在把上半平面 \(D\) 一一地映为多角形域 \(G\) 的双全纯映射 \(w=f(z)\)，它在 \(\overline{D}\) 上连续，且把实轴一一地映为 \(\partial G\)。
\end{lemma}
\begin{proof}
任取 \(a \in D\)，则分式线性变换
\begin{align*}
\zeta = \frac{z-a}{z-\overline{a}}
\end{align*}
把 \(D\) 一一地映为 \(|\zeta|<1\)，它在 \(\overline{D}\) 上连续，且把实轴一一地映为 \(|\zeta|=1\)。根据 Riemann 映射定理和边界对应原理，存在双全纯函数 \(w=g(\zeta)\)，它把 \(|\zeta|<1\) 一一地映为 \(G\)，且 \(g\) 在 \(|\zeta| \leqslant 1\) 上连续，并把 \(|\zeta|=1\) 一一地映为 \(\partial G\)。于是，函数
\begin{align*}
w = g\left( \frac{z-a}{z-\overline{a}} \right) = f(z)
\end{align*}
即符合引理的要求.
\end{proof}

\begin{lemma}\label{lemma:复变引理7.4.2}
设 \(\Omega\) 是具有角点 \(w_0\) 的域，在 \(w_0\) 的邻域中，\(\Omega\) 的边界由两个直线段构成，它们的交角为 \(\alpha\pi\)。设双全纯映射 \(w=f(z)\) 把上半平面映为 \(\Omega\)，把实轴上的点 \(z_0\) 映为 \(w_0\)，那么在 \(z_0\) 的邻域内，\(f\) 可表示为
\begin{align}
f(z) = w_0 + (z-z_0)^\alpha (c_0 + c_1(z-z_0) + \cdots), \quad c_0 \neq 0. \label{eq:1}
\end{align}
\end{lemma}
\begin{proof}
令 \(\eta=(w-w_0)^{\frac{1}{\alpha}}\)，它可以在 \(\Omega\) 中分出单值的全纯分支，它把 \(w_0\) 映为原点，且把 \(\Omega\) 的两条边界线所张的角扩大为 \(\pi\)。因此
\begin{align}
\eta = \eta(z) = (f(z)-w_0)^{\frac{1}{\alpha}}, \label{eq:2}
\end{align}
把 \(z_0\) 的邻域在上半平面中的那部分变到 \(\eta\) 平面上 \(\eta=0\) 的一个邻域的某个半平面，对应于 \(z\) 平面中实轴上的线段的是一段直线段。由引理7.4.1知，它连续到实轴上，故由 Schwarz 对称原理，\(\eta=\eta(z)\) 可以全纯开拓到 \(z_0\) 的整个邻域中，因而在 \(z_0\) 的邻域中有展开式
\begin{align}
\eta(z) = b_1(z-z_0) + b_2(z-z_0)^2 + \cdots, \quad b_1 \neq 0. \label{eq:3}
\end{align}
这里没有常数项是因为 \(\eta(z_0)=0\)，\(b_1 \neq 0\) 是因为 \(b_1=\eta'(z_0) \neq 0\)。由 \eqref{eq:2} 式和 \eqref{eq:3} 式即得
\begin{align}
f(z) = w_0 + (\eta(z))^\alpha 
= w_0 + (z-z_0)^\alpha \{ b_1 + b_2(z-z_0) + \cdots \}^\alpha. \label{eq:4}
\end{align}
如果记 \(g(z)=b_1+b_2(z-z_0)+\cdots\)，由于 \(g(z_0)=b_1 \neq 0\)，故在 \(z_0\) 的邻域中 \((g(z))^\alpha\) 能分出单值的全纯分支，把 \((g(z))^\alpha\) 在 \(z_0\) 的邻域中展开成幂级数
\begin{align*}
(g(z))^\alpha = c_0 + c_1(z-z_0) + \cdots.
\end{align*}
再由 \eqref{eq:4} 式即得 \eqref{eq:1} 式.
\end{proof}

\begin{theorem}[Schwarz-Christoffel定理]\label{theorem:Schwarz-Christoffel定理}
设双全纯函数 \(w=f(z)\) 把上半平面 \(D\) 一一地映为多角形域 \(G\)，且 \(f\) 在 \(\overline{D}\) 上连续。如果 \(G\) 在其顶点 \(w_k(k=1,\cdots,n)\) 处的顶角是 \(\alpha_k\pi\) (\(0<\alpha_k<2\))，实轴上与 \(w_k\) 对应的点是 \(a_k(k=1,\cdots,n)\)，\(-\infty < a_1 < \cdots < a_n < \infty\)，那么 \(f\) 可表示为
\begin{align}
f(z) = C \int_{z_0}^{z} (z-a_1)^{\alpha_1-1} \cdots (z-a_n)^{\alpha_n-1} \mathrm{d}z + C_1, \label{eq:Schwarz-Christoffel公式}
\end{align}
其中 \(z_0, C, C_1\) 是三个常数。上式称为\textbf{Schwarz-Christoffel公式}.
\end{theorem}
\begin{proof}
由于 \(f\) 把实轴一一地映为 \(G\) 的边界，所以当 \(z\) 在 \((a_k, a_{k+1})\) 中变动时，\(f(z)\) 在线段 \(w_k w_{k+1}\) 上变动，即 \(f\) 把直线段映为直线段。根据 \hyperref[theorem:双全纯映射的Schwarz对称原理]{Schwarz对称原理}，\(f\) 可以跨过 \((a_k, a_{k+1})\) 全纯开拓到下平面 \(D'\) 中去，即可以得一函数 \(f_k\)，它满足：
\begin{enumerate}[(1)]
\item \(f_k\) 在 \(D \cup D' \cup (a_k, a_{k+1})\) 中全纯；
\item 当 \(z \in D\) 时，\(f_k(z)=f(z)\)，\(f_k\) 把下半平面 \(D'\) 映成 \(G\) 关于线段 \(w_k w_{k+1}\) 对称的多角形域 \(G'\)；
\item 当 \(z \in D \cup D' \cup (a_k, a_{k+1})\) 时，\(f_k'(z) \neq 0\)。
\end{enumerate}
让 \(k=1,\cdots,n\)，我们就得到 \(n\) 个函数 \(f_1,\cdots,f_n\)，它们在 \(D\) 内都等于 \(f\)，而在 \(D'\) 中是不相等的。我们来研究 \(f_k\) 和 \(f_{k+1}\) 在 \(D'\) 中的关系。任取 \(z_0 \in D'\)，则 \(\overline{z_0} \in D\)，于是 \(f_k(z_0)\) 是 \(f(\overline{z_0})\) 关于线段 \(w_k w_{k+1}\) 的对称点，\(f_{k+1}(z_0)\) 是 \(f(\overline{z_0})\) 关于线段 \(w_{k+1} w_{k+2}\) 的对称点。因此，\(f_k(z_0)\) 可以看成是由 \(f_{k+1}(z_0)\) 经过两条直线的对称变换得到的。显然，有
\begin{align*}
f_k(z_0) - w_{k+1} = \mathrm{e}^{2i\theta} (f(\overline{z_0}) - w_{k+1}) = \mathrm{e}^{2i(\theta+\varphi)} (f_{k+1}(z_0) - w_{k+1}) = \mathrm{e}^{2i\alpha_{k+1}\pi} (f_{k+1}(z_0) - w_{k+1}).
\end{align*}

\begin{figure}[H]
\centering
\includegraphics[width=0.6\textwidth]{复变1图7.9.png}
\caption{}
\label{figure:复变1图7.9}
\end{figure}
由于 \(z_0\) 是 \(D'\) 中的任意点，故可设
\begin{align*}
f_k(z) = \alpha_k f_{k+1}(z) + \beta_k, \quad z \in D',
\end{align*}
其中 \(\alpha_k=\mathrm{e}^{2i\alpha_{k+1}\pi}, \beta_k=(1-\mathrm{e}^{2i\alpha_{k+1}\pi})w_{k+1}\)。所以
\begin{align*}
f_k'(z) = \alpha_k f_{k+1}'(z), \quad f_k''(z) = \alpha_k f_{k+1}''(z).
\end{align*}
因而
\begin{align*}
\frac{f_k''(z)}{f_k'(z)} = \frac{f_{k+1}''(z)}{f_{k+1}'(z)}, \quad z \in D'.
\end{align*}
这说明 \(f_k(k=1,\cdots,n)\) 在 \(D'\) 中虽然都不相等，但 \(\frac{f_k''}{f_k'}(k=1,\cdots,n)\) 在 \(D'\) 中却都是一样的，它们在 \(D\) 中当然都等于 \(\frac{f''}{f'}\)。这样，我们就得到了一个单值函数 \(g\)，它在 \(D'\) 中等于 \(\frac{f_k''}{f_k'}\)，在 \(D\) 中
\begin{align*}
g(z) = \frac{f''(z)}{f'(z)}.
\end{align*}
故 \(g\) 在 \(C\) 中除去 \(a_1,\cdots,a_n\) 外是全纯的。下面我们证明 \(a_k(k=1,\cdots,n)\) 是 \(g\) 的 \(1\) 阶极点。事实上，由\cref{lemma:复变引理7.4.2}，\(f\) 在 \(a_k\) 的邻域中可展开为
\begin{align*}
f(z) = w_k + (z-a_k)^{\alpha_k} h(z), \quad h(a_k) \neq 0,
\end{align*}
这里 \(h\) 是 \(a_k\) 的邻域中的全纯函数。于是
\begin{align*}
f'(z) &= \alpha_k (z-a_k)^{\alpha_k-1} h(z) + (z-a_k)^{\alpha_k} h'(z) \\
&= (z-a_k)^{\alpha_k-1} [\alpha_k h(z) + (z-a_k)h'(z)] \\
&= (z-a_k)^{\alpha_k-1} q(z),
\end{align*}
这里 \(q(z)=\alpha_k h(z)+(z-a_k)h'(z)\) 是 \(a_k\) 的邻域中的全纯函数，而且 \(q(a_k)=\alpha_k h(a_k) \neq 0\)。因而
\begin{align*}
f''(z) = (\alpha_k-1)(z-a_k)^{\alpha_k-2} q(z) + (z-a_k)^{\alpha_k-1} q'(z).
\end{align*}
于是，在 \(a_k\) 的邻域中有
\begin{align}
g(z) = \frac{f''(z)}{f'(z)} = \frac{\alpha_k-1}{z-a_k} + \frac{q'(z)}{q(z)}. \label{eq:6}
\end{align}
因为 \(\frac{q'(z)}{q(z)}\) 在 \(a_k\) 的邻域中全纯，可以展开成幂级数，所以 \eqref{eq:6} 式就是 \(g\) 在 \(a_k\) 处的 Laurent 展开式，它说明 \(a_k\) 是 \(g\) 的 \(1\) 阶极点，\(g\) 在 \(a_k\) 处的残数为 \(\alpha_k-1\)。

再看 \(g\) 在 \(z=\infty\) 处的情况。由于 \(f\) 把 \(\infty\) 映入多角形的边界，因此 \(f\) 在 \(\infty\) 处全纯，它在 \(\infty\) 的邻域中有展开式
\begin{align*}
f(z) = b_0 + \frac{b_m}{z^m} + \frac{b_{m+1}}{z^{m+1}} + \cdots, \quad b_m \neq 0,
\end{align*}
\begin{align*}
f'(z) = -\frac{m b_m}{z^{m+1}} - \frac{(m+1)b_{m+1}}{z^{m+2}} - \cdots.
\end{align*}
这里 \(p\) 在 \(z=\infty\) 的邻域内全纯，且 \(p(\infty) \neq 0\)。由此可得
\begin{align*}
f''(z) = \frac{m+1}{z^{m+2}} p(z) - \frac{1}{z^{m+1}} p'(z),
\end{align*}
因而
\begin{align*}
g(z) = \frac{f''(z)}{f'(z)} = -\frac{m+1}{z} + \frac{p'(z)}{p(z)}.
\end{align*}
由此即得
\begin{align}
\lim_{z \to \infty} g(z) = 0. \label{eq:7}
\end{align}
现在令
\begin{align*}
G(z) = g(z) - \sum\limits_{k=1}^{n} \frac{\alpha_k-1}{z-a_k},
\end{align*}
它是一个整函数，由 \eqref{eq:7} 式即知 \(G(z) \equiv 0\)，即
\begin{align*}
g(z) = \sum\limits_{k=1}^{n} \frac{\alpha_k-1}{z-a_k}.
\end{align*}
由于在上半平面中 \(g(z)=\frac{f''(z)}{f'(z)}\)，因而有
\begin{align*}
\frac{f''(z)}{f'(z)} = \sum\limits_{k=1}^{n} \frac{\alpha_k-1}{z-a_k}, \quad z \in D.
\end{align*}
让上式两边沿上半平面内任一条从 \(z_0\) 到 \(z\) 的可求长曲线积分，可得
\begin{align*}
\ln f'(z) = \sum\limits_{k=1}^{n} (\alpha_k-1) \ln(z-a_k) + C' = \ln \left[ C(z-a_1)^{\alpha_1-1} \cdots (z-a_n)^{\alpha_n-1} \right],
\end{align*}
即
\begin{align*}
f'(z) = C(z-a_1)^{\alpha_1-1} \cdots (z-a_n)^{\alpha_n-1},
\end{align*}
由此即得 \eqref{eq:5} 式.
\end{proof}

\begin{theorem}\label{theorem:复变定理7.4.22}
对于实轴上任意三点 \(a_1<a_2<a_3\)，一定存在唯一 \(f\)，它在 \(D\) 中单叶且全纯，在 \(\overline{D}\) 上连续，把 \(D\) 一一地映为多角形域 \(G\)，且把 \(a_1,a_2,a_3\) 分别映为 \(G\) 的三个顶点 \(w_1,w_2,w_3\)。
\end{theorem}
\begin{remark}
在\hyperref[eq:Schwarz-Christoffel公式]{Schwarz-Christoffel 公式}中，用以表达 \(f\) 的是 \(z\) 平面实轴上的点 \(a_1,\cdots,a_n\)。但在具体问题中，通常给出的是多角形的顶点 \(w_k(k=1,\cdots,n)\)，而 \(a_k\) 却是未知的，如何确定 \(a_k\) 是解决问题的关键。在实际解题时，我们往往在实轴上任取三点 \(a_1<a_2<a_3\)，使其与多角形的三个顶点 \(w_1,w_2,w_3\) 相对应，而其他的 \(a_4,\cdots,a_n\) 及 \(C, C_1\) 则由具体问题中的条件来确定。之所以能这样做，其依据正是这个\cref{theorem:复变定理7.4.22}.
\end{remark}
\begin{proof}
根据\cref{lemnma:复变引理7.4.1}，存在 \(w=\varphi(\zeta)\)，它在 \(\zeta\) 平面的上半平面 \(\mathrm{Im}\zeta>0\) 中单叶且全纯，在 \(\mathrm{Im}\zeta \geqslant 0\) 上连续，且把 \(\mathrm{Im}\zeta>0\) 一一地映为 \(G\)，把 \(\mathrm{Im}\zeta=0\) 一一地映为 \(\partial G\)。设 \(\varphi^{-1}(w_j)=\zeta_j, j=1,2,3\)，且满足 \(\zeta_1<\zeta_2<\zeta_3\)，作分式线性变换 \(\zeta=T(z)\)，把 \(a_1,a_2,a_3\) 分别变成 \(\zeta_1,\zeta_2,\zeta_3\)，于是复合函数
\begin{align*}
w = \varphi(T(z)) = f(z)
\end{align*}
便满足定理的要求。

如果还有 \(w=g(z)\) 也满足定理的要求，那么函数 \(h(z)=g^{-1}(f(z))\) 便把上半平面映为上半平面，且把实轴映为实轴。如果记 \(g(\infty)=w'\)，则 \(w'\) 是多角形边界上某一点，而 \(f^{-1}(w')=t, t\) 是实轴上某一点，那么 \(h(t)=\infty\)。现在 \(h\) 在 \((-\infty, t)\) 和 \((t, \infty)\) 上取实数值，根据\hyperref[theorem:双全纯映射的Schwarz对称原理]{Schwarz对称原理}，它可以全纯开拓到下半平面，因而 \(h \in \mathrm{Aut}(C_\infty)\)。由\cref{theorem:定理5.3.5}，\(h\) 是一个分式线性变换。但 \(h(z_j)=z_j, j=1,2,3\)，即 \(h\) 有三个不动点，因而 \(h(z)=z\)，即 \(f(z)=g(z)\)。这就证明了满足定理要求的函数是唯一的.
\end{proof}

\begin{theorem}
设双全纯函数 \(w=f(z)\) 把上半平面 \(D\) 一一地映为多角形域 \(G\)，且 \(f\) 在 \(\overline{D}\) 上连续。如果 \(G\) 在其顶点 \(w_k(k=1,\cdots,n)\) 处的顶角是 \(\alpha_k\pi\) (\(0<\alpha_k<2\))，实轴上与 \(w_k\) 对应的点是 \(a_k(k=1,\cdots,n)\)，\(-\infty < a_1 < \cdots < a_{n-1} < \infty\)，如果 \(a_n=\infty\)，这时 \(f\) 可表示为
\begin{align}
f(z) = C \int_{z_0}^{z} (z-a_1)^{\alpha_1-1} \cdots (z-a_{n-1})^{\alpha_{n-1}-1} \mathrm{d}z + C_1. \label{eq:8}
\end{align}
\end{theorem}
\begin{proof}
任取 \(a<a_1\)，作分式线性变换 \(\zeta=T(z)=\frac{1}{a-z}\)，容易看出，它把 \(\mathrm{Im}z>0\) 映为 \(\mathrm{Im}\zeta>0\)，把实轴上的 \(a_1<a_2<\cdots<a_{n-1}, a_n=\infty\) 映为 \(b_1<b_2<\cdots<b_{n-1}<b_n=0\)，这里 \(b_j=T(a_j), j=1,\cdots,n\)。于是，函数 \(F(\zeta)=(f \circ T^{-1})(\zeta)\) 把上半平面 \(\mathrm{Im}\zeta>0\) 双全纯地映为多角形域 \(G\)，且把 \(b_1,\cdots,b_n\) 分别映为 \(w_1,\cdots,w_n\)。根据\hyperref[theorem:Schwarz-Christoffel定理]{Schwarz-Christoffel定理}，\(F\) 可表示为
\begin{align*}
F(\zeta) = C' \int_{\zeta_0}^{\zeta} (\tau-b_1)^{\alpha_1-1} \cdots (\tau-b_{n-1})^{\alpha_{n-1}-1} \tau^{\alpha_n-1} \mathrm{d}\tau + C_1.
\end{align*}
作变量代换 \(\tau=\frac{1}{a-\eta}\)，因为 \(b_k=\frac{1}{a-a_k}, k=1,\cdots,n-1\)，所以
\begin{align*}
\tau-b_k = \frac{\eta-a_k}{(a-\eta)(a-a_k)}, \quad k=1,\cdots,n-1.
\end{align*}
上式就变成
\begin{align*}
F(\zeta) = C' \int_{T^{-1}(\zeta_0)}^{T^{-1}(\zeta)} (\eta-a_1)^{\alpha_1-1} \cdots (\eta-a_{n-1})^{\alpha_{n-1}-1} \cdot \frac{1}{(a-\eta)^{\alpha_1+\cdots+\alpha_n-(n-2)}} \mathrm{d}\eta + C_1.
\end{align*}
这里 \(C=\frac{C'}{(a-a_1)^{\alpha_1-1}\cdots(a-a_{n-1})^{\alpha_{n-1}-1}}\)。由于 \(\alpha_1+\cdots+\alpha_n=n-2\)，所以
\begin{align*}
F(\zeta) = C \int_{T^{-1}(\zeta_0)}^{T^{-1}(\zeta)} (\eta-a_1)^{\alpha_1-1} \cdots (\eta-a_{n-1})^{\alpha_{n-1}-1} \mathrm{d}\eta + C_1.
\end{align*}
由此即得
\begin{align*}
f(z)=F(T(z)) = C \int_{z_0}^{z} (\eta-a_1)^{\alpha_1-1} \cdots (\eta-a_{n-1})^{\alpha_{n-1}-1} \mathrm{d}\eta + C_1.
\end{align*}
这就是要证明的.
\end{proof}

\begin{theorem}
如果多角形有一个或几个顶点在无穷远处，如果把这个无穷远处的顶角规定为过该点两条边在有限点处的交角乘以 \(-1\)，那么 \hyperref[eq:Schwarz-Christoffel公式]{Schwarz-Christoffel公式}仍然成立。
\end{theorem}
\begin{proof}
\begin{figure}[H]
\centering
\includegraphics[width=0.3\textwidth]{复变1图7.7.png}
\caption{}
\label{figure:复变1图7.7}
\end{figure}
设 \(w_k=\infty\)，如\cref{figure:复变1图7.7}所示，在 \(w_{k-1}w_k\) 和 \(w_k w_{k+1}\) 上分别取 \(w_k'\) 和 \(w_k''\)，连接 \(w_k'\) 和 \(w_k''\) 后得一 \(n+1\) 角形 \(G'\)。于是，把上半平面映为 \(G'\) 的公式为
\begin{align*}
w = C \int_{z_0}^{z} (z-a_1)^{\alpha_1-1} \cdots (z-a_k')^{\alpha_k'-1} (z-a_k'')^{\alpha_k''-1} \cdots (z-a_n)^{\alpha_n-1} \mathrm{d}z + C_1,
\end{align*}
这里 \(a_k'\) 和 \(a_k''\) 是 \(w_k'\) 和 \(w_k''\) 在实轴上的原像。现在让线段 \(w_k' w_k''\) 平行移动到无穷远去，\(G'\) 就趋于 \(G\)，而 \(a_k'\) 和 \(a_k''\) 就趋于 \(\infty\) 点在实轴上的原像 \(a_k\)，上面公式中的被积函数
\begin{align*}
(z-a_k')^{\alpha_k'-1} (z-a_k'')^{\alpha_k''-1} \to (z-a_k)^{\alpha_k'+\alpha_k''-2}.
\end{align*}
如果直线 \(w_{k-1}w_k'\) 和 \(w_k'' w_{k+1}\) 在有限点处的交角为 \(\beta_k\pi\)，那么易知 \(\alpha_k'+\alpha_k''+\beta_k=1\)，所以 \(\alpha_k'+\alpha_k''-2=-1-\beta_k\)。若记 \(a_k=-\beta_k\)，则有
\begin{align*}
\alpha_k'+\alpha_k''-2 = a_k-1.
\end{align*}
这样，上面的公式仍可写成
\begin{align*}
w = C \int_{z_0}^{z} (z-a_1)^{\alpha_1-1} \cdots (z-a_k)^{\alpha_k-1} \cdots (z-a_n)^{\alpha_n-1} \mathrm{d}z + C_1,
\end{align*}
这里的 \(a_k\pi\) 是顶点在无穷远处的两条边在有限点处的交角乘以 \(-1\)。综上所述，我们得到了定理.
\end{proof}

\begin{example}
设 \(G\) 是以 \(w_1, w_2, w_3\) 为顶点，相应的顶角为 \(\alpha_1\pi, \alpha_2\pi, \alpha_3\pi\) (\(\alpha_1+\alpha_2+\alpha_3=1\)) 的三角形域，求出把上半平面映为 \(G\) 的保角变换。
\end{example}
\begin{solution}
设所求的变换为 \(w=f(z)\)。在实轴上取三点 \(a_1=0, a_2=1, a_3=\infty\)，使得 \(f(a_j)=w_j, j=1,2,3\)。由公式 \eqref{eq:8}，有
\begin{align}
w = C \int_{0}^{z} \zeta^{\alpha_1-1} (1-\zeta)^{\alpha_2-1} \mathrm{d}\zeta + C_1. \label{eq:9}
\end{align}
由 \(f(0)=w_1\)，得 \(C_1=w_1\)。再由 \(f(1)=w_2\)，得
\begin{align*}
C = \frac{w_2-w_1}{\int_{0}^{1} t^{\alpha_1-1} (1-t)^{\alpha_2-1} \mathrm{d}t} = \frac{w_2-w_1}{B(\alpha_1,\alpha_2)}.
\end{align*}
这里 \(B\) 表示 Beta 函数。由于 \(\alpha_1+\alpha_2+\alpha_3=1\)，所以
\begin{align*}
B(\alpha_1,\alpha_2) = \frac{\Gamma(\alpha_1)\Gamma(\alpha_2)}{\Gamma(\alpha_1+\alpha_2)} = \frac{\Gamma(\alpha_1)\Gamma(\alpha_2)}{\Gamma(1-\alpha_3)}.
\end{align*}
由余元公式，有
\begin{align*}
\Gamma(\alpha_3)\Gamma(1-\alpha_3) = \frac{\pi}{\sin \alpha_3\pi},
\end{align*}
所以
\begin{align*}
B(\alpha_1,\alpha_2) = \frac{1}{\pi} \Gamma(\alpha_1)\Gamma(\alpha_2)\Gamma(\alpha_3)\sin\alpha_3\pi.
\end{align*}
于是
\begin{align*}
C = \frac{\pi(w_2-w_1)}{\Gamma(\alpha_1)\Gamma(\alpha_2)\Gamma(\alpha_3)\sin\alpha_3\pi}.
\end{align*}
将 \(C\) 和 \(C_1\) 的值代入 \eqref{eq:9} 式，即得所求的变换。
\end{solution}

\begin{example}
求一保角变换，把上半平面映为域 \(G=\{w: -\frac{\pi}{2}<\mathrm{Re}w<\frac{\pi}{2}, \mathrm{Im}w>0\}\)。
\end{example}
\begin{solution}
把 \(G\) 看成一个三角形，它的三个顶点为 \(w_1=-\frac{\pi}{2}, w_2=\frac{\pi}{2}, w_3=\infty\)，对应的三个角分别为 \(\frac{\pi}{2}, \frac{\pi}{2}\) 和 \(0\)。在实轴上取三个点 \(a_1=-1, a_2=1, a_3=\infty\)，由公式 \eqref{eq:8} 即知，把上半平面映为域 \(G\) 的变换为
\begin{align*}
f(z) = C \int_{0}^{z} (z+1)^{-\frac{1}{2}} (z-1)^{-\frac{1}{2}} \mathrm{d}z + C_1 = C \int_{0}^{z} \frac{\mathrm{d}z}{\sqrt{z^2-1}} + C_1 = C'\arcsin z + C_1.
\end{align*}
由 \(f(-1)=-\frac{\pi}{2}, f(1)=\frac{\pi}{2}\)，得 \(C_1=0, C'=1\)。故所求的变换为
\begin{align*}
w = \arcsin z.
\end{align*}

\begin{example}
求一保角变换，把带状域 \(D=\{z: -\pi<\mathrm{Im}z<\pi\}\) 映为\cref{figure:复变2图7.9}(b) 所示的域 \(G\)。
\end{example}
\begin{solution}
\begin{figure}[H]
\centering
\includegraphics[width=0.6\textwidth]{复变2图7.9.png}
\caption{}
\label{figure:复变2图7.9}
\end{figure}
\begin{figure}[H]
\centering
\includegraphics[width=0.6\textwidth]{复变2图7.10.png}
\caption{}
\label{figure:复变2图7.10}
\end{figure}
由对称性，我们只需考虑把 \(0<\mathrm{Im}z<\pi\) 映为\cref{figure:复变2图7.9}(b) 中上半部分域的变换。因为 \(\zeta=\mathrm{e}^z\) 可以把 \(0<\mathrm{Im}z<\pi\) 映为上半平面，故只需研究把上半平面映为\cref{figure:复变2图7.10}(b) 中的那个域 \(G'\)。

把域 \(G'\) 看成一个三角形，它的三个顶点为 \(w_1=\infty, w_2=\infty, w_3=hi\)，相应的三个角分别为 \(\alpha_1=0, \alpha_2=-\beta\) 和 \(\alpha_3=\beta+1\)。令 \(a_1=0, a_2=\infty, a_3=-1\)，由定理 7.4.6，所求的变换为
\begin{align}
w = f(z) = C \int_{-1}^{z} z^{-1}(z+1)^\beta \mathrm{d}z + C_1. \label{eq:10}
\end{align}
因为 \(f(-1)=hi\)，所以 \(C_1=hi\)。

现设法确定常数 \(C\)。以 \(a_1=0\) 为中心、\(\rho\) 为半径作圆，与实轴交于 \(z_1\) 和 \(z_2\)。因为 \(z_1 \in (-1,0)=(a_3,a_1), z_2 \in (0,\infty)=(a_1,a_2)\)，所以 \(f(z_1) \in w_3 w_1, f(z_2) \in w_1 w_2\)，因而
\begin{align}
\mathrm{Im}(f(z_2)-f(z_1)) = -h. \label{eq:11}
\end{align}
另一方面，如果记 \(\gamma_\rho=\{z: z=\rho \mathrm{e}^{i\theta}, 0 \leqslant \theta \leqslant \pi\}\)，则当 \(z \in \gamma_\rho\) 时，有
\begin{align*}
(1+z)^\beta = (1+\rho \mathrm{e}^{i\theta})^\beta = 1 + \beta \rho \mathrm{e}^{i\theta} + \cdots = 1 + O(\rho).
\end{align*}
于是，从 \eqref{eq:10} 式可得
\begin{align*}
f(z_2)-f(z_1) &= C \int_{z_1}^{z_2} \frac{(1+z)^\beta}{z} \mathrm{d}z = -C \int_{\gamma_\rho} \frac{(1+z)^\beta}{z} \mathrm{d}z \\
&= -C \int_{0}^{\pi} \frac{(1+\rho \mathrm{e}^{i\theta})^\beta}{\rho \mathrm{e}^{i\theta}} \rho i \mathrm{e}^{i\theta} \mathrm{d}\theta = -C\pi i + O(\rho).
\end{align*}
令 \(\rho \to 0\)，即得
\begin{align}
\mathrm{Im}(f(z_2)-f(z_1)) = -C\pi. \label{eq:12}
\end{align}
比较 \eqref{eq:11} 式和 \eqref{eq:12} 式，即得
\begin{align*}
C = \frac{h}{\pi}.
\end{align*}
至此，得到把上半平面 \(\mathrm{Im}\zeta>0\) 映为 \(G'\) 的变换为
\begin{align}
w = f(\zeta) = \frac{h}{\pi} \int_{-1}^{\zeta} \frac{(1+z)^\beta}{z} \mathrm{d}z + ih. \label{eq:13}
\end{align}
所以，\(w=f(\mathrm{e}^z)\) 是把 \(0<\mathrm{Im}z<\pi\) 映为 \(G'\) 的变换。当 \(z\) 在实轴上时，\(\mathrm{e}^z\) 在正实轴上，因而 \(f(\mathrm{e}^z)\) 取实数值，根据对称原理，\(w=f(\mathrm{e}^z)\) 把 \(-\pi<\mathrm{Im}z<\pi\) 映为 \(G\)。

下面来看两个特例：
\begin{enumerate}[(1)]
\item 如果 \(\beta=1\)，那么
\begin{align*}
f(\zeta) = \frac{h}{\pi} \int_{-1}^{\zeta} \left( 1+\frac{1}{z} \right) \mathrm{d}z + ih = \frac{h}{\pi} (\ln \zeta + \zeta + 1),
\end{align*}
所以 \(w=f(\mathrm{e}^z) = \frac{h}{\pi} (\mathrm{e}^z + z + 1)\)，它把 \(-\pi<\mathrm{Im}z<\pi\) 一一地映为全平面除去两条半直线(\cref{figure:复变2图7.11}) \(\{z: z=x\pm hi, -\infty<x<0\}\)。

\caption{}
\label{figure:复变2图7.11}
\end{figure}

\item 如果 \(\beta=\frac{1}{2}\)，那么
\begin{align*}
f(\zeta) = \frac{h}{\pi} \int_{-1}^{\zeta} \frac{\sqrt{z+1}}{z} \mathrm{d}z + ih = \frac{2h}{\pi} \left[ \sqrt{\zeta+1} + \ln(\sqrt{\zeta+1}-1) - \frac{1}{2} \ln \zeta \right],
\end{align*}
所以 \(w=f(\mathrm{e}^z) = \frac{2h}{\pi} \left[ \sqrt{\mathrm{e}^z+1} + \ln(\sqrt{\mathrm{e}^z+1}-1) - \frac{z}{2} \right]\)，它把 \(-\pi<\mathrm{Im}z<\pi\) 一一地映为\cref{figure:复变2图7.12}所示的域。

\end{figure}
\end{enumerate}
\end{solution}
