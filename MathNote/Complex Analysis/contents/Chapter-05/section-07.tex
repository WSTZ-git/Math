\section{特殊域上的Mittag-Leffler定理、Weierstrass因子分解定理和Blaschke定理}

\begin{definition}
设 \(\{\gamma_n\}\) 是一列可求长的简单闭曲线，\(l_n = \int_{\gamma_n} |\mathrm{d}z|\) 是 \(\gamma_n\) 的长度，\(d_n \triangleq d(0, \gamma_n) = \inf\limits_{z \in \gamma_n} |z|\) 是 \(\gamma_n\) 与原点 \(O\) 之间的距离. 若满足
\begin{enumerate}[(i)]
\item \(\gamma_n\) 总是位于 \(\gamma_{n+1}\) 的内部，原点 \(O\) 位于 \(\gamma_1\) 的内部；
\item \(d_n \to \infty \quad (n \to \infty)\)；
\item \(\left\{ \frac{l_n}{d_n} \right\}\) 有界，
\end{enumerate}
则称 \(\{\gamma_n\}\) 是\textbf{正则曲线列}.
\end{definition}

\begin{theorem}\label{thm:meromorphic_expan}
设 \(\{\gamma_n\}\) 是正则曲线列，\(f\) 是 \(\mathbb{C}\) 上的亚纯函数. 若
\begin{enumerate}[(i)]
\item \(f\) 的全部互不相同的极点 \(\{a_n\}\) 都是 \(f\) 的 1 阶极点，记 \(c_n = \mathrm{Res}(f, a_n)\)；
\item 原点 \(O\) 不是 \(f\) 的极点；
\item \(f\) 在 \(\bigcup\limits_{n=1}^\infty \gamma_n\) 上有界，
\end{enumerate}
则
\begin{align} \label{eq:1}
f(z) = f(0) + \sum_{n=1}^\infty c_n \left( \frac{1}{z-a_n} + \frac{1}{a_n} \right),
\end{align}
其中，\eqref{eq:1} 式右端在 \(\mathbb{C} \setminus \{a_n : n \in \mathbb{N}\}\) 上内闭一致收敛.
\end{theorem}

\begin{proof}
对每个 \(\gamma_n\)，令 \(D_n\) 是由 \(\gamma_n\) 所围成的单连通域. 设 \(z \in (B(0, R) \setminus \{0\}) \setminus \{a_n : n \in \mathbb{N}\}\)，取 \(m\) 使得 \(B(0, R) \subset D_m\)，考虑积分
\begin{align*}
I_m = \frac{1}{2\pi \mathrm{i}} \int_{\gamma_m} \frac{f(\zeta)}{\zeta(\zeta-z)} \mathrm{d}\zeta.
\end{align*}
显然 \(g(\zeta) = \frac{f(\zeta)}{\zeta(\zeta-z)}\) 在 \(D_m\) 中的全部极点为 \(0, z, a_n \in D_m\)，\(g\) 在这些极点处的留数分别为 \(-\frac{f(0)}{z}, \frac{f(z)}{z}, \frac{c_n}{a_n(a_n-z)}\). 由\hyperref[theorem:留数定理(残数定理)-定理5.4.9]{留数定理}便知，当 \(z \in (B(0, R) \setminus \{0\}) \setminus \{a_n : n \in \mathbb{N}\}\) 时，有
\begin{align*}
I_m = -\frac{f(0)}{z} + \frac{f(z)}{z} + \sum_{a_n \in D_m} \frac{c_n}{a_n(a_n-z)},
\end{align*}
或
\begin{align*}
f(z) = f(0) + \sum_{a_n \in D_m} c_n \left( \frac{1}{z-a_n} + \frac{1}{a_n} \right) + z I_m.
\end{align*}
注意到
\begin{align*}
|z I_m| \leqslant \frac{R}{2\pi} \int_{\gamma_m} \frac{|f(\zeta)|}{|\zeta| |\zeta-z|} |\mathrm{d}\zeta| \leqslant \frac{R}{2\pi} \frac{M l_m}{d_m(d_m - R)} \to 0 \quad (m \to \infty),
\end{align*}
其中 \(M = \sup\limits_{\zeta \in \bigcup_{n=1}^\infty \gamma_n} |f(\zeta)| < \infty\)，便知
\begin{align*}
f(0) + \sum_{n=1}^\infty c_n \left( \frac{1}{z-a_n} + \frac{1}{a_n} \right)
\end{align*}
在 \(\mathbb{C} \setminus \{a_n : n \in \mathbb{N}\}\) 上内闭一致收敛于 \(f(z)\).
\end{proof}

\begin{theorem} \label{thm:entire_product}
设 \(\{\gamma_n\}\) 是正则曲线列，\(f\) 是整函数. 若
\begin{enumerate}[(i)]
\item \(f\) 的全部互不相同的零点集为 \(\{a_n\}\)，在每个 \(a_n\) 处的零点阶数为 \(k_n\)；
\item \(f(0) \neq 0\)；
\item \(\frac{f'(z)}{f(z)}\) 在 \(\bigcup\limits_{n=1}^\infty \gamma_n\) 上有界，
\end{enumerate}
则
\begin{align} \label{eq:2}
f(z) = f(0) \mathrm{e}^{\frac{f'(0)}{f(0)} z} \prod_{n=1}^\infty \left( 1 - \frac{z}{a_n} \right)^{k_n} \mathrm{e}^{\frac{k_n z}{a_n}},
\end{align}
其中\eqref{eq:2} 式右端在 \(\mathbb{C} \setminus \{a_n : n \in \mathbb{N}\}\) 上内闭一致收敛.
\end{theorem}
\begin{proof}
对 \(\mathbb{C}\) 上的亚纯函数 \(\frac{f'(z)}{f(z)}\) 应用定理 \cref{thm:meromorphic_expan}，并注意到
\begin{align*}
\mathrm{Res}\left( \frac{f'(z)}{f(z)}, a_n \right) = k_n,
\end{align*}
便知
\begin{align*}
\frac{f'(0)}{f(0)} + \sum_{n=1}^\infty \left( \frac{k_n}{z-a_n} + \frac{k_n}{a_n} \right)
\end{align*}
在 \(\mathbb{C} \setminus \{a_n : n \in \mathbb{N}\}\) 上内闭一致收敛于 \(\frac{f'(z)}{f(z)}\). 于是
\begin{align} \label{eq:3}
\ln \frac{f(z)}{f(0)} = \int_0^z \frac{f'(\zeta)}{f(\zeta)} \mathrm{d}\zeta = \frac{f'(0)}{f(0)} z + \sum_{n=1}^\infty \int_0^z \left( \frac{k_n}{\zeta - a_n} + \frac{k_n}{a_n} \right) \mathrm{d}\zeta,
\end{align}
其中，积分是沿 \(\mathbb{C} \setminus \{a_n : n \in \mathbb{N}\}\) 中连接 \(O\) 和 \(z\) 的任意可求长曲线进行的，并且 \eqref{eq:3} 式右端在 \(\mathbb{C} \setminus \{a_n : n \in \mathbb{N}\}\) 上内闭一致收敛. 因此
\begin{align*}
f(z) = f(0) \mathrm{e}^{\frac{f'(0)}{f(0)} z} \prod_{n=1}^\infty \left( 1 - \frac{z}{a_n} \right)^{k_n} \mathrm{e}^{\frac{k_n z}{a_n}}.
\end{align*}
\end{proof}

\begin{theorem}\label{thm:blaschke}
设 \(\{a_n\}\) 是 \(B(0, R) \setminus \{0\}\) 中互不相同的点列，\(\{k_n\}\) 是一列自然数. 若
\begin{align*}
\sum_{n=1}^\infty k_n (R - |a_n|) < \infty,
\end{align*}
则
\begin{align}\label{eq:4225ggdrere}
\prod_{j=1}^n \left( \frac{R(a_j - z)}{R^2 - \overline{a_j} z} \right)^{k_j} \left( \frac{|a_j|}{a_j} \right)^{k_j}
\end{align}
必在 \(B(0, R)\) 上内闭一致收敛于一个全纯映射 \(f: B(0, R) \to B(0, 1)\)，使得 \(f\) 恰以 \(\{a_n\}\) 为其零点集，且 \(f\) 在每个 \(a_n\) 处的零点阶数恰为 \(k_n\).
\end{theorem}
\begin{proof}
由所给条件知 \(\{a_n\}\) 在 \(B(0, R)\) 中无极限点，因而 \(B(0, R) \setminus \{a_n : n \in \mathbb{N}\}\) 是域，并且
\begin{align*}
\sum_{n=1}^\infty \frac{k_n (R^2 - |a_n|^2)}{(R^2 - \overline{a_n} z)(z - a_n)}
\end{align*}
在 \(B(0, R) \setminus \{a_n : n \in \mathbb{N}\}\) 上内闭一致收敛于 \(g \in H(B(0, R) \setminus \{a_n : n \in \mathbb{N}\})\). 显然 \(g\) 是 \(B(0, R)\) 上的亚纯函数，恰以 \(\{a_n\}\) 为其极点集，且在每一个 \(a_n\) 处的留数为 \(k_n\). 令
\begin{align} \label{eq:5}
F(z) = \int_0^z g(\zeta) \mathrm{d}\zeta = \sum_{n=1}^\infty \int_0^z \frac{k_n (R^2 - |a_n|^2)}{(R^2 - \overline{a_n} \zeta)(\zeta - a_n)} \mathrm{d}\zeta = \sum_{n=1}^\infty k_n \int_0^z \left( \frac{1}{\zeta - a_n} + \frac{\overline{a_n}}{R^2 - \overline{a_n} \zeta} \right) \mathrm{d}\zeta,
\end{align}
其中，积分是沿 \(B(0, R) \setminus \{a_n : n \in \mathbb{N}\}\) 中连接 \(O\) 和 \(z\) 的可求长曲线进行的，并且 \eqref{eq:5} 式右端在 \(B(0, R) \setminus \{a_n : n \in \mathbb{N}\}\) 上内闭一致收敛. 显然 \(F(z)\) 是 \(B(0, R) \setminus \{a_n : n \in \mathbb{N}\}\) 上的多值全纯函数，但在同一点处的任意两个函数值之差为 \(2\pi \mathrm{i}\) 的整数倍，故
\begin{align*}
h(z) = \mathrm{e}^{F(z)} = \prod_{n=1}^\infty \mathrm{e}^{k_n \int_0^z \left( \frac{1}{\zeta - a_n} + \frac{\overline{a_n}}{R^2 - \overline{a_n} \zeta} \right) \mathrm{d}\zeta} = \prod_{n=1}^\infty \left( \frac{R(a_n - z)}{R^2 - \overline{a_n} z} \right)^{k_n} \left( \frac{R}{a_n} \right)^{k_n}
\end{align*}
是 \(B(0, R)\) 上的全纯函数，\(h\) 恰以 \(\{a_n\}\) 为其零点集，且在每一个 \(a_n\) 处的零点阶数恰为 \(k_n\). 再注意到 \(\prod\limits_{n=1}^\infty \left( \frac{|a_n|}{R} \right)^{k_n}\) 是一个正数，便知 \eqref{eq:4225ggdrere} 式在 \(B(0, R)\) 上内闭一致收敛于 \(h(z) \prod\limits_{n=1}^\infty \left( \frac{|a_n|}{R} \right)^{k_n}\). 最后，由于对任意 \(z \in B(0, R)\)，\(\left| \frac{R(a_n - z)}{R^2 - \overline{a_n} z} \right| < 1\)，故若令
\begin{align*}
f(z) = h(z) \prod_{n=1}^\infty \left( \frac{|a_n|}{a_n} \right)^{k_n} = \prod_{n=1}^\infty \left( \frac{R(a_n - z)}{R^2 - \overline{a_n} z} \right)^{k_n} \left( \frac{|a_n|}{a_n} \right)^{k_n},
\end{align*}
则
\begin{align*}
f(B(0, R)) \subset B(0, 1).
\end{align*}
\end{proof}

\begin{example} \label{ex:cot}
用极点处 Laurent 级数的主要部分表示 \(\text{ctg} z - \frac{1}{z}\).
\end{example}
\begin{solution}
\(f(z) = \text{ctg} z - \frac{1}{z}\) 的全部极点 \(\pm n\pi \quad (n \in \mathbb{N})\) 都是 1 阶的，且 \(\mathrm{Res}(f, \pm n\pi) = 1, f(0) = 0\). 令 \(\gamma_n\) 为以 \(O\) 为中心，以 \((2n-1)\pi\) 为边长，并且平行于坐标轴的正方形折线，则 \(\{\gamma_n\}\) 是正则曲线列. 注意到
\begin{align*}
\left| \text{ctg}\left( \pm \left( n - \frac{1}{2} \right) \pi + \mathrm{i}y \right) \right| = \left| \frac{\mathrm{e}^{+(2n-1)\pi} \mathrm{e}^{-2y} + 1}{\mathrm{e}^{\pm(2n-1)\pi} \mathrm{e}^{-2y} - 1} \right| = \frac{\mathrm{e}^{2|y|} - 1}{\mathrm{e}^{2|y|} + 1} \leqslant 1,
\end{align*}
并且
\begin{align*}
\left| \text{ctg}\left( x \pm \mathrm{i}\left( n - \frac{1}{2} \right)\pi \right) \right| = \left| \frac{1 + \mathrm{e}^{-\mathrm{i}2x} \mathrm{e}^{\pm(2n-1)\pi}}{1 - \mathrm{e}^{-\mathrm{i}2x} \mathrm{e}^{\pm(2n-1)\pi}} \right| \leqslant \frac{\mathrm{e}^{(2n-1)\pi} + 1}{\mathrm{e}^{(2n-1)\pi} - 1} \leqslant \frac{\mathrm{e}^\pi + 1}{\mathrm{e}^\pi - 1}.
\end{align*}
因而 \(\text{ctg} z - \frac{1}{z}\) 在 \(\bigcup\limits_{n=1}^\infty \gamma_n\) 上有界. 故由定理 \cref{thm:meromorphic_expan} 得
\begin{align*}
\text{ctg} z - \frac{1}{z} = \sum_{n=1}^\infty \left[ \left( \frac{1}{z - n\pi} + \frac{1}{n\pi} \right) + \left( \frac{1}{z + n\pi} - \frac{1}{n\pi} \right) \right] = \sum_{n=1}^\infty \frac{2z}{z^2 - n^2\pi^2},
\end{align*}
或者
\begin{align*}
\text{ctg} z = \frac{1}{z} + \sum_{n=1}^\infty \frac{2z}{z^2 - n^2\pi^2}.
\end{align*}
\end{solution}

\begin{example} \label{ex:sin}
求 \(\sin z\) 的因子分解.
\end{example}
\begin{solution}
整函数 \(f(z) = \frac{\sin z}{z}\) 的全部零点 \(\pm n\pi \quad (n \in \mathbb{N})\) 都是 1 阶的，且 \(f(0) = 1\). 设 \(\{\gamma_n\}\) 是\cref{ex:cot} 中给出的正则曲线列，则
\begin{align*}
\frac{f'(z)}{f(z)} = \text{ctg} z - \frac{1}{z}
\end{align*}
在 \(\bigcup\limits_{n=1}^\infty \gamma_n\) 上有界. 故由\cref{thm:entire_product} 得
\begin{align*}
\frac{\sin z}{z} = \prod_{n=1}^\infty \left[ \left( 1 - \frac{z}{n\pi} \right) \mathrm{e}^{\frac{z}{n\pi}} \right] \left[ \left( 1 + \frac{z}{n\pi} \right) \mathrm{e}^{-\frac{z}{n\pi}} \right] = \prod_{n=1}^\infty \left( 1 - \frac{z^2}{n^2 \pi^2} \right),
\end{align*}
或者
\begin{align*}
\sin z = z \prod_{n=1}^\infty \left( 1 - \frac{z^2}{n^2 \pi^2} \right).
\end{align*}
\end{solution}
