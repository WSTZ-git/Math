\section{拓扑空间}

\begin{definition}
设 $X$ 是一个拓扑空间.
\begin{enumerate}[(1)]
\item 若 $X$ 不能表示为两个非空不相交的开集之并，则称 $X$ 是\textbf{连通的}.
\item 若 $A\subset X$ 作为子空间是连通的，则称 $A$ 为 $X$ 中的\textbf{连通子集}.
\end{enumerate}
\end{definition}

\begin{definition}
设 $X$ 是一个拓扑空间. 若对任意 $x,y\in X$，存在连续映射 $\gamma:[0,1]\to X$ 使得 $\gamma(0)=x$，$\gamma(1)=y$，则称 $X$ 是\textbf{道路连通的}. 子集 $A\subset X$ 若作为子空间道路连通，则称为 $X$ 中的\textbf{道路连通子集}.
\end{definition}

\begin{proposition}\label{proposition:道路连通空间一定是连通空间}
道路连通空间一定是连通空间.
\end{proposition}
\begin{proof}
设 $X$ 道路连通. 反设 $X$ 不连通，则可写成 $X=U\cup V$，其中 $U,V$ 为非空不相交开集. 取 $x\in U$，$y\in V$，由道路连通性，存在连续 $\gamma:[0,1]\to X$ 连接 $x$ 与 $y$. 则 $\gamma([0,1])$ 连通，但 $\gamma([0,1])\cap U$ 与 $\gamma([0,1])\cap V$ 是非空不相交的相对开集，其并集为 $\gamma([0,1])$，矛盾. 故 $X$ 连通.
\end{proof}

\begin{proposition}\label{proposition:道路连通空间一定是连通空间-1}
设 $\Omega\subset \mathbb{R}^n$ 是非空开集. 若 $\Omega$ 连通，则 $\Omega$ 道路连通.
\end{proposition}
\begin{remark}
由\refpro{proposition:道路连通空间一定是连通空间}和\refpro{proposition:道路连通空间一定是连通空间-1}知，$\mathbb{R}^n$ 中的非空开集连通当且仅当道路连通.
\end{remark}
\begin{proof}
固定 $x_0\in \Omega$. 定义
\begin{align*}
A \triangleq \{\, x\in \Omega \mid \text{存在道路在 }\Omega\text{ 内连接 }x_0\text{ 与 }x\,\}.
\end{align*}
显然 $A$ 非空（$x_0\in A$）. 下面证明 $A$ 既是 $\Omega$ 的开集又是闭集.

先证 $A$ 是开集：任取 $x\in A$，因 $\Omega$ 开，存在 $r>0$ 使得 $B(x,r)\subset \Omega$. 对任意 $y\in B(x,r)$，由于球 $B(x,r)$ 是凸集，线段连接 $x$ 与 $y$ 完全位于 $B(x,r)\subset \Omega$，故可从 $x_0$ 经道路到 $x$，再沿线段到 $y$，得 $y\in A$. 所以 $B(x,r)\subset A$，$A$ 为开集.

再证 $A$ 是闭集（在 $\Omega$ 中）：设 $z\in \Omega\setminus A$. 因 $\Omega$ 开，取 $r>0$ 使 $B(z,r)\subset \Omega$. 若存在 $y\in B(z,r)\cap A$，则从 $x_0$ 到 $y$ 有道路，再从 $y$ 到 $z$ 沿线段（在球内），可得 $z\in A$，矛盾. 故 $B(z,r)\cap A=\varnothing$，即 $\Omega\setminus A$ 为开集，所以 $A$ 在 $\Omega$ 中为闭集.

因 $\Omega$ 连通，且 $A$ 非空、既开又闭，故 $A=\Omega$. 于是 $\Omega$ 道路连通.
\end{proof}

\begin{definition}
设 $X$ 是一个拓扑空间.
\begin{enumerate}[(1)]
\item 若 $C\subset X$ 是连通的，且不存在 $X$ 中的连通子集 $D$ 使得 $C\subsetneq D$，则称 $C$ 为 $X$ 的一个\textbf{极大连通子集}.
\item 等价地，在 $X$ 上定义关系：$x\sim y$ 当且仅当存在一个包含 $x$ 与 $y$ 的连通子集. 该关系的每个等价类称为 $X$ 的一个\textbf{连通分支}.
\end{enumerate}
\end{definition}

\begin{proposition}\label{proposition:连通分支的基本性质}
设 $X$ 为拓扑空间,则
\begin{enumerate}[(1)]
\item\label{proposition:连通分支的基本性质-1} 连通分支一定是连通的.
\item\label{proposition:连通分支的基本性质-2} $E\subset X$是连通分支等价于$E$是$X$ 的极大连通子集；
\item\label{proposition:连通分支的基本性质-3} 不同连通分支不相交；
\item\label{proposition:连通分支的基本性质-4} 所有连通分支的并等于 $X$.
\end{enumerate}
\end{proposition}
\begin{proof}
\begin{enumerate}[(1)]
\item 由连通分支的定义知,每个等价类都是连通的,故任意两点可由连通子集相连，进而整个集合连通. 因此结论成立.

\item 先证每个等价类都是极大连通子集. 设 $E$ 是一个等价类. 任取 $x\in E$，因为 $x\sim x$，且单点集连通，所以 $E$ 非空. 对任意 $y,z\in E$，有 $x\sim y$ 且 $x\sim z$，因此存在连通集 $A$ 含 $x,y$，连通集 $B$ 含 $x,z$. 由于 $A\cup B$ 连通且含 $y,z$，故 $y\sim z$，从而 $E$ 是连通的（任意两点由连通子集相连）.

若 $E$ 不是极大的，则存在连通子集 $F$ 满足 $E\subsetneq F$. 取 $f\in F\setminus E$，任取 $e\in E$，则 $F$ 是包含 $e$ 与 $f$ 的连通集，故 $e\sim f$，与 $f\notin E$ 矛盾. 所以 $E$ 是极大连通子集，即连通分支.

反之，设 $C$ 是极大连通子集. 任取 $x\in C$，则 $C$ 是包含 $x$ 与 $y$ 的连通集，故对任意 $y\in C$ 有 $x\sim y$，即 $C$ 含于 $x$ 的等价类 $E_x$. 而 $E_x$ 连通（如上所证），且 $C\subset E_x$，由 $C$ 的极大性得 $C=E_x$. 因此每个极大连通子集恰为一个等价类.

\item 若两个连通分支 $C_1,C_2$ 相交，则 $C_1\cup C_2$ 是连通子集（两个连通集相交，并集连通），且 $C_1\subsetneq C_1\cup C_2$，这与 $C_1$ 的极大性矛盾.
\item 对任意 $x\in X$，由等价关系知 $x$ 属于某个等价类，即属于某个连通分支，因此并集为 $X$.
\end{enumerate}
\end{proof}

\begin{proposition}\label{proposition:Rn中开集的连通分支必是开集}
设 $\Omega\subset \mathbb{R}^n$ 是开集，$C$ 是 $\Omega$ 的一个连通分支，则
\begin{enumerate}[(1)]
\item\label{proposition:Rn中开集的连通分支必是开集-1} $C$是道路连通的.

\item\label{proposition:Rn中开集的连通分支必是开集-2} $C$是 $\mathbb{R}^n$ 中的开集.

\item\label{proposition:Rn中开集的连通分支必是开集-1-3} 在子空间拓扑下，$C$ 既是 $\Omega$ 的开子集也是 $\Omega$ 的闭子集.
\end{enumerate}
\end{proposition}
\begin{proof}
\begin{enumerate}[(1)]
\item 由\refpro{proposition:道路连通空间一定是连通空间-1}和\rrefpro{proposition:连通分支的基本性质}{proposition:连通分支的基本性质-1}立得.

\item 设 $C$ 是 $\Omega$ 的一个连通分支，任取 $x\in C$. 由于 $\Omega$ 是开集，存在 $r>0$ 使得 $B(x,r)\subset \Omega$. 因为 $B(x,r)$ 是连通的且与 $C$ 相交于 $x$，所以 $C\cup B(x,r)$ 是连通的. 由 $C$ 的极大性得 $C\cup B(x,r)\subset C$，从而 $B(x,r)\subset C$. 故 $C$ 是开集.

\item $C$ 是开集已由(1)得到. 又因 $\Omega\setminus C$ 是其余连通分支的并集，每个分支都是开集，故 $\Omega\setminus C$ 是开集，从而 $C$ 在 $\Omega$ 中是闭集.
\end{enumerate}
\end{proof}

\begin{proposition}\label{proposition:Rn中连通分支的边界包含于原集合的边界}
设 $\Omega\subset \mathbb{R}^n$ 是开集，$C$ 是 $\Omega$ 的一个连通分支，则
\begin{align*}
\partial C \subset \partial \Omega .
\end{align*}
即连通分支的边界包含于原集合的边界.
\end{proposition}
\begin{remark}
对 $\mathbb{R}^n$ 中的开集，连通性与道路连通性等价，因此同一连通分支内的任意两点都可由连续曲线连接.
\end{remark}
\begin{proof}
因为 $C\subset \Omega$，取闭包得 $\overline{C}\subset \overline{\Omega}$，故 $\partial C\subset \overline{C}\subset \overline{\Omega}$. 下面证明 $\partial C$ 不含 $\Omega$ 的内点.
假设存在 $x\in \partial C\cap \Omega$. 由\refpro{proposition:Rn中开集的连通分支必是开集}知$\Omega$ 是开集且 $C$ 是开集，因为开集不含边界点,所以$x\notin C$. 因此 $x$ 属于 $\Omega\setminus C$. 又因为连通分支构成 $\Omega$ 的划分，$x$ 必属于另一个连通分支 $C'$. 由于 $C'$ 也是开集，存在 $r>0$ 使得 $B(x,r)\subset C'$. 于是 $B(x,r)\cap C=\varnothing$，这与 $x\in \partial C$ 矛盾!故 $\partial C\cap \Omega=\varnothing$，结合 $\partial C\subset \overline{\Omega}$ 得 $\partial C\subset \overline{\Omega}\setminus \Omega=\partial \Omega$.
\end{proof}




\begin{definition}[连续映射]\label{definition:topology-连续映射-Q7m2R9}
设 $(X,\tau_X)$ 和 $(Y,\tau_Y)$ 是拓扑空间, $f:X\to Y$ 是映射. 对任意集合 $V\subseteq Y$, 定义 $V$ 在 $f$ 下的原像为
\begin{align*}
f^{-1}(V)\triangleq\{x\in X:f(x)\in V\}.
\end{align*}
如果对每个 $Y$ 中的开集 $V$, 原像 $f^{-1}(V)$ 都是 $X$ 中的开集, 即
\begin{align*}
V\in\tau_Y\Longrightarrow f^{-1}(V)\in\tau_X,
\end{align*}
则称 $f$ 是\textbf{连续映射}.
这里的 $f^{-1}(V)$ 表示集合的原像, 并不要求 $f$ 存在反函数.
\end{definition}

\label{topology-theorem:连续映射的基本性质}

\begin{definition}[紧集]\label{definition:topology-紧集-L4p8N2}
设 $X$ 是拓扑空间, $K\subseteq X$. 如果对于任意一族 $X$ 中的开集 $\{U_\alpha\}_{\alpha\in A}$, 只要
\begin{align*}
K\subseteq\bigcup\limits_{\alpha\in A}U_\alpha,
\end{align*}
就存在有限指标集 $A_0\subseteq A$, 使得
\begin{align*}
K\subseteq\bigcup\limits_{\alpha\in A_0}U_\alpha,
\end{align*}
则称 $K$ 是 $X$ 中的紧集.

也就是说, $K$ 的任意开覆盖都有有限子覆盖. 当 $K=X$ 时, 称 $X$ 是紧空间.
\end{definition}

\begin{proposition}[连续映射下闭集的原像是闭集]\label{proposition:连续映射下闭集的原像是闭集}
设 $X,Y$ 为拓扑空间, $f:X\to Y$ 为连续映射.
若 $F\subseteq Y$ 是闭集, 则其原像
\begin{align*}
f^{-1}(F)\triangleq\{x\in X:f(x)\in F\}
\end{align*}
是 $X$ 中的闭集.
\end{proposition}
\begin{proof}
由于 $F$ 是 $Y$ 中的闭集, $Y\setminus F$ 是 $Y$ 中的开集.
由 $f$ 的连续性, $f^{-1}(Y\setminus F)$ 是 $X$ 中的开集.
另一方面, 根据原像的定义,有
\begin{align*}
f^{-1}(Y\setminus F)
=\{x\in X:f(x)\notin F\}=X\setminus\{x\in X:f(x)\in F\}=X\setminus f^{-1}(F).
\end{align*}
因此, $f^{-1}(F)$ 在 $X$ 中的补集是开集, 所以 $f^{-1}(F)$ 是 $X$ 中的闭集.
\end{proof}

\begin{theorem}[紧集的连续像是紧集]\label{theorem:topology-紧集的连续像是紧集}
设 $X$ 和 $Y$ 是拓扑空间, $f:X\to Y$ 是连续映射. 如果 $K\subseteq X$ 是紧集, 则
\begin{align*}
f(K)\triangleq\{f(x):x\in K\}
\end{align*}
是 $Y$ 中的紧集.
\end{theorem}
\begin{proof}
根据紧集的定义, 只需证明 $f(K)$ 的任意开覆盖都有有限子覆盖.

\noindent$\bm{Step}\bm{1}$.{\heiti 利用连续性, 将像集的开覆盖拉回到原空间.}

任取 $f(K)$ 的一个开覆盖 $\{V_\alpha\}_{\alpha\in A}$, 其中每个 $V_\alpha$ 都是 $Y$ 中的开集, 且
\begin{align*}
f(K)\subseteq\bigcup\limits_{\alpha\in A}V_\alpha.
\end{align*}
对每个 $\alpha\in A$, 令
\begin{align*}
U_\alpha\triangleq f^{-1}(V_\alpha).
\end{align*}
由于 $f$ 连续, 而 $V_\alpha$ 是 $Y$ 中的开集, 根据连续映射的定义, $U_\alpha$ 是 $X$ 中的开集.

下面验证这些原像覆盖 $K$. 任取 $x\in K$, 则 $f(x)\in f(K)$. 由上述覆盖关系, 存在 $\alpha\in A$, 使得
\begin{align*}
f(x)\in V_\alpha.
\end{align*}
根据原像的定义, 这等价于
\begin{align*}
x\in f^{-1}(V_\alpha)=U_\alpha.
\end{align*}
因此
\begin{align*}
K\subseteq\bigcup\limits_{\alpha\in A}U_\alpha.
\end{align*}
这就说明 $\{U_\alpha\}_{\alpha\in A}$ 是 $K$ 的一个开覆盖.

\noindent$\bm{Step}\bm{2}$.{\heiti 利用 $K$ 的紧性选取有限子覆盖.}

由于 $K$ 是紧集, 上述开覆盖存在有限子覆盖. 因此, 存在有限指标集 $A_0\subseteq A$, 使得
\begin{align*}
K
\subseteq\bigcup\limits_{\alpha\in A_0}U_\alpha
=\bigcup\limits_{\alpha\in A_0}f^{-1}(V_\alpha).
\end{align*}

\noindent$\bm{Step}\bm{3}$.{\heiti 验证对应的有限个开集覆盖像集.}

下面证明, 同一组指标对应的开集 $\{V_\alpha\}_{\alpha\in A_0}$ 覆盖 $f(K)$.

任取 $y\in f(K)$. 根据像集的定义, 存在 $x\in K$, 使得 $y=f(x)$. 由前一步的有限覆盖关系, 存在 $\alpha\in A_0$, 使得
\begin{align*}
x\in f^{-1}(V_\alpha).
\end{align*}
于是
\begin{align*}
y=f(x)\in V_\alpha.
\end{align*}
因此
\begin{align*}
f(K)\subseteq\bigcup\limits_{\alpha\in A_0}V_\alpha.
\end{align*}
由于 $A_0$ 是有限集, 这就是原开覆盖的一个有限子覆盖.

综上, $f(K)$ 的任意开覆盖都有有限子覆盖, 所以 $f(K)$ 是 $Y$ 中的紧集.
\end{proof}



\begin{definition}[同胚]
设 \(X,Y\) 为拓扑空间, \(f:X\to Y\) 为映射. 若 \(f\) 是双射, 且 \(f\) 与 \(f^{-1}:Y\to X\) 均连续, 则称 \(f\) 为从 \(X\) 到 \(Y\) 的\textbf{同胚映射}, 或称 \(f\) 为\textbf{同胚}. 若存在从 \(X\) 到 \(Y\) 的同胚映射, 则称 \(X\) 与 \(Y\) \textbf{同胚}.
\end{definition}

\begin{definition}[微分同胚]
设 \(U\subseteq\mathbb R^n\), \(V\subseteq\mathbb R^m\) 为非空开集, \(k\in\mathbb N\cup\{0\}\cup\{\infty\}\). 若映射 \(f:U\to V\) 是双射, 且
\begin{align*}
f\in C^k(U),\qquad f^{-1}\in C^k(V),
\end{align*}
则称 \(f\) 为从 \(U\) 到 \(V\) 的 \(C^k\) 同胚, 亦称 \textbf{\(C^k\) 微分同胚}. 当 \(k=0\) 时, \(C^0\) 同胚就是通常的同胚; 当 \(k\geqslant 1\) 时, \(C^k\) 同胚通常称为 \(C^k\) 微分同胚;当$k=\infty$时,称 \(f\) 为从 \(U\) 到 \(V\) 的 \(C^\infty\) 微分同胚, 亦称\textbf{光滑微分同胚}.
\end{definition}


\begin{theorem}\label{theorem:同胚保持集合内点闭包边界}
设 \(X,Y\) 为拓扑空间, \(f:X\to Y\) 为同胚, \(A\subseteq X\). 则 \(f\) 把 \(X\) 中开集映为 \(Y\) 中开集, 把 \(X\) 中闭集映为 \(Y\) 中闭集, 且
\begin{align*}
f(\operatorname{int}A)=\operatorname{int}f(A),\qquad
f(\overline A)=\overline{f(A)},\qquad
f(\partial A)=\partial f(A).
\end{align*}
\end{theorem}
\begin{proof}
\noindent$\mathbf{Step}\bm{1}.${\heiti (证明同胚保持开集与闭集)}
任取 \(X\) 中开集 \(U\). 由 \(f^{-1}\) 连续知 \(f(U)=(f^{-1})^{-1}(U)\) 是 \(Y\) 中开集, 故 \(f\) 把开集映为开集. 若 \(F\subseteq X\) 为闭集, 则 \(X\setminus F\) 为开集, 故 \(f(X\setminus F)=Y\setminus f(F)\) 为开集, 从而 \(f(F)\) 为闭集.

\noindent$\mathbf{Step}\bm{2}.${\heiti (证明内部保持)}
先证 \(f(\operatorname{int}A)\subseteq\operatorname{int}f(A)\). 由于 \(\operatorname{int}A\) 是 \(X\) 中开集且 \(\operatorname{int}A\subseteq A\), 由 Step1 知 \(f(\operatorname{int}A)\) 是 \(Y\) 中开集, 且 \(f(\operatorname{int}A)\subseteq f(A)\). 由内部的最大性得 \(f(\operatorname{int}A)\subseteq\operatorname{int}f(A)\). 再证反向包含. 由于 \(\operatorname{int}f(A)\) 是 \(Y\) 中开集且 \(\operatorname{int}f(A)\subseteq f(A)\), 由 \(f^{-1}\) 连续知 \(f^{-1}(\operatorname{int}f(A))\) 是 \(X\) 中开集, 且由 \(f\) 为双射得 \(f^{-1}(\operatorname{int}f(A))\subseteq f^{-1}(f(A))=A\). 由内部的最大性得 \(f^{-1}(\operatorname{int}f(A))\subseteq\operatorname{int}A\). 两边作用 \(f\) 得 \(\operatorname{int}f(A)\subseteq f(\operatorname{int}A)\). 故等号成立.

\noindent$\mathbf{Step}\bm{3}.${\heiti (证明闭包保持)}
先证 \(\overline{f(A)}\subseteq f(\overline A)\). 由于 \(\overline A\) 是 \(X\) 中闭集, 由 Step1 知 \(f(\overline A)\) 是 \(Y\) 中闭集, 且 \(f(A)\subseteq f(\overline A)\). 由闭包的最小性得 \(\overline{f(A)}\subseteq f(\overline A)\). 再证 \(f(\overline A)\subseteq\overline{f(A)}\). 由于 \(\overline{f(A)}\) 是 \(Y\) 中闭集, 由 \(f\) 连续知 \(f^{-1}(\overline{f(A)})\) 是 \(X\) 中闭集, 且 \(A\subseteq f^{-1}(\overline{f(A)})\). 由闭包的最小性得 \(\overline A\subseteq f^{-1}(\overline{f(A)})\). 两边作用 \(f\) 得 \(f(\overline A)\subseteq\overline{f(A)}\). 故等号成立.

\noindent$\mathbf{Step}\bm{4}.${\heiti (证明边界保持)}
由边界定义 \(\partial A=\overline A\setminus\operatorname{int}A\). 因为 \(f\) 是双射, 对任意 \(C,D\subseteq X\) 有 \(f(C\setminus D)=f(C)\setminus f(D)\). 于是
\begin{align*}
f(\partial A)
=f(\overline A\setminus\operatorname{int}A)
=f(\overline A)\setminus f(\operatorname{int}A)
=\overline{f(A)}\setminus\operatorname{int}f(A)
=\partial f(A).
\end{align*}
综上, 同胚保持内部, 闭包和边界.
\end{proof}
