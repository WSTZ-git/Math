\section{$L^p$空间的结构}

\subsection{$L^p(E)(1\leqslant p<+\infty)$是完备距离空间}

\begin{theorem}
对于$f,g\in L^p(E)$，定义
\begin{align*}
d(f,g)=\|f-g\|_p,\quad 1\leqslant p\leqslant+\infty,
\end{align*}
则$(L^p(E),d)$是一个距离(度量)空间，仍简记为$L^p(E)$.
\end{theorem}
\begin{remark}
我们将$L^p(E)$中几乎处处相等的函数视为同一元素，即当$f,g\in L^p(E)$且$f(x)=g(x)$ a.e. $x\in E$时，称$f$与$g$为$L^p(E)$中的同一元素。或者说用几乎处处相等作为等价关系把$L^p(E)$中的元分成等价类.例如,在$E$上几乎处处等于零的函数全体称为$L^p(E)$中的零元.
\end{remark}
\begin{proof}
\begin{enumerate}[(i)]
\item $d(f,g)\geqslant0$，且$d(f,g)=0$当且仅当$f(x)=g(x)$ a.e. $x\in E$，即$f$与$g$为同一元素。

\item $d(f,g)=d(g,f)$。

\item 由\hyperref[theorem:实变函数--Minkowski不等式]{Minkowski不等式}，
\begin{align*}
\|f-g\|_p=\|f-h+h-g\|_p
\leqslant\|f-h\|_p+\|g-h\|_p,
\end{align*}
即$d(f,g)\leqslant d(f,h)+d(h,g)$。
\end{enumerate}
\end{proof}

\begin{definition}
设$\{f_k\}\subseteq L^p(E)$。若存在$f\in L^p(E)$，使得
\begin{align*}
\lim\limits_{k\to\infty}d(f_k,f)=\lim\limits_{k\to\infty}\|f_k-f\|_p=0,
\end{align*}
则称$\{f_k\}$依$L^p(E)$的意义\textbf{收敛于}$f$，$\{f_k\}$为$L^p(E)$中的\textbf{收敛列},$f$为$\{f_k\}$在$L^p(E)$中的\textbf{极限}。
\end{definition}

\begin{proposition}
\begin{enumerate}[(i)]
\item 若$\lim\limits_{k\to\infty}\|f_k-f\|_p=0$且$\lim\limits_{k\to\infty}\|f_k-g\|_p=0$，则$f=g$，即$f(x)=g(x)$ a.e. $x\in E$。
\item 若$\lim\limits_{k\to\infty}\|f_k-f\|_p=0$，则$\lim\limits_{k\to\infty}\|f_k\|_p=\|f\|_p$，因为$|\|f_k\|_p-\|f\|_p|\leqslant\|f_k-f\|_p$。
\end{enumerate}
\end{proposition}

\begin{definition}
设$\{f_k\}\subseteq L^p(E)$。若$\lim\limits_{k,j\to\infty}\|f_k-f_j\|_p=0$，则称$\{f_k\}$是$L^p(E)$中的\textbf{基本}（或$\mathbf{Cauchy}$）\textbf{列}。
\end{definition}

\begin{theorem}[Riesz-Fischer定理]\label{theorem:Riesz-Fischer定理的另一种常用形式}
$L^p(E)$是完备的距离空间。
\end{theorem}
\begin{proof}
当$1\leqslant p<+\infty$时，设$\{f_k\}\subseteq L^p(E)$满足$\lim\limits_{k,j\to\infty}\|f_j-f_k\|_p=0$。对任给$\sigma>0$，令$E_{j,k}(\sigma)=\{x\in E:|f_j(x)-f_k(x)|\geqslant\sigma\}$，则
\begin{align*}
\sigma(m(E_{j,k}(\sigma)))^{\frac{1}{p}}\leqslant\left(\int_{E_{j,k}(\sigma)}|f_j(x)-f_k(x)|^p\mathrm{d}x\right)^{\frac{1}{p}}
\leqslant\left(\int_E|f_j(x)-f_k(x)|^p\mathrm{d}x\right)^{\frac{1}{p}}=\|f_j-f_k\|_p.
\end{align*}
故$\lim\limits_{j,k\to\infty}m(E_{j,k}(\sigma))=0$，即$\{f_k\}$是依测度Cauchy列。由\refthe{theorem:定理3.16}，存在$E$上几乎处处有限的可测函数$f(x)$，使得$\{f_k\}$依测度收敛于$f(x)$。再由\refthe{theorem:Riesz定理}，存在子列$\{f_{k_i}\}$，使得$\lim\limits_{i\to\infty}f_{k_i}(x)=f(x)$ a.e. $x\in E$。

由\hyperref[theorem:Fatou引理]{Fatou引理}，
\begin{align*}
\int_E|f_k(x)-f(x)|^p\mathrm{d}x=\int_E\lim\limits_{i\to\infty}|f_k(x)-f_{k_i}(x)|^p\mathrm{d}x
\leqslant\liminf\limits_{i\to\infty}\int_E|f_k(x)-f_{k_i}(x)|^p\mathrm{d}x,
\end{align*}
故$\lim\limits_{k\to\infty}\int_E|f_k(x)-f(x)|^p\mathrm{d}x=0$，即$\lim\limits_{k\to\infty}\|f_k-f\|_p=0$。由$\|f\|_p\leqslant\|f-f_k\|_p+\|f_k\|_p$，知$f\in L^p(E)$。

当$p=+\infty$时，设$\{f_k\}\subseteq L^\infty(E)$满足$\lim\limits_{k,j\to\infty}\|f_k-f_j\|_\infty=0$。对任意$k,j$，有$|f_k(x)-f_j(x)|\leqslant\|f_k-f_j\|_\infty$ a.e. $x\in E$，故存在零测集$Z$，使得对所有$k,j$，$|f_k(x)-f_j(x)|\leqslant\|f_k-f_j\|_\infty$，$x\notin Z$。于是存在$f(x)$，使得$\lim\limits_{k\to\infty}f_k(x)=f(x)$，$x\in E\setminus Z$。

对任给$\varepsilon>0$，取自然数$N$，当$j,k>N$时，$\|f_k-f_j\|_\infty<\varepsilon$。当$k>N$且$x\in E\setminus Z$时，
\begin{align*}
|f_k(x)-f(x)|=\lim\limits_{j\to\infty}|f_k(x)-f_j(x)|\leqslant\varepsilon,
\end{align*}
故$\|f_k-f\|_\infty\leqslant\varepsilon$，即$\lim\limits_{k\to\infty}\|f_k-f\|_\infty=0$，且$f\in L^\infty(E)$。
\end{proof}


\subsection{$L^p(E)(1\leqslant p<+\infty)$是可分空间}

\begin{definition}
设$\Gamma$是$L^p(E)$的子集。若对任意$f\in L^p(E)$及$\varepsilon>0$，存在$g\in\Gamma$，使得$\|f-g\|_p<\varepsilon$，则称$\Gamma$在$L^p(E)$中\textbf{稠密}；若$L^p(E)$中存在可数稠密子集，则称$L^p(E)$是\textbf{可分的}。
\end{definition}

\begin{lemma}\label{lemma:实变函数--紧支集连续函数逼近Lp}
设$f\in L^p(E)(1\leqslant p<+\infty)$，则对任意$\varepsilon>0$，有
\begin{enumerate}[(i)]
\item 存在$\mathbb{R}^n$上具有紧支集的连续函数$g(x)$，使得
\begin{align*}
\int_E|f(x)-g(x)|^p\mathrm{d}x<\varepsilon;
\end{align*}
\item 存在$\mathbb{R}^n$上具有紧支集的阶梯函数$\varphi(x)=\sum\limits_{i=1}^k c_i\chi_{I_i}(x)$（每个$I_i$是二进方体），使得
\begin{align*}
\int_E|f(x)-\varphi(x)|^p\mathrm{d}x<\varepsilon.
\end{align*}
\end{enumerate}
\end{lemma}
\begin{lemma}\label{lemma:实变函数--紧支集连续函数逼近Lp}
设 \(E\subseteq\mathbb{R}^n\) 可测，\(f\in L^p(E)\)（\(1\leqslant p<+\infty\)）。则对任意 \(\varepsilon>0\)，有
\begin{enumerate}[(i)]
\item 存在 \(\mathbb{R}^n\) 上具有紧支集的连续函数 \(g\)，使得
\begin{align*}
\int_E |f(x)-g(x)|^p\,\mathrm{d}x<\varepsilon;
\end{align*}
\item 存在 \(\mathbb{R}^n\) 上具有紧支集的阶梯函数 \(\varphi(x)=\sum\limits_{i=1}^k c_i\chi_{I_i}(x)\)（每个 \(I_i\) 是二进方体），使得
\begin{align*}
\int_E |f(x)-\varphi(x)|^p\,\mathrm{d}x<\varepsilon.
\end{align*}
\end{enumerate}
\end{lemma}
\begin{proof}

令 \(\widetilde f\) 为 \(f\) 在 \(\mathbb{R}^n\) 上的零延拓，即
\begin{align*}
\widetilde f(x)\triangleq
\begin{cases}
f(x), & x\in E,\\
0, & x\in\mathbb{R}^n\setminus E.
\end{cases}
\end{align*}
则 \(\widetilde f\in L^p(\mathbb{R}^n)\)，且对任意在 \(\mathbb{R}^n\) 上定义的函数 \(\psi\)，有
\begin{align*}
\int_E |f-\psi|^p\,\mathrm{d}x
=\int_E |\widetilde f-\psi|^p\,\mathrm{d}x
\leqslant\int_{\mathbb{R}^n} |\widetilde f-\psi|^p\,\mathrm{d}x.
\end{align*}
因此只需证明 \(E=\mathbb{R}^n\) 且 \(f\in L^p(\mathbb{R}^n)\) 的情形。

固定 \(\varepsilon>0\)。令 \(\delta\triangleq\varepsilon^{\frac{1}{p}}\)，则只需找到相应的函数使得其与 \(\widetilde f\) 的 \(L^p\) 范数小于 \(\delta\)。

(i)因为 \(\widetilde f\in L^p(\mathbb{R}^n)\)，存在 \(R>0\)，使得
\begin{align*}
\int_{|x|>R}|\widetilde f(x)|^p\,\mathrm{d}x<\left(\frac{\delta}{4}\right)^p.
\end{align*}
令 \(f_R\triangleq\widetilde f\chi_{B(0,R)}\)，则
\begin{align*}
\|\widetilde f-f_R\|_{L^p(\mathbb{R}^n)}<\frac{\delta}{4}.
\end{align*}
由于\hyperref[theorem:简单函数逼近定理]{简单函数逼近定理}知，存在在$\mathbb{R}^n$中具有紧支集的简单函数 \(s\)，使得
\begin{align*}
\|f_R-s\|_{L^p(\mathbb{R}^n)}<\frac{\delta}{4}.
\end{align*}
设
\begin{align*}
s=\sum_{j=1}^m a_j\chi_{A_j},
\end{align*}
其中 \(A_j\) 是 \(\mathbb{R}^n\) 中的可测集，且 \(A_j\subseteq B(0,R+1)\)。

对每个 \(j\)，因为 \(A_j\) 可测，由\cref{corollary:集合可测的充要条件2}知,存在紧集 \(K_j\subseteq A_j\) 和开集 \(O_j\supseteq A_j\)，使得
\begin{align*}
m(O_j\setminus K_j)<\left(\frac{\delta}{4m(|a_j|+1)}\right)^p.
\end{align*}
构造连续函数 \(h_j\)，满足
\begin{align*}
0\leqslant h_j\leqslant 1,\quad
h_j=1\ \text{在}\ K_j\ \text{上},\quad
h_j=0\ \text{在}\ O_j\ \text{外}.
\end{align*}
则
\begin{align*}
\|\chi_{A_j}-h_j\|_{L^p(\mathbb{R}^n)}^p
=\int_{O_j\setminus K_j}|\chi_{A_j}-h_j|^p\,\mathrm{d}x
\leqslant m(O_j\setminus K_j)
<\left(\frac{\delta}{4m(|a_j|+1)}\right)^p,
\end{align*}
故
\begin{align*}
\|\chi_{A_j}-h_j\|_{L^p(\mathbb{R}^n)}
<\frac{\delta}{4m(|a_j|+1)}.
\end{align*}
令
\begin{align*}
g\triangleq\sum_{j=1}^m a_j h_j,
\end{align*}
则 \(g\) 具有紧支集且连续。并且
\begin{align*}
\|s-g\|_{L^p(\mathbb{R}^n)}
&\leqslant\sum_{j=1}^m |a_j|\,\|\chi_{A_j}-h_j\|_{L^p(\mathbb{R}^n)}
<\sum_{j=1}^m |a_j|\frac{\delta}{4m(|a_j|+1)}
\leqslant\frac{\delta}{4}.
\end{align*}
于是
\begin{align*}
\|\widetilde f-g\|_{L^p(\mathbb{R}^n)}
&\leqslant\|\widetilde f-f_R\|_{L^p(\mathbb{R}^n)}
+\|f_R-s\|_{L^p(\mathbb{R}^n)}
+\|s-g\|_{L^p(\mathbb{R}^n)}
\\
&<\frac{\delta}{4}+\frac{\delta}{4}+\frac{\delta}{4}
=\frac{3\delta}{4}<\delta.
\end{align*}
因此
\begin{align*}
\int_E |f-g|^p\,\mathrm{d}x
\leqslant\|\widetilde f-g\|_{L^p(\mathbb{R}^n)}^p
<\delta^p=\varepsilon.
\end{align*}
故 (i) 成立。

(ii)由 (i)，存在紧支连续函数 \(g\)，使得
\begin{align*}
\|\widetilde f-g\|_{L^p(\mathbb{R}^n)}<\frac{\delta}{2}.
\end{align*}
因为 \(g\) 紧支且连续，所以 \(g\) 在 \(\mathbb{R}^n\) 上一致连续。设 \(K\triangleq\operatorname{spt}g\)。取 \(\eta>0\) 足够小，使得
\begin{align*}
\eta^p\,m(K+\overline{B(0,1)})<\left(\frac{\delta}{2}\right)^p.
\end{align*}
由一致连续性，存在 \(\rho>0\)，使得当 \(|x-y|<\rho\) 时，
\begin{align*}
|g(x)-g(y)|<\eta.
\end{align*}
取足够大的整数 \(N\)，使得二进方体
\begin{align*}
I_{\mathbf{k}}^{(N)}\triangleq\prod_{i=1}^n\left[\frac{k_i}{2^N},\frac{k_i+1}{2^N}\right),\quad
\mathbf{k}=(k_1,\cdots,k_n)\in\mathbb{Z}^n
\end{align*}
的直径小于 \(\rho\)。令
\begin{align*}
\varphi(x)\triangleq\sum_{\mathbf{k}} g(\xi_{\mathbf{k}})\chi_{I_{\mathbf{k}}^{(N)}}(x),
\end{align*}
其中 \(\xi_{\mathbf{k}}\) 是 \(I_{\mathbf{k}}^{(N)}\) 中任意固定点，求和仅对与 \(K+\overline{B(0,1)}\) 相交的方体进行。则 \(\varphi\) 是紧支阶梯函数，且对每个 \(x\in\mathbb{R}^n\)，若 \(x\in I_{\mathbf{k}}^{(N)}\)，则
\begin{align*}
|g(x)-\varphi(x)|=|g(x)-g(\xi_{\mathbf{k}})|<\eta.
\end{align*}
因此
\begin{align*}
\|g-\varphi\|_{L^p(\mathbb{R}^n)}^p
=\int_{K+\overline{B(0,1)}}|g(x)-\varphi(x)|^p\,\mathrm{d}x
\leqslant\eta^p\,m(K+\overline{B(0,1)})
<\left(\frac{\delta}{2}\right)^p,
\end{align*}
故
\begin{align*}
\|g-\varphi\|_{L^p(\mathbb{R}^n)}<\frac{\delta}{2}.
\end{align*}
于是
\begin{align*}
\|\widetilde f-\varphi\|_{L^p(\mathbb{R}^n)}
\leqslant\|\widetilde f-g\|_{L^p(\mathbb{R}^n)}+\|g-\varphi\|_{L^p(\mathbb{R}^n)}
<\frac{\delta}{2}+\frac{\delta}{2}=\delta.
\end{align*}
因此
\begin{align*}
\int_E |f-\varphi|^p\,\mathrm{d}x
\leqslant\|\widetilde f-\varphi\|_{L^p(\mathbb{R}^n)}^p
<\delta^p=\varepsilon.
\end{align*}
故 (ii) 成立。
\end{proof}

\begin{theorem}
$L^p(E)(1\leqslant p<+\infty)$是可分空间。
\end{theorem}
\begin{proof}
先考虑$E=\mathbb{R}^n$，$f\in L^p(\mathbb{R}^n)$。对任意$\varepsilon>0$，由\reflem{lemma:实变函数--紧支集连续函数逼近Lp}，存在阶梯函数$\varphi(x)=\sum\limits_{i=1}^k c_i\chi_{I_i}(x)$，使得$\|f-\varphi\|_p<\varepsilon/2$。不妨设$|c_i|<M$，$m(I_i)\leqslant M^p$，选取有理数$r_i$，使得$|r_i|<M$且$|c_i-r_i|<\frac{\varepsilon}{2kM}$，令$\psi(x)=\sum\limits_{i=1}^k r_i\chi_{I_i}(x)$，则
\begin{align*}
\|\varphi-\psi\|_p&=\left\|\sum\limits_{i=1}^k c_i\chi_{I_i}-\sum\limits_{i=1}^k r_i\chi_{I_i}\right\|_p
\leqslant\sum\limits_{i=1}^k |c_i-r_i|\|\chi_{I_i}\|_p\\
&\leqslant\frac{\varepsilon}{2kM}\sum\limits_{i=1}^k (m(I_i))^{\frac{1}{p}}\leqslant\frac{\varepsilon}{2kM}\cdot kM=\frac{\varepsilon}{2},
\end{align*}
故$\|f-\psi\|_p\leqslant\|f-\varphi\|_p+\|\varphi-\psi\|_p<\varepsilon$。所有形如$\psi$的阶梯函数构成可数集$\Gamma$，是$L^p(\mathbb{R}^n)$的可数稠密子集。

对一般可测集$E$，设$f\in L^p(E)$，令$f_1(x)=f(x)$（$x\in E$），$f_1(x)=0$（$x\notin E$），则$f_1\in L^p(\mathbb{R}^n)$。对$\varepsilon>0$，存在$g\in\Gamma$，使得$\left(\int_{\mathbb{R}^n}|f_1(x)-g(x)|^p\mathrm{d}x\right)^{\frac{1}{p}}<\varepsilon$，从而$\left(\int_E|f(x)-g(x)|^p\mathrm{d}x\right)^{\frac{1}{p}}<\varepsilon$。将$\Gamma$中函数限制在$E$上，得到的集合$\Gamma'$是$L^p(E)$的可数稠密子集。
\end{proof}

\begin{corollary}
若$1\leqslant p<+\infty$，$1\leqslant r\leqslant+\infty$，则$L^p(E)\cap L^r(E)$在$L^p(E)$中稠密。
\end{corollary}

\begin{lemma}[紧支集连续函数在全体空间上一致连续]\label{lemma:紧支连续一致连续-9a2b}
设 \(g\in C_c(\mathbb{R}^n)\),则 \(g\) 在 \(\mathbb{R}^n\) 上一致连续。
\end{lemma}
\begin{proof}

设 \(K\triangleq \operatorname{spt} g\)。则 \(K\) 是紧集，且 \(g\) 在 \(K\) 外恒为 \(0\)。由于 \(g\) 连续，而 \(K\) 为闭集，且$g\equiv 0$,所以在边界 \(\partial K\) 上有 \(g=0\),否则与$g$的连续性矛盾!
固定 \(\varepsilon>0\)。因为 \(g\) 在紧集 \(K\) 上连续，由 Cantor 定理，\(g\) 在 \(K\) 上一致连续。因此存在 \(\delta_1>0\)，使得对任意 \(x,y\in K\)，当 \(|x-y|<\delta\) 时，有
\begin{align}\label{eq::jiOJIOJFIO2323}
|g(x)-g(y)|<\frac{\varepsilon}{2}.
\end{align}
现在任取 \(x,y\in\mathbb{R}^n\) 满足 \(|x-y|<\delta\)。分三种情形讨论。

情形一：\(x,y\in K\)。由\eqref{eq::jiOJIOJFIO2323}式得
\begin{align*}
|g(x)-g(y)|<\frac{\varepsilon}{2}<\varepsilon.
\end{align*}

情形二：\(x,y\notin K\)。此时 \(g(x)=g(y)=0\)，故
\begin{align*}
|g(x)-g(y)|=0<\varepsilon.
\end{align*}

情形三：\(x\in K\)，\(y\notin K\)或\(x\notin K\)，\(y\in K\)。不妨设 \(x\in K\)，\(y\notin K\)。由于 \(y\notin K\) 且 \(K\) 为闭集，故 \(x\neq y\)。线段 \([x,y]\) 从 \(K\) 内一点出发到 \(K\) 外一点，必与边界 \(\partial K\) 相交。设交点为 \(z\in\partial K\),则由\(z\in\partial K\)知\(g(z)=0\)。又因为 \(|x-z|<\delta\) 且 \(x,z\in K\)，由\eqref{eq::jiOJIOJFIO2323}式得
\begin{align*}
|g(x)-g(z)|<\frac{\varepsilon}{2}.
\end{align*}
所以
\begin{align*}
|g(x)-g(y)|=|g(x)|\leqslant |g(x)-g(z)|+|g(z)|<\frac{\varepsilon}{2}+0=\frac{\varepsilon}{2}<\varepsilon.
\end{align*}
对称情形同理。

综上，对任意 \(x,y\in\mathbb{R}^n\)，只要 \(|x-y|<\delta\)，就有
\begin{align*}
|g(x)-g(y)|<\varepsilon.
\end{align*}
故 \(g\) 在 \(\mathbb{R}^n\) 上一致连续。
\end{proof}

\begin{theorem}[\(L^p(\mathbb{R}^n)\)范数的平均连续性]\label{theorem:Lp范数下平均连续性}
若$f\in L^p(\mathbb{R}^n)(1\leqslant p<+\infty)$，则
\begin{align}
\lim\limits_{t\to0}\int_{\mathbb{R}^n}|f(x+t)-f(x)|^p\mathrm{d}x=0.\label{eq::--9ru90RFUJIJR3ruu0-5}
\end{align}
\end{theorem}
\begin{proof}

固定 \(\varepsilon>0\)。由\cref{lemma:实变函数--紧支集连续函数逼近Lp}知\(C_c(\mathbb{R}^n)\) 在 \(L^p(\mathbb{R}^n)\) 中稠密，存在 \(g\in C_c(\mathbb{R}^n)\)，使得
\begin{align}\label{eq::jJFIOJIOWI23rji2jrsji}
\|f-g\|_{L^p(\mathbb{R}^n)}<\frac{\varepsilon}{3}.
\end{align}
下面先证明对 \(g\) 有
\begin{align}
\lim\limits_{t\to 0}\int_{\mathbb{R}^n}|g(x+t)-g(x)|^p\,\mathrm{d}x=0.
\label{eq:紧支连续函数平均连续-7x3k}
\end{align}
因为 \(g\in C_c(\mathbb{R}^n)\)，所以由\eqref{lemma:紧支连续一致连续-9a2b}知\(g\) 在 \(\mathbb{R}^n\) 上一致连续。于是对任意 \(\eta>0\)，存在 \(\delta>0\)，使得当 \(|t|<\delta\) 时，对一切 \(x\in\mathbb{R}^n\)，有
\begin{align*}
|g(x+t)-g(x)|<\eta.
\end{align*}
设 \(K\triangleq \operatorname{spt} g\)，则 \(K\) 紧。取 \(0<\delta\leqslant 1\)，则当 \(|t|<\delta\) 时，
\begin{align*}
K\cup(K-t)\subseteq K+\overline{B(0,1)}.
\end{align*}
因此
\begin{align*}
\int_{\mathbb{R}^n}|g(x+t)-g(x)|^p\,\mathrm{d}x
=\int_{K\cup(K-t)}|g(x+t)-g(x)|^p\,\mathrm{d}x
\leqslant \eta^p\,m\bigl(K\cup(K-t)\bigr)
\leqslant \eta^p\,m\bigl(K+\overline{B(0,1)}\bigr).
\end{align*}
由于 \(m\bigl(K+\overline{B(0,1)}\bigr)<\infty\)，令 \(\eta\to 0\)，即得 \eqref{eq:紧支连续函数平均连续-7x3k}式。

现在设\(f\in L^p(\mathbb{R}^n)\)。由 \eqref{eq:紧支连续函数平均连续-7x3k}式，存在 \(\delta_1>0\)，使得当 \(|t|<\delta_1\) 时，
\begin{align}\label{eqJFIOJIWOJIOUJWIOJFWWWWW}
\|g(\cdot+t)-g(\cdot)\|_{L^p(\mathbb{R}^n)}<\frac{\varepsilon}{3}.
\end{align}
对任意 \(t\in\mathbb{R}^n\)，有
\begin{align*}
\|f(\cdot+t)-f(\cdot)\|_{L^p(\mathbb{R}^n)}
\leqslant \|f(\cdot+t)-g(\cdot+t)\|_{L^p(\mathbb{R}^n)}
+\|g(\cdot+t)-g(\cdot)\|_{L^p(\mathbb{R}^n)}
+\|g(\cdot)-f(\cdot)\|_{L^p(\mathbb{R}^n)}.
\end{align*}
由积分变量替换，得
\begin{align}\label{eqJIOFJIOJIOWIJOIJOIOJJJJJ}
\|f(\cdot+t)-g(\cdot+t)\|_{L^p(\mathbb{R}^n)}^p
=\int_{\mathbb{R}^n}|f(x+t)-g(x+t)|^p\,\mathrm{d}x
=\int_{\mathbb{R}^n}|f(y)-g(y)|^p\,\mathrm{d}y
=\|f-g\|_{L^p(\mathbb{R}^n)}^p,
\end{align}
故由\eqref{eq::jJFIOJIOWI23rji2jrsji}式可得
\begin{align}\label{eqJIOFJIOJIOWIJOIJOIOJJJJJ-1}
\|f(\cdot+t)-g(\cdot+t)\|_{L^p(\mathbb{R}^n)}
=\|f-g\|_{L^p(\mathbb{R}^n)}
<\frac{\varepsilon}{3}.
\end{align}
于是当 \(|t|<\delta_1\) 时，结合\eqref{eqJFIOJIWOJIOUJWIOJFWWWWW}\eqref{eqJIOFJIOJIOWIJOIJOIOJJJJJ}\eqref{eqJIOFJIOJIOWIJOIJOIOJJJJJ-1}式有
\begin{align*}
\|f(\cdot+t)-f(\cdot)\|_{L^p(\mathbb{R}^n)}
<\frac{\varepsilon}{3}+\frac{\varepsilon}{3}+\frac{\varepsilon}{3}
=\varepsilon.
\end{align*}
因此
\begin{align*}
\lim\limits_{t\to 0}\|f(\cdot+t)-f(\cdot)\|_{L^p(\mathbb{R}^n)}=0.
\end{align*}
等价地，有
\begin{align*}
\lim\limits_{t\to 0}\int_{\mathbb{R}^n}|f(x+t)-f(x)|^p\,\mathrm{d}x=0.
\end{align*}
\end{proof}

\begin{theorem}[局部\(L^p\)范数的平均连续性]\label{theorem:Lp局部平均连续性-8v2q}
设 \(1\leqslant p<\infty\), \(f\in L^p_{\mathrm{loc}}(\mathbb{R}^n)\). 则对任意紧集 \(K\subseteq\mathbb{R}^n\),有
\begin{align*}
\lim\limits_{h\to 0}\int_K|f(x+h)-f(x)|^p\,\mathrm{d}x=0.
\end{align*}
\end{theorem}
\begin{remark}
\(p=\infty\) 时结论一般不成立. 取 \(f=\chi_{(0,\infty)}\in L^\infty(\mathbb{R})\). 则对任意 \(h\neq 0\),
\begin{align*}
\|f(\cdot+h)-f(\cdot)\|_{L^\infty(\mathbb{R})}=1,
\end{align*}
故 \(L^\infty\) 中平移不连续.
\end{remark}
\begin{proof}
取 \(\varphi\in C_c^\infty(\mathbb{R}^n)\) 使得 \(\varphi=1\) 在 \(K+\overline{B(0,1)}\) 上. 令 \(g\triangleq\varphi f\). 则 \(g\in L^p(\mathbb{R}^n)\). 由 \hyperref[theorem:Lp范数下平均连续性]{\(L^p(\mathbb{R}^n)\)范数的平均连续性},得
\begin{align*}
\lim\limits_{h\to 0}\int_{\mathbb{R}^n}|g(x+h)-g(x)|^p\,\mathrm{d}x=0.
\end{align*}
当 \(|h|<1\) 且 \(x\in K\) 时, \(x+h\in K+\overline{B(0,1)}\), 故 \(g(x+h)=f(x+h)\) 且 \(g(x)=f(x)\). 因此
\begin{align*}
\int_K|f(x+h)-f(x)|^p\,\mathrm{d}x
=\int_K|g(x+h)-g(x)|^p\,\mathrm{d}x
\leqslant\int_{\mathbb{R}^n}|g(x+h)-g(x)|^p\,\mathrm{d}x,
\end{align*}
令 \(h\to 0\) 即得结论.
\end{proof}

\begin{example}
若$f\in L^p(\mathbb{R}^n)(1\leqslant p<+\infty)$，则
\begin{align*}
\lim\limits_{|t|\to\infty}\int_{\mathbb{R}^n}|f(x)+f(x-t)|^p\mathrm{d}x=2\int_{\mathbb{R}^n}|f(x)|^p\mathrm{d}x.
\end{align*}
\end{example}

\begin{proof}
对任给$\varepsilon>0$，作分解$f(x)=g(x)+h(x)$，其中$g(x)$是$\mathbb{R}^n$上具有紧支集的连续函数，且$\|h\|_p<\varepsilon/4$。存在$M>0$，当$|t|\geqslant M$时，$g(x)$与$g(x-t)$的支集不相交，故
\begin{align*}
\int_{\mathbb{R}^n}|g(x)+g(x-t)|^p\mathrm{d}x=\int_{\mathbb{R}^n}|g(x)|^p\mathrm{d}x+\int_{\mathbb{R}^n}|g(x-t)|^p\mathrm{d}x=2\int_{\mathbb{R}^n}|g(x)|^p\mathrm{d}x.
\end{align*}

由分解式，$|\|f\|_p-\|g\|_p|\leqslant\|h\|_p<\varepsilon/4$。令$f_t(x)=f(x-t)$，$g_t(x)=g(x-t)$，则
\begin{align*}
|\|f+f_t\|_p-\|g+g_t\|_p|\leqslant\|h+h_t\|_p\leqslant2\|h\|_p<\frac{\varepsilon}{2},
\end{align*}
故当$|t|\geqslant M$时，
\begin{align*}
|\|f+f_t\|_p-2^{\frac{1}{p}}\|g\|_p|<\frac{\varepsilon}{2}.
\end{align*}

因此，
\begin{align*}
|\|f+f_t\|_p-2^{\frac{1}{p}}\|f\|_p|\leqslant|\|f+f_t\|_p-2^{\frac{1}{p}}\|g\|_p|+|2^{\frac{1}{p}}\|g\|_p-2^{\frac{1}{p}}\|f\|_p|
<\frac{\varepsilon}{2}+\frac{\varepsilon}{2}=\varepsilon.
\end{align*}
\end{proof}

\begin{example}
$f(x)$在$\mathbb{R}^n$中任一测度有限的可测集上均可积的充分必要条件是：存在$f_1\in L^1(\mathbb{R}^n)$，$f_2\in L^\infty(\mathbb{R}^n)$，使得
\begin{align*}
f(x)=f_1(x)+f_2(x),\quad x\in\mathbb{R}^n.
\end{align*}
\end{example}
\begin{proof}

{\heiti 必要性：}令$A_k=\{x\in\mathbb{R}^n:k^2<|f(x)|\leqslant(k+1)^2\}$（$k=1,2,\cdots$），则存在$k_0$，使得$\sum\limits_{k=k_0}^\infty k^2m(A_k)<+\infty$。否则，若对任意$j$，$\sum\limits_{k=j}^\infty k^2m(A_k)=+\infty$，则
\begin{enumerate}[(i)]
\item 若存在$\{k_i\}$，使得$k_i^2m(A_{k_i})\geqslant1$，则存在$B_i\subseteq A_{k_i}$，使得$k_i^2m(B_i)=1$，故$m\left(\bigcup\limits_{i=1}^\infty B_i\right)<+\infty$，但$|f(x)|$在$\bigcup\limits_{i=1}^\infty B_i$上不可积，矛盾；
\item 若存在$K$，当$k\geqslant K$时，$k^2m(A_k)<1$，则$m\left(\bigcup\limits_{k=K}^\infty A_k\right)<+\infty$，但$|f(x)|$在$\bigcup\limits_{k=K}^\infty A_k$上的积分为$+\infty$，矛盾。
\end{enumerate}

令$A=\bigcup\limits_{k=k_0}^\infty A_k$，$f_1(x)=f(x)\chi_A(x)$，$f_2(x)=f(x)\chi_{A^c}(x)$，则$f(x)=f_1(x)+f_2(x)$，其中$f_1\in L^1(\mathbb{R}^n)$，$f_2\in L^\infty(\mathbb{R}^n)$。

{\heiti 充分性：}若$f(x)=f_1(x)+f_2(x)$，则$f(x)$可测，且对任意$m(E)<+\infty$，
\begin{align*}
\int_E|f(x)|\mathrm{d}x\leqslant\int_E|f_1(x)|\mathrm{d}x+\int_E|f_2(x)|\mathrm{d}x<\infty.
\end{align*}
\end{proof}
