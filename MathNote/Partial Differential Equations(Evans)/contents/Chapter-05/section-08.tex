\subsection{Poincaré 不等式}\label{subsection:ch05-边界情形}

\begin{proposition}\label{proposition:ch05--examoi-11}
设 $U$ 连通，$u\in W^{1,p}(U)$ 满足
\begin{align*}
Du=0\quad\text{a.e. 于}\ U.
\end{align*}
证明 $u$ 在 $U$ 中几乎处处为常数。
\end{proposition}
\begin{proof}

由于 $u\in W^{1,p}(U)$ 且 $Du=0$ a.e. 于 $U$, 对任意 $i=1,\cdots,n$ 和任意 $\varphi\in C_c^\infty(U)$, 有
\begin{align*}
\int_U u\,\partial_i\varphi\,\mathrm{d}x
= -\int_U \partial_i u\,\varphi\,\mathrm{d}x
=0.
\end{align*}
因此对任意 $\varphi\in C_c^\infty(U)$, 有
\begin{align}\label{eqwr:Njkojioejioio2}
\int_U u\,D\varphi\,\mathrm{d}x=0.
\end{align}

先证明一个局部事实.取任意开球 $B=B(x_0,r)$ 使得 $\overline{B}\subseteq U$. 令 $\eta_\varepsilon$ 为标准磨光核, 并取 $0<\varepsilon<\mathrm{dist}(B,\partial U)$. 定义
\begin{align*}
u_\varepsilon(x)=(\eta_\varepsilon*u)(x)=\int_U \eta_\varepsilon(x-y)u(y)\,\mathrm{d}y,\quad x\in B.
\end{align*}
则 $u_\varepsilon\in C^\infty(B)$, 且由\eqref{eqwr:Njkojioejioio2}式, 对 $x\in B$ 有
\begin{align*}
D u_\varepsilon(x)
=\int_U u(y)D_x\eta_\varepsilon(x-y)\,\mathrm{d}y
=-\int_U u(yD_y\eta_\varepsilon(x-y))\,\mathrm{d}y
=0.
\end{align*}
故 $u_\varepsilon$ 在连通的 $B$ 上为常数. 由于 $u\in L^1_{\mathrm{loc}}(U)$, 当 $\varepsilon\to0^+$ 时 $u_\varepsilon\to u$ 于 $L^1(B)$,故 $\varepsilon\to0^+$ 时,$u_{\varepsilon_j}\to u$ a.e. 于 $B$. 而每个 $u_{\varepsilon}$ 在 $B$ 上为常数, 故 $u$ 在 $B$ 上 a.e. 等于某个常数.

固定 $x_0\in U$. 令 $\mathcal{S}$为所有满足存在有限个开球 $B_1,\cdots,B_m$ 以及常数 $c$,使得
\begin{enumerate}[(1)]
\item 对每个 $j=1,\cdots,m$, $\overline{B_j}\subseteq U$, 且 $u=c$ a.e. 于 $B_j$;
\item $x_0\in B_1$, $x\in B_m$;
\item $B_j\cap B_{j+1}\neq\varnothing$, $j=1,\cdots,m-1$.
\end{enumerate}
的点$x\in U$构成的集合.
由局部事实知$x_0\in \mathcal{S}$, 故 $\mathcal{S}\neq\varnothing$.

若 $x\in \mathcal{S}$, 取上述链末端的球 $B_m$. 对每个 $y\in B_m$, 由于 $B_m$ 是开球, 存在球 $B_y$ 使得 $y\in B_y$ 且 $\overline{B_y}\subseteq B_m$. 由 $u=c$ a.e. 于 $B_m$, 得 $u=c$ a.e. 于 $B_y$. 于是 $B_1,\cdots,B_m,B_y$ 是一条连接 $x_0$ 与 $y$ 的链, 故 $y\in \mathcal{S}$. 因此 $B_m\subseteq \mathcal{S}$, 即 $\mathcal{S}$ 是 $U$ 中的开集.

若 $x\in U$ 且存在 $x_k\in \mathcal{S}$ 使得 $x_k\to x$. 由局部事实, 存在球 $B$ 使得 $x\in B$, $\overline{B}\subseteq U$, 且 $u=d$ a.e. 于 $B$, 其中 $d$ 为某常数. 对充分大的 $k$, $x_k\in B$. 由于 $x_k\in \mathcal{S}$, 存在链 $B_1,\cdots,B_m$ 和常数 $c$ 连接 $x_0$ 与 $x_k$, 且 $u=c$ a.e. 于每个 $B_j$. 特别地, $x_k\in B_m$. 于是 $B_m\cap B$ 是非空开集, 故其 Lebesgue 测度为正, 从而 $c=d$. 因此 $B_1,\cdots,B_m,B$ 是一条连接 $x_0$ 与 $x$ 的链, 且所有球上 $u=c$ a.e., 故 $x\in \mathcal{S}$. 因此 $\mathcal{S}$ 在 $U$ 中闭.

由于 $U$ 连通, 且 $\mathcal{S}$ 非空、既开又闭, 故 $\mathcal{S}=U$(否则$U=\mathcal{S} \sqcup \left( U\backslash \mathcal{S} \right) $与连通定义矛盾!). 于是存在常数 $c$, 使得对每个 $x\in U$, 存在球 $B_x$ 满足 $x\in B_x$, $\overline{B_x}\subseteq U$, 且 $u=c$ a.e. 于 $B_x$. 这些球覆盖 $U$, 取可数子覆盖 $\{B_{x_j}\}_{j=1}^{\infty}$, 则
\begin{align*}
U=\bigcup\limits_{j=1}^{\infty} B_{x_j},
\end{align*}
且每个 $B_{x_j}$ 上 $u=c$ a.e.. 因此 $u=c$ a.e. 于 $U$.
\end{proof}

\begin{theorem}[Poincaré 不等式]\label{theorem:ch05-Poincare}
设 $U$ 是$\mathbb{R}^{n}$的具有$C^1$边界$\partial U$的有界连通开子集,$1\leqslant p\leqslant\infty$。则存在只依赖于 $n$、$p$ 与 $U$的常数 $C$，使对每个函数 $u\in W^{1,p}(U)$ 有
\begin{align}
\|u-(u)_{U}\|_{L^{p}(U)}\leqslant C\|Du\|_{L^{p}(U)}.
\end{align}
其中$(u)_{U}=\fint_{U}u\,\mathrm{d}y$，即 $u$ 在 $U$ 上的平均。
\end{theorem}
\begin{remark}
此式的意义在于右端只出现 $u$ 的梯度。
\end{remark}
\begin{proof}

反证。若所述估计不成立，则对每个整数 $k=1,2,\cdots$ 存在函数 $u_{k}\in W^{1,p}(U)$ 满足
\begin{align}\label{eq:iojiowjwoioioii}
\|u_{k}-(u_{k})_{U}\|_{L^{p}(U)}>k\|Du_{k}\|_{L^{p}(U)}.
\end{align}
我们归一化，定义
\begin{align*}
v_{k}\triangleq\frac{u_{k}-(u_{k})_{U}}{\|u_{k}-(u_{k})_{U}\|_{L^{p}(U)}}\qquad (k=1,2,\cdots).
\end{align*}
则
\begin{align*}
(v_{k})_{U}=0,\qquad \|v_{k}\|_{L^{p}(U)}=1;
\end{align*}
且上述不等式\eqref{eq:iojiowjwoioioii}蕴含
\begin{align}\label{eq::jiojwjjwiwIJIWD}
\|Dv_{k}\|_{L^{p}(U)}<\frac{1}{k}\qquad (k=1,2,\cdots).
\end{align}
特别地，函数 $\{v_{k}\}_{k=1}^{\infty}$ 在 $W^{1,p}(U)$ 中有界。

由\hyperref[theorem:Rellich--Kondrachov紧性定理]{Rellich--Kondrachov紧性定理}，存在子序列 $\{v_{k_{j}}\}_{j=1}^{\infty}\subset\{v_{k}\}_{k=1}^{\infty}$ 及函数 $v\in L^{p}(U)$ 使
\begin{align}
v_{k_{j}}\to v\quad\text{于}\ L^{p}(U).
\end{align}
由归一化式得
\begin{align*}
(v)_{U}=0,\qquad \|v\|_{L^{p}(U)}=1.
\end{align*}
另一方面，\eqref{eq::jiojwjjwiwIJIWD}式蕴含对每个 $i=1,\cdots,n$ 及 $\phi\in C^{\infty}_{c}(U)$，
\begin{align*}
\int_{U}v\phi_{x_{i}}\,\mathrm{d}x
=\lim_{k_{j}\to\infty}\int_{U}v_{k_{j}}\phi_{x_{i}}\,\mathrm{d}x
=-\lim_{k_{j}\to\infty}\int_{U}v_{k_{j},x_{i}}\phi\,\mathrm{d}x=0.
\end{align*}
因而 $v\in W^{1,p}(U)$，且 $Dv(x)=0$ a.e.$\,x\in U$。于是由于 $U$ 连通和\cref{proposition:ch05--examoi-11}知，$v$ 为常数。然而此结论与 $\|v\|_{L^{p}(U)}=1$、$(v)_{U}=0$ 矛盾!
\end{proof}

\begin{theorem}[球上的 Poincaré 不等式]\label{theorem:ch05-球Poincare}
设 $1\leqslant p\leqslant\infty$。则存在只依赖于 $n$ 与 $p$的常数 $C$，使对每个球 $B(x,r)\subset\mathbb{R}^{n}$ 及每个函数 $u\in W^{1,p}(B^{0}(x,r))$ 有
\begin{align}
\|u-(u)_{x,r}\|_{L^{p}(B(x,r))}\leqslant Cr\|Du\|_{L^{p}(B(x,r))},
\end{align}
其中$(u)_{x,r}=\fint_{B(x,r)}u\,\mathrm{d}y$，即 $u$ 在球 $B(x,r)$ 上的平均.
\end{theorem}
\begin{remark}
如果直接在$B(x,r)$上使用\hyperref[theorem:ch05-Poincare]{Poincaré不等式}得到的常数$C$会与$B(x,r)$相关.
\end{remark}
\begin{proof}
$U=B^{0}(0,1)$ 的情形由\hyperref[theorem:ch05-Poincare]{Poincaré不等式}即得。一般地，若 $u\in W^{1,p}(B^{0}(x,r))$，写
\begin{align*}
v(y)\triangleq u(x+ry)\qquad (y\in B(0,1)).
\end{align*}
则 $v\in W^{1,p}(B^{0}(0,1))$，且
\begin{align*}
\|v-(v)_{0,1}\|_{L^{p}(B(0,1))}\leqslant C\|Dv\|_{L^{p}(B(0,1))}.
\end{align*}
作变量代换，再由\hyperref[theorem:ch05-Poincare]{Poincaré不等式}即得所述估计。
\end{proof}

\begin{definition}\label{definition:BMO}
设 $u\in L^1_{\mathrm{loc}}(\mathbb{R}^n)$。若
\begin{align*}
[u]_{BMO(\mathbb{R}^n)} \triangleq \sup\limits_{B(x,r)\subseteq \mathbb{R}^n} \left\{ \fint_{B(x,r)} |u - (u)_{x,r}| \mathrm{d}y \right\} < \infty,
\end{align*}
则称 $u$ 属于\textbf{有界平均振荡空间}$BMO(\mathbb{R}^n)$，并称 $[u]_{BMO(\mathbb{R}^n)}$ 为 $u$ 的\textbf{$BMO$半范数}。其中 $(u)_{x,r} = \fint_{B(x,r)} u \mathrm{d}y$ 表示 $u$ 在球 $B(x,r)$ 上的平均。
\end{definition}

\begin{theorem}\label{theorem:W1n嵌入BMO}
设 $u\in W^{1,n}(\mathbb{R}^n)\cap L^1(\mathbb{R}^n)$,则 $u\in BMO(\mathbb{R}^n)$，且存在只依赖于 $n$ 的常数 $C$，使得
\begin{align}\label{eq:BMO估计}
[u]_{BMO(\mathbb{R}^n)} \leqslant C \|Du\|_{L^n(\mathbb{R}^n)}.
\end{align}
\end{theorem}
\begin{proof}
任取球 $B(x,r)\subseteq \mathbb{R}^n$。由\hyperref[theorem:ch05-球Poincare]{球上的 Poincaré 不等式（$p=1$ 的情形）}得
\begin{align*}
\fint_{B(x,r)} |u - (u)_{x,r}| \mathrm{d}y \leqslant C r \fint_{B(x,r)} |Du| \mathrm{d}y.
\end{align*}
再由\hyperref[Real Analysis-theorem:实变函数--Hölder不等式]{Hölder不等式}得
\begin{align*}
\fint_{B(x,r)} |Du| \mathrm{d}y \leqslant \left( \fint_{B(x,r)} |Du|^n \mathrm{d}y \right)^{\frac{1}{n}}.
\end{align*}
注意到 $|B(x,r)| = \alpha(n) r^n$，其中 $\alpha(n)$ 为单位球体积，于是
\begin{align*}
r \left( \fint_{B(x,r)} |Du|^n \mathrm{d}y \right)^{\frac{1}{n}}
= r \left( \frac{1}{\alpha(n) r^n} \int_{B(x,r)} |Du|^n \mathrm{d}y \right)^{\frac{1}{n}}
= \frac{1}{(\alpha(n))^{\frac{1}{n}}} \left( \int_{B(x,r)} |Du|^n \mathrm{d}y \right)^{\frac{1}{n}}.
\end{align*}
因此
\begin{align*}
\fint_{B(x,r)} |u - (u)_{x,r}| \mathrm{d}y \leqslant \frac{C}{(\alpha(n))^{\frac{1}{n}}} \left( \int_{\mathbb{R}^n} |Du|^n \mathrm{d}y \right)^{\frac{1}{n}}.
\end{align*}
对所有的球 $B(x,r)\subseteq \mathbb{R}^n$ 取上确界，即得
\begin{align*}
[u]_{BMO(\mathbb{R}^n)} \leqslant C \|Du\|_{L^n(\mathbb{R}^n)}.
\end{align*}
故 $u\in BMO(\mathbb{R}^n)$。
\end{proof}


\subsection{差商}\label{subsection:ch05-差商}

\subsubsection{差商与 $W^{1,p}$}\label{subsubsection:ch05-差商W1p}

\begin{definition}\label{definition:ch05-差商}
设$U\subset \mathbb{R}^n$是开集,$u:U\to\mathbb{R}$ 为局部可和函数，$V\subset \subset U$。
步长为 $h$ 的第 $i$ 差商为
\begin{align*}
D_{i}^{h}u(x)=\frac{u(x+he_{i})-u(x)}{h}\qquad (i=1,\cdots,n),
\end{align*}
其中 $x\in V$，$h\in\mathbb{R}$，$0<|h|<\operatorname{dist}(V,\partial U)$。
并且$D^{h}u\triangleq(D_{1}^{h}u,\cdots,D_{n}^{h}u)$。
\end{definition}

\begin{theorem}[差商的分部积分公式]\label{theorem:ch05-差商的分部积分公式}
设 $U\subseteq\mathbb{R}^n$ 为开集, $i\in\{1,\cdots,n\}$, $h\neq 0$. 设 $u\in L^1_{\mathrm{loc}}(U)$, $v\in C_c^\infty(U)$, 且
\begin{align*}
0<|h|<\frac{1}{2}\operatorname{dist}(\operatorname{spt}v,\partial U).
\end{align*}
定义差商
\begin{align*}
D_i^h v(x)\triangleq\frac{v(x+h e_i)-v(x)}{h},\qquad
D_i^{-h}u(x)\triangleq\frac{u(x)-u(x-h e_i)}{h}.
\end{align*}
则
\begin{align*}
\int_U u(x)D_i^h v(x)\,\mathrm{d}x
=-\int_U D_i^{-h}u(x)v(x)\,\mathrm{d}x.
\end{align*}
\end{theorem}
\begin{proof}
由 $v\in C_c^\infty(U)$ 及 $0<|h|<\frac{1}{2}\operatorname{dist}(\operatorname{spt}v,\partial U)$, 可知对任意 $x\in\operatorname{spt}v$, 有 $x+h e_i\in U$ 且 $x-h e_i\in U$. 因此下列积分均有意义. 直接计算得
\begin{align*}
\int_U u(x)D_i^h v(x)\,\mathrm{d}x
=\frac{1}{h}\int_U u(x)\left(v(x+h e_i)-v(x)\right)\mathrm{d}x=\frac{1}{h}\int_U u(x)v(x+h e_i)\,\mathrm{d}x
-\frac{1}{h}\int_U u(x)v(x)\,\mathrm{d}x.
\end{align*}
对第一个积分作变量替换 $y=x+h e_i$, 则 $x=y-h e_i$, $\mathrm{d}x=\mathrm{d}y$. 由于 $v$ 的支集在 $U$ 中, 当 $y\notin U$ 时 $v(y)=0$, 故积分区域仍可取为 $U$, 从而
\begin{align*}
\int_U u(x)v(x+h e_i)\,\mathrm{d}x
=\int_U u(y-h e_i)v(y)\,\mathrm{d}y.
\end{align*}
代回上式, 得
\begin{align*}
\int_U u(x)D_i^h v(x)\,\mathrm{d}x
&=\frac{1}{h}\int_U u(x-h e_i)v(x)\,\mathrm{d}x
-\frac{1}{h}\int_U u(x)v(x)\,\mathrm{d}x\\
&=-\int_U \frac{u(x)-u(x-h e_i)}{h}v(x)\,\mathrm{d}x\\
&=-\int_U D_i^{-h}u(x)v(x)\,\mathrm{d}x.
\end{align*}
\end{proof}

\begin{theorem}[差商与弱导数]\label{theorem:ch05-差商弱导数}
设$U$是$\mathbb{R}^n$中的开集,对每个 $V\subset \subset U$.
\begin{enumerate}[(i)]
\item 设 $1\leqslant p<\infty$，$u\in W^{1,p}(U)$。则
\begin{align}
\|D^{h}u\|_{L^{p}(V)}\leqslant C\|Du\|_{L^{p}(U)}
\end{align}
其中常数 $C$ 及一切 $0<|h|<\frac{1}{2}\operatorname{dist}(V,\partial U)$。
\item 设 $1<p<\infty$，$u\in L^{p}(V)$，且存在常数 $C$ 使
\begin{align}\label{eq::jijiowjiwiw23323rwe}
\|D^{h}u\|_{L^{p}(V)}\leqslant C
\end{align}
对一切 $0<|h|<\frac{1}{2}\operatorname{dist}(V,\partial U)$ 成立。则
\begin{align*}
u\in W^{1,p}(V),
\end{align*}
且 $\|Du\|_{L^{p}(V)}\leqslant C$。
\end{enumerate}
\end{theorem}
\begin{remark}
断言 (ii) 对 $p=1$ 不成立（见习题 12）。$p=\infty$ 见下文\cref{theorem:ch05-W1infty刻画}。
\end{remark}
\begin{remark}
\textbf{邻近边界的差商:}\cref{theorem:ch05-差商弱导数}的变形甚至在$V\subset \subset U$ 不成立时仍然成立。例如若 $U$ 为开半球 $B^{0}(0,1)\cap\{x_{n}>0\}$，$V=B^{0}(0,\frac{1}{2})\cap\{x_{n}>0\}$，则对 $i=1,\cdots,n-1$ 有
\begin{align*}
\int_{V}|D_{i}^{h}u|^{p}\,\mathrm{d}x\leqslant\int_{U}|u_{x_{i}}|^{p}\,\mathrm{d}x.
\end{align*}
证明与这个定理证明类似。我们将在第 6 章\cref{subsection:ch05-差商} 中用到此注。
\end{remark}
\begin{proof}

1. 设 $1\leqslant p<\infty$，暂时设 $u$ 光滑。则对每个 $x\in V$,$i=1,\cdots,n$,$0<|h|<\frac{1}{2}\operatorname{dist}(V,\partial U)$，由微分学基本定理有
\begin{align*}
u\left( x+he_i \right) -u\left( x \right) =\int_0^1{\frac{\mathrm{d}}{\mathrm{d}t}u\left( x+the_i \right) \,\mathrm{d}t}=h\int_0^1{u_{x_i}\left( x+the_i \right) \,\mathrm{d}t}\Longrightarrow D^h_i u(x)=\int_0^1{u_{x_i}\left( x+the_i \right) \,\mathrm{d}t}.
\end{align*}
从而再由\hyperref[Real Analysis-theorem:实变函数--Hölder不等式]{Hölder不等式}得
\begin{align*}
|D^h u(x)|^p
&\leqslant \left( \sum_{i=1}^n |D_i^h u(x)| \right)^p
\leqslant n^{p-1} \sum_{i=1}^n |D_i^h u(x)|^p \\
&\leqslant n^{p-1} \sum_{i=1}^n \int_0^1 |u_{x_i}(x + t h e_i)|^p \,\mathrm{d}t \\
&\leqslant n^{p-1} \sum_{i=1}^n \int_0^1 |Du(x + t h e_i)|^p \,\mathrm{d}t.
\end{align*}
两边在 $V$ 上积分并结合\hyperref[Real Analysis-theorem:实变函数-Fubini定理]{Fubini定理}，即得
\begin{align*}
\int_V |D^h u|^p \,\mathrm{d}x
\leqslant n^{p-1} \sum_{i=1}^n \int_0^1 \int_V |Du(x + t h e_i)|^p \,\mathrm{d}x \,\mathrm{d}t.
\end{align*}
故
\begin{align*}
\int_{V}|D^{h}u|^{p}\,\mathrm{d}x\leqslant C\int_{U}|Du|^{p}\,\mathrm{d}x.
\end{align*}
此估计在 $u$ 光滑时成立.取开集 $W$ 使得 $V\subset\subset W\subset\subset U$。由\hyperref[theorem:ch05-局部光滑逼近]{光滑函数的局部逼近}，存在一列光滑函数 $u_m\in C^\infty(W)$，使得
\begin{align*}
u_m\to u\quad\text{于}\ W^{1,p}(W).
\end{align*}
于是 $u_m\to u$ 于 $L^p(W)$，且 $Du_m\to Du$ 于 $L^p(W)$。对每个 $m$，由于 $u_m$ 光滑，由前面的光滑情形估计可得
\begin{align*}
\|D^h u_m\|_{L^p(V)}\leqslant C\|Du_m\|_{L^p(W)}.
\end{align*}
另一方面，由差商的定义，当$h$充分小时,有
\begin{align*}
\|D^h u_m-D^h u\|_{L^p(V)}
&\leqslant \frac{1}{|h|}\left(\|u_m-u\|_{L^p(V+h e_i)}+\|u_m-u\|_{L^p(V)}\right)\\
&\leqslant \frac{2}{|h|}\|u_m-u\|_{L^p(W)}\to 0,\quad m\to 0.
\end{align*}
故 $D^h u_m\to D^h u$ 于 $L^p(V)$。在不等式
\begin{align*}
\|D^h u_m\|_{L^p(V)}\leqslant C\|Du_m\|_{L^p(W)}
\end{align*}
中令 $m\to\infty$，由 $L^p$ 范数的连续性得
\begin{align*}
\|D^h u\|_{L^p(V)}\leqslant C\|Du\|_{L^p(W)}\leqslant C\|Du\|_{L^p(U)}.
\end{align*}
这就完成了对任意 $u\in W^{1,p}(U)$ 的证明。

2. 现设估计对一切 $0<|h|<\frac{1}{2}\operatorname{dist}(V,\partial U)$ 及某常数 $C$ 成立。取 $i=1,\cdots,n$，$\phi\in C^{\infty}_{c}(V)$，注意对充分小的 $h$，由\hyperref[theorem:ch05-差商的分部积分公式]{差商的分部积分公式}有
\begin{align}\label{eqr::JIjiwjijjJFEW8334435JIOFJA-1}
\int_{V}u(D_{i}^{h}\phi)\,\mathrm{d}x=-\int_{V}(D_{i}^{-h}u)\phi\,\mathrm{d}x.
\end{align}
估计式\eqref{eq::jijiowjiwiw23323rwe}蕴含
\begin{align*}
\sup_{h}\|D_{i}^{-h}u\|_{L^{p}(V)}<\infty;
\end{align*}
又由$1<p<\infty$和\cref{Functional Analysis-proposition:Lp对偶与自反性}知$L^p(V)$是自反空间，再根据\cref{Functional Analysis-proposition:自反空间中弱列紧性等价与有界},存在函数 $v_{i}\in L^{p}(V)$ 及子列 $h_{k}\to 0$ 使
\begin{align}\label{eqr::JIjiwjijjJFEW8334435JIOFJA-2}
D_{i}^{-h_{k}}u\rightharpoonup v_{i}\quad\text{弱收敛于}\ L^{p}(V).
\end{align}
于是由$\phi\in C^{\infty}_{c}(V)$以及\eqref{eqr::JIjiwjijjJFEW8334435JIOFJA-1}\eqref{eqr::JIjiwjijjJFEW8334435JIOFJA-2}式可得
\begin{align*}
\int_{V}u\phi_{x_{i}}\,\mathrm{d}x
=\int_{U}u\phi_{x_{i}}\,\mathrm{d}x
=\lim_{h_{k}\to 0}\int_{U}uD_{i}^{h_{k}}\phi\,\mathrm{d}x=-\lim_{h_{k}\to 0}\int_{V}D_{i}^{-h_{k}}u\,\phi\,\mathrm{d}x
=-\int_{V}v_{i}\phi\,\mathrm{d}x
=-\int_{U}v_{i}\phi\,\mathrm{d}x.
\end{align*}
故在弱意义下 $v_{i}=u_{x_{i}}$（$i=1,\cdots,n$），从而 $Du\in L^{p}(V)$。由于 $u\in L^{p}(V)$，我们推得 $u\in W^{1,p}(V)$。
\end{proof}



\subsubsection{Lipschitz 连续函数与 $W^{1,\infty}$}\label{subsubsection:ch05-W1infty}

\begin{theorem}[$W^{1,\infty}$ 的刻画（局部情形）]\label{theorem:ch05-W1infty局部刻画}
设 $U \subseteq \mathbb{R}^n$ 为开集，$f: U \to \mathbb{R}$。则
$f\ \text{在}\ U\ \text{中局部 Lipschitz 连续当且仅当}f \in W^{1,\infty}_{\mathrm{loc}}(U).$
特别地,$f$的Lipschitz 常数不超过 \(\|Df\|_{L^\infty(W)}\)
\end{theorem}
\begin{proof}

{\heiti 必要性:}
设 $f$ 在 $U$ 中局部 Lipschitz 连续 . 固定 $i\in\{1,\cdots,n\}$ . 任取开集 $V \subset\subset W \subset\subset U$ , 取 $0<|h|<\operatorname{dist}(V,\partial W)$ , 定义差商
\begin{align}\label{eq:局部差商定义}
D_{i}^{-h}f(x)\triangleq\frac{f(x)-f(x-he_i)}{h},\qquad x\in V.
\end{align}
由 $f$ 在 $W$ 上的 Lipschitz 条件 , 有
\begin{align*}
\sup_{0<|h|<\operatorname{dist}(V,\partial W)}\|D_{i}^{-h}f\|_{L^\infty(V)}
\leqslant \operatorname{Lip}(f|_W)<\infty.
\end{align*}
因此差商族 $\{D_{i}^{-h}f\}_{h\neq 0}$ 在 $L^\infty(V)$ 中一致有界 , 从而在 $L^2(V)$ 中也一致有界 . 由\cref{Functional Analysis-proposition:Lp对偶与自反性}知 $L^2(V)$ 是自反的 , 再根据\cref{Functional Analysis-proposition:自反空间中弱列紧性等价与有界} , 存在子列 $h_k\to 0$ 及函数 $g_i\in L^2(V)$ , 使得
\begin{align*}
D_{i}^{-h_k}f\rightharpoonup g_i\quad\text{在}\ L^2(V)\ \text{中弱收敛}.
\end{align*}
另一方面，由 $\|D_i^{-h}f\|_{L^\infty(\mathbb{R}^n)}\leqslant \operatorname{Lip}(f|_W)$ 对一切 $h\neq 0$ 成立，得
\begin{align*}
\left|\int_{\mathbb{R}^n}\phi D_i^{-h_k}f\,\mathrm{d}x\right|
\leqslant \|D_i^{-h_k}f\|_{L^\infty(\mathbb{R}^n)}\int_{\mathbb{R}^n}|\phi|\,\mathrm{d}x
\leqslant \operatorname{Lip}(f|_W)\int_{\mathbb{R}^n}|\phi|\,\mathrm{d}x.
\end{align*}
令 $k\to\infty$，即得
\begin{align*}
\left|\int_{\mathbb{R}^n}g_i\,\phi\,\mathrm{d}x\right|
\leqslant \operatorname{Lip}(u)\int_{\mathbb{R}^n}|\phi|\,\mathrm{d}x.
\end{align*}
由于\cref{Real Analysis-theorem:L无穷空间范数对偶刻画}以及$C_c^\infty(\mathbb{R}^n)$ 在 $L^1(\mathbb{R}^n)$ 中稠密，有
\begin{align*}
\|g_i\|_{L^\infty(\mathbb{R}^n)}
=\sup_{\substack{\phi\in C_c^\infty(\mathbb{R}^n)\\ \|\phi\|_{L^1(\mathbb{R}^n)}= 1}}
\left|\int_{\mathbb{R}^n}g_i\,\phi\,\mathrm{d}x\right|.
\end{align*}
因此综上可得
\begin{align}\label{eq:局部弱极限L无穷界}
\|g_i\|_{L^\infty(V)}\leqslant \operatorname{Lip}(f|_W).
\end{align}

任取 $\phi\in C_c^\infty(V)$ . 由\hyperref[theorem:ch05-差商的分部积分公式]{差商的分部积分公式} , 有
\begin{align}\label{eq:局部差商分部积分}
\int_{V}f(x)D_{i}^{h}\phi(x)\,\mathrm{d}x
=-\int_{V}D_{i}^{-h}f(x)\phi(x)\,\mathrm{d}x.
\end{align}
其中 $D_{i}^{h}\phi(x)=\frac{\phi(x+he_i)-\phi(x)}{h}$ . 因为 $\phi$ 光滑且具有紧支集 , 所以当 $h_k\to 0$ 时 , $D_{i}^{h_k}\phi\to \phi_{x_i}$ 在 $L^\infty(V)$ 中成立 . 对 \eqref{eq:局部差商分部积分} 式令 $h=h_k\to 0$ , 左端由\hyperref[Real Analysis-theorem:控制收敛定理]{控制收敛定理}收敛到 $\int_V f\phi_{x_i}\,\mathrm{d}x$ ; 右端由弱收敛定义收敛到 $-\int_V g_i\phi\,\mathrm{d}x$ . 于是
\begin{align*}
\int_{V}f\phi_{x_i}\,\mathrm{d}x=-\int_{V}g_i\phi\,\mathrm{d}x.
\end{align*}
上式对一切 $\phi\in C_c^\infty(V)$ 成立 , 故 $g_i$ 是 $f$ 关于 $x_i$ 的弱导数 , 即 $g_i=f_{x_i}$ 在弱意义下成立 . 由于 $g_i\in L^\infty(V)$ , 且这对任意 $V\subset\subset U$ 均成立 , 所以 $f\in W^{1,\infty}_{\mathrm{loc}}(U)$ .

{\heiti 充分性：}
设 \(f\in W^{1,\infty}_{\mathrm{loc}}(U)\)。任取闭球 \(B\subset\subset U\)，取开集 \(W\) 使得 \(B\subset\subset W\subset\subset U\)。令 \(f^\varepsilon\triangleq\eta_\varepsilon*f\) 为通常的磨光函数，其中 \(\varepsilon<\operatorname{dist}(B,\partial W)\)。
由\hyperref[theorem:appC-光滑算子的性质]{光滑算子的性质}知 \(f^\varepsilon\in C^\infty(W)\)，且对任意 \(i=1,\cdots,n\)，有
\begin{align*}
D_if^\varepsilon(x)=(\eta_\varepsilon*D_if)(x).
\end{align*}
根据\hyperref[theorem:appC-光滑算子的性质]{光滑算子的性质}，有
\begin{align}\label{eq:磨光导数一致界}
\|Df^\varepsilon\|_{L^\infty(B)}\leqslant \|Df\|_{L^\infty(W)}<\infty.
\end{align}
因为 \(f^\varepsilon\) 是 \(C^\infty\) 的，对任意 \(x,y\in B\)，由微积分基本定理有
\begin{align*}
f^\varepsilon(x)-f^\varepsilon(y)=\int_0^1 Df^\varepsilon(y+t(x-y))\,\mathrm{d}t\cdot(x-y).
\end{align*}
两边取绝对值，并利用\eqref{eq:磨光导数一致界}式，得到
\begin{align*}
|f^\varepsilon(x)-f^\varepsilon(y)|
&\leqslant \|Df^\varepsilon\|_{L^\infty(B)}|x-y|
\leqslant \|Df\|_{L^\infty(W)}|x-y|.
\end{align*}
因此对于固定的 \(x,y\in B\)，令 \(\varepsilon\to 0^+\)，由磨光函数的性质 \(f^\varepsilon\to f\) 在 \(L^1_{\mathrm{loc}}(U)\) 中成立，故由\cref{Real Analysis-corollary:非负可测函数积分为0则几乎处处为0}知,存在子列几乎处处收敛，从而
\begin{align*}
|f(x)-f(y)|\leqslant \|Df\|_{L^\infty(W)}|x-y|.
\end{align*}
这说明 \(f|_B\) 是 Lipschitz 连续的，Lipschitz 常数不超过 \(\|Df\|_{L^\infty(W)}\)。由于闭球 \(B\subset\subset U\) 是任意的，故 \(f\) 在 \(U\) 中局部 Lipschitz 连续。
\end{proof}

\begin{theorem}[$W^{1,\infty}$ 的刻画]\label{theorem:ch05-W1infty刻画}
设 $U$是$\mathbb{R}^n$中的有界开集，$\partial U$ 为 $C^{1}$ 类。则 $u:U\to\mathbb{R}$ Lipschitz 连续当且仅当 $u\in W^{1,\infty}(U)$。
\end{theorem}
\begin{remark}
若 $n<p<\infty$，则每个函数 $u\in W^{1,p}$ 都属于 $C^{0,1-\frac{n}{p}}$；但一个以小于 1 的指数H\"older连续的函数未必属于任何Sobolev空间 $W^{1,p}$。
\end{remark}
\begin{proof}

1. 先设 $U=\mathbb{R}^{n}$ 且 $u$ 有紧支集。设 $u\in W^{1,\infty}(\mathbb{R}^{n})$。则 $u_{\varepsilon}\triangleq\eta_{\varepsilon}*u$（$\eta_{\varepsilon}$ 为通常磨光子）光滑，且满足
\begin{align*}
\begin{cases}
u_{\varepsilon}\to u\ \text{一致地，当}\ \varepsilon\to 0,\\
\|Du_{\varepsilon}\|_{L^{\infty}(\mathbb{R}^{n})}\leqslant\|Du\|_{L^{\infty}(\mathbb{R}^{n})}.
\end{cases}
\end{align*}
任取两点 $x,y\in\mathbb{R}^{n}$，$x\neq y$。我们有
\begin{align*}
u_{\varepsilon}(x)-u_{\varepsilon}(y)
=\int_{0}^{1}\frac{\mathrm{d}}{\mathrm{d}t}u_{\varepsilon}(tx+(1-t)y)\,\mathrm{d}t=\int_{0}^{1}Du_{\varepsilon}(tx+(1-t)y)\,\mathrm{d}t\cdot(x-y),
\end{align*}
因而
\begin{align*}
|u_{\varepsilon}(x)-u_{\varepsilon}(y)|
\leqslant\|Du_{\varepsilon}\|_{L^{\infty}(\mathbb{R}^{n})}|x-y|
\leqslant\|Du\|_{L^{\infty}(\mathbb{R}^{n})}|x-y|.
\end{align*}
令 $\varepsilon\to 0$，得
\begin{align*}
|u(x)-u(y)|\leqslant\|Du\|_{L^{\infty}(\mathbb{R}^{n})}|x-y|.
\end{align*}
故 $u$ Lipschitz 连续。

2. 现设 $u:\mathbb{R}^{n}\to\mathbb{R}$ 是 Lipschitz 连续的，记其 Lipschitz 常数为 $\operatorname{Lip}(u)$。固定 $i\in\{1,\cdots,n\}$。对任意 $h\neq 0$，定义差商
\begin{align*}
D_{i}^{-h}u(x)\triangleq\frac{u(x)-u(x-he_{i})}{h}.
\end{align*}
由 Lipschitz 条件，有
\begin{align*}
|D_{i}^{-h}u(x)|\leqslant\operatorname{Lip}(u),\quad \forall x\in\mathbb{R}^{n},
\end{align*}
故
\begin{align*}
\|D_{i}^{-h}u\|_{L^{\infty}(\mathbb{R}^{n})}\leqslant\operatorname{Lip}(u).
\end{align*}
因此差商族 $\{D_{i}^{-h}u\}_{h\neq 0}$ 在 $L^{\infty}(\mathbb{R}^{n})$ 中一致有界，从而在 $L^{2}_{\mathrm{loc}}(\mathbb{R}^{n})$ 中也一致有界。由\cref{Functional Analysis-proposition:Lp对偶与自反性}知$L^{2}_{\mathrm{loc}}(\mathbb{R}^{n})$ 是自反的，再根据\cref{Functional Analysis-proposition:自反空间中弱列紧性等价与有界}, 存在子列$h_{k}\to 0$ 及函数 $v_{i}\in L^{2}_{\mathrm{loc}}(\mathbb{R}^{n})$，使得对每个紧集 $K\subset\mathbb{R}^{n}$，有
\begin{align*}
D_{i}^{-h_{k}}u\rightharpoonup v_{i}\quad\text{在}\ L^{2}(K)\ \text{中弱收敛}.
\end{align*}
对任意 $\phi\in C_c^\infty(\mathbb{R}^n)$，由弱收敛 $D_i^{-h_k}u\rightharpoonup v_i$ 于 $L^2_{\mathrm{loc}}(\mathbb{R}^n)$，有
\begin{align*}
\int_{\mathbb{R}^n}\phi D_i^{-h_k}u\,\mathrm{d}x
\to \int_{\mathbb{R}^n}v_i\,\phi\,\mathrm{d}x.
\end{align*}
另一方面，由 $\|D_i^{-h}u\|_{L^\infty(\mathbb{R}^n)}\leqslant \operatorname{Lip}(u)$ 对一切 $h\neq 0$ 成立，得
\begin{align*}
\left|\int_{\mathbb{R}^n}\phi D_i^{-h_k}u\,\mathrm{d}x\right|
\leqslant \|D_i^{-h_k}u\|_{L^\infty(\mathbb{R}^n)}\int_{\mathbb{R}^n}|\phi|\,\mathrm{d}x
\leqslant \operatorname{Lip}(u)\int_{\mathbb{R}^n}|\phi|\,\mathrm{d}x.
\end{align*}
令 $k\to\infty$，即得
\begin{align*}
\left|\int_{\mathbb{R}^n}v_i\,\phi\,\mathrm{d}x\right|
\leqslant \operatorname{Lip}(u)\int_{\mathbb{R}^n}|\phi|\,\mathrm{d}x.
\end{align*}
由于\cref{Real Analysis-theorem:L无穷空间范数对偶刻画}以及$C_c^\infty(\mathbb{R}^n)$ 在 $L^1(\mathbb{R}^n)$ 中稠密，有
\begin{align*}
\|v_i\|_{L^\infty(\mathbb{R}^n)}
=\sup_{\substack{\phi\in C_c^\infty(\mathbb{R}^n)\\ \|\phi\|_{L^1(\mathbb{R}^n)}= 1}}
\left|\int_{\mathbb{R}^n}v_i\,\phi\,\mathrm{d}x\right|.
\end{align*}
因此综上可得
\begin{align*}
\|v_i\|_{L^\infty(\mathbb{R}^n)}\leqslant \operatorname{Lip}(u).
\end{align*}

任取 $\phi\in C^{\infty}_{c}(\mathbb{R}^{n})$。由\hyperref[theorem:ch05-差商的分部积分公式]{差商的分部积分公式}得
\begin{align*}
\int_{\mathbb{R}^{n}}u(x)D_{i}^{h}\phi(x)\,\mathrm{d}x
=-\int_{\mathbb{R}^{n}}D_{i}^{-h}u(x)\phi(x)\,\mathrm{d}x,
\end{align*}
因为 $\phi$ 光滑且具有紧支集，所以当 $h_{k}\to 0$ 时，$D_{i}^{h_{k}}\phi\to\phi_{x_{i}}$ 在 $L^{\infty}(\mathbb{R}^{n})$ 中成立。于是由\hyperref[Real Analysis-theorem:控制收敛定理]{控制收敛定理}和弱收敛定义得
\begin{align*}
\int_{\mathbb{R}^{n}}u\phi_{x_{i}}\,\mathrm{d}x
=\lim_{k\to\infty}\int_{\mathbb{R}^{n}}uD_{i}^{h_{k}}\phi\,\mathrm{d}x=-\lim_{k\to\infty}\int_{\mathbb{R}^{n}}D_{i}^{-h_{k}}u\,\phi\,\mathrm{d}x
=-\int_{\mathbb{R}^{n}}v_{i}\phi\,\mathrm{d}x.
\end{align*}
上式对一切 $\phi\in C^{\infty}_{c}(\mathbb{R}^{n})$ 成立，故 $v_{i}$ 是 $u$ 关于 $x_{i}$ 的弱导数，即 $v_{i}=u_{x_{i}}$ 在弱意义下成立。由于 $v_{i}\in L^{\infty}(\mathbb{R}^{n})$，所以 $u\in W^{1,\infty}(\mathbb{R}^{n})$。

3. 在 $U$ 有界、$\partial U$ 为 $C^{1}$ 类的一般情形，由\hyperref[theorem:ch05-扩张定理]{扩张定理}，取 $p=\infty$，存在有界线性扩张算子
\begin{align*}
E:W^{1,\infty}(U)\to W^{1,\infty}(\mathbb{R}^{n}),
\end{align*}
使得对每个 $u\in W^{1,\infty}(U)$，有 $Eu=u$ a.e. 于 $U$，且 $Eu$ 在 $\mathbb{R}^{n}$ 中具有紧支集，并且
\begin{align*}
\|Eu\|_{W^{1,\infty}(\mathbb{R}^{n})}\leqslant C\|u\|_{W^{1,\infty}(U)}.
\end{align*}

若 $u\in W^{1,\infty}(U)$，则 $Eu\in W^{1,\infty}(\mathbb{R}^{n})$ 且具有紧支集。由第 1 步的结论，$Eu$ 是 Lipschitz 连续的。因此 $u=Eu|_{U}$ 也是 Lipschitz 连续的。

反之，若 $u:U\to\mathbb{R}$ 是 Lipschitz 连续的，由 \hyperref[theorem:实值-Kirszbraun扩张定理]{Kirszbraun扩张定理}，$u$ 可以扩张为 $\mathbb{R}^{n}$ 上的 Lipschitz 函数 $\overline{u}$，且 $\operatorname{Lip}(\overline{u})=\operatorname{Lip}(u)$。由第 2 步知 $\overline{u}\in W^{1,\infty}(\mathbb{R}^{n})$。限制在 $U$ 上得 $u=\overline{u}|_{U}\in W^{1,\infty}(U)$。

综上，$u$ Lipschitz 连续当且仅当 $u\in W^{1,\infty}(U)$。
\end{proof}




\subsection{几乎处处可微性}\label{subsection:ch05-a.e.可微}

\begin{definition}\label{definition:ch05-可微}
函数 $u:U\to\mathbb{R}$ 在 $x\in U$ 处可微，若存在 $a\in\mathbb{R}^{n}$ 使当 $y\to x$ 时
\begin{align}
u(y)=u(x)+a\cdot(y-x)+o(|y-x|).
\end{align}
换言之，有
\begin{align*}
\lim_{y\to x}\frac{|u(y)-u(x)-a\cdot(y-x)|}{|y-x|}=0.
\end{align*}
容易验证 $a$ 若存在则唯一。此后我们写 $Du(x)$ 表示 $a$，并称 $Du$ 为 $u$ 的\textbf{梯度}。
\end{definition}

\begin{theorem}[几乎处处可微]\label{theorem:ch05-a.e.可微}
设 $u\in W^{1,p}_{\mathrm{loc}}(U)$，其中 $n<p\leqslant\infty$。则 $u$ 在 $U$ 中几乎处处可微，且其梯度几乎处处等于其弱梯度。
\end{theorem}
\begin{remark}
回忆\cref{theorem:ch05-W1p有界估计},我们总把 $u$ 与其连续等价形等同.
\end{remark}
\begin{proof}

1. 先设 $n<p<\infty$。由\cref{corollary:ch05-Morrey局部估计}可得,对任何 $C^{1}$ 函数 $v$ 和$r>0$成立
\begin{align}\label{eq::jijiojiojiijji323399090ps}
|v(y)-v(x)|\leqslant Cr^{1-\frac{n}{p}}\left(\int_{B(x,2r)}|Dv(z)|^{p}\,\mathrm{d}z\right)^{\frac{1}{p}}\quad (y\in B(x,r)),
\end{align}
因而由逼近对任何 $v\in W^{1,p}$ 也成立。

2. 取 $u\in W^{1,p}_{\mathrm{loc}}(U)$。现对几乎处处的$x\in U$，由\hyperref[theorem:appE-Lebesgue微分1]{Lebesgue微分定理}可得，当$r\to 0$时,有
\begin{align}\label{eq::jijiojiojiijji323399090ps-1}
\fint_{B(x,r)}|Du(x)-Du(z)|^{p}\,\mathrm{d}z\to 0,
\end{align}
其中 $Du$ 如常表示 $u$ 的弱导数。固定任一这样的点 $x$，在估计中令
\begin{align*}
v(y)\triangleq u(y)-u(x)-Du(x)\cdot(y-x),
\end{align*}
其中$r=|x-y|.$当$r$充分小时,由\eqref{eq::jijiojiojiijji323399090ps}\eqref{eq::jijiojiojiijji323399090ps-1}式有
\begin{align*}
|u(y)-u(x)-Du(x)\cdot(y-x)|
&\leqslant Cr^{1-\frac{n}{p}}\left(\int_{B(x,2r)}|Du(x)-Du(z)|^{p}\,\mathrm{d}z\right)^{\frac{1}{p}}\\
&\leqslant Cr\left(\fint_{B(x,2r)}|Du(x)-Du(z)|^{p}\,\mathrm{d}z\right)^{\frac{1}{p}}\\
&=o(r)=o(|x-y|).
\end{align*}
故 $u$ 在 $x$ 处可微，且其梯度在 $x$ 处等于其弱梯度。

3. 若 $p=\infty$，即 $u\in W^{1,\infty}_{\mathrm{loc}}(U)$。取任意 $q$ 满足 $n<q<\infty$。我们首先证明 $u\in W^{1,q}_{\mathrm{loc}}(U)$。事实上，对任意紧集 $K\subset U$，由于 $u\in L^\infty(K)$ 且弱导数 $Du\in L^\infty(K)$，而 $K$ 的 Lebesgue 测度 $|K|<\infty$，故
\begin{align*}
\|u\|_{L^q(K)} \leqslant \|u\|_{L^\infty(K)} |K|^{\frac{1}{q}} < \infty,\qquad
\|Du\|_{L^q(K)} \leqslant \|Du\|_{L^\infty(K)} |K|^{\frac{1}{q}} < \infty.
\end{align*}
因此 $u\in W^{1,q}(K)$。由 $K$ 的任意性，$u\in W^{1,q}_{\mathrm{loc}}(U)$。于是对 $q$ 应用第 1、2 步的结论（注意 $n<q<\infty$），可知 $u$ 在 $U$ 中几乎处处可微，且其梯度几乎处处等于其弱梯度。由\hyperref[lemma:ch05-弱导数唯一性]{弱导数唯一性}知,弱梯度在 $W^{1,\infty}$ 与 $W^{1,q}$ 意义下一致，故 $p=\infty$ 时定理成立。
\end{proof}

\begin{theorem}[Rademacher 定理]\label{theorem:ch05-Rademacher}设 $u$ 在 $U$ 中局部 Lipschitz 连续。则 $u$ 在 $U$ 中几乎处处可微。
\end{theorem}
\begin{proof}
由\cref{theorem:ch05-W1infty局部刻画}和\cref{theorem:ch05-a.e.可微}的$p=\infty$情形立得.
\end{proof}

\subsection{Hardy 不等式}\label{subsection:ch05-Hardy}

\begin{theorem}[Hardy不等式]\label{theorem:ch05-Hardy}
设 $n\geqslant 3$ 且 $r>0$。设 $u\in H^1(B(0,r))$。则 $\frac{u}{|x|}\in L^2(B(0,r))$，且满足
\begin{align}\label{eq:Hardy-ineq-17}
\int_{B(0,r)} \frac{u^2}{|x|^2} \mathrm{d}x \leqslant 4 \int_{B(0,r)} \left( |Du|^2 + \frac{C u^2}{r^2} \right) \mathrm{d}x,
\end{align}
其中
\begin{align*}
C = \frac{n}{2(n-2 + \sqrt{(n-3)(n-1)})} \in \left(\frac{1}{4}, \frac{3}{2}\right].
\end{align*}
\end{theorem}
\begin{remark}
这不是\hyperref[theorem:Gagliardo--Nirenberg--Sobolev不等式]{Gagliardo--Nirenberg--Sobolev不等式}的推论.
\end{remark}
\begin{proof}
先设 $u\in C^\infty(B(0,r))$。注意到
\begin{align*}
D\left(\frac{1}{|x|}\right) = -\frac{x}{|x|^3}.
\end{align*}
由\hyperref[Basis of Analytics-theorem:Green恒等式]{Green恒等式\ref{Basis of Analytics-theorem:Green恒等式-3}}，对 $\bm{F} = \frac{u^2 x}{|x|}$ 和$g = \frac{1}{|x|}$，有
\begin{align*}
\int_{B(0,r)} \bm{F}\cdot \nabla g \,\mathrm{d}x
= \int_{\partial B(0,r)} g \bm{F}\cdot \nu \,\mathrm{d}S - \int_{B(0,r)} g \nabla\cdot \bm{F} \,\mathrm{d}x,
\end{align*}
其中$v=\frac{x}{|x|}$ 是 $\partial B(0,r)$ 的单位外法向量.
因为 $\nabla g = -\frac{x}{|x|^3}$，所以
\begin{align*}
\int_{B(0,r)} \bm{F}\cdot \nabla g \,\mathrm{d}x
= -\int_{B(0,r)} \frac{u^2 x}{|x|}\cdot \frac{x}{|x|^3} \,\mathrm{d}x
= -\int_{B(0,r)} \frac{u^2}{|x|^2} \,\mathrm{d}x.
\end{align*}
另一方面，直接计算得
\begin{align*}
\nabla\cdot \bm{F}
= \nabla\cdot \left( \frac{u^2 x}{|x|} \right)
= \frac{2u x\cdot Du}{|x|} + \frac{(n-1)u^2}{|x|}.
\end{align*}
在边界 $|x|=r$ 上，$\bm{F}\cdot \nu = u^2$，且 $g = \frac{1}{r}$，故
\begin{align*}
\int_{\partial B(0,r)} g \bm{F}\cdot \nu \,\mathrm{d}S
= \int_{\partial B(0,r)} \frac{u^2}{r} \,\mathrm{d}S.
\end{align*}
因此
\begin{align*}
-\int_{B(0,r)} \frac{u^2}{|x|^2} \,\mathrm{d}x
= \int_{\partial B(0,r)} \frac{u^2}{r} \,\mathrm{d}S
- \int_{B(0,r)} \frac{1}{|x|} \left( \frac{2u x\cdot Du}{|x|} + \frac{(n-1)u^2}{|x|} \right) \mathrm{d}x.
\end{align*}
整理得
\begin{align*}
\int_{B(0,r)} \frac{(n-2)u^2}{|x|^2} \,\mathrm{d}x
= -\int_{B(0,r)} \frac{2u x\cdot Du}{|x|^2} \,\mathrm{d}x
+ \int_{\partial B(0,r)} \frac{u^2}{r} \,\mathrm{d}S.
\end{align*}
应用\hyperref[Basis of Analytics-theorem:Gauss-Green公式(散度定理)]{Gauss-Green定理}于上式右边第二个积分，得
\begin{align}\label{eq:trace-18}
\int_{\partial B(0,r)} \frac{u^2}{r} \,\mathrm{d}S
= \int_{\partial B(0,r)} \frac{u^2 x}{r^2}\cdot \nu \,\mathrm{d}S
= \int_{B(0,r)} \nabla\cdot \left( \frac{u^2 x}{r^2} \right) \mathrm{d}x
= \int_{B(0,r)} \frac{n u^2 + 2u(x\cdot Du)}{r^2} \,\mathrm{d}x.
\end{align}
代入前式得
\begin{align*}
\int_{B(0,r)} \frac{(n-2)u^2}{|x|^2} \,\mathrm{d}x
= \int_{B(0,r)} \frac{n u^2}{r^2} \,\mathrm{d}x
+ \int_{B(0,r)} \frac{2u}{|x|^2} \left( \frac{|x|^2}{r^2} - 1 \right) x\cdot Du \,\mathrm{d}x.
\end{align*}
对上式右边第二个积分项应用\hyperref[theorem:带参的Cauchy不等式]{带参数 $\varepsilon$ 的 Cauchy 不等式}得
\begin{align*}
&\int_{B(0,r)}{\frac{2u}{|x|^2}\left( \frac{|x|^2}{r^2}-1 \right) x\cdot Du\,\mathrm{d}x}\leqslant \int_{B(0,r)}{\frac{2\left| u \right|}{|x|^2}\cdot \left| \frac{|x|^2}{r^2}-1 \right|\cdot \left| x \right|\cdot \left| Du \right|\,\mathrm{d}x}
\\
&\leqslant \int_{B(0,r)}{\frac{\left| u \right|}{|x|}\cdot 2\left| Du \right|\,\mathrm{d}x}\leqslant \varepsilon \int_{B(0,r)}{\frac{u^2}{|x|^2}\,\mathrm{d}x}+\frac{1}{\varepsilon}\int_{B(0,r)}{|Du|^2\,\mathrm{d}x}.
\end{align*}
于是
\begin{align*}
\int_{B(0,r)} \frac{(n-2)u^2}{|x|^2} \,\mathrm{d}x
\leqslant \int_{B(0,r)} \frac{n u^2}{r^2} \,\mathrm{d}x
+ \varepsilon \int_{B(0,r)} \frac{u^2}{|x|^2} \,\mathrm{d}x
+ \frac{1}{\varepsilon} \int_{B(0,r)} |Du|^2 \,\mathrm{d}x.
\end{align*}
移项得
\begin{align*}
\int_{B(0,r)} \frac{(n-2-\varepsilon)u^2}{|x|^2} \,\mathrm{d}x
\leqslant \frac{1}{\varepsilon} \int_{B(0,r)} |Du|^2 \,\mathrm{d}x
+ \int_{B(0,r)} \frac{n u^2}{r^2} \,\mathrm{d}x.
\end{align*}
因 $n\geqslant 3$，令 $\varepsilon(n-2-\varepsilon) = \lambda > 0$，只要
\begin{align*}
\varepsilon = \frac{n-2 - \sqrt{(n-2)^2 - 4\lambda}}{2} > 0
\end{align*}
即可。
\begin{align*}
\int_{B(0,r)} \frac{u^2}{|x|^2} \,\mathrm{d}x
\leqslant \frac{1}{\lambda} \left( \int_{B(0,r)} |Du|^2 \,\mathrm{d}x
+ n\varepsilon \int_{B(0,r)} \frac{u^2}{r^2} \,\mathrm{d}x \right)
.
\end{align*}
固定$n$,$\lambda=\varepsilon(n-2-\varepsilon)$是关于$\varepsilon$的二元函数,最大值为$\lambda_{\text{max}}=\frac{(n-2)^2}{4}$.因为不等式要对所有$n\geqslant 3$成立,所以$\lambda$最大只可取 $\lambda = \frac{1}{4}$，此时令
$4n\varepsilon = 4C$，则
\begin{align*}
C = \frac{n}{2(n-2 + \sqrt{(n-3)(n-1)})}.
\end{align*}
再代入上式得
\begin{align*}
\int_{B(0,r)} \frac{u^2}{|x|^2} \,\mathrm{d}x
\leqslant 4 \int_{B(0,r)} |Du|^2 \,\mathrm{d}x
+ 4n\varepsilon \int_{B(0,r)} \frac{u^2}{r^2} \,\mathrm{d}x= 4 \int_{B(0,r)} \left( |Du|^2 + \frac{C u^2}{r^2} \right) \mathrm{d}x.
\end{align*}
最后，对一般 $u\in H^1(B(0,r))$，由\hyperref[theorem:ch05-边界光滑逼近]{边界光滑函数的整体逼近定理},可取 $u_m\in C^\infty(\overline{B(0,r)})$ 使得 $u_m\to u$ 于 $H^1$。对每个 $u_m$ 上述不等式成立。由\hyperref[Real Analysis-theorem:Fatou引理]{Fatou引理}得
\begin{align*}
\int_{B(0,r)} \frac{u^2}{|x|^2} \,\mathrm{d}x
\leqslant \varliminf_{m\to\infty} \int_{B(0,r)} \frac{u_m^2}{|x|^2} \,\mathrm{d}x \leqslant \lim_{m\to\infty} 4 \int_{B(0,r)} \left( |Du_m|^2 + \frac{C u_m^2}{r^2} \right) \mathrm{d}x
= 4 \int_{B(0,r)} \left( |Du|^2 + \frac{C u^2}{r^2} \right) \mathrm{d}x.
\end{align*}
\end{proof}
\begin{remark}
应用\hyperref[theorem:带参的Cauchy不等式]{带参数 $\varepsilon$ 的 Cauchy 不等式}于 \eqref{eq:trace-18} 式右边得迹不等式
\begin{align*}
\frac{1}{r} \int_{\partial B(0,r)} u^2 \mathrm{d}S
\leqslant C \int_{B(0,r)} \left( |Du|^2 + \frac{u^2}{r^2} \right) \mathrm{d}x,
\end{align*}
其中
\begin{align*}
C = \frac{n + \sqrt{n^2+4}}{2}.
\end{align*}
它不限于 $n\geqslant 3$。习题 7 是另一种得到迹不等式的方法。习题 16 是 \eqref{eq:Hardy-ineq-17} 中 $r=\infty$ 时的变形的 Hardy 不等式。
\end{remark}


\subsection{Fourier变换方法}\label{subsection:ch05-Fourier}

下面我们利用Fourier变换（\cref{section:ch04-变换方法}）给出空间 $H^{k}(\mathbb{R}^{n})$ 的另一种刻画。本小节所有函数均为复值。

\begin{theorem}[用 Fourier 变换刻画 $H^k$]\label{theorem:Fourier刻画Hk}
设 $k$ 为非负整数。
函数 $u\in L^2(\mathbb{R}^n)$ 属于 $H^k(\mathbb{R}^n)$ 当且仅当
\begin{align}\label{eq:Fourier-Hk-19}
(1+|y|^k)\hat{u}\in L^2(\mathbb{R}^n).
\end{align}
此外, 对 $\forall u\in H^k(\mathbb{R}^n)$,
\begin{align}\label{eq:Fourier-Hk-20}
\frac{1}{\sqrt{\max\{k,1\}}}\|u\|_{H^k(\mathbb{R}^n)}\leqslant \|(1+|y|^k)\hat{u}\|_{L^2(\mathbb{R}^n)}\leqslant \sqrt{2}\|u\|_{H^k(\mathbb{R}^n)}.
\end{align}
\end{theorem}
\begin{proof}

{\heiti 1. } 当 $k=0$ 时定理自然成立. 现设 $k\geqslant 1$. 首先, 假设 $u\in H^k(\mathbb{R}^n)$. 则由 $H^k(\mathbb{R}^n)$ 的定义, 对每一多重指标 $|\alpha|\leqslant k$, 我们有 $D^\alpha u\in L^2(\mathbb{R}^n)$. 若 $u\in C_c^k(\mathbb{R}^n)$ 有紧支集, 根据\hyperref[proposition:Fourier 变换的基本性质2]{Fourier变换性质\ref{proposition:Fourier 变换的基本性质2-2}}得
\begin{align}\label{eq:Fourier-Hk-21}
(\text{i}y)^\alpha\hat{u}=\widehat{D^\alpha u}.
\end{align}
当 $u\in H^k(\mathbb{R}^n)$ 时, 由\hyperref[theorem:ch05-边界光滑逼近]{边界光滑函数的整体逼近定理},取一列光滑函数 $u_m\in C_c^\infty(\mathbb{R}^n)$ 使得 $u_m\to u$ 于 $H^k(\mathbb{R}^n)$, 并用光滑函数逼近可推出公式\eqref{eq:Fourier-Hk-21} 对一般的 $u\in H^k(\mathbb{R}^n)$ 也成立. 故对每个 $|\alpha|\leqslant k$, $(\text{i}y)^\alpha\hat{u}\in L^2(\mathbb{R}^n)$. 由\eqref{eq:Fourier-Hk-21}式及\hyperref[theorem:Plancherel定理]{Plancherel定理}得
\begin{align*}
\int_{\mathbb{R} ^n}{|y|^{2k}|\hat{u}|^2\,\mathrm{d}y}&\overset{?}{=}\int_{\mathbb{R} ^n}{\sum_{|\alpha |=k}{\left| y^{\alpha} \right|^2\left| \hat{u} \right|^2\,\mathrm{d}x}}=\int_{\mathbb{R} ^n}{\sum_{|\alpha |=k}{\left| \left( \mathrm{i}y \right) ^{\alpha}\hat{u} \right|^2\,\mathrm{d}x}}
\\
&=\int_{\mathbb{R} ^n}{\sum_{|\alpha |=k}{|D^{\alpha}u|^2\,\mathrm{d}x}}=\int_{\mathbb{R} ^n}{|D^ku|^2\,\mathrm{d}x}<\infty .
\end{align*}
所以, 因 $(1+|y|^k)^2\leqslant 2(1+|y|^{2k})$, 我们有
\begin{align*}
\int_{\mathbb{R}^n}(1+|y|^k)^2|\hat{u}|^2\,\mathrm{d}y
\leqslant 2\int_{\mathbb{R}^n}(1+|y|^{2k})|\hat{u}|^2\,\mathrm{d}y
\leqslant 2\|u\|_{H^k(\mathbb{R}^n)}^2<\infty.
\end{align*}
从而 $(1+|y|^k)\hat{u}\in L^2(\mathbb{R}^n)$. 同时由上式还可得
\begin{align*}
\|(1+|y|^k)\hat{u}\|_{L^2(\mathbb{R}^n)}\leqslant \sqrt{2}\|u\|_{H^k(\mathbb{R}^n)}.
\end{align*}
这就是 \eqref{eq:Fourier-Hk-20} 式的右半边不等式.

{\heiti 2. } 反之, 若 $(1+|y|^k)\hat{u}\in L^2(\mathbb{R}^n)$, 且 $|\alpha|\leqslant k$, 则当 $k\geqslant 1$ 时,有
\begin{align}\label{eq:Fourier-Hk-22}
\sum_{s=0}^k\int_{\mathbb{R}^n}|y|^{2s}|\hat{u}|^2\,\mathrm{d}y
\leqslant k\int_{\mathbb{R}^n}(1+|y|^k)^2|\hat{u}|^2\,\mathrm{d}y.
\end{align}
令
\begin{align*}
u_\alpha\triangleq ((\text{i}y)^\alpha\hat{u})^\vee.
\end{align*}
则对每一 $\phi\in C_c^\infty(\mathbb{R}^n)$, 由\hyperref[theorem:Plancherel定理]{Plancherel定理}及\hyperref[proposition:Fourier 变换的基本性质2]{Fourier变换性质\ref{proposition:Fourier 变换的基本性质2-2}}得
\begin{align*}
\int_{\mathbb{R}^n}(D^\alpha\phi)\bar{u}\,\mathrm{d}x
=\int_{\mathbb{R}^n}\widehat{(D^\alpha\phi)}\bar{\hat{u}}\,\mathrm{d}y
=\int_{\mathbb{R}^n}(\text{i}y)^\alpha\hat{\phi}\bar{\hat{u}}\,\mathrm{d}y
=(-1)^{|\alpha|}\int_{\mathbb{R}^n}\hat{\phi}\overline{(\text{i}y)^\alpha\hat{u}}\,\mathrm{d}y
=(-1)^{|\alpha|}\int_{\mathbb{R}^n}\phi\bar{u}_\alpha\,\mathrm{d}x.
\end{align*}
因而在弱意义下 $u_\alpha=D^\alpha u$, 所以
\begin{align*}
\sum_{s=0}^k\int_{\mathbb{R}^n}|D^su|^2\,\mathrm{d}x
=\sum_{s=0}^k\int_{\mathbb{R}^n}|y|^{2s}|\hat{u}|^2\,\mathrm{d}y.
\end{align*}
由 \eqref{eq:Fourier-Hk-22} 得 $u\in H^k(\mathbb{R}^n)$, 且由上式与 \eqref{eq:Fourier-Hk-22} 有
\begin{align*}
\|u\|_{H^k(\mathbb{R}^n)}^2
=\sum_{|\alpha|\leqslant k}\int_{\mathbb{R}^n}|D^\alpha u|^2\,\mathrm{d}x
\leqslant \max\{k,1\}\int_{\mathbb{R}^n}(1+|y|^k)^2|\hat{u}|^2\,\mathrm{d}y.
\end{align*}
故有 \eqref{eq:Fourier-Hk-20} 式的左半边估计式. 
\end{proof}

\begin{lemma}\label{lemma:多项式等价}
设 $k$ 为非负整数。则存在只依赖于 $n$ 与 $k$ 的常数 $c_1,c_2>0$，使得对一切 $y\in\mathbb{R}^n$，
\begin{align}\label{eq:多项式等价}
c_1(1+|y|^k)^2
\leqslant \sum_{|\alpha|\leqslant k}|y^\alpha|^2
\leqslant c_2(1+|y|^k)^2.
\end{align}
\end{lemma}
\begin{proof}
记
\begin{align*}
P(y)\triangleq\sum_{|\alpha|\leqslant k}|y^\alpha|^2,\qquad Q(y)\triangleq(1+|y|^k)^2.
\end{align*}
则 $P,Q$ 都是连续函数，且 $P(y)\geqslant 1$，$Q(y)\geqslant 1$。

{\heiti 上界. } 对每个 $|\alpha|\leqslant k$，有 $|y^\alpha|\leqslant (1+|y|)^{|\alpha|}\leqslant (1+|y|)^k$。因此
\begin{align*}
P(y)\leqslant \sum_{|\alpha|\leqslant k}(1+|y|)^{2k}
\leqslant N_{n,k}(1+|y|)^{2k},
\end{align*}
其中 $N_{n,k}$ 表示满足 $|\alpha|\leqslant k$ 的多重指标个数，只依赖于 $n,k$。又因为
\begin{align*}
(1+|y|)^{2k}\leqslant 2^{2k}(1+|y|^{2k})\leqslant 2^{2k}(1+|y|^k)^2,
\end{align*}
故取 $c_2\triangleq 2^{2k}N_{n,k}$，即得
\begin{align*}
P(y)\leqslant c_2(1+|y|^k)^2.
\end{align*}

{\heiti 下界. } 先证存在常数 $A>0$ 使得
\begin{align}\label{eq:齐次控制}
|y|^{2k}\leqslant A\sum_{|\alpha|=k}|y^\alpha|^2.
\end{align}
事实上，由多项式展开，
\begin{align*}
|y|^{2k}=\left(\sum_{i=1}^n y_i^2\right)^k
=\sum_{|\alpha|=k}\frac{k!}{\alpha!}y^{2\alpha}
=\sum_{|\alpha|=k}\frac{k!}{\alpha!}|y^\alpha|^2
\leqslant k!\sum_{|\alpha|=k}|y^\alpha|^2.
\end{align*}
故可取 $A=k!$。于是
\begin{align*}
|y|^k\leqslant \sqrt{k!}\left(\sum_{|\alpha|=k}|y^\alpha|^2\right)^{1/2}.
\end{align*}
令 $X\triangleq\sum_{|\alpha|=k}|y^\alpha|^2\geqslant 0$，则
\begin{align*}
Q(y)&=(1+|y|^k)^2
=1+2|y|^k+|y|^{2k}
\leqslant 1+2\sqrt{k!}\sqrt{X}+k!X.
\end{align*}
因为 $2\sqrt{X}\leqslant 1+X$，所以
\begin{align*}
1+2\sqrt{k!}\sqrt{X}+k!X
&\leqslant 1+\sqrt{k!}(1+X)+k!X\\
&=(1+\sqrt{k!})+( \sqrt{k!}+k!)X\\
&\leqslant (1+\sqrt{k!}+k!)(1+X).
\end{align*}
又因为
\begin{align*}
P(y)=\sum_{|\alpha|\leqslant k}|y^\alpha|^2
\geqslant 1+\sum_{|\alpha|=k}|y^\alpha|^2=1+X,
\end{align*}
故
\begin{align*}
Q(y)\leqslant (1+\sqrt{k!}+k!)P(y).
\end{align*}
取 $c_1\triangleq \frac{1}{1+\sqrt{k!}+k!}$，即得
\begin{align*}
c_1(1+|y|^k)^2\leqslant \sum_{|\alpha|\leqslant k}|y^\alpha|^2.
\end{align*}
\end{proof}

\begin{theorem}[用Fourier变换刻画 $H^k$]\label{theorem:Fourier刻画Hk-严谨}
设 $k$ 为非负整数。则 $u\in H^k(\mathbb{R}^n)$ 当且仅当
\begin{align*}
(1+|y|^k)\hat{u}\in L^2(\mathbb{R}^n).
\end{align*}
此外，存在只依赖于 $n$ 与 $k$ 的常数 $C_1,C_2>0$，使得对一切 $u\in H^k(\mathbb{R}^n)$，
\begin{align}\label{eq:Fourier-Hk-等价范数11}
C_1\|u\|_{H^k(\mathbb{R}^n)}
\leqslant \|(1+|y|^k)\hat{u}\|_{L^2(\mathbb{R}^n)}
\leqslant C_2\|u\|_{H^k(\mathbb{R}^n)}.
\end{align}
\end{theorem}
\begin{proof}

{\heiti 1. } 当 $k=0$ 时定理自然成立. 现设 $k\geqslant 1$. 首先, 假设 $u\in H^k(\mathbb{R}^n)$. 则由 $H^k(\mathbb{R}^n)$ 的定义, 对每一多重指标 $|\alpha|\leqslant k$, 我们有 $D^\alpha u\in L^2(\mathbb{R}^n)$. 若 $u\in C_c^k(\mathbb{R}^n)$ 有紧支集, 根据\hyperref[theorem:1222-Fourier变换的性质]{Fourier变换性质\ref{theorem:1222-Fourier变换的性质-1}}得
\begin{align}\label{eq:Fourier-Hk-211}
(\text{i}y)^\alpha\hat{u}=\widehat{D^\alpha u}.
\end{align}
当 $u\in H^k(\mathbb{R}^n)$ 时, 由\hyperref[theorem:ch05-边界光滑逼近]{边界光滑函数的整体逼近定理}, 取一列光滑函数 $u_m\in C_c^\infty(\mathbb{R}^n)$ 使得 $u_m\to u$ 于 $H^k(\mathbb{R}^n)$, 并用光滑函数逼近可推出公式 \eqref{eq:Fourier-Hk-211} 对一般的 $u\in H^k(\mathbb{R}^n)$ 也成立. 故对每个 $|\alpha|\leqslant k$, $(\text{i}y)^\alpha\hat{u}\in L^2(\mathbb{R}^n)$. 由 \eqref{eq:Fourier-Hk-211} 式及\hyperref[theorem:Plancherel定理]{Plancherel定理}得
\begin{align*}
\int_{\mathbb{R}^n}\sum_{|\alpha|=k}|y^\alpha|^2|\hat{u}|^2\,\mathrm{d}y
&=\int_{\mathbb{R}^n}\sum_{|\alpha|=k}|(\text{i}y)^\alpha\hat{u}|^2\,\mathrm{d}y
=\int_{\mathbb{R}^n}\sum_{|\alpha|=k}|\widehat{D^\alpha u}|^2\,\mathrm{d}y \\
&=\int_{\mathbb{R}^n}\sum_{|\alpha|=k}|D^\alpha u|^2\,\mathrm{d}x
=\int_{\mathbb{R}^n}|D^k u|^2\,\mathrm{d}x<\infty.
\end{align*}
因此, 由\cref{lemma:多项式等价}, 存在只依赖于 $n$ 与 $k$ 的常数 $c_1>0$, 使得
\begin{align*}
(1+|y|^k)^2\leqslant c_1\sum_{|\alpha|\leqslant k}|y^\alpha|^2.
\end{align*}
于是
\begin{align*}
\int_{\mathbb{R}^n}(1+|y|^k)^2|\hat{u}|^2\,\mathrm{d}y
\leqslant c_1\int_{\mathbb{R}^n}\sum_{|\alpha|\leqslant k}|y^\alpha|^2|\hat{u}|^2\,\mathrm{d}y
\leqslant c_1\|u\|_{H^k(\mathbb{R}^n)}^2<\infty.
\end{align*}
从而 $(1+|y|^k)\hat{u}\in L^2(\mathbb{R}^n)$. 同时由上式还可得
\begin{align*}
\|(1+|y|^k)\hat{u}\|_{L^2(\mathbb{R}^n)}\leqslant C_2\|u\|_{H^k(\mathbb{R}^n)},
\end{align*}
其中 $C_2=\sqrt{c_1}$. 这就是 \eqref{eq:Fourier-Hk-等价范数11} 式的右半边不等式.

{\heiti 2. } 反之, 若 $(1+|y|^k)\hat{u}\in L^2(\mathbb{R}^n)$, 且 $|\alpha|\leqslant k$, 则当 $k\geqslant 1$ 时, 由\cref{lemma:多项式等价}, 存在只依赖于 $n$ 与 $k$ 的常数 $c_2>0$, 使得
\begin{align}\label{eq:Fourier-Hk-222}
\sum_{|\alpha|\leqslant k}|y^\alpha|^2
\leqslant c_2(1+|y|^k)^2.
\end{align}
令
\begin{align*}
u_\alpha\triangleq ((\text{i}y)^\alpha\hat{u})^\vee.
\end{align*}
则对每一 $\phi\in C_c^\infty(\mathbb{R}^n)$, 由\hyperref[theorem:Plancherel定理]{Plancherel定理}及\hyperref[theorem:1222-Fourier变换的性质]{Fourier变换性质\ref{theorem:1222-Fourier变换的性质-1}\ref{theorem:1222-Fourier变换的性质-2}}得
\begin{align*}
\int_{\mathbb{R}^n}(D^\alpha\phi)\bar{u}\,\mathrm{d}x
=\int_{\mathbb{R}^n}\widehat{(D^\alpha\phi)}\bar{\hat{u}}\,\mathrm{d}y
=\int_{\mathbb{R}^n}(\text{i}y)^\alpha\hat{\phi}\bar{\hat{u}}\,\mathrm{d}y =(-1)^{|\alpha|}\int_{\mathbb{R}^n}\hat{\phi}\overline{(\text{i}y)^\alpha\hat{u}}\,\mathrm{d}y
=(-1)^{|\alpha|}\int_{\mathbb{R}^n}\phi\bar{u}_\alpha\,\mathrm{d}x.
\end{align*}
因而在弱意义下 $u_\alpha=D^\alpha u$, 所以
\begin{align*}
\sum_{|\alpha|\leqslant k}\int_{\mathbb{R}^n}|D^\alpha u|^2\,\mathrm{d}x
=\sum_{|\alpha|\leqslant k}\int_{\mathbb{R}^n}|y^\alpha|^2|\hat{u}|^2\,\mathrm{d}y.
\end{align*}
由 \eqref{eq:Fourier-Hk-222} 得 $u\in H^k(\mathbb{R}^n)$, 且由上式与 \eqref{eq:Fourier-Hk-222} 有
\begin{align*}
\|u\|_{H^k(\mathbb{R}^n)}^2
=\sum_{|\alpha|\leqslant k}\int_{\mathbb{R}^n}|D^\alpha u|^2\,\mathrm{d}x
\leqslant c_2\int_{\mathbb{R}^n}(1+|y|^k)^2|\hat{u}|^2\,\mathrm{d}y.
\end{align*}
故有 \eqref{eq:Fourier-Hk-等价范数11} 式的左半边估计式, 其中 $C_1=\frac{1}{\sqrt{c_2}}$. 证毕.
\end{proof}

\begin{definition}\label{definition:ch05-Hs}
设 $0<s<\infty$，$u\in L^{2}(\mathbb{R}^{n})$。则 $u\in H^{s}(\mathbb{R}^{n})$ 若 $(1+|y|^{s})\widehat{u}\in L^{2}(\mathbb{R}^{n})$。对非整数 $s$，令
\begin{align*}
\|u\|_{H^{s}(\mathbb{R}^{n})}\triangleq\|(1+|y|^{s})\widehat{u}\|_{L^{2}(\mathbb{R}^{n})}.
\end{align*}
\end{definition}
