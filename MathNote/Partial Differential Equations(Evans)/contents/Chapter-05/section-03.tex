\section{逼近}\label{section:ch05-逼近}

\subsection{内部光滑逼近}\label{subsection:ch05-内部逼近}

\begin{theorem}[光滑函数的局部逼近]\label{theorem:ch05-局部光滑逼近}
设$U\subseteq \mathbb{R}^n$是开集,固定正整数 $k$ 与 $1\leqslant p<\infty$，记$U_{\varepsilon}=\{x\in U\mid \operatorname{dist}(x,\partial U)>\varepsilon\}$.再设 $u\in W^{k,p}(U)$，其中 $1\leqslant p<\infty$，令
\begin{align*}
u_{\varepsilon}\triangleq \begin{cases}
\eta _{\varepsilon}*u,&x\in U_{\varepsilon},\\
0,&x\in U\backslash U_{\varepsilon}.\\
\end{cases}
\end{align*}
则
\begin{enumerate}[(i)]
\item 对每个 $\varepsilon>0$，$u_{\varepsilon}\in C^{\infty}(U_{\varepsilon})$；
\item 当 $\varepsilon\to 0$ 时，$u_{\varepsilon}\to u$ 于 $W^{k,p}_{\mathrm{loc}}(U)$。
\end{enumerate}
\end{theorem}
\begin{proof}

(i)在\hyperref[theorem:appC-光滑算子的性质]{光滑算子的性质}中已证,下证(ii).
我们断言：若 $|\alpha|\leqslant k$，则
\begin{align}
D^{\alpha}u_{\varepsilon}=\eta_{\varepsilon}*D^{\alpha}u,\quad\text{在}\ U_{\varepsilon}\text{中}.
\label{align:ch05-磨光导数}
\end{align}
即光滑函数 $u_{\varepsilon}$ 的通常的$\alpha$ 阶偏导数是 $u$ 的 $\alpha$ 阶弱偏导数的 $\varepsilon$-光滑化。
为证此，对固定的$x\in U_{\varepsilon}$,$\varepsilon>0$与多重指标 \(|\alpha|\leqslant k\),利用\hyperref[theorem:appC-光滑算子的性质-1]{光滑算子的性质\ref{theorem:appC-光滑算子的性质-1}}
可得
\begin{align}\label{eq::jifjiojiwjiIOJ9200--392022-1}
D^{\alpha}u_{\varepsilon}\left( x \right) = D^{\alpha}\int_U{\eta _{\varepsilon}\left( x-y \right) u\left( y \right) \,\mathrm{d}y}=\int_U{u\left( y \right) D_{x}^{\alpha}\eta _{\varepsilon}\left( x-y \right) \,\mathrm{d}y}.
\end{align}
令$F(x,y)\triangleq\eta_\varepsilon(x-y)$. 由于$\eta_\varepsilon\in C^\infty(\mathbb{R}^n)$, 故$F\in C^\infty(\mathbb{R}^{2n})$.
令$z=x-y$. 因为
\begin{align*}
\partial_{x_i}z_j=\delta_{ij},\quad\partial_{y_i}z_j=-\delta_{ij},
\end{align*}
其中$\delta_{ij}$为 Kronecker 符号, 所以对每个$1\leqslant i\leqslant n$, 多元链式法则给出
\begin{align*}
\partial_{x_i}F(x,y)
&=\sum\limits_{j=1}^{n}(\partial_j\eta_\varepsilon)(x-y)\delta_{ij}
=(\partial_i\eta_\varepsilon)(x-y),\\
\partial_{y_i}F(x,y)
&=-\sum\limits_{j=1}^{n}(\partial_j\eta_\varepsilon)(x-y)\delta_{ij}
=-(\partial_i\eta_\varepsilon)(x-y).
\end{align*}
这里$\partial_i\eta_\varepsilon$表示磨光子对第$i$个自变量的偏导数. 因而
\begin{align*}
\partial_{x_i}F=-\partial_{y_i}F.
\end{align*}
下面对$|\alpha|$归纳, 得到
\begin{align*}
D_x^\alpha F=(-1)^{|\alpha|}D_y^\alpha F.
\end{align*}
当$|\alpha|=0$时, 等式显然成立. 假设对所有$|\beta|=m$等式均成立. 对任意$|\alpha|=m+1$, 选取满足$\alpha_i\geqslant1$的$i$, 并令$\beta\triangleq\alpha-e_i$, 其中$e_i$为第$i$个坐标单位向量. 则$|\beta|=m$. 由于$F$光滑, 关于$x$和$y$的混合偏导数可以交换次序, 故
\begin{align*}
D_x^\alpha F
&=\partial_{x_i}(D_x^\beta F)=(-1)^m\partial_{x_i}(D_y^\beta F)=(-1)^mD_y^\beta(\partial_{x_i}F)\\
&=(-1)^{m+1}D_y^\beta(\partial_{y_i}F)=(-1)^{m+1}D_y^\alpha F.
\end{align*}
归纳可知, 对任意多指标$\alpha$, 都有
\begin{align}\label{eq::jifjiojiwjiIOJ9200--392022-2}
D_x^\alpha[\eta_\varepsilon(x-y)]
=(-1)^{|\alpha|}D_y^\alpha[\eta_\varepsilon(x-y)].
\end{align}
从而结合\eqref{eq::jifjiojiwjiIOJ9200--392022-1}\eqref{eq::jifjiojiwjiIOJ9200--392022-2}式可得
\begin{align*}
D^{\alpha}u_{\varepsilon}\left( x \right) = \int_Uu(y)D_x^\alpha[\eta_\varepsilon(x-y)]\,\mathrm{d}y
=(-1)^{|\alpha|}
\int_Uu(y)D_y^\alpha[\eta_\varepsilon(x-y)]\,\mathrm{d}y.
\end{align*}
现对固定的 $x\in U_{\varepsilon}$，函数 $\phi(y)\triangleq\eta_{\varepsilon}(x-y)$ 属于 $C^{\infty}_{c}(U)$。因而 $\alpha$ 阶弱偏导数的定义给出
\begin{align*}
\int_{U}u(y)D_{y}^{\alpha}\eta_{\varepsilon}(x-y)\,\mathrm{d}y
=(-1)^{|\alpha|}\int_{U}\eta_{\varepsilon}(x-y)D^{\alpha}u(y)\,\mathrm{d}y.
\end{align*}
于是
\begin{align*}
D^{\alpha}u_{\varepsilon}(x)
=(-1)^{|\alpha|+|\alpha|}\int_{U}\eta_{\varepsilon}(x-y)D^{\alpha}u(y)\,\mathrm{d}y
=[\eta_{\varepsilon}*D^{\alpha}u](x).
\end{align*}
这就证明了\eqref{align:ch05-磨光导数}式。
现取开集 $V\subset \subset U$。由\eqref{align:ch05-磨光导数}式及\hyperref[theorem:appC-光滑算子的性质]{光滑算子的性质\ref{theorem:appC-光滑算子的性质-4}}，当 $\varepsilon\to 0$ 时，对每个 $|\alpha|\leqslant k$ 都有 $D^{\alpha}u_{\varepsilon}\to D^{\alpha}u$ 于 $L^{p}(V)$。从而
\begin{align*}
\|u_{\varepsilon}-u\|_{W^{k,p}(V)}^{p}
=\sum_{|\alpha|\leqslant k}\|D^{\alpha}u_{\varepsilon}-D^{\alpha}u\|_{L^{p}(V)}^{p}\to 0,\quad \varepsilon\to 0.
\end{align*}
这就证明了断言 (ii)。
\end{proof}

\subsection{光滑函数的逼近}\label{subsection:ch05-光滑函数逼近}

\begin{lemma}[光滑单位分解]\label{lemma:ch05-光滑单位分解-p7R3}
设 $U\subseteq\mathbb{R}^{n}$ 为开集, $\{V_i\}_{i=0}^{\infty}$ 是 $U$ 的可数开覆盖, 且 $V_i\subset \subset U$. 假设该覆盖在 $U$ 中局部有限, 即每个 $x\in U$ 都有一个邻域, 仅与有限多个 $V_i$ 相交. 则存在 $\zeta_i\in C_c^\infty(V_i)$, 使得
\begin{align*}
0\leqslant\zeta_i\leqslant1,\quad
\sum\limits_{i=0}^{\infty}\zeta_i(x)=1,\quad\forall x\in U.
\end{align*}
这里将每个 $\zeta_i$ 在 $V_i$ 外延拓为零. 并且, 每个紧集 $K\subseteq U$ 只与有限多个 $\operatorname{spt}\zeta_i$ 相交, 即上述和在 $U$ 中是局部有限和.
\end{lemma}
\begin{remark}
特别地,若$\{V_i\}_{i=0}^{N}$ 是 $U$ 的有限开覆盖,则取$V_{N+1},V_{N+2},\cdots$为空集,此时也有上面的结论.
\end{remark}
\begin{proof}

若 $U=\varnothing$, 取所有 $\zeta_i=0$ 即可. 以下设 $U\neq\varnothing$.
先构造从属于各 $V_i$ 的紧集覆盖. 对 $x\in U$, 定义
\begin{align*}
a_i(x)\triangleq\operatorname{dist}(x,\mathbb{R}^{n}\setminus V_i),\quad
A(x)\triangleq\max_{i\geqslant0}a_i(x).
\end{align*}
每个 $a_i$ 都是连续函数, 且在 $V_i$ 外等于零. 由于覆盖局部有限, 每一点附近只有有限多个 $a_i$ 可能非零, 因而上述最大值可以取到, 且 $A$ 在 $U$ 上连续. 又因为 $\{V_i\}$ 覆盖 $U$, 所以 $A(x)>0$ 对每个 $x\in U$ 成立.

令
\begin{align*}
K_i\triangleq\left\{x\in U:a_i(x)\geqslant\frac{A(x)}{2}\right\}.
\end{align*}
由 $A(x)>0(\forall x\in U)$ 可知 $K_i\subseteq V_i$. 连续性表明 $K_i$ 在 $U$ 中相对闭, 而 $\overline{V_i}$ 是包含在 $U$ 中的紧集, 因而 $K_i$ 是紧集. 对任意 $x\in U$, 选取使 $a_i(x)=A(x)$ 的指标 $i$, 即得 $x\in K_i$. 因此
\begin{align*}
U=\bigcup\limits_{i=0}^{\infty}K_i.
\end{align*}
接着构造截断函数. 取非负标准光滑化子 $\eta\in C_c^\infty(B(0,1))$, 满足
\begin{align*}
\int_{\mathbb{R}^{n}}\eta(x)\,\mathrm{d}x=1,\quad
\eta_r(x)\triangleq\frac{1}{r^{n}}\eta\left(\frac{x}{r}\right)\quad(r>0).
\end{align*}
若 $K_i=\varnothing$, 令 $\varphi_i=0$. 若 $K_i\neq\varnothing$, 则
\begin{align*}
b_i\triangleq\operatorname{dist}(K_i,\mathbb{R}^{n}\setminus V_i)>0.
\end{align*}
取 $r_i=\frac{b_i}{4}$, 并定义
\begin{align*}
E_i\triangleq\{y\in\mathbb{R}^{n}:\operatorname{dist}(y,K_i)<2r_i\},\quad
\varphi_i\triangleq\eta_{r_i}*\chi_{E_i}.
\end{align*}
由于 $E_i$ 有界, 其特征函数可积, 故由\hyperref[theorem:appC-光滑算子的性质]{光滑算子的性质\ref{theorem:appC-光滑算子的性质-1}}可知 $\varphi_i\in C^\infty(\mathbb{R}^{n})$. 又由 $\eta_{r_i}\geqslant0$ 及其积分等于 $1$, 有 $0\leqslant\varphi_i\leqslant1$. 若 $\operatorname{dist}(x,K_i)<r_i$, 则对所有 $z\in\operatorname{spt}\eta_{r_i}$, 即$z\in B(0,r_i)$.再利用三角不等式有
\begin{align*}
\operatorname{dist}(x-z,K_i)
\leqslant\operatorname{dist}(x,K_i)+|z|
<2r_i\Longrightarrow x-z\in E_i.
\end{align*}
因此
\begin{align*}
\varphi _i(x)=\int_{\mathbb{R} ^n}{\eta _{r_i}(z)\chi _{E_i}(x-z)\,\mathrm{d}z}=\int_{\mathbb{R} ^n}{\eta _{r_i}(z)\,\mathrm{d}z}=1.
\end{align*}
特别地, $\varphi_i=1$ 在 $K_i$ 的一个邻域中成立. 同时, 由\hyperref[theorem:appC-光滑算子的性质-6]{光滑算子的性质\ref{theorem:appC-光滑算子的性质-6}},再结合$3r_i<b_i$可得
\begin{align*}
\mathrm{spt}\,\varphi _i\subseteq \mathrm{spt}\,\chi _{E_i}+\overline{B\left( 0,r_i \right) }\subseteq \overline{E_i}+\overline{B\left( 0,r_i \right) }\subseteq \{x\in \mathbb{R} ^n:\mathrm{dist}(x,K_i)\leqslant 3r_i\}\subseteq V_i.
\end{align*}
因为上式左侧包含于一个有界闭集,所以 $\varphi_i\in C_c^\infty(V_i)$.

最后令
\begin{align*}
\Phi(x)\triangleq\sum\limits_{i=0}^{\infty}\varphi_i(x),\quad
\zeta_i(x)\triangleq\frac{\varphi_i(x)}{\Phi(x)}\quad(x\in U).
\end{align*}
对任意紧集 $K\subseteq U$,由紧集定义知,存在$N\in \mathbb{N}$,$x_i\in K,\varepsilon_i>0(i=1,2,\cdots,N)$,使得
\begin{align*}
K\subseteq \bigcup_{i=1}^N{B\left( x_i,\varepsilon _i \right)}.
\end{align*}
又由$\{V_i\}$覆盖$U$且覆盖局部有限,于是$B\left( x_i,\varepsilon _i \right)$只与有限多个$V_i$相交,又$\operatorname{spt}\varphi_i\subseteq V_i$,故$B\left( x_i,\varepsilon _i \right)$只与有限多个$\operatorname{spt}\zeta_i$ 相交.因此$K$也只与有限多个$\operatorname{spt}\zeta_i$ 相交,即$\Phi$ 是局部有限的光滑函数之和, 因而 $\Phi\in C^\infty(U)$. 又因为 $\{K_i\}$ 覆盖 $U$, 每个 $x\in U$ 处至少有一个 $\varphi_i(x)=1$, 所以 $\Phi(x)\geqslant1$. 因此 $\zeta_i\in C^\infty(U)$, 且
\begin{align*}
0\leqslant\zeta_i\leqslant1,\quad
\operatorname{spt}\zeta_i\subseteq\operatorname{spt}\varphi_i\subset \subset V_i,\quad
\sum\limits_{i=0}^{\infty}\zeta_i(x)=1.
\end{align*}
又因为$\varphi_i\in C_c^\infty(V_i)$,所以$\zeta_i\in C_c^\infty(V_i)$. 
\end{proof}

\begin{theorem}[光滑函数的整体逼近]\label{theorem:ch05-整体光滑逼近}设 $U\subset \mathbb{R}^n$是有界开集，并设 $u\in W^{k,p}(U)$，其中 $k\in \mathbb{N},1\leqslant p<\infty$。则存在函数 $u_{m}\in C^{\infty}(U)\cap W^{k,p}(U)$ 使
\begin{align*}
u_{m}\to u,\quad\text{在}\ W^{k,p}(U)\text{中}.
\end{align*}
\end{theorem}
\begin{remark}
我们并未断言 $u_{m}\in C^{\infty}(\overline{U})$（但见下文\cref{theorem:ch05-边界光滑逼近}）。
\end{remark}
\begin{proof}

若 $U=\varnothing$, 结论显然成立. 下设$U\neq \varnothing$.

1.构造局部有限的开覆盖并应用\hyperref[lemma:ch05-光滑单位分解-p7R3]{光滑单位分解引理}.
记 $d(x)\triangleq\operatorname{dist}(x,\partial U)$, 并令
\begin{align*}
U_i\triangleq\left\{x\in U:d(x)>\frac{1}{i}\right\}
\quad(i=1,2,\cdots).
\end{align*}
距离函数连续, 且 $d(x)>0$ 对每个 $x\in U$ 成立, 所以 $U_i$ 是递增的开集族, 且
\begin{align*}
U=\bigcup\limits_{i=1}^{\infty}U_i.
\end{align*}
由于 $U$ 有界, $\overline{U_i}$ 是紧集. 若 $x\in\overline{U_i}$, 则 $x\in\overline{U}$ 且 $d(x)\geqslant\frac{1}{i}>0$, 因而 $x$ 不在 $\partial U$ 上, 从而 $x\in U$. 因此
\begin{align*}
\overline{U_i}
\subseteq\left\{x\in U:d(x)\geqslant\frac{1}{i}\right\}
\subseteq U_{i+1}.
\end{align*}
特别地, $U_i\subset \subset U$.
具体取 $V_0\triangleq U_3$, $W_0\triangleq U_4$, 并对 $i\geqslant1$ 定义
\begin{align*}
V_i\triangleq U_{i+3}\setminus\overline{U_{i+1}},\quad
W_i\triangleq U_{i+4}\setminus\overline{U_i}.
\end{align*}
注意到这些集合都是开集, 且 $\overline{V_0}\subseteq W_0$. 对 $i\geqslant1$, 有
\begin{align*}
\overline{V_i}
\subseteq
\left\{x\in U:\frac{1}{i+3}\leqslant d(x)\leqslant\frac{1}{i+1}\right\}
\subseteq W_i.
\end{align*}
又因为 $\overline{W_i}\subseteq\overline{U_{i+4}}$, 所以 $W_i\subset \subset U$, 从而 $V_i\subset \subset W_i\subset \subset U$ 对所有 $i\geqslant0$ 成立.

下面验证 $\{V_i\}_{i=0}^{\infty}$ 覆盖 $U$. 若 $x\in U_3$, 则 $x\in V_0$. 若 $x\notin U_3$,由最小数原理可取满足 $x\in U_j$ 的最小正整数 $j$, 则$j\geqslant4$, 且
\begin{align*}
\frac{1}{j}<d(x)\leqslant\frac{1}{j-1}<\frac{1}{j-2}.
\end{align*}
因此 $x\in U_j\setminus\overline{U_{j-2}}=V_{j-3}$. 故
\begin{align*}
U=\bigcup\limits_{i=0}^{\infty}V_i.
\end{align*}
再验证局部有限性. 任取紧集 $K\subseteq U$, 由\cref{proposition:开集中的紧集距离边界距离必大于0}可选取正整数 $N$, 使得 $K\subseteq U_N$. 对每个 $i\geqslant N$, 有 $K\subseteq U_i$, 而 $W_i\subseteq U\setminus\overline{U_i}$, 所以 $K\cap W_i=\varnothing$. 因此每个紧子集只与有限多个 $W_i$ 相交. 在每个点附近取一个包含于 $U$ 的闭球$B$,从而$B$只与有限多个 $W_i$ 相交,即知 $\{W_i\}$ 在 $U$ 中局部有限. 由 $V_i\subseteq W_i$, $\{V_i\}$ 也在 $U$ 中局部有限.

由\hyperref[lemma:ch05-光滑单位分解-p7R3]{光滑单位分解引理}, 存在光滑单位分解 $\{\zeta_i\}_{i=0}^{\infty}$, 满足
\begin{align}
0\leqslant\zeta_i\leqslant1,\quad
\zeta_i\in C_c^\infty(V_i),\quad
\sum\limits_{i=0}^{\infty}\zeta_i=1,\quad i=0,1,\cdots.
\label{align:ch05-单位分解-a8L2}
\end{align}
这些函数均按零延拓理解为 $U$ 上的光滑函数.并且,每个紧集 $K\subseteq U$ 只与有限多个 $\operatorname{spt}\zeta_i$ 相交.

2.利用局部逼近定理磨光各个乘积.
由\hyperref[theorem:ch05-弱导数性质-4]{弱导数性质\ref{theorem:ch05-弱导数性质-4}}知, $\zeta_i u\in W^{k,p}(U)$, 且其 $k$ 阶以内的弱导数的支集都包含于 $\operatorname{spt}\zeta_i$. 由于 $\operatorname{spt}\zeta_i$ 是包含于 $W_i$ 的紧集, 且 $W_i\subset\subset U$, 当 $\varepsilon>0$ 充分小时, 有
\begin{align*}
\overline{W_i}\subseteq U_\varepsilon,\quad
\operatorname{spt}\zeta_i+\overline{B(0,\varepsilon)}
\subseteq W_i,\quad i=0,1,\cdots.
\end{align*}
考虑$\zeta_i u$ 在 $U_\varepsilon$ 中的局部磨光,即
\begin{align*}
u_{i,\varepsilon}\left( x \right) \triangleq \begin{cases}
\bigl( \eta _{\varepsilon}*\left( \zeta _iu \right) \bigr) \left( x \right) ,&x\in U_{\varepsilon},\\
0,&x\in U\backslash U_{\varepsilon}.\\
\end{cases}\quad i=0,1,\cdots.
\end{align*}
由\hyperref[theorem:ch05-局部光滑逼近]{光滑函数的局部逼近定理}知,$u_{i,\varepsilon}\in C^{\infty}(U_{\varepsilon})$,进而$u_{i,\varepsilon}\in C^{\infty}(W_i)$. 又由\hyperref[theorem:appC-光滑算子的性质-6]{光滑算子的性质\ref{theorem:appC-光滑算子的性质-6}}可知
\begin{align}\label{eq::jiowjioWJioFJIOUWE-12132}
\mathrm{spt}\,u_{i,\varepsilon}\subseteq \mathrm{spt}\,\zeta _i+\overline{B(0,\varepsilon )}\subseteq W_i,\quad i=0,1,\cdots.
\end{align}
从而 $u_{i,\varepsilon}\in C_c^\infty(W_i)$.于是在 $W_i$ 上应用\hyperref[theorem:ch05-局部光滑逼近]{光滑函数的局部逼近定理}, 并利用 $u_{i,\varepsilon}$ 与 $\zeta_i u$ 及其 $k$ 阶以内的导数都在 $U\setminus W_i$ 上几乎处处为零, 得
\begin{align*}
\|u_{i,\varepsilon}-\zeta_i u\|_{W^{k,p}(U)}
=\|u_{i,\varepsilon}-\zeta_i u\|_{W^{k,p}(W_i)}
\to0\quad(\varepsilon\to0),\quad i=0,1,\cdots.
\end{align*}
故对每个 $\delta>0$,可选取充分小的 $\varepsilon_i>0$, 使得
\begin{align}
\begin{cases}
u_i\triangleq u_{i,\varepsilon _i}\in C_{c}^{\infty}(W_i),\\
\parallel u_i-\zeta _iu\parallel _{W^{k,p}(U)}\leqslant \frac{\delta}{2^{i+1}}.\\
\end{cases}\quad i=0,1,\cdots.
\label{align:ch05-逼近误差-e6Q2}
\end{align}

3.求和并构造逼近序列.
令
\begin{align*}
v\triangleq\sum\limits_{i=0}^{\infty}u_i.
\end{align*}
由\eqref{eq::jiowjioWJioFJIOUWE-12132}式知$\operatorname{spt}u_i\subseteq W_i$ 且 $\{W_i\}$ 在 $U$ 中局部有限, 该和在每一点附近都至多与有限个$W_i$相交,进而也至多与有限个$\text{spt}\,u_i$相交,即该和在每一点附近至多只有有限项非零,因而 $v\in C^\infty(U)$.
注意到$u=\sum_{i=0}^{\infty}\zeta_{i}u$,再由\eqref{align:ch05-逼近误差-e6Q2}式有
\begin{align*}
\left\| v-u \right\| _{W^{k,p}\left( U \right)}=\left\| \sum_{i=0}^{\infty}{\left( u_i-\zeta _iu \right)} \right\| _{W^{k,p}\left( U \right)}\leqslant \sum_{i=0}^{\infty}{\left\| u_i-\zeta _iu \right\| _{W^{k,p}\left( U \right)}}\leqslant \delta \sum_{i=0}^{\infty}{\frac{1}{2^{i+1}}}=\delta .
\end{align*}
对每个正整数 $m$, 取 $\delta=\frac{1}{m}$, 将对应的 $v$ 记为 $u_m$，就完成证明了.
\end{proof}

\subsection{直到边界的光滑整体逼近}\label{subsection:ch05-边界逼近}

\begin{theorem}[由直到边界光滑的函数作整体逼近]\label{theorem:ch05-边界光滑逼近}
设 $U\subseteq \mathbb{R}^n$是有界开集且 $\partial U$ 为 $C^{1}$， $u\in W^{k,p}(U)$，其中 $k\in \mathbb{N},1\leqslant p<\infty$。则存在函数 $u_{m}\in C^{\infty}(\overline{U})$ 使
\begin{align*}
u_{m}\to u,\quad\text{在}\ W^{k,p}(U)\text{中}.
\end{align*}
特别地,若 $u\in W^{k,p}(U)\cap C(\overline{U})$，则$u_{m}\in C^{\infty}(\overline{U})$在 $\overline{U}$ 上一致收敛于 $u$。
\end{theorem}
\begin{remark}
想法是我们已向上平移足够距离，从而“在 $U$ 内部有磨光的余地”。
\end{remark}
\begin{proof}

1. 固定任一点 $x_{0}\in\partial U$。由于 $\partial U$ 为 $C^{1}$，根据\cref{definition:appC-Ck边界}，存在半径 $r>0$ 及 $C^{1}$ 函数 $\gamma:\mathbb{R}^{n-1}\to\mathbb{R}$(必要时重新标记坐标轴)，使得
\begin{align*}
U\cap B(x_{0},r)=\{x\in B(x_{0},r)\mid x_{n}>\gamma(x_{1},\cdots,x_{n-1})\}.
\end{align*}
令 $V\triangleq U\cap B\left( x_0,\frac{r}{2} \right) $。

2. 定义平移点
\begin{align*}
x_{\varepsilon}\triangleq x+\lambda\varepsilon e_{n}\qquad (x\in V,\ \varepsilon>0),
\end{align*}
并注意对某个固定的充分大的数 $\lambda>0$，对一切 $x\in V$ 及一切充分小的 $\varepsilon>0$,有$V+\lambda\varepsilon e_n\subseteq U_\varepsilon$,且球 $B(x_{\varepsilon},\varepsilon)$都含于 $U\cap B(x_{0},r)$中,见\cref{figure:ch05-wiq12}.
\begin{figure}[H]
\centering
\includegraphics[width=0.5\textwidth]{ch05-wiq12.png}
\caption{}
\label{figure:ch05-wiq12}
\end{figure}
现定义 $u_{\varepsilon}(x)\triangleq u(x_{\varepsilon})$（$x\in V$）；这是函数 $u$ 沿 $e_{n}$ 方向平移距离 $\lambda\varepsilon$ 所得。再令 $v_{\varepsilon}=\begin{cases}
\eta _{\varepsilon}*u_{\varepsilon}\left( x \right) ,&x\in V,\\
0,&x\notin V.\\
\end{cases}$。由\hyperref[theorem:appC-光滑算子的性质-1]{光滑算子的性质\ref{theorem:appC-光滑算子的性质-1}}可知$v_{\varepsilon}\in C^{\infty}(\overline{V})$。

3. 我们现在断言
\begin{align}
v_{\varepsilon}\to u,\quad\text{在}\ W^{k,p}(V)\text{中}.
\label{align:ch05-边界逼近收敛}
\end{align}
为确认此点，取任意多重指标 $\alpha$ 满足 $|\alpha|\leqslant k$。则
\begin{align}\label{align:ch05-边界逼近收敛1}
\|D^{\alpha}v_{\varepsilon}-D^{\alpha}u\|_{L^{p}(V)}
\leqslant\|D^{\alpha}v_{\varepsilon}-D^{\alpha}u_{\varepsilon}\|_{L^{p}(V)}
+\|D^{\alpha}u_{\varepsilon}-D^{\alpha}u\|_{L^{p}(V)}.
\end{align}
因为\hyperref[Real Analysis-theorem:Lp范数下平均连续性]{平移在 $L^{p}$ 范数下连续},所以右端第二项随 $\varepsilon\to 0$ 趋于零；下面证明第一项也随 $\varepsilon\to 0$ 趋于零.将$D^\alpha u$零延拓到整个 \(\mathbb{R}^n\)，仍记为$D^\alpha u$。在 \(V\) 上，由\hyperref[theorem:appC-光滑算子的性质]{光滑算子的性质\ref{theorem:appC-光滑算子的性质-1}}有
\begin{align*}
D^{\alpha}v_{\varepsilon}\left( x \right) -D^{\alpha}u_{\varepsilon}\left( x \right) &=\int_{\mathbb{R} ^n}{\eta _{\varepsilon}\left( y \right) D^{\alpha}u\left( x+\lambda \varepsilon e_n-y \right) \,\mathrm{d}y}-D^{\alpha}u\left( x+\lambda \varepsilon e_n \right) 
\\
&=\int_{\mathbb{R} ^n}{\eta _{\varepsilon}\left( y \right) \left[ D^{\alpha}u\left( x+\lambda \varepsilon e_n-y \right) -D^{\alpha}u\left( x+\lambda \varepsilon e_n \right) \right] \,\mathrm{d}y}.
\end{align*}
利用\hyperref[Basis of Analytics-theorem:Minkowski(闵可夫斯基)不等式]{Minkowski不等式}，再结合\(\operatorname{spt}\eta_\varepsilon\subset B(0,\varepsilon)\)，得到
\begin{align*}
&\left\| D^{\alpha}v_{\varepsilon}-D^{\alpha}u_{\varepsilon} \right\| _{L^p\left( V \right)}\leqslant \int_{B\left( 0,\varepsilon \right)}{\left\| D^{\alpha}u\left( x+\lambda \varepsilon e_n-y \right) -D^{\alpha}u\left( x+\lambda \varepsilon e_n \right) \right\| _{L^p\left( V \right)}\,\mathrm{d}y}
\\
&\leqslant \int_{B\left( 0,\varepsilon \right)}{\left\| D^{\alpha}u\left( x-y \right) -D^{\alpha}u\left( x \right) \right\| _{L^p\left( \mathbb{R} ^n \right)}\,\mathrm{d}y}\leqslant \underset{\left| y \right|\leqslant \varepsilon}{\mathrm{sup}}\left\| D^{\alpha}u\left( x-y \right) -D^{\alpha}u\left( x \right) \right\| _{L^p\left( \mathbb{R} ^n \right)}.
\end{align*}
再利用\hyperref[Real Analysis-theorem:Lp范数下平均连续性]{平移在 \(L^p(\mathbb{R}^n)\) 中保持范数不变}，可得
\begin{align*}
\underset{\varepsilon \rightarrow 0}{\lim}\left\| D^{\alpha}v_{\varepsilon}-D^{\alpha}u_{\varepsilon} \right\| _{L^p\left( V \right)}&\leqslant \underset{\varepsilon \rightarrow 0}{\lim}\underset{\left| y \right|\leqslant \varepsilon}{\mathrm{sup}}\left\| D^{\alpha}u\left( x-y \right) -D^{\alpha}u\left( x \right) \right\| _{L^p\left( \mathbb{R} ^n \right)}\\
&=\underset{y\rightarrow 0}{\overline{\lim }}\left\| D^{\alpha}u\left( x-y \right) -D^{\alpha}u\left( x \right) \right\| _{L^p\left( \mathbb{R} ^n \right)}=0.
\end{align*}
故\eqref{align:ch05-边界逼近收敛1}式随$\varepsilon\to 0$趋于零,进而\eqref{align:ch05-边界逼近收敛}式成立.

4. 对任意$\delta>0$。由于 $\partial U$ 紧，再结合\eqref{align:ch05-边界逼近收敛}式,可找到有限个点 $x_{i}^{0}\in\partial U$、半径 $r_{i}>0$、相应集合 $V_{i}=U\cap B^0\left( x_{i}^{0},\frac{r_i}{2} \right) $ 及函数 $v_{i}\in C^{\infty}(\overline{V}_{i})$（$i=1,\cdots,N$），使 $\partial U\subset\bigcup_{i=1}^{N}B^0\left( x_{i}^{0},\frac{r_i}{2} \right) $ 且
\begin{align}
\|v_{i}-u\|_{W^{k,p}(V_{i})}\leqslant\delta.
\label{align:ch05-边界局部误差}
\end{align}
取开集 $V_{0}\subset \subset U$ 使 $U\subset\bigcup_{i=0}^{N}V_{i}$，并用\hyperref[theorem:ch05-局部光滑逼近]{光滑函数的局部逼近定理}选函数 $v_{0}\in C^{\infty}(\overline{V}_{0})$ 满足
\begin{align}
\|v_{0}-u\|_{W^{k,p}(V_{0})}\leqslant\delta.
\label{align:ch05-内部逼近误差}
\end{align}

5. 现设 $\{\zeta_{i}\}_{i=0}^{N}$ 为 $\overline{U}$ 上从属于开集族 $\left\{ V_0,B^0\left( x_{1}^{0},\frac{r_1}{2} \right) ,\cdots ,B^0\left( x_{N}^{0},\frac{r_N}{2} \right) \right\} $ 的\hyperref[lemma:ch05-光滑单位分解-p7R3]{光滑单位分解},满足
\begin{align*}
0\leqslant\zeta_i\leqslant1,\quad
\zeta_i\in C_c^\infty(V_i),\quad
\sum\limits_{i=0}^{N}\zeta_i=1,\quad i=0,1,\cdots,N.
\end{align*}
由$\zeta_i\in C_c^\infty(V_i)$可知$\zeta_i\in C^\infty(\overline{V_i})$.定义 $v\triangleq\sum_{i=0}^{N}\zeta_{i}v_{i}$。则显然 $v\in C^{\infty}(\overline{U})$。此外，由于 $u=\sum_{i=0}^{N}\zeta_{i}u$，利用\hyperref[theorem:ch05-弱导数性质]{弱导数性质\ref{theorem:ch05-弱导数性质-4}}和\eqref{align:ch05-边界局部误差}式与\eqref{align:ch05-内部逼近误差}式可知,对每个 $|\alpha|\leqslant k$,有
\begin{align*}
&\left\| D^{\alpha}v-D^{\alpha}u \right\| _{L^p\left( U \right)}\leqslant \sum_{i=0}^N{\left\| D^{\alpha}\left( \zeta _iv_i \right) -D^{\alpha}\left( \zeta _iu \right) \right\| _{L^p\left( V_i \right)}}=\sum_{i=0}^N{\left\| D^{\alpha}\left( \zeta _i\left( v_i-u \right) \right) \right\| _{L^p\left( V_i \right)}}
\\
&=\sum_{i=0}^N{\left\| \sum_{\beta \leqslant \alpha}{\left( \begin{array}{c}
\alpha\\
\beta\\
\end{array} \right)}D^{\beta}\left( \zeta _i \right) D^{\alpha -\beta}\left( v_i-u \right) \right\| _{L^p\left( V_i \right)}}\leqslant \sum_{i=0}^N{\sum_{\beta \leqslant \alpha}{\left( \begin{array}{c}
\alpha\\
\beta\\
\end{array} \right)}\left\| D^{\beta}\left( \zeta _i \right) \right\| _{L^{\infty}\left( V_i \right)}\left\| D^{\alpha -\beta}\left( v_i-u \right) \right\| _{L^p\left( V_i \right)}}
\\
&\leqslant C\sum_{i=0}^N{\left\| v_i-u \right\| _{W^{k,p}\left( V_i \right)}}\left( \sum_{\beta \leqslant \alpha}{\left( \begin{array}{c}
\alpha\\
\beta\\
\end{array} \right)}\left\| D^{\beta}\left( \zeta _i \right) \right\| _{L^{\infty}\left( V_i \right)} \right) \leqslant C(N+1)\delta ,
\end{align*}
其中$C\triangleq \underset{\substack{0\leqslant i\leqslant N\\
\left| \alpha \right|\leqslant k\\}}{\max}\sum_{\beta \leqslant \alpha}{\left( \begin{array}{c}
\alpha\\
\beta\\
\end{array} \right)}\left\| D^{\beta}\left( \zeta _i \right) \right\| _{L^{\infty}\left( V_i \right)}.$于是
\begin{align*}
\left\| v-u \right\| _{W^{k,p}\left( U \right)}=\sum_{\left| \alpha \right|\leqslant k}{\left\| D^{\alpha}v-D^{\alpha}u \right\| _{L^p\left( U \right)}}\leqslant \sum_{\left| \alpha \right|\leqslant k}{C\left( N+1 \right) \delta}.
\end{align*}
对每个正整数 $m$, 取 $\delta=\frac{1}{m}$, 将对应的 $v$ 记为 $u_m$，就完成证明了.

特别地,设 \(u\in W^{k,p}(U)\cap C(\overline U)\),则由$\overline{U}$是紧集知\(u\) 在 \(\overline U\) 上一致连续，即对任意 \(\eta>0\)，存在 \(\delta>0\)，使得当 \(|x-y|<\delta\) 时，有
\begin{align*}
|u(x)-u(y)|<\eta.
\end{align*}
在步骤 2 中，取 \(\varepsilon>0\) 充分小，使得 \(\lambda\varepsilon+\varepsilon<\delta\)。则对 \(x\in V\)，有
\begin{align*}
|u_\varepsilon(x)-u(x)|=|u(x+\lambda\varepsilon e_n)-u(x)|<\eta,
\end{align*}
且由\hyperref[theorem:appC-光滑算子的性质-1]{光滑算子的性质}知
\begin{align*}
|v_{\varepsilon}(x)-u(x)|=\left| \int_{\mathbb{R} ^n}{\eta _{\varepsilon}(y)\left[ u(x+\lambda \varepsilon e_n-y)-u(x) \right] \,\mathrm{d}y} \right|\leqslant \int_{\mathbb{R} ^n}{\eta _{\varepsilon}(y)|u}(x+\lambda \varepsilon e_n-y)-u(x)|\,\mathrm{d}y\leqslant \eta .
\end{align*}
故 \(v_\varepsilon\to u\) 在 \(V\) 上一致收敛。在步骤4中，当 \(\varepsilon\) 充分小时，
\begin{align*}
|(\eta_\varepsilon*u)(x)-u(x)|
\leqslant\int_{\mathbb R^n}\eta_\varepsilon(y)|u(x-y)-u(x)|\,\mathrm{d}y
\leqslant\eta.
\end{align*}
因此在选取 \(v_i\) 时，除要求
\begin{align*}
\|v_i-u\|_{W^{k,p}(V_i)}\leqslant\delta
\end{align*}
外，还可同时要求
\begin{align*}
\sup_{V_i}|v_i-u|\leqslant\delta,\quad i=0,1,\cdots,N.
\end{align*}
最后由 \(\sum\limits_{i=0}^N\zeta_i=1\)，对 \(x\in\overline U\) 有
\begin{align*}
|v(x)-u(x)|
&=\left|\sum_{i=0}^N\zeta_i(x)(v_i(x)-u(x))\right|
\leqslant\sum_{i=0}^N\zeta_i(x)|v_i(x)-u(x)|
\leqslant\delta.
\end{align*}
故 \(v\) 在 \(\overline U\) 上一致收敛于 \(u\)。
\end{proof}
