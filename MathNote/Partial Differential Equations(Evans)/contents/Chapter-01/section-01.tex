\section{偏微分方程}\label{section:ch01-pde}

本章综述求解偏微分方程所涉及的主要理论问题。

为阅读后续讨论，读者最好先翻阅\cref{chapter:appA-记号} 中给出的记号，特别是偏导数的多重指标记号。

一个偏微分方程（PDE）是含有一个未知的多元函数及其某些偏导数的方程。利用\cref{chapter:appA-记号} 中说明的记号，我们可以符号地写出一个典型的 PDE 如下。取定整数 $k\geqslant 1$，设 $U$ 为 $\mathbb{R}^n$ 的一个开子集。

\begin{definition}\label{definition:ch01-k阶pde}
形如
\begin{align}\label{align:ch01-pde-general}
F(D^k u(x), D^{k-1}u(x), \cdots, Du(x), u(x), x)=0 \qquad (x\in U),
\end{align}
的表达式称为一个 $k$ 阶偏微分方程，其中
\begin{align*}
F:\mathbb{R}^{n^k}\times\mathbb{R}^{n^{k-1}}\times\cdots\times\mathbb{R}^n\times\mathbb{R}\times U\to\mathbb{R}
\end{align*}
是给定的，而
\begin{align*}
u:U\to\mathbb{R}
\end{align*}
是未知函数。
\end{definition}

所谓求解该 PDE，就是要找出所有满足\eqref{align:ch01-pde-general} 的 $u$，这或许只是在那些于 $\partial U$ 的某一部分 $\Gamma$ 上满足某些辅助边界条件的函数中去找。所谓求解，理想地说，就是得到简单、显式的解；若做不到这一点，则至少要推导出解的存在性及其他性质。

\begin{definition}\label{definition:ch01-方程分类}
\begin{enumerate}[(i)]
\item 若偏微分方程\eqref{align:ch01-pde-general} 具有形式
\begin{align*}
\sum_{|\alpha|\leqslant k} a_\alpha(x) D^\alpha u = f(x),
\end{align*}
其中 $a_\alpha$（$|\alpha|\leqslant k$）与 $f$ 是给定函数，则称它为\textbf{线性}的。若 $f\equiv 0$，则称此线性 PDE 是齐次的。
\item 若 PDE \eqref{align:ch01-pde-general} 具有形式
\begin{align*}
\sum_{|\alpha|=k} a_\alpha(x) D^\alpha u + a_0(D^{k-1}u, \cdots, Du, u, x)=0,
\end{align*}
则称它为\textbf{半线性}的。
\item 若 PDE \eqref{align:ch01-pde-general} 具有形式
\begin{align*}
\sum_{|\alpha|=k} a_\alpha(D^{k-1}u, \cdots, Du, u, x) D^\alpha u + a_0(D^{k-1}u, \cdots, Du, u, x)=0,
\end{align*}
则称它为\textbf{拟线性}的。
\item 若 PDE \eqref{align:ch01-pde-general} 非线性地依赖于最高阶导数，则称它为\textbf{完全非线性}的。
\end{enumerate}
\end{definition}

通俗地说，偏微分方程组就是若干个未知多元函数所对应的若干个 PDE 的集合。

\begin{definition}\label{definition:ch01-方程组}
设 $k$ 为正整数。形如
\begin{align}\label{align:ch01-system-general}
\mathbf{F}(D^k\mathbf{u}(x), D^{k-1}\mathbf{u}(x), \cdots, D\mathbf{u}(x), \mathbf{u}(x), x)=0 \qquad (x\in U),
\end{align}
的表达式称为一个 $k$ 阶偏微分方程组，其中
\begin{align*}
\mathbf{F}:\mathbb{R}^{mn^k}\times\mathbb{R}^{mn^{k-1}}\times\cdots\times\mathbb{R}^{mn}\times\mathbb{R}^m\times U\to\mathbb{R}^m
\end{align*}
是给定的，而
\begin{align*}
\mathbf{u}:U\to\mathbb{R}^m,\qquad \mathbf{u}=(u^1,\cdots,u^m)
\end{align*}
是未知函数。
\end{definition}

这里我们假定方程组中数量与未知函数 $(u^1,\cdots,u^m)$ 相同，即含有同样数目的 $m$ 个标量方程。这是最常见的情形，不过其他方程组的方程个数也可能少于或多于未知函数个数。方程组按显而易见的方式分类为线性、半线性等。

\textbf{记号.} 我们用``PDE''同时作为单数``偏微分方程''与复数``偏微分方程组''的缩写。

\begin{theorem}[向量值函数求导的基本性质]\label{theorem:向量求导基本性质}
设 $U\subseteq\mathbb{R}^n$ 为开集，$x\in U$。
\begin{enumerate}[(1)]
\item 设 $A$ 为 $m\times n$ 常矩阵，$b\in\mathbb{R}^m$。定义 $f:U\to\mathbb{R}^m$ 为
\begin{align*}
f(x)\triangleq Ax+b.
\end{align*}
则 $f$ 在 $U$ 上可微，且其 Jacobi 矩阵为
\begin{align*}
Df(x)=A,\qquad \forall x\in U.
\end{align*}
特别地，恒等映射 $f(x)=x$ 的 Jacobi 矩阵为 $Df(x)=I_n$。常值映射$f:U\to\mathbb{R}^m$ 的 Jacobi 矩阵为$Df(x)=0_{m\times n}$.

\item 设 $a\in\mathbb{R}^n$ 为常向量，定义 $f:U\to\mathbb{R}$ 为
\begin{align*}
f(x)\triangleq a^T x.
\end{align*}
则 $f$ 在 $U$ 上可微，且其梯度为
\begin{align*}
\nabla f(x)=a,\qquad \forall x\in U.
\end{align*}

\item 设 $A$ 为 $n\times n$ 常矩阵，定义 $f:U\to\mathbb{R}$ 为
\begin{align*}
f(x)\triangleq x^T A x.
\end{align*}
则 $f$ 在 $U$ 上可微，且其梯度为
\begin{align*}
\nabla f(x)=(A+A^T)x,\qquad \forall x\in U.
\end{align*}
若约定 $\frac{\partial f}{\partial x}$ 为行向量，则
\begin{align*}
\frac{\partial f}{\partial x}=x^T(A+A^T).
\end{align*}

\item 设 $f:U\to\mathbb{R}^m$ 在 $x$ 处可微，$g:f(U)\to\mathbb{R}^p$ 在 $f(x)$ 处可微。则复合函数 $g\circ f$ 在 $x$ 处可微，且
\begin{align*}
D(g\circ f)(x)=Dg(f(x))Df(x).
\end{align*}

\item 设 $f_1,\cdots,f_k:U\to\mathbb{R}^m$ 均在 $x$ 处可微，$c_1,\cdots,c_k\in\mathbb{R}$。则
\begin{align*}
f\triangleq\sum\limits_{i=1}^k c_i f_i
\end{align*}
在 $x$ 处可微，且
\begin{align*}
Df(x)=\sum\limits_{i=1}^k c_i Df_i(x).
\end{align*}
\end{enumerate}
\end{theorem}
\begin{proof}
\begin{enumerate}[(1)]
\item 对任意 $h\in\mathbb{R}^n$，有
\begin{align*}
f(x+h)-f(x)=A(x+h)+b-(Ax+b)=Ah.
\end{align*}
由于 $Ah$ 关于 $h$ 是线性的，且 $|Ah|\leqslant\|A\||h|$，故
\begin{align*}
f(x+h)=f(x)+Ah+o(|h|)\qquad (|h|\to 0).
\end{align*}
因此 $f$ 在 $x$ 处可微，且 $Df(x)=A$。

\item 对任意 $h\in\mathbb{R}^n$，有
\begin{align*}
f(x+h)-f(x)=a^T(x+h)-a^T x=a^T h.
\end{align*}
故
\begin{align*}
f(x+h)=f(x)+a^T h+o(|h|)\qquad (|h|\to 0).
\end{align*}
因此 $f$ 在 $x$ 处可微，且 $\nabla f(x)=a$。

\item 展开
\begin{align*}
f(x+h)=(x+h)^T A(x+h)=x^T A x+x^T A h+h^T A x+h^T A h.
\end{align*}
由于 $h^T A h$ 是 $h$ 的二次项，故 $h^T A h=o(|h|)$。又 $x^T A h$ 和 $h^T A x$ 均为 $h$ 的线性项，且 $h^T A x=x^T A^T h$。因此
\begin{align*}
f(x+h)=f(x)+(x^T A+x^T A^T)h+o(|h|).
\end{align*}
故梯度（列向量）为
\begin{align*}
\nabla f(x)=(A+A^T)x.
\end{align*}
若 $\frac{\partial f}{\partial x}$ 为行向量，则
\begin{align*}
\frac{\partial f}{\partial x}=x^T(A+A^T).
\end{align*}

\item 由可微性定义，存在线性映射 $Df(x)$ 和 $Dg(f(x))$，使得
\begin{align*}
f(x+h)&=f(x)+Df(x)h+o(|h|),\\
g(f(x)+k)&=g(f(x))+Dg(f(x))k+o(|k|).
\end{align*}
取 $k=Df(x)h+o(|h|)$，则 $|k|=O(|h|)$。代入得
\begin{align*}
(g\circ f)(x+h)&=g(f(x)+Df(x)h+o(|h|))\\
&=g(f(x))+Dg(f(x))(Df(x)h+o(|h|))+o(|h|)\\
&=g(f(x))+Dg(f(x))Df(x)h+o(|h|).
\end{align*}
故 $g\circ f$ 在 $x$ 处可微，且
\begin{align*}
D(g\circ f)(x)=Dg(f(x))Df(x).
\end{align*}

\item 由导数的线性性直接得到：
\begin{align*}
f(x+h)-f(x)=\sum\limits_{i=1}^k c_i(f_i(x+h)-f_i(x))
=\sum\limits_{i=1}^k c_i(Df_i(x)h+o(|h|))
=\left(\sum\limits_{i=1}^k c_i Df_i(x)\right)h+o(|h|).
\end{align*}
故
\begin{align*}
Df(x)=\sum\limits_{i=1}^k c_i Df_i(x).
\end{align*}
\end{enumerate}
\end{proof}
