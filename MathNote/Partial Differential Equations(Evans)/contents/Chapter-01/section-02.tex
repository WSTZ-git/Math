\section{例题}\label{section:ch01-examples}

目前尚没有关于一切偏微分方程可解性的一般理论。鉴于可用 PDE 建模的物理、几何与概率现象如此丰富多彩，这样一种理论也几乎不可能存在。相反，研究集中于各类特定的、在数学内外均有重要应用的偏微分方程，并希望从这些 PDE 的来源中得到关于其解的线索。

下面列出当前研究中许多具体而有意义的偏微分方程。此清单只是为了让读者熟悉各种著名 PDE 的名称与形式。为最清楚地展现这些方程的数学结构，我们大多把相关物理常数取为 1。后文将讨论其中许多 PDE 的来源与解释。

通篇取 $x\in U$，其中 $U$ 是 $\mathbb{R}^n$ 的开子集，且 $t\geqslant 0$。又记 $Du=D_xu=(u_{x_1},\cdots,u_{x_n})$ 为 $u$ 关于空间变量 $x=(x_1,\cdots,x_n)$ 的梯度。变量 $t$ 总表示时间。

\subsection{单个偏微分方程}\label{subsection:ch01-single}

\textbf{a. 线性方程.}
\begin{enumerate}
\item Laplace方程
\begin{align*}
\Delta u=\sum_{i=1}^n u_{x_i x_i}=0.
\end{align*}
\item Helmholtz 方程（特征值方程）
\begin{align*}
-\Delta u=\lambda u.
\end{align*}
\item 线性输运方程
\begin{align*}
u_t+\sum_{i=1}^n b_i u_{x_i}=0.
\end{align*}
\item Liouville 方程
\begin{align*}
u_t+\sum_{i=1}^n (b_i u)_{x_i}=0.
\end{align*}
\item 热方程（扩散方程）
\begin{align*}
u_t-\Delta u=0.
\end{align*}
\item Schrödinger 方程
\begin{align*}
i u_t+\Delta u=0.
\end{align*}
\item Kolmogorov 方程
\begin{align*}
u_t-\sum_{i,j=1}^n a_{ij} u_{x_i x_j}+\sum_{i=1}^n b_i u_{x_i}=0.
\end{align*}
\item Fokker--Planck 方程
\begin{align*}
u_t-\sum_{i,j=1}^n (a_{ij}u)_{x_i x_j}-\sum_{i=1}^n (b_i u)_{x_i}=0.
\end{align*}
\item 波动方程
\begin{align*}
u_{tt}-\Delta u=0.
\end{align*}
\item Klein--Gordon 方程
\begin{align*}
u_{tt}-\Delta u+m^2 u=0.
\end{align*}
\item 电报方程
\begin{align*}
u_{tt}+2d u_t-u_{xx}=0.
\end{align*}
\item 一般波动方程
\begin{align*}
u_{tt}-\sum_{i,j=1}^n a_{ij} u_{x_i x_j}+\sum_{i=1}^n b_i u_{x_i}=0.
\end{align*}
\item Airy 方程
\begin{align*}
u_t+u_{xxx}=0.
\end{align*}
\item 梁方程
\begin{align*}
u_{tt}+u_{xxxx}=0.
\end{align*}
\end{enumerate}

\textbf{b. 非线性方程.}
\begin{enumerate}
\item 程函方程
\begin{align*}
|Du|=1.
\end{align*}
\item 非线性 Poisson 方程
\begin{align*}
-\Delta u=f(u).
\end{align*}
\item $p$-Laplace 方程
\begin{align*}
\operatorname{div}(|Du|^{p-2}Du)=0.
\end{align*}
\item 极小曲面方程
\begin{align*}
\operatorname{div}\!\left(\frac{Du}{(1+|Du|^2)^{1/2}}\right)=0.
\end{align*}
\item Monge--Amp\`ere 方程
\begin{align*}
\det(D^2 u)=f.
\end{align*}
\item Hamilton--Jacobi 方程
\begin{align*}
u_t+H(Du,x)=0.
\end{align*}
\item 标量守恒律
\begin{align*}
u_t+\operatorname{div}F(u)=0.
\end{align*}
\item 无黏 Burgers 方程
\begin{align*}
u_t+u u_x=0.
\end{align*}
\item 标量反应--扩散方程
\begin{align*}
u_t-\Delta u=f(u).
\end{align*}
\item 渗流方程
\begin{align*}
u_t-\Delta(u^\gamma)=0.
\end{align*}
\item 非线性波动方程
\begin{align*}
u_{tt}-\Delta u+f(u)=0.
\end{align*}
\item Korteweg--de Vries（KdV）方程
\begin{align*}
u_t+u u_x+u_{xxx}=0.
\end{align*}
\item 非线性 Schrödinger 方程
\begin{align*}
i u_t+\Delta u=f(|u|^2)u.
\end{align*}
\end{enumerate}

\subsection{偏微分方程组}\label{subsection:ch01-systems}

\textbf{a. 线性方程组.}
\begin{enumerate}
\item 线性弹性力学的平衡方程
\begin{align*}
\mu\Delta\mathbf{u}+(\lambda+\mu)D(\operatorname{div}\mathbf{u})=0.
\end{align*}
\item 线性弹性力学的发展方程
\begin{align*}
\mathbf{u}_{tt}-\mu\Delta\mathbf{u}-(\lambda+\mu)D(\operatorname{div}\mathbf{u})=0.
\end{align*}
\item Maxwell 方程
\begin{align*}
\begin{cases}
\mathbf{E}_t=\operatorname{curl}\mathbf{B},\\
\mathbf{B}_t=-\operatorname{curl}\mathbf{E},\\
\operatorname{div}\mathbf{B}=\operatorname{div}\mathbf{E}=0.
\end{cases}
\end{align*}
\end{enumerate}

\textbf{b. 非线性方程组.}
\begin{enumerate}
\item 守恒律方程组
\begin{align*}
\mathbf{u}_t+\operatorname{div}\mathbf{F}(\mathbf{u})=0.
\end{align*}
\item 反应--扩散方程组
\begin{align*}
\mathbf{u}_t-\Delta\mathbf{u}=\mathbf{f}(\mathbf{u}).
\end{align*}
\item 不可压无黏流的 Euler 方程
\begin{align*}
\begin{cases}
\mathbf{u}_t+\mathbf{u}\cdot D\mathbf{u}=-Dp,\\
\operatorname{div}\mathbf{u}=0.
\end{cases}
\end{align*}
\item 不可压黏性流的 Navier--Stokes 方程
\begin{align*}
\begin{cases}
\mathbf{u}_t+\mathbf{u}\cdot D\mathbf{u}-\Delta\mathbf{u}=-Dp,\\
\operatorname{div}\mathbf{u}=0.
\end{cases}
\end{align*}
\end{enumerate}

更为详尽的有趣 PDE 清单见 Zwillinger \cite{Zw}。
