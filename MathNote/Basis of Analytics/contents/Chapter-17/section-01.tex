\section{Fourier级数及基本性质}

\begin{definition}
设 $f$ 是 $[a,b]$ 上的 Riemann 可积函数，记 $L=b-a$。定义其 Fourier 系数为
\[
a_n = \frac{2}{L}\int_a^b f(x)\cos\left(\frac{2\pi n x}{L}\right)\,\mathrm{d}x,\quad n=0,1,2,\cdots,
\]
\[
b_n = \frac{2}{L}\int_a^b f(x)\sin\left(\frac{2\pi n x}{L}\right)\,\mathrm{d}x,\quad n=1,2,3,\cdots.
\]
称
\[
\frac{a_0}{2}+\sum_{n=1}^{\infty}\left(a_n\cos\left(\frac{2\pi n x}{L}\right)+b_n\sin\left(\frac{2\pi n x}{L}\right)\right)
\]
为 $f$ 在 $[a,b]$ 上的\textbf{Fourier 级数}，也常记为
\[
f(x)\sim \frac{a_0}{2}+\sum_{n=1}^{\infty}\left(a_n\cos\left(\frac{2\pi n x}{L}\right)+b_n\sin\left(\frac{2\pi n x}{L}\right)\right).
\]
特别地,当 $[a,b]=[-\pi,\pi]$ 时，$L=2\pi$，上述公式退化为
\[
a_n = \frac{1}{\pi}\int_{-\pi}^{\pi} f(x)\cos(nx)\,\mathrm{d}x,\quad n=0,1,2,\cdots,
\]
\[
b_n = \frac{1}{\pi}\int_{-\pi}^{\pi} f(x)\sin(nx)\,\mathrm{d}x,\quad n=1,2,3,\cdots,
\]
此时
\[
f(x)\sim \frac{a_0}{2}+\sum_{n=1}^{\infty}\bigl(a_n\cos(nx)+b_n\sin(nx)\bigr).
\]
\end{definition}

\begin{definition}
若两个函数 $\varphi$ 与 $\psi$ 在 $[a, b]$ 上可积，且
$\int_a^b \varphi(x)\psi(x)\,\mathrm{d}x = 0$，
则称函数 $\varphi$ 与 $\psi$ 在 $[a, b]$ 上是\textbf{正交的}。
\end{definition}

\begin{proposition}[三角函数系的正交性]\label{proposition:三角函数系的正交性}
三角函数系
$$\{1, \cos x, \sin x, \cos 2x, \sin 2x, \cdots\}$$
在 $[-\pi, \pi]$ 上具有正交性，即任意两个不同函数的乘积在 $[-\pi, \pi]$ 上的积分都等于零，而每个函数的平方在 $[-\pi, \pi]$ 上的积分都不等于零。即
\begin{gather*}
\int_{-\pi}^{\pi} \cos nx \,\mathrm{d}x = \int_{-\pi}^{\pi} \sin nx \,\mathrm{d}x = 0, \\
\int_{-\pi}^{\pi} \cos mx \cos nx \,\mathrm{d}x = 0 \quad (m \neq n), \\
\int_{-\pi}^{\pi} \sin mx \sin nx \,\mathrm{d}x = 0 \quad (m \neq n), \\
\int_{-\pi}^{\pi} \cos^2 nx \,\mathrm{d}x = \int_{-\pi}^{\pi} \sin^2 nx \,\mathrm{d}x = \pi, \quad \int_{-\pi}^{\pi} 1^2 \,\mathrm{d}x = 2\pi.
\end{gather*}
也称上述三角函数系在 $[-\pi, \pi]$ 上是\textbf{正交函数系}。
\end{proposition}

\begin{lemma}[Riemann-Lebesgue引理]\label{lemma:Riemann-Lebesgue引理}
设 $f(x)\in R[a,b]$，则对$\forall p\in \mathbb{N}$,有
\begin{gather*}
\lim_{n\to+\infty}\int_a^b f(x)\sin px\,\mathrm{d}x=0,
\qquad
\lim_{n\to+\infty}\int_a^b f(x)\cos px\,\mathrm{d}x=0.
\end{gather*}
\end{lemma}
\begin{proof}

先考虑 $\psi(x)$ 有界的情况，这时 $\psi(x)$ Riemann 可积。
对于任意给定的 $\varepsilon > 0$，由\subcref{theorem:Riemann可积的充要条件}{theorem:Riemann可积的充要条件-3}，存在着一种划分
$a = x_0 < x_1 < x_2 < \cdots < x_n = b$，
满足
\begin{align*}
\sum\limits_{i=1}^n \omega_i \Delta x_i < \frac{\varepsilon}{2},
\end{align*}
这里 $\Delta x_i = x_i - x_{i-1}$，$\omega_i$ 是 $\psi(x)$ 在 $[x_{i-1}, x_i]$ 中的振幅。

对于这种固定的划分，记 $m_i$ 是 $\psi(x)$ 在 $[x_{i-1}, x_i]$ 中的下确界，并取实数
$P = \frac{4}{\varepsilon} \left( \sum\limits_{i=1}^n |m_i| \right) > 0$，
则当 $p > P$ 时，有
\begin{align*}
\frac{2}{p} \left( \sum\limits_{i=1}^n |m_i| \right) < \frac{\varepsilon}{2}.
\end{align*}
于是，对于任意给定的 $\varepsilon > 0$，存在实数 $P > 0$，当 $p > P$ 时，有
\begin{align*}
\left| \int_a^b \psi(x) \sin p x \,\mathrm{d}x \right|
&= \left| \sum_{i=1}^n \int_{x_{i-1}}^{x_i} \psi(x) \sin p x \,\mathrm{d}x \right|
= \left| \sum_{i=1}^n \int_{x_{i-1}}^{x_i} (\psi(x) - m_i) \sin p x \,\mathrm{d}x + \sum_{i=1}^n m_i \int_{x_{i-1}}^{x_i} \sin p x \,\mathrm{d}x \right| \\
&\leqslant \sum_{i=1}^n \int_{x_{i-1}}^{x_i} |\psi(x) - m_i| \cdot |\sin p x| \,\mathrm{d}x + \sum_{i=1}^n |m_i| \left| \int_{x_{i-1}}^{x_i} \sin p x \,\mathrm{d}x \right| \\
&\leqslant \sum_{i=1}^n \omega_i \Delta x_i + \frac{2}{p} \left( \sum_{i=1}^n |m_i| \right) < \varepsilon.
\end{align*}

再考虑 $\psi(x)$ 无界的情况，这时 $\psi(x)$ 绝对可积。
不妨假设 $b$ 是 $\psi(x)$ 的唯一奇点。由无界函数反常积分绝对收敛的定义，对于任意给定的 $\varepsilon > 0$，存在 $\delta > 0$，当 $\eta < \delta$ 时，
\begin{align*}
\int_{b-\eta}^{b} |\psi(x)| \,\mathrm{d}x < \frac{\varepsilon}{2}.
\end{align*}
固定 $\eta$，则 $\psi(x)$ 在 $[a, b-\eta]$ 上 Riemann 可积，应用上面的结论，存在实数 $P>0$，当 $p>P$ 时，
\begin{align*}
\left| \int_a^{b-\eta} \psi(x) \sin p x \,\mathrm{d}x \right| < \frac{\varepsilon}{2}.
\end{align*}
因此
\begin{align*}
\left| \int_a^b \psi(x) \sin p x \,\mathrm{d}x \right|
\leqslant \left| \int_a^{b-\eta} \psi(x) \sin p x \,\mathrm{d}x \right| + \int_{b-\eta}^b |\psi(x)| \,\mathrm{d}x 
< \varepsilon.
\end{align*}
所以无论对哪一种情况，都有
\begin{align*}
\lim_{p \to +\infty} \int_a^b \psi(x) \sin p x \,\mathrm{d}x = 0.
\end{align*}

同理可证
\begin{align*}
\lim_{p \to +\infty} \int_a^b \psi(x) \cos p x \,\mathrm{d}x = 0.
\end{align*}
\end{proof}

\begin{corollary}[局部性原理]
可积或绝对可积函数 $f(x)$ 的 Fourier 级数在 $x$ 点是否收敛只与 $f(x)$ 在 $(x-\delta, x+\delta)$ 的性质有关，这里 $\delta$ 是任意小的正常数。
\end{corollary}
\begin{proof}
由于对任意给定的 $\delta > 0$，
$\frac{f(x+u) + f(x-u)}{2\sin \frac{u}{2}}$
关于 $u$ 在 $[\delta, \pi]$ 可积或绝对可积，由 Riemann 引理，
\begin{align*}
\lim_{m \to \infty} \int_{\delta}^\pi [f(x+u) + f(x-u)] \frac{\sin \frac{2m+1}{2}u}{2\sin \frac{u}{2}} \,\mathrm{d}u = 0.
\end{align*}

因此，若将 $S_m(x)$ 的表达式中积分区间分成 $[0, \delta]$ 和 $[\delta, \pi]$ 两部分，则当 $m \to \infty$ 时，
$S_m(x)$ 的敛散性显然只与
$$\frac{1}{\pi} \int_0^\delta [f(x+u) + f(x-u)] \frac{\sin \frac{2m+1}{2}u}{2\sin \frac{u}{2}} \,\mathrm{d}u$$
有关，而这个积分只涉及 $f(x)$ 在 $(x-\delta, x+\delta)$ 的性质。
\end{proof}

\begin{corollary}
设函数 $\psi(u)$ 在 $[0,\delta]$ 上可积或绝对可积，则成立
\begin{align*}
\lim_{m \to \infty} \int_0^\delta \psi(u) \frac{\sin \frac{2m+1}{2}u}{2\sin \frac{u}{2}} \,\mathrm{d}u = \frac{\pi}{2} \psi(0+).
\end{align*}
\end{corollary}

\begin{definition}[分段连续]
设函数 $f$ 在闭区间 $[a,b]$ 上有定义。若存在 $a = x_0 < x_1 < \cdots < x_m = b$，使得 $f$ 在每个开区间 $(x_{i-1}, x_i)$ 上连续，且
\begin{gather*}
f(x_0^+),f(x_m^-)\text{存在且有限},
\\
f(x_i^-),f(x_i^+)\text{存在且有限},i=1,2,\cdots,m-1.
\end{gather*}
则称 $f$ 在 $[a,b]$ 上\textbf{分段连续}.
\end{definition}

\begin{definition}[分段光滑]
设函数 $f$ 在闭区间 $[a,b]$ 上分段连续。若 $f$ 的导函数 $f'$ 在 $[a,b]$ 上分段连续，则称 $f$ 在 $[a,b]$ 上\textbf{分段光滑}.
\end{definition}

\begin{definition}
设函数 $f$ 在 $[a, b]$（或 $(a, b)$）上有定义。如果在 $[a, b]$（或 $(a, b)$）上存在有限个点
\begin{align*}
a = x_0 < x_1 < x_2 < \cdots < x_N = b,
\end{align*}
使得 $f$ 在每个区间 $(x_{i-1}, x_i)$（$i = 1, 2, \cdots, N$）上是单调函数，则称 $f$ 在 $[a, b]$（或 $(a, b)$）上\textbf{分段单调}。
\end{definition}

\begin{definition}
设点 $x$ 是函数 $f(x)$ 的连续点或第一类不连续点，若对于充分小的正数 $\delta$，存在常数 $L > 0$ 和 $\alpha \in (0, 1]$，使得成立
\begin{align*}
|f(x \pm u) - f(x)| < L u^\alpha \quad (0 < u < \delta),
\end{align*}
则称 $f(x)$ 在点 $x$ 处满足指数为 $\alpha \in (0, 1]$ 的 \textbf{Hölder条件}（当 $\alpha = 1$ 也称为 \textbf{Lipschitz条件}）。
\end{definition}

\begin{theorem}[Dirichlet引理]\label{theorem:Dirichlet引理}
设函数 $\psi(u)$ 在 $[0,\delta]$ 上单调，则成立
\begin{align*}
\lim_{p \to +\infty} \int_0^\delta \frac{\psi(u) - \psi(0+)}{u} \sin p u \,\mathrm{d}u = 0.
\end{align*}
\end{theorem}
\begin{remark}
Dirichlet 引理也经常表达为等价形式
\begin{align*}
\lim_{p \to +\infty} \int_{0}^{\delta} \psi(u) \frac{\sin p u}{u} \,\mathrm{d}u = \frac{\pi}{2} \psi(0+).
\end{align*}
如果 $\psi(u)$ 是分段单调有界函数，易知 Dirichlet 引理依然成立。
\end{remark}
\begin{proof}

不妨设 $\psi(x)$ 单调增加。于是对任意给定的 $\varepsilon > 0$，存在 $\eta \in (0, \delta)$，当 $u \in (0, \eta]$ 时，有
\begin{align*}
0 \leqslant \psi(u) - \psi(0+) < \varepsilon.
\end{align*}
将积分分为两部分
\begin{align*}
\int_{0}^{\delta} \frac{\psi(u) - \psi(0+)}{u} \sin p u \,\mathrm{d}u
= \int_{0}^{\eta} \frac{\psi(u) - \psi(0+)}{u} \sin p u \,\mathrm{d}u + \int_{\eta}^{\delta} \frac{\psi(u) - \psi(0+)}{u} \sin p u \,\mathrm{d}u.
\end{align*}

对等式右边的第一项，由\hyperref[theorem:积分中值定理]{积分中值定理}，存在 $\xi \in [0, \eta]$，
\begin{align*}
\left| \int_{0}^{\eta} \frac{\psi(u) - \psi(0+)}{u} \sin p u \,\mathrm{d}u \right|
= \left| \psi(\eta) - \psi(0+) \right| \cdot \left| \int_{\xi}^{\eta} \frac{\sin p u}{u} \,\mathrm{d}u \right| 
< \left| \int_{\xi}^{\eta} \frac{\sin p u}{u} \,\mathrm{d}u \right| \cdot \varepsilon = \left| \int_{p\xi}^{p\eta} \frac{\sin u}{u} \,\mathrm{d}u \right| \cdot \varepsilon.
\end{align*}

利用含参变量积分中已经得到的结论
$\int_{0}^{\infty} \frac{\sin x}{x} \,\mathrm{d}x = \frac{\pi}{2}$，
可知存在与 $p$ 无关的常数 $K$，使得
$\left| \int_{p\xi}^{p\eta} \frac{\sin u}{u} \,\mathrm{d}u \right| < K$，
即
\begin{align*}
\left| \int_{0}^{\eta} \frac{\psi(u) - \psi(0+)}{u} \sin p u \,\mathrm{d}u \right| < K\varepsilon.
\end{align*}

而对右边的第二项，由于
$\frac{\psi(u) - \psi(0+)}{u}$
在 $[\eta, \delta]$ 上显然是可积或绝对可积的，由 Riemann 引理，存在常数 $P > 0$，当 $p > P$ 时，有
\begin{align*}
\left| \int_{\eta}^{\delta} [\psi(u) - \psi(0+)] \frac{\sin p u}{u} \,\mathrm{d}u \right| < \varepsilon.
\end{align*}

综合上述两项估计，即知结论成立。
\end{proof}

\begin{theorem}
设函数 $f(x)$ 在 $[-\pi, \pi]$ 上可积或绝对可积，且满足下列两个条件之一，则 $f(x)$ 的 Fourier 级数在点 $x$ 处收敛于
$\frac{f(x+)+f(x-)}{2}$.
\begin{enumerate}[(1)]
\item \textbf{Dirichlet--Jordan 判别法:} $f(x)$ 在点 $x$ 的某个邻域 $O(x, \delta)$ 上是分段单调有界函数；
\item \textbf{Dini--Lipschitz 判别法:} $f(x)$ 在点 $x$ 处满足指数为 $\alpha \in (0, 1]$ 的 Hölder 条件.
\end{enumerate}
\end{theorem}
\begin{proof}

当 $f(x)$ 满足条件 (1) 时，由 \hyperref[theorem:Dirichlet引理]{Dirichlet引理}知
\begin{align*}
\lim_{p \to +\infty} \int_0^\delta \frac{f(x + u) - f(x+)}{u} \sin p u \,\mathrm{d}u = 0, \\
\lim_{p \to +\infty} \int_0^\delta \frac{f(x - u) - f(x-)}{u} \sin p u \,\mathrm{d}u = 0.
\end{align*}

两式相加，即有
\begin{align*}
\lim_{p \to +\infty} \int_0^\delta \left[ f(x + u) + f(x - u) - f(x+) - f(x-) \right] \frac{\sin p u}{u} \,\mathrm{d}u = 0.
\end{align*}

当 $f(x)$ 满足条件 (2) 时，在 $(0, \delta)$ 上，有
$\frac{|f(x \pm u) - f(x \pm)|}{u} < \frac{L}{u^{1-\alpha}} \quad (0 < \alpha \leqslant 1)$，
所以
\begin{align*}
\frac{\varphi(u, x)}{u} = \frac{f(x + u) - f(x+)}{u} + \frac{f(x - u) - f(x-)}{u}
\end{align*}
在 $[0, \delta]$ 可积或绝对可积，由 \hyperref[lemma:Riemann-Lebesgue引理]{Riemann引理}知
\begin{align*}
\lim_{p \to +\infty} \int_0^\delta \left[ f(x + u) + f(x - u) - f(x+) - f(x-) \right] \frac{\sin p u}{u} \,\mathrm{d}u = 0.
\end{align*}

因此无论哪种情况，$f(x)$ 的 Fourier 级数在点 $x$ 处均收敛于
$\frac{f(x+) + f(x-)}{2}$.
\end{proof}

\begin{corollary}
若 $f(x)$ 在 $[-\pi, \pi]$ 上可积或绝对可积，在点 $x$ 处两个单侧导数 $f'_+(x)$ 和 $f'_-(x)$ 都存在，或更进一步，只要两个拟单侧导数
$\lim\limits_{h \to 0^+} \frac{f(x+h)-f(x+)}{h}$ 和 $\lim\limits_{h \to 0^+} \frac{f(x-h)-f(x-)}{-h}$ 存在，
则 $f(x)$ 的 Fourier 级数在点 $x$ 处收敛于
$\frac{f(x^+) + f(x^-)}{2}$.
\end{corollary}

\begin{theorem}[Fourier级数的逐项积分定理]\label{theorem:Fourier级数的逐项积分定理}
设 $f$ 在 $[-\pi,\pi]$ 上Riemann可积（或绝对可积），其 Fourier 级数为
\[
f(x)\sim\frac{a_0}{2}+\sum_{n=1}^{\infty}(a_n\cos nx + b_n\sin nx).
\]
则对任意 $c,x\in[-\pi,\pi]$，有
\[
\int_{c}^{x}f(t)\,\mathrm{d}t
= \int_{c}^{x}\frac{a_0}{2}\,\mathrm{d}t
+ \sum_{n=1}^{\infty}\int_{c}^{x}\bigl(a_n\cos nt + b_n\sin nt\bigr)\,\mathrm{d}t,
\]
其中右边的级数在 $[-\pi,\pi]$ 上一致收敛。
\end{theorem}

\begin{theorem}[Fourier级数的逐项微分定理]\label{theorem:Fourier级数的逐项微分定理}
设 $f$ 在 $[-\pi,\pi]$ 上连续，且 $f(-\pi)=f(\pi)$。若 $f$ 在 $[-\pi,\pi]$ 上除有限个点外可导，且导函数 $f'$ 在 $[-\pi,\pi]$ 上Riemann可积。设 $f$ 的 Fourier级数为
\[
f(x)\sim\frac{a_0}{2}+\sum_{n=1}^{\infty}(a_n\cos nx + b_n\sin nx),
\]
则 $f'$ 的 Fourier 级数为
\[
f'(x)\sim \frac{\mathrm{d}}{\mathrm{d}x}\left(\frac{a_0}{2}\right)
+\sum_{n=1}^{\infty}\frac{\mathrm{d}}{\mathrm{d}x}\bigl(a_n\cos nx + b_n\sin nx\bigr)
= \sum_{n=1}^{\infty}\bigl(-a_n n\sin nx + b_n n\cos nx\bigr).
\]
\end{theorem}
\begin{remark}
当 $f$ 在 $[-\pi,\pi]$ 上除有限个点外可导时,$f'$在无定义的点可任意补充定义，不影响其可积性.
\end{remark}

\begin{corollary}
若三角级数
$$
\frac{a_0}{2}+\sum_{n=1}^{\infty}(a_n\cos nx + b_n\sin nx)
$$
是某个在 $[-\pi,\pi]$ 上Riemann可积函数的 Fourier 级数，则级数 $\sum_{n=1}^{\infty}\frac{b_n}{n}$ 收敛。
\end{corollary}

\begin{theorem}[Bessel不等式]\label{theorem:Bessel不等式}
设 $f$ 在 $[-\pi,\pi]$ 上平方可积,即 $f^2$是Riemann可积的，则$f$的Fourier 系数$\{a_n\}_{n=0}^{\infty},\{b_n\}_{n=0}^{\infty}$满足
\[
\frac{a_0^2}{2}+\sum_{n=1}^{\infty}(a_n^2 + b_n^2)\leqslant \frac{1}{\pi}\int_{-\pi}^{\pi}f^2(x)\,\mathrm{d}x.
\]
\end{theorem}
\begin{remark}
该不等式表明 Fourier 系数的平方和收敛。
\end{remark}

\begin{theorem}[Parseval恒等式]\label{theorem:Parseval恒等式}
设 $f$ 在 $[-\pi,\pi]$ 上平方可积,即 $f^2$是Riemann可积的，则$f$的Fourier 系数$\{a_n\}_{n=0}^{\infty},\{b_n\}_{n=0}^{\infty}$满足
\[
\frac{a_0^2}{2}+\sum_{n=1}^{\infty}(a_n^2 + b_n^2)=\frac{1}{\pi}\int_{-\pi}^{\pi}f^2(x)\,\mathrm{d}x.
\]
\end{theorem}

\begin{definition}
若函数序列 $\psi_n(x)$ 满足
\begin{align*}
\lim_{n\rightarrow \infty} \left\| f\left( x \right) -\psi _n\left( x \right) \right\| _{2}^{2}=\lim_{n\rightarrow \infty} \int_a^b{\left| f\left( x \right) -\psi _n\left( x \right) \right|^2\mathrm{d}x}=0,
\end{align*}
这里 $f(x)$ 是某一个固定函数，则称 $\psi_n(x)$ 按范数 $\|\cdot\|_2$ \textbf{平方收敛}于 $f(x)$，简称 $\psi_n(x)$ \textbf{平方收敛}于 $f(x)$.
\end{definition}

\begin{corollary}[Fourier级数的平方收敛性质]
设 $f(x)$ 在 $[-\pi, \pi]$ 上可积或平方可积，则 $f(x)$ 的 Fourier 级数的部分和函数序列$\{S_n\}$平方收敛于 $f(x)$,即
\begin{align*}
\lim_{n\rightarrow \infty} \left\| f\left( x \right) -S_n\left( x \right) \right\| _{2}^{2}=\lim_{n\rightarrow \infty} \int_a^b{\left| f\left( x \right) -S_n\left( x \right) \right|^2\mathrm{d}x}=0.
\end{align*}
\end{corollary}

\begin{lemma}\label{lemma:Fourier级数与其导函数的系数关系}
设 $f$ 在 $[-\pi,\pi]$ 上连续可微，且 $f(-\pi)=f(\pi)$。记 $\{a_n\}_{n=0}^{\infty},\{b_n\}_{n=0}^{\infty}$ 为 $f$ 的 Fourier 系数，$\{a_n'\}_{n=0}^{\infty},\{b_n'\}_{n=0}^{\infty}$ 为 $f'$ 的 Fourier 系数。则
\[
a_0' = 0,\qquad a_n' = n b_n,\qquad b_n' = -n a_n\quad (n=1,2,\cdots).
\]
\end{lemma}
\begin{remark}
分部积分的条件，需要$f$的导函数$f'$在积分区域上连续．
\end{remark}
\begin{proof}
由于$f$为$[-\pi,\pi]$上的连续可微函数，因此$f'\in C([-\pi,\pi])$．又$f(\pi)=f(-\pi)$，故
\begin{align*}
a_{0}'&=\frac{1}{\pi}\int_{-\pi}^{\pi}f'(x)\mathrm{d}x=\frac{1}{\pi}f(x)\Big|_{-\pi}^{\pi}=0,\\
a_{n}'&=\frac{1}{\pi}\int_{-\pi}^{\pi}f'(x)\cos nx\mathrm{d}x=\frac{1}{\pi}f(x)\cos nx\Big|_{-\pi}^{\pi}+\frac{n}{\pi}\int_{-\pi}^{\pi}f(x)\sin nx\mathrm{d}x=nb_n(n=1,2,\cdots),\\
b_{n}'&=\frac{1}{\pi}\int_{-\pi}^{\pi}f'(x)\sin nx\mathrm{d}x=\frac{1}{\pi}f(x)\sin nx\Big|_{-\pi}^{\pi}-\frac{n}{\pi}\int_{-\pi}^{\pi}f(x)\cos nx\mathrm{d}x=-na_n(n=1,2,\cdots),
\end{align*}
因此结论得证．
\end{proof}

\begin{example}
设 $f \in R[-\pi, \pi]$，$f$ 以 $2\pi$ 为周期，记
\begin{align*}
a_n = \frac{1}{\pi} \int_{-\pi}^{\pi} f(x) \cos nx\,\mathrm{d}x,\quad b_n = \frac{1}{\pi} \int_{-\pi}^{\pi} f(x) \sin nx\,\mathrm{d}x.
\end{align*}
证明：
\begin{align*}
\sum\limits_{n=1}^{\infty} (a_n^2 + b_n^2) < +\infty.
\end{align*}
\end{example}
\begin{proof}
令
\[
S_N(x)=\frac{a_0}{2}+\sum_{n=1}^N (a_n\cos nx+b_n\sin nx).
\]
由\hyperref[proposition:三角函数系的正交性]{三角函数系的正交性}，计算可得
\begin{align*}
\int_{-\pi}^{\pi} (f(x)-S_N(x))^2\,\mathrm{d}x
&= \int_{-\pi}^{\pi} f^2(x)\,\mathrm{d}x
- \pi\left(\frac{a_0^2}{2}+\sum_{n=1}^N (a_n^2+b_n^2)\right).
\end{align*}
由于左端积分非负，因此
\[
\frac{a_0^2}{2}+\sum_{n=1}^N (a_n^2+b_n^2)
\le \frac{1}{\pi}\int_{-\pi}^{\pi} f^2(x)\,\mathrm{d}x.
\]
又因为 \(f\in R[-\pi,\pi]\)，所以 \(f\) 在 \([-\pi,\pi]\) 上有界，从而 \(f^2\) 也 Riemann 可积，故右端积分是一个有限的常数。于是部分和序列
\[
\left\{\frac{a_0^2}{2}+\sum_{n=1}^N (a_n^2+b_n^2)\right\}_{N=1}^{\infty}
\]
单调递增且有上界，因而收敛。特别地，
\[
\sum_{n=1}^{\infty}(a_n^2+b_n^2) < +\infty.
\]
这就完成了证明。
\end{proof}

\begin{example}
设$f$以$2\pi$为周期且具有二阶连续的导函数,证明$f$的Fourier级数在$(-\infty,+\infty)$上一致收敛于$f$.
\end{example}
\begin{proof}
因为$f(x)$是以$2\pi$为周期的具有二阶连续导数的函数，故$f(x)$,$f'(x)$可展开成傅里叶级数，不妨设
\begin{align*}
f(x) &=\frac{a_0}{2}+\sum_{n=1}^{\infty}(a_n\cos nx+b_n\sin nx),\quad f'(x)=\frac{a_{0}'}{2}+\sum_{n=1}^{\infty}(a_{n}'\cos nx+b_{n}'\sin nx).
\end{align*}
先证$\frac{|a_0|}{2}+\sum_{n=1}^{\infty}(|a_n|+|b_n|)$收敛．由\reflem{lemma:Fourier级数与其导函数的系数关系}可知
$$a_{0}'=0,a_{n}'=nb_n,b_{n}'=-nb_n(n=1,2,\cdots),$$
从而
\begin{align}
|a_n|+|b_n|=\frac{|b_{n}'|}{n}+\frac{|a_{n}'|}{n}\leqslant \frac{1}{2}\left[(b_{n}')^2+\frac{1}{n^2}\right]+\frac{1}{2}\left[(a_{n}')^2+\frac{1}{n^2}\right]
=\frac{1}{n^2}+\frac{1}{2}\left[(a_{n}')^2+(b_{n}')^2\right].\label{100.103}
\end{align}
又由\hyperref[theorem:Bessel不等式]{Bessel不等式}可知
$$\frac{a_{0}'}{2}+\sum_{n=1}^{\infty}\left[(a_{n}')^2+(b_{n}')^2\right]\leqslant \frac{1}{\pi}\int_{-\pi}^{\pi}[f'(x)]^2\mathrm{d}x<+\infty.$$
故$\sum_{n=1}^{\infty}\left[(a_{n}')^2+(b_{n}')^2\right]$收敛．再结合$\sum_{n=1}^{\infty}\frac{1}{n^2}$收敛及\eqref{100.103}式可知$\sum_{n=1}^{\infty}(|a_n|+|b_n|)$收敛，进而$\frac{|a_0|}{2}+\sum_{n=1}^{\infty}(|a_n|+|b_n|)$收敛．注意到
$$\left|\frac{a_0}{2}+\sum_{n=1}^{\infty}(a_n\cos nx+b_n\sin nx)\right|\leqslant \frac{|a_0|}{2}+\sum_{n=1}^{\infty}(|a_n|+|b_n|),\forall x\in(-\infty,+\infty).$$
因此由Weierstrass判别法可知，$\frac{a_0}{2}+\sum_{n=1}^{\infty}(a_n\cos nx+b_n\sin nx)$在$(-\infty,+\infty)$上一致收敛，即$f$的Fourier级数在$(-\infty,+\infty)$上一致收敛于$f$．

\end{proof}
