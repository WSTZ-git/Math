\section{常用初等不等式}

\begin{proposition}[指数/对数函数不等式]\label{proposition:指数对数函数不等式}
\begin{enumerate}[(1)]
\item\label{proposition:指数对数函数不等式1} $\ln \left( 1+x \right) <\frac{x}{\sqrt{1+x}},x>0\iff \ln x<\sqrt{x}-\frac{1}{\sqrt{x}},x>1.$

\item \label{proposition:指数对数函数不等式3} $e^x+e^y-2<e^{x+y}-1,\forall x,y>0.$

\item \label{proposition:指数对数函数不等式4} $e^{-x}\leqslant \frac{1}{1+x}\leqslant e^{x^2-x},\forall x>0.$
\end{enumerate}
\end{proposition}
\begin{proof}
\begin{enumerate}[(1)]
\item 令\(f(x)=\ln(1 + x)-\frac{x}{\sqrt{1 + x}}\)，\(x\geqslant 0\)，则
\begin{align*}
f'\left( x \right) =\frac{2\sqrt{1+x}-x-2}{2\left( 1+x \right) ^{\frac{3}{2}}}=-\frac{1+x-2\sqrt{1+x}+1}{2\left( 1+x \right) ^{\frac{3}{2}}}=-\frac{\left( \sqrt{1+x}-1 \right) ^2}{2\left( 1+x \right) ^{\frac{3}{2}}}<0,\forall x>0.
\end{align*}
故\(f\)在\((0,+\infty)\)上严格单调递减，又\(f\in C[0,+\infty)\)，因此\(f\)在\([0,+\infty)\)上也严格单调递减。从而
\[
f(x)\leqslant f(0)=0,\forall x>0.
\]
即\(\ln(1 + x)<\frac{x}{\sqrt{1 + x}},x>0\).

\item 注意到
\[
(e^x-1)(e^y-1)>0,\forall x,y>0,
\]
故
\begin{align*}
e^x+e^y<e^{x+y}+1\Longrightarrow e^x+e^y-2<e^{x+y}-1,\forall x,y>0.
\end{align*}

\item 由$e^x\geqslant 1+x,\forall x>0$可得
\begin{align*}
e^{-x}&=\frac{1}{e^x}\leqslant \frac{1}{1+y},\forall x>0.
\\
e^{x^2-x}&\geqslant 1+x^2-x=\frac{1+x^3}{1+x}\geqslant \frac{1}{1+x},\forall x>0.
\end{align*}
\end{enumerate}
\end{proof}

\begin{proposition}[经典初等不等式]\label{proposition:经典初等不等式}
\begin{enumerate}[(1)]
\item\label{proposition:经典初等不等式-1} $\left| x^{\alpha}-y^{\alpha} \right|\leqslant \left| x-y \right|^{\alpha},\quad \forall x,y\geqslant 0,\,\,0<\alpha \leqslant 1.$

\item\label{proposition:经典初等不等式-2} 设\(a,b\geqslant0\),则
\begin{align}\label{equation-12.777}
\begin{cases}
a^p + b^p\leqslant(a + b)^p\leqslant2^{p - 1}(a^p + b^p),& p\geqslant1,p\leqslant0\\
a^p + b^p\geqslant(a + b)^p\geqslant2^{p - 1}(a^p + b^p),& 0 < p < 1
\end{cases}
\end{align}
\end{enumerate}
\end{proposition}
\begin{note}
不等式左右是奇次对称的，我们可以设\(t = \frac{a}{b}\in[0,1]\)，于是\eqref{equation-12.777}两边同时除以$b^p$得
\[
\begin{cases}
t^p + 1\leqslant(t + 1)^p\leqslant2^{p - 1}(t^p + 1),& p\geqslant1,p\leqslant0\\
t^p + 1\geqslant(t + 1)^p\geqslant2^{p - 1}(t^p + 1),& 0 < p < 1
\end{cases}.
\]
\end{note}
\begin{proof}
\begin{enumerate}[(1)]
\item {\color{blue}证法一:}不妨设$x\geqslant y>0$,令$t=\frac{x}{y}$,$u=t-1$,则
\[
\left| x^{\alpha}-y^{\alpha} \right|\leqslant \left| x-y \right|^{\alpha}\Longleftrightarrow t^{\alpha}-1\leqslant \left( t-1 \right)^{\alpha}\Longleftrightarrow \left( u+1 \right)^{\alpha}\leqslant u^{\alpha}+1.
\]
再由\hyperref[theorem:Bernoulli不等式]{Bernoulli不等式}知上式成立.

{\color{blue}证法二:}不妨设 $y \geqslant x \geqslant 0$，则只须证 $y^{\alpha} - x^{\alpha} \leqslant \left( y - x \right)^{\alpha}$。则只须证 $\left( \frac{y}{x} \right)^{\alpha} - 1 \leqslant \left( \frac{y}{x} - 1 \right)^{\alpha}$。故只须证
\[
t^{\alpha} - 1 \leqslant \left( t - 1 \right)^{\alpha}, \forall t \geqslant 1.
\]
令 $g\left( t \right) = t^{\alpha} - 1 - \left( t - 1 \right)^{\alpha}$，则
\(
g'\left( t \right) = \alpha t^{\alpha - 1} - \alpha \left( t - 1 \right)^{\alpha - 1} \leqslant 0.
\)
从而 $g\left( t \right) \leqslant g\left( 1 \right) = 0$，$\forall t \geqslant 1$。故
\(
t^{\alpha} - 1 \leqslant \left( t - 1 \right)^{\alpha},\forall t \geqslant 1.
\)

\item 考虑\(f(t)\triangleq\frac{(t + 1)^p}{1 + t^p},t\in[0,1]\)，我们有
\[
f'(t)=p(t + 1)^{p - 1}\frac{1 - t^{p - 1}}{(1 + t^p)^2}
\begin{cases}
\geqslant0,& p\geqslant1,p\leqslant0\\
<0,& 0 < p < 1
\end{cases}
\]
于是
\[
\begin{cases}
2^{p - 1}=f(1)\geqslant f(t)\geqslant f(0)=1,& p\geqslant1,p\leqslant0\\
2^{p - 1}=f(1)\leqslant f(t)\leqslant f(0)=1,& 0 < p < 1
\end{cases}
\]
这就完成了证明.
\end{enumerate}
\end{proof}

\begin{theorem}\label{theorem:三角函数相关初等不等式}
\begin{enumerate}[(1)]
\item\label{theorem:三角函数相关初等不等式-1} 对 $\forall x \in \mathbb{R}$, 有
\begin{align*}
|\sin x| \leqslant |x|.
\end{align*}

\item\label{theorem:三角函数相关初等不等式-2} 对 $\forall x,y \in \mathbb{R}$, 有
\begin{align*}
|\sin x-\sin y|\leqslant |x-y|.
\end{align*}

\item\label{theorem:三角函数相关初等不等式-3} 对 $\forall x,y \in \mathbb{R}$, 有
\begin{align*}
|\cos x-\cos y|\leqslant |x-y|.
\end{align*}

\item\label{theorem:三角函数相关初等不等式-4} 对 $\forall x \in (0,\frac{\pi}{2})$, 有
\begin{align*}
\sin x < x < \tan x.
\end{align*}

\item\label{theorem:三角函数相关初等不等式-5} 对 $\forall x,y \in \mathbb{R}$, 有
\begin{align*}
|\arctan x-\arctan y|\leqslant |x-y|.
\end{align*}

\item\label{theorem:三角函数相关初等不等式-6} (\textbf{Jordan不等式}) 对 $\forall x \in [0,\frac{\pi}{2}]$, 有
\begin{align*}
\frac{2}{\pi}x \leqslant \sin x \leqslant x.
\end{align*}

\item\label{theorem:三角函数相关初等不等式-7} 对 $\forall x \in \mathbb{R}$, 有
\begin{align*}
\cos x \geqslant 1 - \frac{x^2}{2}.
\end{align*}
\end{enumerate}
\end{theorem}
\begin{proof}
\begin{enumerate}[(1)]
\item 当 $x=0$ 时, 显然成立. 当 $x \neq 0$ 时, 由Lagrange中值定理, 存在 $\xi$ 介于 $0$ 与 $x$ 之间, 使得
\begin{align*}
|\sin x| = |\cos \xi||x| \leqslant |x|.
\end{align*}
\item 由和差化积公式, 有
\begin{align*}
|\sin x-\sin y| = 2\left|\cos\frac{x+y}{2}\sin\frac{x-y}{2}\right| \leqslant 2\left|\sin\frac{x-y}{2}\right| \leqslant 2\left|\frac{x-y}{2}\right| = |x-y|.
\end{align*}
\item 由和差化积公式, 有
\begin{align*}
|\cos x-\cos y| = 2\left|\sin\frac{x+y}{2}\sin\frac{x-y}{2}\right| \leqslant 2\left|\sin\frac{x-y}{2}\right| \leqslant 2\left|\frac{x-y}{2}\right| = |x-y|.
\end{align*}
\item 令 $f(x)=x-\sin x$, 则 $f'(x)=1-\cos x \geqslant 0$, 且 $f(0)=0$, 故当 $x>0$ 时 $f(x)>0$, 即 $\sin x < x$. 令 $g(x)=\tan x-x$, 则 $g'(x)=\sec^2 x-1=\tan^2 x>0$, 且 $g(0)=0$, 故当 $x>0$ 时 $g(x)>0$, 即 $x<\tan x$. 综上 $\sin x<x<\tan x$.
\item 由Lagrange中值定理, 存在 $\xi$ 介于 $x,y$ 之间, 使得
\begin{align*}
|\arctan x-\arctan y| = \left|\frac{1}{1+\xi^2}\right||x-y| \leqslant |x-y|.
\end{align*}
\item 令 $f(x)=\sin x-\frac{2}{\pi}x$, 则 $f'(x)=\cos x-\frac{2}{\pi}$, $f''(x)=-\sin x \leqslant 0$. 故 $f'(x)$ 在 $[0,\frac{\pi}{2}]$ 上单调递减. 又 $f'(0)=1-\frac{2}{\pi}>0$, $f'(\frac{\pi}{2})=-\frac{2}{\pi}<0$, 故存在唯一 $x_0 \in (0,\frac{\pi}{2})$ 使得 $f'(x_0)=0$. 从而 $f(x)$ 在 $[0,x_0]$ 上单调递增, 在 $[x_0,\frac{\pi}{2}]$ 上单调递减. 由于 $f(0)=0$, $f(\frac{\pi}{2})=0$, 故 $f(x) \geqslant 0$, 即 $\sin x \geqslant \frac{2}{\pi}x$. 结合 $\sin x \leqslant x$, 可得 Jordan 不等式.
\item 令 $g(x)=\cos x-1+\frac{x^2}{2}$, 则 $g'(x)=x-\sin x$. 由 (1) 知当 $x \geqslant 0$ 时 $x-\sin x \geqslant 0$, 故 $g(x)$ 在 $[0,+\infty)$ 上单调递增. 又 $g(0)=0$, 故 $g(x) \geqslant 0$, 即 $\cos x \geqslant 1-\frac{x^2}{2}$. 由于 $\cos x$ 与 $1-\frac{x^2}{2}$ 均为偶函数, 故对 $\forall x \in \mathbb{R}$, 不等式成立.
\end{enumerate}
\end{proof}
