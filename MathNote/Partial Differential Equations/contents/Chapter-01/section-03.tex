\section{适定性问题}

\begin{definition}
如果一个偏微分方程定解问题满足下列条件：
\begin{enumerate}[(1)]
\item 它的解存在；
\item 它的解惟一；
\item 它的解连续地依赖定解条件和定解问题中的已知函数，
\end{enumerate}
则称这个定解问题是\textbf{适定的}；否则称这个定解问题是\textbf{不适定的}。
\end{definition}

当求解位势方程、热方程和波动方程的定解问题时，我们总是先假设定解问题的解具有非常好的性质（如光滑性）；接着我们求出定解问题的解的表达式。这样的解我们称为定解问题的\textbf{形式解}。然后我们再严格证明在定解条件满足一定的要求时，所得到的形式解的确是定解问题的古典解，从而获得解的存在性。在研究位势方程、热方程和波动方程的定解问题的解的惟一性和稳定性时，我们先导出一些关于解的\textbf{先验估计}；然后再利用这些估计推导出定解问题的解的惟一性和稳定性。这样，我们就基本上回答了关于位势方程、热方程和波动方程的一些定解问题的适定性问题。

\begin{definition}
设 $\Omega\subset \mathbb{R}^n$,我们用 $\overline{\Omega}$ 表示它的闭包，$\partial\Omega$ 表示它的边界。同时，我们用
\begin{align*}
\mathbb{R}_+^n = \{x=(x_1,\cdots,x_n)\in\mathbb{R}^n \mid x_n > 0\}
\end{align*}
表示 $n$ 维欧氏空间 $\mathbb{R}^n$ 中的上半空间。由于上半空间 $\mathbb{R}_+^n$ 的边界
\begin{align*}
\partial\mathbb{R}_+^n = \{x=(x_1,\cdots,x_n)\in\mathbb{R}^n \mid x_n=0\}
\end{align*}
是 $n-1$ 维欧氏空间，我们也记 $\partial\mathbb{R}_+^n = \mathbb{R}^{n-1}$。当 $n=1$ 时，我们将 $\mathbb{R}_+^1$ 简记为 $\mathbb{R}_+$，因而 $\mathbb{R}_+^{n+1} = \mathbb{R}^n \times \mathbb{R}_+$。
\end{definition}

\begin{theorem}\label{theorem:Rn中球体积和面积公式}
\begin{enumerate}[(1)]
\item 我们用 $B(x,r)$ 表示 $n$ 维欧氏空间 $\mathbb{R}^n$ 中以 $x$ 为中心，以 $r$ 为半径的开球，其体积为
\begin{align*}
\int_{B\left( 0,r \right)}{\mathrm{d}x}=\alpha(n)r^n = \frac{2\pi^{\frac{n}{2}}}{n\Gamma\left(\frac{n}{2}\right)}r^n,
\end{align*}
其中 $\alpha (n)\triangleq \int_{B(0,1)}{\mathrm{d}x}$ 表示 $\mathbb{R}^n$ 上单位球的体积.

\item 我们用 $S^{n-1}(x,r)=\partial B(x,r)$ 表示球 $B(x,r)$ 的边界，即 $n-1$ 维球面，其面积为
\begin{align*}
\int_{\mathbb{S}^{n-1}\left( 0,r \right)}{\mathrm{d}x}=\omega(n)r^{n-1} = n\alpha(n)r^{n-1}=\frac{2\pi^{\frac{n}{2}}}{n\Gamma\left(\frac{n}{2}\right)}r^{n-1},
\end{align*}
其中 $w\left( n \right) \triangleq \int_{\mathbb{S} ^{n-1}(0,1)}{\mathrm{d}x}$ 表示 $\mathbb{R}^n$ 上单位球的面积.
\end{enumerate}
\end{theorem}
\begin{proof}
\begin{enumerate}[(1)]
\item 先计算单位球的体积.假设结论对小于$n$的情况都成立,现在考虑$n$的情形.
\begin{align*}
\alpha (n)&=|B(0,1)|=\int_{x_{1}^{2}+\cdots +x_{n-1}^{2}+z^2\leqslant 1}{\mathrm{d}x_1\cdots \mathrm{d}x_{n-1}\mathrm{d}z}\\
&=\int_{-1}^1{\mathrm{d}z\int_{x_{1}^{2}+\cdots +x_{n-1}^{2}\leqslant 1-z^2}{\mathrm{d}x_1\cdots \mathrm{d}x_{n-1}}}=\int_{-1}^1{\left| B_{n-1}\left( (0,\cdots ,0,z),\sqrt{1-z^2} \right) \right|\mathrm{d}z}\\
&=2\int_0^1{\alpha (n}-1)(1-z^2)^{\frac{n-1}{2}}\mathrm{d}z\xlongequal{u=z^2}2\alpha (n-1)\int_0^1{(1}-u)^{\frac{n-1}{2}}\frac{1}{2}u^{-\frac{1}{2}}\mathrm{d}u\\
&=\alpha (n-1)B\left( \frac{n+1}{2},\frac{1}{2} \right) =\alpha (n-1)\frac{\Gamma \left( \frac{n+1}{2} \right) \Gamma \left( \frac{1}{2} \right)}{\Gamma \left( \frac{n+2}{2} \right)}\\
&=\alpha \left( n-2 \right) \frac{\Gamma \left( \frac{n}{2} \right) \Gamma \left( \frac{1}{2} \right)}{\Gamma \left( \frac{n+1}{2} \right)}\frac{\Gamma \left( \frac{n+1}{2} \right) \Gamma \left( \frac{1}{2} \right)}{\Gamma \left( \frac{n+2}{2} \right)}=\alpha \left( 1 \right) \prod_{k=2}^n{\frac{\Gamma \left( \frac{k+1}{2} \right) \Gamma \left( \frac{1}{2} \right)}{\Gamma \left( \frac{k+2}{2} \right)}}\\
&=\frac{\Gamma \left( \frac{3}{2} \right) \left[ \Gamma \left( \frac{1}{2} \right) \right] ^{n-1}}{\Gamma \left( \frac{n+2}{2} \right)}\cdot \frac{2\pi ^{\frac{1}{2}}}{\Gamma (\frac{1}{2})}=\frac{\frac{1}{2}\left[ \Gamma \left( \frac{1}{2} \right) \right] ^n}{\Gamma \left( \frac{n}{2}+1 \right)}\cdot 2\quad (\Gamma (s+1)=s\Gamma (s),s>0)\\
&=\frac{\pi ^{\frac{n}{2}}}{\Gamma \left( \frac{n}{2}+1 \right)}=\frac{2\pi ^{\frac{n}{2}}}{n\Gamma (\frac{n}{2})}\quad \left( \Gamma (s)\Gamma (1-s)=\frac{\pi}{\sin \pi s},0<s<1 \right) .\\
\end{align*}

现在对$\forall r>0$,考虑$\mathbb{R}^{n}$上的变换
\begin{align*}
T:B(0,1)\to B(0,r),\quad x\mapsto ry.
\end{align*}
则这个变换的Jacobian行列式的绝对值为
\begin{align*}
|J| = \left|\left| \frac{\partial x_i}{\partial y_j} \right|_{n\times n} \right|=r^n.
\end{align*}
故
\begin{align*}
\int_{B\left( 0,r \right)}{\mathrm{d}x}=\int_{B\left( 0,1 \right)}{\left| J \right|\,\mathrm{d}y}=r^n\int_{B\left( 0,1 \right)}{\mathrm{d}x}=\alpha \left( n \right) r^n.
\end{align*}

\item 定义球坐标变换
\[
\Phi_n:(r,\theta_1,\cdots,\theta_{n-1})\mapsto(x_1,\cdots,x_n)\in\mathbb{R}^n
\]
如下:
\[
\begin{cases}
x_1=r\cos\theta_1,\\
x_2=r\sin\theta_1\cos\theta_2,\\
x_3=r\sin\theta_1\sin\theta_2\cos\theta_3,\\
\cdots\\
x_{n-1}=r\sin\theta_1\cdots\sin\theta_{n-2}\cos\theta_{n-1},\\
x_n=r\sin\theta_1\cdots\sin\theta_{n-2}\sin\theta_{n-1},
\end{cases}
\]
其中 $r>0$，$\theta_1,\cdots,\theta_{n-2}\in(0,\pi)$，$\theta_{n-1}\in(0,2\pi)$.
先用数学归纳法证该变换的雅可比行列式的绝对值为
\begin{align}
|\det J\Phi_n|=\left|\det\frac{\partial(x_1,\cdots,x_n)}{\partial(r,\theta_1,\cdots,\theta_{n-1})}\right|
=r^{n-1}\sin^{n-2}\theta_1\,\sin^{n-3}\theta_2\cdots\sin\theta_{n-2}.\label{eq:jacobianRn}
\end{align}
对 $n$ 进行归纳.
当 $n=2$ 时，映射为极坐标
\[
x_1=r\cos\theta_1,\qquad x_2=r\sin\theta_1,
\]
其雅可比行列式为
\[
|\det J\Phi_2|=\begin{vmatrix}
\cos\theta_1 & -r\sin\theta_1\\
\sin\theta_1 & r\cos\theta_1
\end{vmatrix}=r,
\]
与 \eqref{eq:jacobianRn} 一致.
假设结论对 $n-1$ 成立，考虑 $n$ 维情形. 将 $\Phi_n$ 写成
\[
x_1=r\cos\theta_1,\qquad (x_2,\cdots,x_n)=r\sin\theta_1\cdot y,
\]
其中 $y=(y_1,\cdots,y_{n-1})\in\mathbb{S}^{n-2}\subset\mathbb{R}^{n-1}$ 由角度 $\theta_2,\cdots,\theta_{n-1}$ 参数化，即
\[
y=\Phi_{n-1}(1,\theta_2,\cdots,\theta_{n-1})
\]
是 $n-1$ 维单位球面上的点. 记 $J_y$ 为 $y$ 关于 $\theta_2,\cdots,\theta_{n-1}$ 的Jacobian矩阵,即半径为 $1$ 的 $n-1$ 维球坐标变换的Jacobian矩阵. 则 $\Phi_n$ 的Jacobian矩阵为分块矩阵
\begin{align*}
J\Phi_n=
\begin{pmatrix}
\cos\theta_1 & -r\sin\theta_1 & \bm{0}^T \\
\sin\theta_1\, y & r\cos\theta_1\, y & r\sin\theta_1\cdot J_y
\end{pmatrix},
\end{align*}
其中第一行是标量，第二行是 $(n-1)\times n$ 子矩阵.按最后一列展开得
\begin{align*}
|\det J\Phi_n|
=\left|\det\begin{pmatrix}
\cos\theta_1 & -r\sin\theta_1\\
\sin\theta_1 & r\cos\theta_1
\end{pmatrix}\right|
\cdot \left|\det(r\sin\theta_1 J_y)\right| 
=r\cdot (r\sin\theta_1)^{n-2}|\det J_y|.
\end{align*}
由归纳假设知
\[
|\det J_y|=\sin^{n-3}\theta_2\,\sin^{n-4}\theta_3\cdots\sin\theta_{n-2}.
\]
代入得
\[
|\det J\Phi_n|
=r\cdot r^{n-2}\sin^{n-2}\theta_1\cdot \sin^{n-3}\theta_2\cdots\sin\theta_{n-2}
=r^{n-1}\sin^{n-2}\theta_1\,\sin^{n-3}\theta_2\cdots\sin\theta_{n-2}.
\]
故由数学归纳法知\eqref{eq:jacobianRn}式成立.
再记
\begin{align*}
J(\theta)=\sin^{n-2}\theta_1\,\sin^{n-3}\theta_2\cdots\sin\theta_{n-2},\quad \mathrm{d}\sigma(\omega)
\triangleq J(\theta)\,\mathrm{d}\theta_1\cdots\mathrm{d}\theta_{n-1},
\end{align*}
又由\eqref{eq:jacobianRn}式可得,$\mathbb{R}^n$中的体积微元为
\begin{align}
\mathrm{d}x=|\det J\Phi_n|\,\mathrm{d}r\mathrm{d}\theta_1\cdots\mathrm{d}\theta_{n-1}=r^{n-1}\,\mathrm{d}r\mathrm{d}\sigma(\omega).\label{eq:volumeelement2}
\end{align}
特别地, 取$\Phi_n$中的$r=1$, 则由\eqref{eq:jacobianRn}式知, $\mathbb{R}^n$上的变换
\begin{align*}
\Phi_n(1): (1,\theta_1,\cdots,\theta_{n-1}) \mapsto (x_1,\cdots,x_n),
\end{align*}
的 Jacobian 行列式的绝对值为
\begin{align*}
J(\theta)=\sin^{n-2}\theta_1\,\sin^{n-3}\theta_2\cdots \sin\theta_{n-2}.
\end{align*}
于是
\begin{align}
w(n)=\int_{\mathbb{S}^{n-1}(0,1)} \mathrm{d}x
=\int_{\mathbb{S}^{n-1}(0,1)} J(\theta)\,\mathrm{d}\theta_1\cdots\mathrm{d}\theta_{n-1}
=\int_{\mathbb{S}^{n-1}(0,1)} \mathrm{d}\sigma(\omega).\label{eq:ijijiiiwwwjeeeee}
\end{align}
现在计算单位球 $B(0,1)$ 的体积. 利用 \eqref{eq:volumeelement2}\eqref{eq:ijijiiiwwwjeeeee}有
\begin{align*}
\alpha(n)=\int_{B(0,1)}\mathrm{d}x
=\int_0^1\int_{\mathbb{S}^{n-1}(0,1)} r^{n-1}\,\mathrm{d}\sigma(\omega)\,\mathrm{d}r=\frac{1}{n}\int_{\mathbb{S}^{n-1}(0,1)}\mathrm{d}\sigma(\omega)=\frac{\omega(n)}{n}.
\end{align*}
整理即得
\[
\omega(n)=n\alpha(n).
\]
现在对$\forall r>0$,考虑$\mathbb{R}^{n-1}$上的变换
\begin{align*}
T:\mathbb{S}^{n-1}\left( 0,1 \right)\to \mathbb{S}^{n-1}\left( 0,r\right),\quad x\mapsto ry.
\end{align*}
则这个变换的Jacobian行列式的绝对值为
\begin{align*}
|J| = \left|\left| \frac{\partial x_i}{\partial y_j} \right|_{n-1\times n-1} \right|=r^{n-1}.
\end{align*}
故
\begin{align*}
\int_{\mathbb{S}^{n-1}\left( 0,r \right)}{\mathrm{d}x}=\int_{\mathbb{S}^{n-1}\left( 0,1 \right)}{\left| J \right|\,\mathrm{d}y}=r^{n-1}\int_{\mathbb{S}^{n-1}\left( 0,1 \right)}{\mathrm{d}x}=\omega \left( n \right) r^{n-1}.
\end{align*}
\end{enumerate}
\end{proof}
