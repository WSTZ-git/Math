\begin{theorem}[向量值函数求导的基本性质]\label{theorem:向量求导基本性质}
设 $U\subseteq\mathbb{R}^n$ 为开集，$x\in U$。
\begin{enumerate}[(1)]
\item 设 $A$ 为 $m\times n$ 常矩阵，$b\in\mathbb{R}^m$。定义 $f:U\to\mathbb{R}^m$ 为
\begin{align*}
f(x)\triangleq Ax+b.
\end{align*}
则 $f$ 在 $U$ 上可微，且其 Jacobi 矩阵为
\begin{align*}
Df(x)=A,\qquad \forall x\in U.
\end{align*}
特别地，恒等映射 $f(x)=x$ 的 Jacobi 矩阵为 $Df(x)=I_n$。常值映射$f:U\to\mathbb{R}^m$ 的 Jacobi 矩阵为$Df(x)=0_{m\times n}$.

\item 设 $a\in\mathbb{R}^n$ 为常向量，定义 $f:U\to\mathbb{R}$ 为
\begin{align*}
f(x)\triangleq a^T x.
\end{align*}
则 $f$ 在 $U$ 上可微，且其梯度为
\begin{align*}
\nabla f(x)=a,\qquad \forall x\in U.
\end{align*}

\item 设 $A$ 为 $n\times n$ 常矩阵，定义 $f:U\to\mathbb{R}$ 为
\begin{align*}
f(x)\triangleq x^T A x.
\end{align*}
则 $f$ 在 $U$ 上可微，且其梯度为
\begin{align*}
\nabla f(x)=(A+A^T)x,\qquad \forall x\in U.
\end{align*}
若约定 $\frac{\partial f}{\partial x}$ 为行向量，则
\begin{align*}
\frac{\partial f}{\partial x}=x^T(A+A^T).
\end{align*}

\item 设 $f:U\to\mathbb{R}^m$ 在 $x$ 处可微，$g:f(U)\to\mathbb{R}^p$ 在 $f(x)$ 处可微。则复合函数 $g\circ f$ 在 $x$ 处可微，且
\begin{align*}
D(g\circ f)(x)=Dg(f(x))Df(x).
\end{align*}

\item 设 $f_1,\cdots,f_k:U\to\mathbb{R}^m$ 均在 $x$ 处可微，$c_1,\cdots,c_k\in\mathbb{R}$。则
\begin{align*}
f\triangleq\sum\limits_{i=1}^k c_i f_i
\end{align*}
在 $x$ 处可微，且
\begin{align*}
Df(x)=\sum\limits_{i=1}^k c_i Df_i(x).
\end{align*}
\end{enumerate}
\end{theorem}
\begin{proof}
\begin{enumerate}[(1)]
\item 对任意 $h\in\mathbb{R}^n$，有
\begin{align*}
f(x+h)-f(x)=A(x+h)+b-(Ax+b)=Ah.
\end{align*}
由于 $Ah$ 关于 $h$ 是线性的，且 $|Ah|\leqslant\|A\||h|$，故
\begin{align*}
f(x+h)=f(x)+Ah+o(|h|)\qquad (|h|\to 0).
\end{align*}
因此 $f$ 在 $x$ 处可微，且 $Df(x)=A$。

\item 对任意 $h\in\mathbb{R}^n$，有
\begin{align*}
f(x+h)-f(x)=a^T(x+h)-a^T x=a^T h.
\end{align*}
故
\begin{align*}
f(x+h)=f(x)+a^T h+o(|h|)\qquad (|h|\to 0).
\end{align*}
因此 $f$ 在 $x$ 处可微，且 $\nabla f(x)=a$。

\item 展开
\begin{align*}
f(x+h)=(x+h)^T A(x+h)=x^T A x+x^T A h+h^T A x+h^T A h.
\end{align*}
由于 $h^T A h$ 是 $h$ 的二次项，故 $h^T A h=o(|h|)$。又 $x^T A h$ 和 $h^T A x$ 均为 $h$ 的线性项，且 $h^T A x=x^T A^T h$。因此
\begin{align*}
f(x+h)=f(x)+(x^T A+x^T A^T)h+o(|h|).
\end{align*}
故梯度（列向量）为
\begin{align*}
\nabla f(x)=(A+A^T)x.
\end{align*}
若 $\frac{\partial f}{\partial x}$ 为行向量，则
\begin{align*}
\frac{\partial f}{\partial x}=x^T(A+A^T).
\end{align*}

\item 由可微性定义，存在线性映射 $Df(x)$ 和 $Dg(f(x))$，使得
\begin{align*}
f(x+h)&=f(x)+Df(x)h+o(|h|),\\
g(f(x)+k)&=g(f(x))+Dg(f(x))k+o(|k|).
\end{align*}
取 $k=Df(x)h+o(|h|)$，则 $|k|=O(|h|)$。代入得
\begin{align*}
(g\circ f)(x+h)&=g(f(x)+Df(x)h+o(|h|))\\
&=g(f(x))+Dg(f(x))(Df(x)h+o(|h|))+o(|h|)\\
&=g(f(x))+Dg(f(x))Df(x)h+o(|h|).
\end{align*}
故 $g\circ f$ 在 $x$ 处可微，且
\begin{align*}
D(g\circ f)(x)=Dg(f(x))Df(x).
\end{align*}

\item 由导数的线性性直接得到：
\begin{align*}
f(x+h)-f(x)=\sum\limits_{i=1}^k c_i(f_i(x+h)-f_i(x))
=\sum\limits_{i=1}^k c_i(Df_i(x)h+o(|h|))
=\left(\sum\limits_{i=1}^k c_i Df_i(x)\right)h+o(|h|).
\end{align*}
故
\begin{align*}
Df(x)=\sum\limits_{i=1}^k c_i Df_i(x).
\end{align*}
\end{enumerate}
\end{proof}

\begin{theorem}[弱导数链式法则]\label{theorem:弱导数链式法则}
设 $\Omega,\Omega'\subseteq\mathbb{R}^n$ 为开集，$\Phi:\Omega'\to\Omega$ 为 $C^1$ 微分同胚，且 $\Phi$ 与 $\Phi^{-1}$ 的 Jacobi 矩阵在各自定义域上局部有界。设 $u\in W^{1,p}(\Omega)$，$1\leqslant p\leqslant\infty$。则 $u\circ\Phi\in W^{1,p}(\Omega')$，且对几乎处处的 $y\in\Omega'$，有
\begin{align*}
D(u\circ\Phi)(y)=Du(\Phi(y))D\Phi(y).
\end{align*}
\end{theorem}
\begin{proof}

记 $\Psi\triangleq\Phi^{-1}:\Omega\to\Omega'$。因为 $\Phi$ 与 $\Psi$ 均为 $C^1$ 微分同胚，所以 $|\det D\Phi|$ 与 $|\det D\Psi|$ 在各自定义域上局部有界且不为零。
首先证明 $u\circ\Phi\in L^p_{\mathrm{loc}}(\Omega')$。任取紧集 $K\subseteq\Omega'$，则 $\Phi(K)\subseteq\Omega$ 为紧集。由变量替换公式，得
\begin{align*}
\int_K |u(\Phi(y))|^p\,\mathrm dy
=\int_{\Phi(K)}|u(x)|^p|\det D\Psi(x)|\,\mathrm dx.
\end{align*}
由于 $u\in L^p_{\mathrm{loc}}(\Omega)$，且 $|\det D\Psi|$ 在 $\Phi(K)$ 上有界，故右端有限，从而 $u\circ\Phi\in L^p_{\mathrm{loc}}(\Omega')$。

下面验证弱导数公式。对任意测试函数 $\varphi\in C_c^\infty(\Omega')$，由弱导数的定义，需要证明
\begin{align}
\int_{\Omega'}u(\Phi(y))D_i\varphi(y)\,\mathrm dy
=-\int_{\Omega'}\sum_{j=1}^n D_j u(\Phi(y))D_i\Phi_j(y)\varphi(y)\,\mathrm dy,
\label{eq:弱链式目标}
\end{align}
其中 $i=1,\cdots,n$。

做变量替换 $x=\Phi(y)$，即 $y=\Psi(x)$，$\mathrm dy=|\det D\Psi(x)|\,\mathrm dx$。令
\begin{align*}
\psi(x)\triangleq\varphi(\Psi(x)).
\end{align*}
则 $\psi\in C_c^\infty(\Omega)$。由链式法则得
\begin{align*}
D_i\varphi(y)=\sum_{j=1}^n D_j\psi(x)D_i\Phi_j(y),
\end{align*}
其中 $y=\Psi(x)$。于是 \eqref{eq:弱链式目标} 式左端变为
\begin{align*}
\int_{\Omega'}u(\Phi(y))D_i\varphi(y)\,\mathrm dy
=\int_\Omega u(x)\sum_{j=1}^n D_j\psi(x)D_i\Phi_j(\Psi(x))|\det D\Psi(x)|\,\mathrm dx
=\sum_{j=1}^n\int_\Omega u(x)a_{ji}(x)D_j\psi(x)\,\mathrm dx,
\end{align*}
其中
\begin{align*}
a_{ji}(x)\triangleq D_i\Phi_j(\Psi(x))|\det D\Psi(x)|.
\end{align*}
注意到 $D\Phi(\Psi(x))=(D\Psi(x))^{-1}$，故
\begin{align*}
D_i\Phi_j(\Psi(x))=\bigl((D\Psi(x))^{-1}\bigr)_{ji}.
\end{align*}
因此
\begin{align*}
a_{ji}(x)=\bigl((D\Psi(x))^{-1}\bigr)_{ji}|\det D\Psi(x)|.
\end{align*}

对每个 $j$，利用 $u$ 的弱导数定义，得
\begin{align*}
-\int_\Omega u(x)a_{ji}(x)D_j\psi(x)\,\mathrm dx
=\int_\Omega D_j u(x)a_{ji}(x)\psi(x)\,\mathrm dx
+\int_\Omega u(x)D_j a_{ji}(x)\psi(x)\,\mathrm dx.
\end{align*}
因此 \eqref{eq:弱链式目标} 式左端等于
\begin{align*}
-\sum_{j=1}^n\int_\Omega u(x)a_{ji}(x)D_j\psi(x)\,\mathrm dx=\sum_{j=1}^n\int_\Omega D_j u(x)a_{ji}(x)\psi(x)\,\mathrm dx
+\sum_{j=1}^n\int_\Omega u(x)D_j a_{ji}(x)\psi(x)\,\mathrm dx.
\end{align*}
下面证明第二项为零，即
\begin{align}
\sum_{j=1}^n D_j a_{ji}(x)=0.
\label{eq:Piola恒等式}
\end{align}
事实上，由 Jacobi 公式，得
\begin{align*}
D_j|\det D\Psi(x)|
=|\det D\Psi(x)|\sum_{k,l=1}^n\bigl((D\Psi(x))^{-1}\bigr)_{lk}D_jD_k\Psi_l(x).
\end{align*}
另一方面，对逆矩阵的元素求导，得
\begin{align*}
D_j\bigl((D\Psi(x))^{-1}\bigr)_{ji}
=-\sum_{k,l=1}^n\bigl((D\Psi(x))^{-1}\bigr)_{jk}D_jD_k\Psi_l(x)\bigl((D\Psi(x))^{-1}\bigr)_{li}.
\end{align*}
于是
\begin{align*}
&D_j a_{ji}(x)
=D_j\left(\bigl((D\Psi(x))^{-1}\bigr)_{ji}|\det D\Psi(x)|\right)\\
&=|\det D\Psi(x)|\left[
D_j\bigl((D\Psi(x))^{-1}\bigr)_{ji}
+\bigl((D\Psi(x))^{-1}\bigr)_{ji}\sum_{k,l=1}^n\bigl((D\Psi(x))^{-1}\bigr)_{lk}D_jD_k\Psi_l(x)
\right]\\
&=|\det D\Psi(x)|\left[
-\sum_{k,l=1}^n\bigl((D\Psi(x))^{-1}\bigr)_{jk}D_jD_k\Psi_l(x)\bigl((D\Psi(x))^{-1}\bigr)_{li}
+\sum_{k,l=1}^n\bigl((D\Psi(x))^{-1}\bigr)_{ji}\bigl((D\Psi(x))^{-1}\bigr)_{lk}D_jD_k\Psi_l(x)
\right].
\end{align*}
对 $j$ 求和，并交换求和次序，得到
\begin{align*}
\sum_{j=1}^n D_j a_{ji}(x)
&=|\det D\Psi(x)|\sum_{k,l=1}^n D_jD_k\Psi_l(x)
\left[
-\sum_{j=1}^n\bigl((D\Psi(x))^{-1}\bigr)_{jk}\bigl((D\Psi(x))^{-1}\bigr)_{li}
+\sum_{j=1}^n\bigl((D\Psi(x))^{-1}\bigr)_{ji}\bigl((D\Psi(x))^{-1}\bigr)_{lk}
\right].
\end{align*}
由于 $D_jD_k\Psi_l=D_kD_j\Psi_l$，且
\begin{align*}
\sum_{j=1}^n\bigl((D\Psi(x))^{-1}\bigr)_{jk}\bigl((D\Psi(x))^{-1}\bigr)_{li}
=\delta_{kl},
\end{align*}
以及
\begin{align*}
\sum_{j=1}^n\bigl((D\Psi(x))^{-1}\bigr)_{ji}\bigl((D\Psi(x))^{-1}\bigr)_{lk}
=\delta_{il},
\end{align*}
括号内两项相等，故其差为零。因此 \eqref{eq:Piola恒等式} 式成立。于是
\begin{align*}
-\sum_{j=1}^n\int_\Omega u(x)a_{ji}(x)D_j\psi(x)\,\mathrm dx
=\sum_{j=1}^n\int_\Omega D_j u(x)a_{ji}(x)\psi(x)\,\mathrm dx.
\end{align*}
换回 $y$ 坐标。由 $x=\Phi(y)$，$\mathrm dx=|\det D\Phi(y)|\,\mathrm dy$，且
\begin{align*}
a_{ji}(\Phi(y))|\det D\Phi(y)|
=D_i\Phi_j(y)|\det D\Psi(\Phi(y))||\det D\Phi(y)|
=D_i\Phi_j(y),
\end{align*}
因为 $|\det D\Psi(\Phi(y))||\det D\Phi(y)|=1$。因此
\begin{align*}
\sum_{j=1}^n\int_\Omega D_j u(x)a_{ji}(x)\psi(x)\,\mathrm dx
=\sum_{j=1}^n\int_{\Omega'}D_j u(\Phi(y))D_i\Phi_j(y)\varphi(y)\,\mathrm dy.
\end{align*}
这正是 \eqref{eq:弱链式目标} 式右端。故
\begin{align*}
\int_{\Omega'}u(\Phi(y))D_i\varphi(y)\,\mathrm dy
=-\int_{\Omega'}\sum_{j=1}^n D_j u(\Phi(y))D_i\Phi_j(y)\varphi(y)\,\mathrm dy.
\end{align*}
由 $i$ 的任意性，得到
\begin{align*}
D(u\circ\Phi)(y)=Du(\Phi(y))D\Phi(y),\qquad \text{a.e. } y\in\Omega'.
\end{align*}
因此 $u\circ\Phi\in W^{1,p}(\Omega')$。定理得证。
\end{proof}

\begin{lemma}\label{lemma:高阶链式法则展开}
设 $Q$ 为 $n$ 阶正交矩阵，$x_0\in\mathbb{R}^n$，$u$ 在包含 $x_0+Q^T y$ 的某个开集上具有直到 $k$ 阶的弱导数。定义
\begin{align*}
\tilde u(y)\triangleq u(x_0+Q^T y).
\end{align*}
则对任意多重指标 $\alpha$ 满足 $|\alpha|\leqslant k$，存在只依赖于 $\alpha$ 和 $Q$ 的常数 $c_{\alpha,\beta}$，使得
\begin{align*}
D^\alpha\tilde u(y)=\sum\limits_{|\beta|\leqslant|\alpha|}c_{\alpha,\beta}D^\beta u(x_0+Q^T y).
\end{align*}
特别地，一阶情形为
\begin{align*}
D_i\tilde u(y)=\sum\limits_{j=1}^n Q_{ij}D_j u(x_0+Q^T y),
\end{align*}
其中 $Q_{ij}$ 为 $Q$ 的 $(i,j)$ 元。
\end{lemma}
\begin{proof}

令
\begin{align*}
x(y)\triangleq x_0+Q^T y.
\end{align*}
则 $x_i(y)=(x_0)_i+\sum\limits_{k=1}^n Q_{ki}y_k$，从而
\begin{align*}
\frac{\partial x_j}{\partial y_i}=Q_{ij}.
\end{align*}
对 $|\alpha|$ 作归纳。
当 $|\alpha|=0$ 时，$\tilde u(y)=u(x(y))$，结论显然成立。
当 $|\alpha|=1$ 时，设 $\alpha=e_i$，由\hyperref[lemma:正交平移变换为等距双射]{正交平移变换为等距双射}知$x(y)$是$C^1$微分同胚,故由\hyperref[theorem:弱导数链式法则]{弱导数链式法则}得
\begin{align*}
D_i\tilde u(y)
&=\sum\limits_{j=1}^n D_j u(x(y))\frac{\partial x_j}{\partial y_i}
=\sum\limits_{j=1}^n Q_{ij}D_j u(x(y)).
\end{align*}
故结论成立。

假设对所有满足 $|\alpha|\leqslant m$ 的多重指标 $\alpha$，结论均成立。现设 $|\alpha|=m+1$。取 $\beta$ 和 $i$ 使得
\begin{align*}
\alpha=\beta+e_i,\qquad |\beta|=m.
\end{align*}
由归纳假设得
\begin{align*}
D^\beta\tilde u(y)=\sum\limits_{|\gamma|\leqslant m}c_{\beta,\gamma}D^\gamma u(x(y)).
\end{align*}
对 $y_i$ 求偏导，得
\begin{align*}
D^\alpha\tilde u(y)
=D_i\left(D^\beta\tilde u(y)\right)
=\sum\limits_{|\gamma|\leqslant m}c_{\beta,\gamma}D_i\left(D^\gamma u(x(y))\right).
\end{align*}
再次使用\hyperref[theorem:弱导数链式法则]{弱导数链式法则}，得
\begin{align*}
D_i\left(D^\gamma u(x(y))\right)
=\sum\limits_{j=1}^n D_j\left(D^\gamma u\right)(x(y))\frac{\partial x_j}{\partial y_i}
=\sum\limits_{j=1}^n Q_{ij}D^{\gamma+e_j}u(x(y)).
\end{align*}
因此
\begin{align*}
D^\alpha\tilde u(y)
=\sum\limits_{|\gamma|\leqslant m}\sum\limits_{j=1}^n c_{\beta,\gamma}Q_{ij}D^{\gamma+e_j}u(x(y)).
\end{align*}
令 $\delta\triangleq\gamma+e_j$，则 $|\delta|\leqslant m+1=|\alpha|$。于是可写成
\begin{align*}
D^\alpha\tilde u(y)
=\sum\limits_{|\delta|\leqslant|\alpha|}c_{\alpha,\delta}D^\delta u(x(y)),
\end{align*}
其中
\begin{align*}
c_{\alpha,\delta}\triangleq\sum\limits_{j=1}^n c_{\beta,\delta-e_j}Q_{ij},
\end{align*}
并约定当 $\delta-e_j$ 不是非负多重指标或 $|\delta-e_j|>m$ 时，$c_{\beta,\delta-e_j}=0$。这些常数只依赖于 $\alpha$ 和 $Q$。归纳完成。
\end{proof}

\begin{lemma}[正交平移变换为等距双射]\label{lemma:正交平移变换为等距双射}
设 $Q$ 为 $n$ 阶正交矩阵，$x_0\in\mathbb{R}^n$，定义
\begin{align*}
T(y)\triangleq x_0+Q^T y,\qquad y\in\mathbb{R}^n.
\end{align*}
则 $T$ 是 $\mathbb{R}^n$ 到自身的等距双射，其逆映射为
\begin{align*}
T^{-1}(x)=Q(x-x_0),\qquad x\in\mathbb{R}^n.
\end{align*}
并且 $T$ 与 $T^{-1}$ 都是 Lipschitz 映射，Lipschitz 常数均为 $1$。
\end{lemma}
\begin{proof}

对任意 $y_1,y_2\in\mathbb{R}^n$，由 $Q^T$ 为正交矩阵且正交矩阵保持范数可知
\begin{align*}
|T(y_1)-T(y_2)|
=|Q^T(y_1-y_2)|
=|y_1-y_2|.
\end{align*}
故 $T$ 为等距映射，特别地 $T$ 是单射。

对任意 $x\in\mathbb{R}^n$，令
$y\triangleq Q(x-x_0),$
则
\begin{align*}
T(y)=x_0+Q^T Q(x-x_0)=x_0+(x-x_0)=x.
\end{align*}
故 $T$ 是满射。因此 $T$ 是双射，且其逆映射为
\begin{align*}
T^{-1}(x)=Q(x-x_0).
\end{align*}
对任意 $x_1,x_2\in\mathbb{R}^n$，同样有
\begin{align*}
|T^{-1}(x_1)-T^{-1}(x_2)|
=|Q(x_1-x_2)|
=|x_1-x_2|.
\end{align*}
所以 $T$ 与 $T^{-1}$ 都是 Lipschitz 映射，Lipschitz 常数均为 $1$。
\end{proof}

\begin{lemma}[Lipschitz映射保持零测集]\label{lemma:Lipschitz映射保持零测集}
设 $f:\mathbb{R}^n\to\mathbb{R}^n$ 为 Lipschitz 映射，Lipschitz 常数为 $L$。若 $N\subseteq\mathbb{R}^n$ 为零测集，则 $f(N)$ 也为零测集。

特别地，若 $f$ 还是双射且 $f^{-1}$ 也是 Lipschitz 映射，则 $N$ 为零测集当且仅当 $f(N)$ 为零测集。
\end{lemma}
\begin{proof}

若 $L=0$，则 $f$ 为常值映射，$f(N)$ 至多含一个点，显然为零测集。以下设 $L>0$。

任取 $\varepsilon>0$。由 $N$ 为零测集及 Lebesgue 外测度的定义，存在可数个闭立方体 $\{Q_i\}_{i=1}^{\infty}$，每个 $Q_i$ 的中心为 $a_i$，半边长为 $r_i>0$，使得
\begin{align*}
N\subseteq\bigcup\limits_{i=1}^{\infty}Q_i,\qquad
\sum\limits_{i=1}^{\infty}|Q_i|<\frac{\varepsilon}{C},
\end{align*}
其中 $|Q_i|=(2r_i)^n$ 表示立方体 $Q_i$ 的体积，而$C\triangleq \omega_n L^n n^{\frac n2}2^{-n},$
$\omega_n$ 为 $\mathbb{R}^n$ 中单位球的体积。

对任意 $x\in N$，存在 $i$ 使得 $x\in Q_i$。由于 $Q_i$ 的中心为 $a_i$，半边长为 $r_i$，故
\begin{align*}
|x-a_i|\leqslant\sqrt{n}\,r_i.
\end{align*}
由 $f$ 的 Lipschitz 连续性，得
\begin{align*}
|f(x)-f(a_i)|\leqslant L|x-a_i|<L\sqrt{n}\,r_i.
\end{align*}
因此
\begin{align*}
f(Q_i)\subseteq B(f(a_i),L\sqrt{n}\,r_i).
\end{align*}
于是
\begin{align*}
f(N)\subseteq\bigcup\limits_{i=1}^{\infty}B(f(a_i),L\sqrt{n}\,r_i).
\end{align*}
注意到
\begin{align}\label{eq::JIOJWIOJFIO}
\sum\limits_{i=1}^{\infty}|B(f(a_i),L\sqrt{n}\,r_i)|
&=\sum\limits_{i=1}^{\infty}\omega_n (L\sqrt{n}\,r_i)^n
=\omega_n L^n n^{\frac n2}\sum\limits_{i=1}^{\infty}r_i^n.
\end{align}
另一方面，
\begin{align*}
\sum\limits_{i=1}^{\infty}|Q_i|
=\sum\limits_{i=1}^{\infty}(2r_i)^n
=2^n\sum\limits_{i=1}^{\infty}r_i^n\Longrightarrow \sum\limits_{i=1}^{\infty}r_i^n
=2^{-n}\sum\limits_{i=1}^{\infty}|Q_i|.
\end{align*}
代入\eqref{eq::JIOJWIOJFIO}式得
\begin{align*}
\sum\limits_{i=1}^{\infty}|B(f(a_i),L\sqrt{n}\,r_i)|
=\omega_n L^n n^{\frac n2}2^{-n}\sum\limits_{i=1}^{\infty}|Q_i|
<C\cdot\frac{\varepsilon}{C}
=\varepsilon.
\end{align*}
因此 $f(N)$ 可以被总体积任意小的可数球列覆盖，故其外测度为零，即 $f(N)$ 为零测集。

最后，若 $f$ 为双射且 $f^{-1}$ 也是 Lipschitz 映射.将上述结论分别应用于 $f$ 与 $f^{-1}$，可知 $N$ 为零测集当且仅当 $f(N)$ 为零测集。
\end{proof}

\begin{lemma}[保持零测集的双射保持 $L^\infty$ 范数]\label{lemma:零测集双射保Linfty范数}
设 $U,\widetilde U\subseteq\mathbb{R}^n$ 为可测集，$T:\widetilde U\to U$ 为双射，且 $T$ 与 $T^{-1}$ 都把零测集映为零测集。则对任意可测函数 $v:U\to\mathbb{R}$，有
\begin{align*}
\|v\circ T\|_{L^\infty(\widetilde U)}=\|v\|_{L^\infty(U)}.
\end{align*}
\end{lemma}
\begin{proof}

对任意 $M\geqslant 0$，定义
\begin{align*}
A_M\triangleq\{x\in U\mid |v(x)|>M\},\qquad
B_M\triangleq\{y\in\widetilde U\mid |v(T(y))|>M\}.
\end{align*}
由 $T$ 为双射，有
\begin{align*}
B_M=T^{-1}(A_M).
\end{align*}
因为 $T$ 与 $T^{-1}$ 都把零测集映为零测集，所以 $A_M$ 为零测集当且仅当 $B_M$ 为零测集。

由 $L^\infty$ 范数的本质确界定义知
\begin{gather*}
\|v\|_{L^\infty(U)}
=\inf\{M\geqslant 0\mid A_M\text{ 为零测集}\},\\
\|v\circ T\|_{L^\infty(\widetilde U)}
=\inf\{M\geqslant 0\mid B_M\text{ 为零测集}\}.
\end{gather*}
由于对每个 $M\geqslant 0$，$A_M$ 为零测集当且仅当 $B_M$ 为零测集，所以上述两个下确界相等。因此
\begin{align*}
\|v\circ T\|_{L^\infty(\widetilde U)}=\|v\|_{L^\infty(U)}.
\end{align*}
\end{proof}

\begin{lemma}[正交变换下 $W^{k,p}$ 范数的等价性]\label{lemma:正交变换Wkp等价}
设 $Q$ 为 $n$ 阶正交矩阵，$x_0\in\mathbb R^n$，$U\subseteq\mathbb R^n$ 为开集，令
\begin{align*}
\widetilde U\triangleq\{y\in\mathbb R^n\mid x_0+Q^T y\in U\}.
\end{align*}
对 $u\in W^{k,p}(U)$，其中 $1\leqslant p\leqslant\infty$，定义
\begin{align*}
\tilde u(y)\triangleq u(x_0+Q^T y),\qquad y\in\widetilde U.
\end{align*}
则 $\tilde u\in W^{k,p}(\widetilde U)$，且存在只依赖于 $n,k$ 的常数 $C_1,C_2>0$，使得
\begin{align*}
C_1\|u\|_{W^{k,p}(U)}\leqslant\|\tilde u\|_{W^{k,p}(\widetilde U)}\leqslant C_2\|u\|_{W^{k,p}(U)}.
\end{align*}
特别地，$u\mapsto\tilde u$ 是 $W^{k,p}(U)$ 到 $W^{k,p}(\widetilde U)$ 的线性同构，且当 $1\leqslant p\leqslant \infty$ 时，保持$L^p$ 范数,即
\begin{align*}
\|\tilde u\|_{L^p(\widetilde U)}=\|u\|_{L^p(U)}.
\end{align*}
\end{lemma}
\begin{proof}

令
\begin{align*}
T(y)\triangleq x_0+Q^T y,\qquad T^{-1}(x)=Q(x-x_0).
\end{align*}
由\hyperref[lemma:正交平移变换为等距双射]{正交平移变换为等距双射}，$T$ 与 $T^{-1}$ 均为 Lipschitz 双射，又由\hyperref[lemma:Lipschitz映射保持零测集]{Lipschitz映射保持零测集},因此 $T$ 与 $T^{-1}$ 都把零测集映为零测集。再由\hyperref[theorem:向量求导基本性质]{向量求导基本性质}知
\begin{align*}
\left| \det \left[ D\left( x_0+Q^Ty \right) \right] \right|=\left| \det Q^T \right|=1,\quad \left| \det \left[ D\left( Q\left( x-x_0 \right) \right) \right] \right|=\left| \det Q \right|=1.
\end{align*}
设 $\alpha$ 为多重指标且 $|\alpha|\leqslant k$，则由\cref{lemma:高阶链式法则展开}，存在只依赖于 $\alpha$ 和 $Q$ 的常数 $c_{\alpha,\beta}$，使得
\begin{align}\label{eqw::JIOjfiojwiojeiowjj23248}
D^\alpha\tilde u(y)=\sum\limits_{|\beta|\leqslant|\alpha|}c_{\alpha,\beta}D^\beta u(x_0+Q^T y).
\end{align}

{\heiti 情形一：$1\leqslant p<\infty$。}
先看 $L^p$ 部分。
由变量替换公式，得
\begin{align*}
\|\tilde u\|_{L^p(\widetilde U)}^p
=\int_{\widetilde U}|u(x_0+Q^T y)|^p\,\mathrm dy
=\int_U|u(x)|^p\left| \det \left[ D\left( x_0+Q^Ty \right) \right] \right|\,\mathrm dx
=\|u\|_{L^p(U)}^p.
\end{align*}
故保持$L^p$ 范数。

再看高阶导数。记$C_\alpha\triangleq \max_{|\beta|\leqslant |\alpha|}\{|c_{\alpha,\beta}|\}$,则由\eqref{eqw::JIOjfiojwiojeiowjj23248}式可得
\begin{align*}
|D^\alpha\tilde u(y)|^p
\leqslant C_\alpha\sum\limits_{|\beta|\leqslant|\alpha|}|D^\beta u(x_0+Q^T y)|^p.
\end{align*}
对 $y\in\widetilde U$ 积分并作变量替换，得到
\begin{align*}
&\int_{\widetilde{U}}{|D^{\alpha}\tilde{u}(y)|^p\,\mathrm{d}y}\leqslant C_{\alpha}\sum_{|\beta |\leqslant |\alpha |}{\int_{\widetilde{U}}{|D^{\alpha}\tilde{u}(y)|^p\,\mathrm{d}y}}\\
&=C_{\alpha}\sum_{|\beta |\leqslant |\alpha |}{\int_U{|D^{\beta}u(x)|^p\left|\det \left[ D\left( x_0+Q^Ty \right) \right]\right| \,\mathrm{d}x}}\\
&=C_{\alpha}\sum_{|\beta |\leqslant |\alpha |}{\int_U{|D^{\beta}u(x)|^p\,\mathrm{d}x}}.
\end{align*}
对所有 $|\alpha|\leqslant k$ 求和，再记$C\triangleq \sum_{|\alpha|\leqslant k}C_\alpha$,则
\begin{align*}
\|\tilde u\|_{W^{k,p}(\widetilde U)}^p
\leqslant C\sum\limits_{|\beta|\leqslant k}\int_U|D^\beta u(x)|^p\,\mathrm dx
=C\|u\|_{W^{k,p}(U)}^p.
\end{align*}
反向不等式由逆变换 $u(x)=\tilde u(Q(x-x_0))$ 同理得到。因此存在只依赖于 $n,k$ 的常数 $C_1,C_2>0$，使得
\begin{align*}
C_1\|u\|_{W^{k,p}(U)}\leqslant\|\tilde u\|_{W^{k,p}(\widetilde U)}\leqslant C_2\|u\|_{W^{k,p}(U)}.
\end{align*}

{\heiti 情形二：$p=\infty$。}
由于 $T$ 与 $T^{-1}$ 都把零测集映为零测集，故由\cref{lemma:零测集双射保Linfty范数}知,对任意可测函数 $v$，有
\begin{align*}
\|v\circ T\|_{L^\infty(\widetilde U)}=\|v\|_{L^\infty(U)}.
\end{align*}
于是
\begin{align*}
\|\tilde u\|_{L^\infty(\widetilde U)}=\|u\|_{L^\infty(U)}.
\end{align*}
故保持$L^\infty$ 范数。

由\eqref{eqw::JIOjfiojwiojeiowjj23248}式和$D^\beta u\in L^\infty(U)(|\beta|\leqslant |\alpha|)$可得,$D^\alpha\tilde u\in L^\infty(\widetilde U)$，并且同样记$C_\alpha\triangleq \max_{|\beta|\leqslant |\alpha|}\{|c_{\alpha,\beta}|\}$,则有
\begin{align*}
\|D^\alpha\tilde u\|_{L^\infty(\widetilde U)}
\leqslant C_\alpha\sum\limits_{|\beta|\leqslant|\alpha|}\|D^\beta u\|_{L^\infty(U)}.
\end{align*}
对所有 $|\alpha|\leqslant k$ 求和，得到
\begin{align*}
\|\tilde u\|_{W^{k,\infty}(\widetilde U)}
\leqslant C\|u\|_{W^{k,\infty}(U)}.
\end{align*}
反向不等式由逆变换 $u(x)=\tilde u(Q(x-x_0))$ 同理得到。因此存在只依赖于 $n,k$ 的常数 $C_1,C_2>0$，使得
\begin{align}\label{eq::fjiosdijojioFJIOJIOSIJO}
C_1\|u\|_{W^{k,\infty}(U)}\leqslant\|\tilde u\|_{W^{k,\infty}(\widetilde U)}\leqslant C_2\|u\|_{W^{k,\infty}(U)}.
\end{align}

定义映射
\begin{align*}
L:W^{k,p}(U)\to W^{k,p}(\widetilde U),\qquad Lu\triangleq \tilde u.
\end{align*}
由定义 $Lu(y)=u(x_0+Q^T y)$，显然 $L$ 是线性的。由\eqref{eq::fjiosdijojioFJIOJIOSIJO}式知，存在 $C_1,C_2>0$，使得
\begin{align*}
C_1\|u\|_{W^{k,p}(U)}\leqslant\|Lu\|_{W^{k,p}(\widetilde U)}\leqslant C_2\|u\|_{W^{k,p}(U)}.
\end{align*}
左端不等式说明 $L$ 是单射：若 $Lu=0$，则 $\|Lu\|=0$，从而 $\|u\|=0$，即 $u=0$。

为证满射，任取 $\tilde u\in W^{k,p}(\widetilde U)$，定义
\begin{align*}
u(x)\triangleq \tilde u(Q(x-x_0)),\qquad x\in U.
\end{align*}
由\eqref{eq::fjiosdijojioFJIOJIOSIJO}式知$u\in W^{k,p}(U)$，且
\begin{align*}
Lu(y)=u(x_0+Q^T y)=\tilde u(Q(x_0+Q^T y-x_0))=\tilde u(y),\qquad y\in\widetilde U.
\end{align*}
故 $L$ 是满射。因此 $L$ 是双射。

最后，逆映射
\begin{align*}
L^{-1}:W^{k,p}(\widetilde U)\to W^{k,p}(U),\qquad L^{-1}\tilde u(x)=\tilde u(Q(x-x_0))
\end{align*}
也是线性的。并且由\eqref{eq::fjiosdijojioFJIOJIOSIJO}式知
\begin{align*}
\|L^{-1}\tilde u\|_{W^{k,p}(U)}\leqslant C_1^{-1}\|\tilde u\|_{W^{k,p}(\widetilde U)},
\end{align*}
故 $L^{-1}$ 有界。因此 $L$ 是 $W^{k,p}(U)$ 到 $W^{k,p}(\widetilde U)$ 的线性同构。
\end{proof}

\begin{lemma}\label{lemma:高阶链式法则多项式表示}
设 $\Omega,\Omega'\subseteq\mathbb{R}^n$ 为开集，$\Phi:\Omega'\to\Omega$ 为 $C^k$ 映射，$u\in C^k(\Omega)$，其中 $k\in\mathbb{N}$。则对任意多重指标 $\alpha$ 满足 $|\alpha|\leqslant k$，有
\begin{align*}
D^\alpha(u\circ\Phi)(y)=\sum\limits_{1\leqslant|\beta|\leqslant|\alpha|}P_{\alpha,\beta}\bigl(\{D^\gamma\Phi(y):1\leqslant|\gamma|\leqslant|\alpha|\}\bigr)D^\beta u(\Phi(y)),
\end{align*}
其中 $P_{\alpha,\beta}$ 是依赖于 $\alpha,\beta$ 的常数系数多项式。当 $|\alpha|=0$ 时，约定$D^0(u\circ\Phi)(y)=u(\Phi(y)).$
\end{lemma}
\begin{proof}

对 $|\alpha|$ 作归纳。
当 $|\alpha|=0$ 时，$D^0(u\circ\Phi)=u\circ\Phi$，结论显然。
当 $|\alpha|=1$ 时，设 $\alpha=e_i$，由\hyperref[theorem:弱导数链式法则]{弱导数链式法则}知
\begin{align*}
D_i(u\circ\Phi)(y)=\sum\limits_{j=1}^n D_j u(\Phi(y))D_i\Phi_j(y).
\end{align*}
取 $P_{e_i,e_j}=D_i\Phi_j$，它是 $D\Phi$ 的一次多项式，故结论成立。

假设对所有满足 $|\alpha|\leqslant m$ 的多重指标结论成立。现设 $|\alpha|=m+1$。取 $\beta$ 和 $i$ 使得
\begin{align*}
\alpha=\beta+e_i,\qquad |\beta|=m.
\end{align*}
由归纳假设，
\begin{align*}
D^\beta(u\circ\Phi)(y)=\sum\limits_{1\leqslant|\gamma|\leqslant m}P_{\beta,\gamma}\bigl(\{D^\delta\Phi(y)\}\bigr)D^\gamma u(\Phi(y)).
\end{align*}
对 $y_i$ 求偏导，由\hyperref[theorem:弱导数链式法则]{弱导数链式法则}和\hyperref[theorem:ch05-弱导数性质-4]{弱导数的Leibniz法则}可得
\begin{align*}
D^\alpha(u\circ\Phi)(y)
&=D_i\left(\sum\limits_{1\leqslant|\gamma|\leqslant m}P_{\beta,\gamma}D^\gamma u(\Phi(y))\right)\\
&=\sum\limits_{1\leqslant|\gamma|\leqslant m}\left(D_i P_{\beta,\gamma}\right)D^\gamma u(\Phi(y))
+\sum\limits_{1\leqslant|\gamma|\leqslant m}P_{\beta,\gamma}D_i\bigl(D^\gamma u(\Phi(y))\bigr).
\end{align*}
对于第二项，再次使用一阶链式法则，
\begin{align*}
D_i\bigl(D^\gamma u(\Phi(y))\bigr)=\sum\limits_{j=1}^n D_j(D^\gamma u)(\Phi(y))D_i\Phi_j(y)
=\sum\limits_{j=1}^n D^{\gamma+e_j}u(\Phi(y))D_i\Phi_j(y).
\end{align*}
因此
\begin{align*}
D^\alpha(u\circ\Phi)(y)
&=\sum\limits_{1\leqslant|\gamma|\leqslant m}\left(D_i P_{\beta,\gamma}\right)D^\gamma u(\Phi(y))
+\sum\limits_{1\leqslant|\gamma|\leqslant m}\sum\limits_{j=1}^n P_{\beta,\gamma}D_i\Phi_j\,D^{\gamma+e_j}u(\Phi(y)).
\end{align*}
第一项中，$D_i P_{\beta,\gamma}$ 仍是关于 $\{D^\delta\Phi\}$ 的多项式，乘以 $D^\gamma u$，其中 $|\gamma|\leqslant m<|\alpha|$。第二项中，$P_{\beta,\gamma}D_i\Phi_j$ 是关于 $\{D^\delta\Phi\}$ 的多项式，乘以 $D^{\gamma+e_j}u$，其中 $|\gamma+e_j|\leqslant m+1=|\alpha|$。因此所有项都可以合并写成
\begin{align*}
D^\alpha(u\circ\Phi)(y)=\sum\limits_{1\leqslant|\delta|\leqslant|\alpha|}P_{\alpha,\delta}\bigl(\{D^\eta\Phi(y):1\leqslant|\eta|\leqslant|\alpha|\}\bigr)D^\delta u(\Phi(y)),
\end{align*}
其中 $P_{\alpha,\delta}$ 是相应多项式的和。归纳完成。
\end{proof}

\begin{theorem}[微分同胚下整数阶 Sobolev 空间的全局同构及其刻画]
\label{theorem:一般微分同胚Wkp等价}
设 $\Omega,\Omega'\subseteq\mathbb{R}^n$ 为非空开集，
$k\geqslant 1$ 为整数，且
$\Phi:\Omega'\to\Omega$ 为 $C^k$ 微分同胚。
记 $\Psi\triangleq\Phi^{-1}$，并以
\[
T_\Phi u\triangleq u\circ\Phi
\]
表示由 $\Phi$ 诱导的拉回映射。
以下四个条件等价：
\begin{enumerate}
    \item[(i)]
    $\Phi$ 与 $\Psi$ 的 $1,\ldots,k$ 阶导数均全局一致有界，即
    \[
    \|D^j\Phi\|_{L^\infty(\Omega')}<\infty,
    \qquad
    \|D^j\Psi\|_{L^\infty(\Omega)}<\infty,
    \qquad 1\leqslant j\leqslant k.
    \]

    \item[(ii)]
    存在有限常数 $M_1,\ldots,M_k$，使得
    \[
    M_j\triangleq
    \|D^j\Phi\|_{L^\infty(\Omega')}<\infty,
    \qquad 1\leqslant j\leqslant k,
    \]
    且
    \[
    \delta\triangleq
    \inf_{y\in\Omega'}|\det D\Phi(y)|>0.
    \]

    \item[(iii)]
    映射
    \[
    T_\Phi:
    W^{k,\infty}(\Omega)\longrightarrow W^{k,\infty}(\Omega')
    \]
    是有界线性同构，即它是有界线性双射，
    且其逆映射也是有界线性映射。

    \item[(iv)]
    对每个整数 $0\leqslant m\leqslant k$ 及每个
    $1\leqslant p\leqslant\infty$，映射
    \[
    T_\Phi:
    W^{m,p}(\Omega)\longrightarrow W^{m,p}(\Omega')
    \]
    都是有界线性同构，且
    \[
    T_\Phi^{-1}v=v\circ\Psi.
    \]
    这里约定 $W^{0,p}=L^p$。
\end{enumerate}

进一步，在条件 \textup{(ii)} 成立时，
对每个 $1\leqslant p\leqslant\infty$，
存在仅依赖于
$n,k,p,M_1,\ldots,M_k,\delta$ 的常数 $C\geqslant 1$，
使得对所有整数 $0\leqslant m\leqslant k$
及所有 $u\in W^{m,p}(\Omega)$，均有
\[
C^{-1}\|u\|_{W^{m,p}(\Omega)}
\leqslant
\|u\circ\Phi\|_{W^{m,p}(\Omega')}
\leqslant
C\|u\|_{W^{m,p}(\Omega)}.
\]
本定理不要求 $\Omega,\Omega'$ 有界、凸或具有边界正则性，
也不要求 $\Phi,\Psi$ 本身有界。
\end{theorem}

\begin{proof}
证明中对 $D^j\Phi$ 和 $D^j\Psi$ 采用相应多重线性映射的算子范数。
由于各阶导数连续，其本质上界与逐点上界一致。

对任意开集 $U$，采用如下等价的 Sobolev 范数：
\[
\|u\|_{W^{m,p}(U)}
\triangleq
\sum_{|\alpha|\leqslant m}
\|D^\alpha u\|_{L^p(U)},
\qquad 1\leqslant p\leqslant\infty.
\]
换用通常的其他等价范数，只会改变依赖于 $n,m,p$ 的常数。

首先说明复合运算在几乎处处相等的函数等价类上定义良好。
由于 $\Phi$ 和 $\Psi$ 均为 $C^1$ 映射，
它们在定义域的每一点附近都是 Lipschitz 映射。
将定义域用可数个这样的邻域覆盖，
并利用 Lipschitz 映射将 Lebesgue 零测集映为零测集这一事实，
可知 $\Phi$ 和 $\Psi$ 都将零测集映为零测集。
因此，若 $u_1=u_2$ 几乎处处成立于 $\Omega$，
则
\[
u_1\circ\Phi=u_2\circ\Phi
\quad\text{几乎处处成立于 }\Omega'.
\]
对与 $\Psi$ 的复合也有相同结论。

\medskip
\noindent
\textbf{第一步：证明 \textup{(i)} 与 \textup{(ii)} 等价。}

设 \textup{(i)} 成立，并记
\[
N_1\triangleq\|D\Psi\|_{L^\infty(\Omega)}.
\]
由逆映射的微分公式，
\[
D\Psi(\Phi(y))=(D\Phi(y))^{-1},
\qquad y\in\Omega'.
\]
于是
\[
|\det D\Phi(y)|
=
\frac{1}{|\det D\Psi(\Phi(y))|}
\geqslant N_1^{-n}>0.
\]
故 \textup{(ii)} 成立。

反之，设 \textup{(ii)} 成立。
对任意可逆矩阵 $A\in\mathbb{R}^{n\times n}$，
由奇异值分解可得
\[
\|A^{-1}\|
\leqslant
\frac{\|A\|^{n-1}}{|\det A|}.
\]
因此
\[
\|D\Psi(x)\|
=
\|(D\Phi(\Psi(x)))^{-1}\|
\leqslant
\frac{M_1^{n-1}}{\delta},
\qquad x\in\Omega.
\]
特别地，
\[
N_1=\|D\Psi\|_{L^\infty(\Omega)}
\leqslant \frac{M_1^{n-1}}{\delta}<\infty.
\]

下面归纳证明 $\Psi$ 的高阶导数也全局有界。
设 $2\leqslant r\leqslant k$，
并已知
\[
N_j\triangleq\|D^j\Psi\|_{L^\infty(\Omega)}<\infty,
\qquad 1\leqslant j\leqslant r-1.
\]
对恒等式
\[
\Phi\circ\Psi=\operatorname{id}_\Omega
\]
求 $r$ 阶导数。由经典高阶链式法则，
\[
(D\Phi\circ\Psi)D^r\Psi+R_r=0,
\]
其中 $R_r$ 是由
\[
D^s\Phi\circ\Psi,
\qquad 2\leqslant s\leqslant r,
\]
以及
\[
D^{\ell_1}\Psi,\ldots,D^{\ell_s}\Psi,
\qquad
\ell_1+\cdots+\ell_s=r,\quad \ell_i\geqslant 1,
\]
通过有限次多重线性复合和求和组成的表达式。
由于 $s\geqslant 2$，
上述每个 $\ell_i$ 都不超过 $r-1$。
因此存在仅依赖于 $n,r$ 的常数 $c_{n,r}$，使得
\[
\|R_r\|_{L^\infty(\Omega)}
\leqslant
c_{n,r}
\sum_{s=2}^{r}M_s
\sum_{\substack{\ell_1+\cdots+\ell_s=r\\ \ell_i\geqslant 1}}
N_{\ell_1}\cdots N_{\ell_s}.
\]
又因为
\[
D^r\Psi
=
-(D\Phi\circ\Psi)^{-1}R_r,
\]
故
\[
N_r
\leqslant
c_{n,r}N_1
\sum_{s=2}^{r}M_s
\sum_{\substack{\ell_1+\cdots+\ell_s=r\\ \ell_i\geqslant 1}}
N_{\ell_1}\cdots N_{\ell_s}
<\infty.
\]
归纳得
\[
\|D^j\Psi\|_{L^\infty(\Omega)}
\leqslant
C_j(n,M_1,\ldots,M_j,\delta),
\qquad 1\leqslant j\leqslant k.
\]
于是 \textup{(i)} 成立。

此外，
\[
|\det D\Phi(y)|\leqslant M_1^n,
\qquad y\in\Omega',
\]
因而
\[
|\det D\Psi(x)|
=
\frac{1}{|\det D\Phi(\Psi(x))|}
\geqslant M_1^{-n},
\qquad x\in\Omega.
\]
这表明，在 \textup{(ii)} 成立时，
逆映射 $\Psi$ 也满足同类型的条件，
且相应常数仅依赖于
$n,k,M_1,\ldots,M_k,\delta$。

\medskip
\noindent
\textbf{第二步：建立局部弱导数链式法则。}

本步骤暂时只使用 $\Phi$ 是 $C^k$ 微分同胚这一事实。
设 $1\leqslant m\leqslant k$，
$1\leqslant p\leqslant\infty$，
且 $u\in W^{m,p}_{\mathrm{loc}}(\Omega)$。

对任意光滑函数 $f$，
反复应用经典链式法则可得：
当 $1\leqslant|\alpha|\leqslant m$ 时，
\[
D^\alpha(f\circ\Phi)
=
\sum_{1\leqslant|\beta|\leqslant|\alpha|}
a_{\alpha,\beta}\,(D^\beta f)\circ\Phi,
\]
其中 $a_{\alpha,\beta}$ 是
$D\Phi,\ldots,D^{|\alpha|}\Phi$
的各分量的通用多项式。
这些多项式仅由 $n,\alpha,\beta$ 决定，
其中每个单项式含有至多 $|\alpha|$ 个导数因子。

现证明相同公式对 $u$ 的弱导数成立。
任取开集 $V\Subset\Omega'$。
由于 $\Phi(\overline V)$ 是 $\Omega$ 中的紧集，
可取 $\eta\in C_c^\infty(\Omega)$，
使得 $\eta=1$ 于 $\Phi(\overline V)$ 的某个邻域。

将 $\eta u$ 在 $\Omega$ 外补零，所得函数记为 $w$。
由弱导数乘积法则、$\eta$ 的紧支集性质，
以及有限测度集合上的
$L^p\subseteq L^1$，可得
\[
w\in W^{m,1}(\mathbb{R}^n).
\]
取标准磨光核 $\rho_\varepsilon$，令
\[
w_\varepsilon=\rho_\varepsilon*w.
\]
则
\[
w_\varepsilon\longrightarrow w
\quad\text{于 }W^{m,1}(\mathbb{R}^n).
\]
由于 $w=u$ 于 $\Phi(\overline V)$ 的某个邻域，
从而
\[
w_\varepsilon\longrightarrow u
\quad\text{于 }W^{m,1}(\Phi(V)).
\]

记
\[
Q_V\triangleq
\sup_{x\in\Phi(\overline V)}|\det D\Psi(x)|<\infty.
\]
由变量替换公式，对任意 $g\in L^1(\Phi(V))$，有
\[
\int_V|g(\Phi(y))|\,dy
=
\int_{\Phi(V)}|g(x)|\,|\det D\Psi(x)|\,dx
\leqslant
Q_V\|g\|_{L^1(\Phi(V))}.
\]
因此
\[
w_\varepsilon\circ\Phi
\longrightarrow u\circ\Phi
\quad\text{于 }L^1(V),
\]
且对所有 $|\beta|\leqslant m$，
\[
(D^\beta w_\varepsilon)\circ\Phi
\longrightarrow (D^\beta u)\circ\Phi
\quad\text{于 }L^1(V).
\]

各系数 $a_{\alpha,\beta}$ 在 $V$ 上有界，
故经典链式法则进一步给出
\[
D^\alpha(w_\varepsilon\circ\Phi)
\longrightarrow
G_\alpha
\quad\text{于 }L^1(V),
\]
其中
\[
G_\alpha
\triangleq
\sum_{1\leqslant|\beta|\leqslant|\alpha|}
a_{\alpha,\beta}\,(D^\beta u)\circ\Phi.
\]

对任意 $\varphi\in C_c^\infty(V)$，
利用上述 $L^1$ 收敛以及
$w_\varepsilon\circ\Phi\in C^m(V)$，有
\begin{align*}
\int_V(u\circ\Phi)D^\alpha\varphi\,dy
&=
\lim_{\varepsilon\to0}
\int_V(w_\varepsilon\circ\Phi)D^\alpha\varphi\,dy\\
&=
(-1)^{|\alpha|}
\lim_{\varepsilon\to0}
\int_VD^\alpha(w_\varepsilon\circ\Phi)\varphi\,dy\\
&=
(-1)^{|\alpha|}\int_VG_\alpha\varphi\,dy.
\end{align*}
所以 $G_\alpha$ 正是 $u\circ\Phi$ 在 $V$ 上的
$\alpha$ 阶弱导数。

当 $p<\infty$ 时，由变量替换公式及系数在 $V$ 上有界，
可知 $G_\alpha\in L^p(V)$；
当 $p=\infty$ 时，
由 $\Psi$ 保持零测集及系数在 $V$ 上有界，
同样得到 $G_\alpha\in L^\infty(V)$。
零阶项也有相同的局部可积性结论。

由于 $V\Subset\Omega'$ 任意，故
\[
u\circ\Phi\in W^{m,p}_{\mathrm{loc}}(\Omega'),
\]
且
\[
D^\alpha(u\circ\Phi)
=
\sum_{1\leqslant|\beta|\leqslant|\alpha|}
a_{\alpha,\beta}\,(D^\beta u)\circ\Phi
\quad\text{几乎处处成立于 }\Omega',
\qquad 1\leqslant|\alpha|\leqslant m.
\]
特别地，这一论证也涵盖 $p=\infty$，
不需要使用 $W^{m,\infty}$ 范数下的光滑逼近。

\medskip
\noindent
\textbf{第三步：证明 \textup{(ii)} 推出 \textup{(iv)}，
并建立全局范数估计。}

以下设 \textup{(ii)} 成立。

当 $1\leqslant p<\infty$ 时，
对任意 $f\in L^p(\Omega)$，
由变量替换公式及
$|\det D\Psi|\leqslant\delta^{-1}$，有
\begin{align*}
\|f\circ\Phi\|_{L^p(\Omega')}^p
&=
\int_{\Omega}|f(x)|^p|\det D\Psi(x)|\,dx\\
&\leqslant
\delta^{-1}\|f\|_{L^p(\Omega)}^p.
\end{align*}
因此
\[
\|f\circ\Phi\|_{L^p(\Omega')}
\leqslant
\delta^{-1/p}\|f\|_{L^p(\Omega)}.
\]
当 $p=\infty$ 时，
由于 $\Phi$ 和 $\Psi$ 均保持零测集，
对于任意 $t\geqslant0$，
集合
\[
\{y\in\Omega':|f(\Phi(y))|>t\}
\]
为零测集，当且仅当
\[
\{x\in\Omega:|f(x)|>t\}
\]
为零测集。因此
\[
\|f\circ\Phi\|_{L^\infty(\Omega')}
=
\|f\|_{L^\infty(\Omega)}.
\]
约定 $1/\infty=0$ 后，
上述两种情形可统一写为
\[
\|f\circ\Phi\|_{L^p(\Omega')}
\leqslant
\delta^{-1/p}\|f\|_{L^p(\Omega)},
\qquad 1\leqslant p\leqslant\infty.
\]

令
\[
M\triangleq1+\sum_{j=1}^kM_j.
\]
由第二步中系数多项式的结构，
对每个 $1\leqslant m\leqslant k$，
存在仅依赖于 $n,m$ 的常数 $c_{n,m}$，使得
\[
\|a_{\alpha,\beta}\|_{L^\infty(\Omega')}
\leqslant c_{n,m}M^m,
\qquad
1\leqslant|\beta|\leqslant|\alpha|\leqslant m.
\]

现取 $u\in W^{m,p}(\Omega)$。
第二步已给出 $u\circ\Phi$ 的局部弱导数公式，
结合上述全局系数界及 $L^p$ 复合估计，得到
\begin{align*}
\|D^\alpha(u\circ\Phi)\|_{L^p(\Omega')}
&\leqslant
\sum_{1\leqslant|\beta|\leqslant|\alpha|}
\|a_{\alpha,\beta}\|_{L^\infty(\Omega')}
\|(D^\beta u)\circ\Phi\|_{L^p(\Omega')}\\
&\leqslant
c_{n,m}M^m\delta^{-1/p}
\sum_{1\leqslant|\beta|\leqslant|\alpha|}
\|D^\beta u\|_{L^p(\Omega)},
\qquad 1\leqslant|\alpha|\leqslant m.
\end{align*}
这里 $p=\infty$ 时仍约定 $\delta^{-1/p}=1$。

因此，上述局部弱导数实际上都属于
$L^p(\Omega')$。
又因为任意紧支集测试函数都支撑于某个
$V\Subset\Omega'$，局部弱导数公式也是全域上的分布导数公式，
从而
\[
u\circ\Phi\in W^{m,p}(\Omega').
\]
对所有 $1\leqslant|\alpha|\leqslant m$ 求和，
再加入零阶项，得到
\[
\|u\circ\Phi\|_{W^{m,p}(\Omega')}
\leqslant
c_{n,m}M^m\delta^{-1/p}
\|u\|_{W^{m,p}(\Omega)}.
\]
适当选取 $c_{n,0}$ 后，
这一估计对 $m=0$ 也成立。

第一步已经证明，$\Psi$ 满足同类型的全局假设：
其 $1,\ldots,k$ 阶导数有界，且
\[
\inf_{x\in\Omega}|\det D\Psi(x)|\geqslant M_1^{-n}>0.
\]
因此，对 $\Psi$ 应用刚刚证明的估计，可得
\[
\|v\circ\Psi\|_{W^{m,p}(\Omega)}
\leqslant
C_-\|v\|_{W^{m,p}(\Omega')},
\qquad v\in W^{m,p}(\Omega'),
\]
其中 $C_-$ 仅依赖于
$n,k,p,M_1,\ldots,M_k,\delta$。

在上式中令 $v=u\circ\Phi$，由
\[
(u\circ\Phi)\circ\Psi=u
\quad\text{几乎处处成立于 }\Omega,
\]
得到
\[
\|u\|_{W^{m,p}(\Omega)}
\leqslant
C_-\|u\circ\Phi\|_{W^{m,p}(\Omega')}.
\]
结合正向估计，便得到所需的双边范数不等式。
由于 $m$ 只取 $0,\ldots,k$ 中的有限多个值，
可将常数取为这些估计常数与 $1$ 的最大值，
使同一个常数适用于所有 $0\leqslant m\leqslant k$。

复合运算显然是线性的，且
\[
T_\Psi T_\Phi=\operatorname{id}_{W^{m,p}(\Omega)},
\qquad
T_\Phi T_\Psi=\operatorname{id}_{W^{m,p}(\Omega')}.
\]
因此 $T_\Phi$ 是有界线性同构，
其有界逆映射为 $T_\Psi$。
于是 \textup{(iv)} 成立。

\medskip
\noindent
\textbf{第四步：证明 \textup{(iii)} 推出 \textup{(i)}。}

设 \textup{(iii)} 成立。
于是存在常数 $A<\infty$，使得
\[
\|u\circ\Phi\|_{W^{k,\infty}(\Omega')}
\leqslant
A\|u\|_{W^{k,\infty}(\Omega)},
\qquad u\in W^{k,\infty}(\Omega).
\]

固定
\[
\chi\in C_c^\infty(\mathbb{R}^n),
\qquad
\chi=1\ \text{于 }B(0,1),
\qquad
\operatorname{supp}\chi\subseteq\overline{B(0,2)}.
\]
对任意 $a\in\Omega$ 及 $1\leqslant i\leqslant n$，
定义
\[
f_{a,i}(x)\triangleq(x_i-a_i)\chi(x-a),
\qquad x\in\Omega.
\]
这些函数是固定全空间光滑函数的平移在 $\Omega$ 上的限制。
由乘积求导公式，
存在仅依赖于 $n,k,\chi$ 的常数 $B$，使得
\[
\|f_{a,i}\|_{W^{k,\infty}(\Omega)}
\leqslant B,
\qquad a\in\Omega,\quad 1\leqslant i\leqslant n.
\]
这里不要求 $f_{a,i}$ 在 $\Omega$ 内具有紧支集。

任取 $y_0\in\Omega'$，令 $a=\Phi(y_0)$。
由 $\Phi$ 的连续性，
存在 $y_0$ 的一个开邻域 $U\subseteq\Omega'$，
使得
\[
\Phi(U)\subseteq B(a,1).
\]
于是，对所有 $y\in U$，
\[
(f_{a,i}\circ\Phi)(y)=\Phi_i(y)-a_i.
\]
因此，对每个 $1\leqslant|\alpha|\leqslant k$，
\[
D^\alpha(f_{a,i}\circ\Phi)(y_0)
=
D^\alpha\Phi_i(y_0).
\]

由于 $f_{a,i}\circ\Phi\in C^k(\Omega')$，
其经典导数与弱导数一致。
连续函数在开集上的本质上界控制其每一点的值，
故
\begin{align*}
|D^\alpha\Phi_i(y_0)|
&=
|D^\alpha(f_{a,i}\circ\Phi)(y_0)|\\
&\leqslant
\|D^\alpha(f_{a,i}\circ\Phi)\|_{L^\infty(\Omega')}\\
&\leqslant
\|f_{a,i}\circ\Phi\|_{W^{k,\infty}(\Omega')}\\
&\leqslant
A\|f_{a,i}\|_{W^{k,\infty}(\Omega)}
\leqslant AB.
\end{align*}
由于 $y_0$、$i$ 和 $\alpha$ 任意，
再利用有限维空间中各范数的等价性，得到
\[
\|D^j\Phi\|_{L^\infty(\Omega')}<\infty,
\qquad 1\leqslant j\leqslant k.
\]

还需证明 $\Psi$ 的各阶导数全局有界。
由 \textup{(iii)}，逆算子
\[
S\triangleq T_\Phi^{-1}:
W^{k,\infty}(\Omega')
\longrightarrow W^{k,\infty}(\Omega)
\]
有界。对任意 $v\in W^{k,\infty}(\Omega')$，
令 $u=Sv$，则
\[
u\circ\Phi=v
\quad\text{几乎处处成立于 }\Omega'.
\]
由于 $\Phi$ 将零测集映为零测集，
与 $\Psi$ 复合可得
\[
u=v\circ\Psi
\quad\text{几乎处处成立于 }\Omega.
\]
因此
\[
S=T_\Psi.
\]
特别地，$T_\Psi$ 是从
$W^{k,\infty}(\Omega')$ 到 $W^{k,\infty}(\Omega)$
的有界复合算子。

对 $\Psi$ 重复上述平移截断函数的论证，得到
\[
\|D^j\Psi\|_{L^\infty(\Omega)}<\infty,
\qquad 1\leqslant j\leqslant k.
\]
故 \textup{(i)} 成立。

最后，\textup{(iv)} 取 $m=k$、$p=\infty$
即得 \textup{(iii)}。
结合以上证明，
\[
\textup{(i)}
\Longleftrightarrow
\textup{(ii)}
\Longrightarrow
\textup{(iv)}
\Longrightarrow
\textup{(iii)}
\Longrightarrow
\textup{(i)},
\]
从而四个条件等价，且所述范数估计成立。
\end{proof}
